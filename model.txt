from __future__ import annotations

from dataclasses import asdict, dataclass
from pathlib import Path
from typing import Literal
import argparse
import re

import pandas as pd


RateFormat = Literal["percent", "decimal"]
ValuationMode = Literal["full", "until_end"]


@dataclass(frozen=True)
class SimulationSummary:
    requested_start_date: pd.Timestamp
    requested_end_date: pd.Timestamp
    effective_valuation_date: pd.Timestamp
    tenor: str
    valuation_mode: str
    initial_amount: float
    final_amount: float
    total_interest: float
    annualized_return_percent: float
    completed_transactions: int
    open_accrual_included: bool


def load_curve(
    file_path: str | Path,
    *,
    sheet_name: str | int = 0,
    date_column: str = "Date",
) -> pd.DataFrame:
    path = Path(file_path)

    if not path.exists():
        raise FileNotFoundError(f"Curve file not found: {path}")

    if path.suffix.lower() in {".xlsx", ".xlsm", ".xls"}:
        curve = pd.read_excel(path, sheet_name=sheet_name)
    elif path.suffix.lower() == ".csv":
        curve = pd.read_csv(path)
    else:
        raise ValueError("Supported curve files: .xlsx, .xlsm, .xls, .csv")

    return prepare_curve(curve, date_column=date_column)


def prepare_curve(
    curve: pd.DataFrame,
    *,
    date_column: str = "Date",
) -> pd.DataFrame:
    curve = curve.copy()

    if date_column in curve.columns:
        curve[date_column] = pd.to_datetime(
            curve[date_column],
            errors="coerce",
        )
        curve = curve.set_index(date_column)
    else:
        curve.index = pd.to_datetime(curve.index, errors="coerce")

    curve = curve[~curve.index.isna()]
    curve.index = pd.DatetimeIndex(curve.index).normalize()
    curve = curve.sort_index()

    if curve.empty:
        raise ValueError("Curve contains no valid dates.")

    if curve.index.has_duplicates:
        raise ValueError("Curve contains duplicate dates.")

    curve.columns = [
        str(column).strip().upper()
        for column in curve.columns
    ]
    curve = curve.apply(pd.to_numeric, errors="coerce")

    return curve


def simulate_rollover(
    curve: pd.DataFrame,
    *,
    start_date: str | pd.Timestamp,
    end_date: str | pd.Timestamp,
    tenor: str,
    amount: float,
    valuation_mode: ValuationMode = "full",
    rate_format: RateFormat = "percent",
) -> tuple[pd.DataFrame, SimulationSummary]:
    """
    valuation_mode="full"
        Include only deposits fully matured on or before end_date.

    valuation_mode="until_end"
        Include completed deposits plus accrued interest on the open
        deposit up to end_date using its original locked tenor rate.

    A term tenor is never changed to ON.
    """
    curve = prepare_curve(curve)

    start = pd.Timestamp(start_date).normalize()
    end = pd.Timestamp(end_date).normalize()
    tenor = tenor.strip().upper()
    valuation_mode = valuation_mode.lower()

    if start >= end:
        raise ValueError("end_date must be after start_date.")

    if amount <= 0:
        raise ValueError("amount must be positive.")

    if valuation_mode not in {"full", "until_end"}:
        raise ValueError(
            "valuation_mode must be 'full' or 'until_end'."
        )

    if tenor not in curve.columns:
        raise ValueError(
            f"Tenor '{tenor}' is not a curve column. "
            f"Available columns: {', '.join(curve.columns)}"
        )

    if start < curve.index.min():
        raise ValueError(
            "The curve starts after the requested start date."
        )

    tenor_offset = parse_tenor(tenor)

    transactions: list[dict[str, object]] = []
    principal = float(amount)
    trade_date = start
    transaction_number = 0
    open_accrual_included = False

    while trade_date < end:
        contractual_maturity = trade_date + tenor_offset

        rate_date, quoted_rate = rate_on_or_before(
            curve,
            trade_date,
            tenor,
        )
        rate_decimal = to_decimal(
            quoted_rate,
            rate_format,
        )

        if contractual_maturity <= end:
            accrual_end = contractual_maturity
            status = "Completed"
        elif valuation_mode == "until_end":
            accrual_end = end
            status = "Open - accrued to end date"
            open_accrual_included = True
        else:
            break

        days = int((accrual_end - trade_date).days)
        interest = principal * rate_decimal * days / 360.0
        accrued_value = principal + interest
        transaction_number += 1

        transactions.append(
            {
                "transaction": transaction_number,
                "status": status,
                "trade_date": trade_date,
                "rate_date": rate_date,
                "contractual_maturity_date": contractual_maturity,
                "accrual_end_date": accrual_end,
                "accrual_days": days,
                "tenor": tenor,
                "rate_input": quoted_rate,
                "rate_percent": rate_decimal * 100.0,
                "start_principal": principal,
                "accrued_interest": interest,
                "accrued_value": accrued_value,
            }
        )

        principal = accrued_value

        if status == "Completed":
            trade_date = contractual_maturity
        else:
            break

    transactions_df = pd.DataFrame(transactions)

    if transactions_df.empty:
        effective_date = start
        final_amount = float(amount)
        completed_count = 0
    else:
        effective_date = pd.Timestamp(
            transactions_df.iloc[-1]["accrual_end_date"]
        )
        final_amount = float(
            transactions_df.iloc[-1]["accrued_value"]
        )
        completed_count = int(
            (transactions_df["status"] == "Completed").sum()
        )

    effective_days = int((effective_date - start).days)
    total_interest = final_amount - amount

    if effective_days > 0:
        annualized_return = (
            (final_amount / amount - 1.0)
            * 360.0
            / effective_days
            * 100.0
        )
    else:
        annualized_return = 0.0

    summary = SimulationSummary(
        requested_start_date=start,
        requested_end_date=end,
        effective_valuation_date=effective_date,
        tenor=tenor,
        valuation_mode=valuation_mode,
        initial_amount=float(amount),
        final_amount=final_amount,
        total_interest=total_interest,
        annualized_return_percent=annualized_return,
        completed_transactions=completed_count,
        open_accrual_included=open_accrual_included,
    )

    return transactions_df, summary


def parse_tenor(tenor: str) -> pd.DateOffset:
    tenor = tenor.upper()

    if tenor == "ON":
        return pd.DateOffset(days=1)

    match = re.fullmatch(r"(\d+)(D|W|M|Y)", tenor)

    if not match:
        raise ValueError(
            f"Unsupported tenor header '{tenor}'. "
            "Use ON or formats such as 1D, 1W, 2M, 6M, 1Y."
        )

    quantity = int(match.group(1))
    unit = match.group(2)

    if quantity <= 0:
        raise ValueError("Tenor quantity must be positive.")

    if unit == "D":
        return pd.DateOffset(days=quantity)

    if unit == "W":
        return pd.DateOffset(weeks=quantity)

    if unit == "M":
        return pd.DateOffset(months=quantity)

    return pd.DateOffset(years=quantity)


def rate_on_or_before(
    curve: pd.DataFrame,
    target_date: pd.Timestamp,
    tenor: str,
) -> tuple[pd.Timestamp, float]:
    valid_rates = curve.loc[
        (curve.index <= target_date) & curve[tenor].notna(),
        tenor,
    ]

    if valid_rates.empty:
        raise ValueError(
            f"No valid {tenor} rate exists on or before "
            f"{target_date.date()}."
        )

    rate_date = valid_rates.index[-1]

    return (
        pd.Timestamp(rate_date),
        float(valid_rates.iloc[-1]),
    )


def to_decimal(
    rate: float,
    rate_format: RateFormat,
) -> float:
    if rate_format == "percent":
        return rate / 100.0

    if rate_format == "decimal":
        return rate

    raise ValueError(
        "rate_format must be 'percent' or 'decimal'."
    )


def export_csv(
    transactions: pd.DataFrame,
    summary: SimulationSummary,
    output_prefix: str | Path,
) -> tuple[Path, Path]:
    """
    Export two CSV files:

        <prefix>_transactions.csv
        <prefix>_summary.csv
    """
    prefix = Path(output_prefix)
    prefix.parent.mkdir(parents=True, exist_ok=True)

    transactions_path = prefix.with_name(
        prefix.name + "_transactions.csv"
    )
    summary_path = prefix.with_name(
        prefix.name + "_summary.csv"
    )

    transactions_to_save = transactions.copy()

    for column in [
        "trade_date",
        "rate_date",
        "contractual_maturity_date",
        "accrual_end_date",
    ]:
        if column in transactions_to_save.columns:
            transactions_to_save[column] = (
                pd.to_datetime(transactions_to_save[column])
                .dt.strftime("%Y-%m-%d")
            )

    summary_data = asdict(summary)

    for key in [
        "requested_start_date",
        "requested_end_date",
        "effective_valuation_date",
    ]:
        summary_data[key] = pd.Timestamp(
            summary_data[key]
        ).strftime("%Y-%m-%d")

    transactions_to_save.to_csv(
        transactions_path,
        index=False,
    )
    pd.DataFrame(
        list(summary_data.items()),
        columns=["metric", "value"],
    ).to_csv(
        summary_path,
        index=False,
    )

    return transactions_path, summary_path


def prompt_user() -> argparse.Namespace:
    print("Rollover return model")

    curve_file = input("Curve file path: ").strip()
    sheet_name = input(
        "Excel sheet name [first sheet]: "
    ).strip()
    date_column = input(
        "Date column [Date]: "
    ).strip() or "Date"
    start_date = input(
        "Start date [YYYY-MM-DD]: "
    ).strip()
    end_date = input(
        "End date [YYYY-MM-DD]: "
    ).strip()
    tenor = input(
        "Tenor matching the column header: "
    ).strip()
    amount = float(
        input("Amount: ").replace(",", "").strip()
    )
    valuation_mode = (
        input("Valuation mode [full/until_end]: ")
        .strip()
        .lower()
        or "full"
    )
    rate_format = (
        input("Rate format [percent/decimal]: ")
        .strip()
        .lower()
        or "percent"
    )
    output_prefix = (
        input("Output prefix [rollover_result]: ")
        .strip()
        or "rollover_result"
    )

    return argparse.Namespace(
        curve_file=curve_file,
        sheet_name=sheet_name or 0,
        date_column=date_column,
        start_date=start_date,
        end_date=end_date,
        tenor=tenor,
        amount=amount,
        valuation_mode=valuation_mode,
        rate_format=rate_format,
        output_prefix=output_prefix,
    )


def parse_arguments() -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description="Calculate term-deposit rollover returns."
    )

    parser.add_argument("--curve-file")
    parser.add_argument("--sheet-name", default=0)
    parser.add_argument("--date-column", default="Date")
    parser.add_argument("--start-date")
    parser.add_argument("--end-date")
    parser.add_argument("--tenor")
    parser.add_argument("--amount", type=float)
    parser.add_argument(
        "--valuation-mode",
        choices=["full", "until_end"],
        default="full",
    )
    parser.add_argument(
        "--rate-format",
        choices=["percent", "decimal"],
        default="percent",
    )
    parser.add_argument(
        "--output-prefix",
        default="rollover_result",
    )

    args = parser.parse_args()

    required_values = [
        args.curve_file,
        args.start_date,
        args.end_date,
        args.tenor,
        args.amount,
    ]

    if any(value in {None, ""} for value in required_values):
        return prompt_user()

    return args


def main() -> None:
    args = parse_arguments()

    curve = load_curve(
        args.curve_file,
        sheet_name=args.sheet_name,
        date_column=args.date_column,
    )

    transactions, summary = simulate_rollover(
        curve,
        start_date=args.start_date,
        end_date=args.end_date,
        tenor=args.tenor,
        amount=args.amount,
        valuation_mode=args.valuation_mode,
        rate_format=args.rate_format,
    )

    transactions_path, summary_path = export_csv(
        transactions,
        summary,
        args.output_prefix,
    )

    print()
    print(f"Final amount: {summary.final_amount:,.2f}")
    print(f"Total interest: {summary.total_interest:,.2f}")
    print(
        "Annualized return: "
        f"{summary.annualized_return_percent:.6f}%"
    )
    print(f"Transactions CSV: {transactions_path}")
    print(f"Summary CSV: {summary_path}")


if __name__ == "__main__":
    main()
