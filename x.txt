The script reproduces the workbook created in ChatGPT using:
- the ON, 1M, 2M, 3M and 6M curve from curve.txt;
- historical FOMC decision dates hard-coded through 17-Jun-2026;
- an analysis window from 03-Jan-2023 through 10-Jul-2026.

The input curve can contain rates either as percentage points, such as 3.5500,

Dependency:
    artifact_tool

Usage:
    python build_sofr_presentation.py

    python build_sofr_presentation.py \
        --curve-file curve.txt \

class CurveData:
    def __init__(self, dates: list[date], rates: dict[str, list[float]]) -> None:
        self.dates = dates
        self.dates_dt = [datetime.combine(d, datetime.min.time()) for d in dates]
        self.rates = rates
        self.last_idx = len(dates) - 1

    def target_date(self, start: date, tenor: str) -> date:
        if tenor == "ON":
            return start + timedelta(days=1)
        return start + relativedelta(months=TENOR_MONTHS[tenor])

    def effective_index(self, target: date) -> int | None:
        idx = bisect_left(self.dates, target)
        return idx if idx < len(self.dates) else None

    def rolled_on_factor(self, start_idx: int, end_idx: int) -> float:
        factor = 1.0
        for idx in range(start_idx, end_idx):
            day_count = (self.dates[idx + 1] - self.dates[idx]).days
            factor *= 1.0 + self.rates["ON"][idx] * day_count / 360.0
        return factor

    def wealth_path(self, start_idx: int, tenor: str) -> list[float | None]:
        values: list[float | None] = [None] * len(self.dates)
        base = 1.0
        current_idx = start_idx
        locked_rate = self.rates[tenor][current_idx]
        maturity_idx = self.effective_index(
            self.target_date(self.dates[current_idx], tenor)
        )

        for idx in range(start_idx, len(self.dates)):
            while (
                maturity_idx is not None
                and idx >= maturity_idx
                and current_idx < idx
            ):
                day_count = (
                    self.dates[maturity_idx] - self.dates[current_idx]
                ).days
                base *= 1.0 + locked_rate * day_count / 360.0
                current_idx = maturity_idx
                if current_idx >= self.last_idx:
                    break
                locked_rate = self.rates[tenor][current_idx]
                maturity_idx = self.effective_index(
                    self.target_date(self.dates[current_idx], tenor)
                )

            elapsed_days = (self.dates[idx] - self.dates[current_idx]).days
            values[idx] = base * (1.0 + locked_rate * elapsed_days / 360.0)

        return values

    def rolling_result(self, start_idx: int, tenor: str) -> float:
        base = 1.0
        current_idx = start_idx
        locked_rate = self.rates[tenor][current_idx]
        maturity_idx = self.effective_index(
            self.target_date(self.dates[current_idx], tenor)
        )

        while (
            maturity_idx is not None
            and maturity_idx <= self.last_idx
            and current_idx < maturity_idx
        ):
            day_count = (
                self.dates[maturity_idx] - self.dates[current_idx]
            ).days
            base *= 1.0 + locked_rate * day_count / 360.0
            current_idx = maturity_idx
            if current_idx >= self.last_idx:
                return base
            locked_rate = self.rates[tenor][current_idx]
            maturity_idx = self.effective_index(
                self.target_date(self.dates[current_idx], tenor)
            )

        elapsed_days = (
            self.dates[self.last_idx] - self.dates[current_idx]
        ).days
        return base * (1.0 + locked_rate * elapsed_days / 360.0)


def _read_text_from_url(url: str) -> str:
    with urllib.request.urlopen(url, timeout=30) as response:
        return response.read().decode("utf-8")


def load_curve(curve_file: str | None, curve_url: str) -> CurveData:
    if curve_file:
        text = Path(curve_file).read_text(encoding="utf-8")
    else:
        text = _read_text_from_url(curve_url)

    reader = csv.DictReader(io.StringIO(text))
    required = ["Date", *TENORS]
    if reader.fieldnames != required:
        raise ValueError(
            f"Expected columns {required}, received {reader.fieldnames}."
        )

    dates: list[date] = []
    rates: dict[str, list[float]] = {tenor: [] for tenor in TENORS}

    for row in reader:
        curve_date = datetime.strptime(row["Date"].strip(), "%Y-%m-%d").date()
        if curve_date > ANALYSIS_END:
            continue

        values = [float(row[tenor]) for tenor in TENORS]
        # GitHub data uses percentage points, e.g. 3.5500. Excel often stores
        # the same rate as 0.0355. Normalize both formats to decimals.
        if max(abs(value) for value in values) > 1.0:
            values = [value / 100.0 for value in values]

        dates.append(curve_date)
        for tenor, value in zip(TENORS, values):
            rates[tenor].append(value)

    if not dates:
        raise ValueError("The curve file contains no usable observations.")
    if dates[-1] != ANALYSIS_END:
        raise ValueError(
            f"Exact reproduction requires curve data through {ANALYSIS_END}; "
            f"last available date is {dates[-1]}."
        )

    return CurveData(dates, rates)


def portfolio_frontier(
    return_matrix: np.ndarray,
    weight_step: float = 0.05,
) -> tuple[
    np.ndarray,
    list[tuple[float, float, np.ndarray]],
    tuple[float, float, np.ndarray],
    tuple[float, float, np.ndarray],
    tuple[float, float, np.ndarray],
]:
    annualized_mean = np.mean(return_matrix, axis=0) * 12.0
    annualized_cov = np.cov(return_matrix, rowvar=False, ddof=1) * 12.0

    units = int(round(1.0 / weight_step))
    portfolios: list[tuple[float, float, np.ndarray]] = []

    for w1 in range(units + 1):
        for w2 in range(units + 1 - w1):
            for w3 in range(units + 1 - w1 - w2):
                w4 = units - w1 - w2 - w3
                weights = np.array([w1, w2, w3, w4], dtype=float) / units
                expected_return = float(weights @ annualized_mean)
                volatility = float(
                    math.sqrt(max(0.0, weights @ annualized_cov @ weights))
                )
                portfolios.append((expected_return, volatility, weights))

    efficient: list[tuple[float, float, np.ndarray]] = []
    best_return = -1e9
    for expected_return, volatility, weights in sorted(
        portfolios,
        key=lambda item: (item[1], -item[0]),
    ):
        if expected_return > best_return + 1e-12:
            efficient.append((expected_return, volatility, weights))
            best_return = expected_return

    minimum_volatility = min(portfolios, key=lambda item: item[1])
    maximum_return = max(portfolios, key=lambda item: item[0])
    best_information_ratio = max(
        portfolios,
        key=lambda item: item[0] / item[1] if item[1] else -1e9,
    )

    return (
        annualized_mean,
        efficient,
        minimum_volatility,
        maximum_return,
        best_information_ratio,
    )


def merged_title(sheet, cell_range: str, text: str) -> None:
    sheet.merge_cells(cell_range)
    first_cell = cell_range.split(":")[0]
    sheet.get_range(first_cell).values = [[text]]
    sheet.get_range(cell_range).format = {
        "fill": DARK,
        "font": {"bold": True, "color": WHITE, "size": 16},
        "row_height": 28,
    }


def merged_section(sheet, cell_range: str, text: str) -> None:
    sheet.merge_cells(cell_range)
    first_cell = cell_range.split(":")[0]
    sheet.get_range(first_cell).values = [[text]]
    sheet.get_range(cell_range).format = {

    }


def weight_description(weights: Iterable[float]) -> str:
    return " / ".join(
        f"{weight:.0%} {tenor}"
        for tenor, weight in zip(TERM_TENORS, weights)
        if weight > 0.001
    )


def build_workbook(curve: CurveData, output_path: str) -> None:
    start_idx = bisect_left(curve.dates, ANALYSIS_START)
    last_idx = curve.last_idx
    last_date = curve.dates[-1]

    # ---------------------------------------------------------------
    # Premium priced versus realized
    # ---------------------------------------------------------------
    premium_summary: list[list[object]] = []
    for tenor in TERM_TENORS:
        entry_values: list[float] = []
        realized_values: list[float] = []

        for idx in range(start_idx, len(curve.dates)):
            maturity_idx = curve.effective_index(
                curve.target_date(curve.dates[idx], tenor)
            )
            if maturity_idx is None:
                break

            day_count = (
                curve.dates[maturity_idx] - curve.dates[idx]
            ).days
            fixed_rate = curve.rates[tenor][idx]
            entry_values.append(
                (fixed_rate - curve.rates["ON"][idx]) * 10000.0
            )
            rolled_equivalent = (
                curve.rolled_on_factor(idx, maturity_idx) - 1.0
            ) * 360.0 / day_count
            realized_values.append(
                (fixed_rate - rolled_equivalent) * 10000.0
            )

        entry = np.array(entry_values)
        realized = np.array(realized_values)
        historical_spreads = np.array(
            [
                (curve.rates[tenor][idx] - curve.rates["ON"][idx])
                * 10000.0
                for idx in range(start_idx, len(curve.dates))
            ]
        )
        current_premium = (
            curve.rates[tenor][-1] - curve.rates["ON"][-1]
        ) * 10000.0

        premium_summary.append(
            [
                tenor,
                len(realized),
                float(entry.mean()),
                float(realized.mean()),
                float(np.median(realized)),
                float(realized.std(ddof=1)),
                float(np.mean(realized > 0)),
                float(np.corrcoef(entry, realized)[0, 1]),
                current_premium,
                float(100.0 * np.mean(historical_spreads <= current_premium)),
            ]
        )

    # ---------------------------------------------------------------
    # Monthly wealth and risk metrics
    # ---------------------------------------------------------------
    wealth_paths = {
        tenor: curve.wealth_path(start_idx, tenor) for tenor in TENORS
    }

    month_end_by_month: dict[tuple[int, int], int] = {}
    for idx in range(start_idx, len(curve.dates)):
        key = (curve.dates[idx].year, curve.dates[idx].month)
        month_end_by_month[key] = idx
    month_end_indices = sorted(month_end_by_month.values())

    monthly_returns = {tenor: [] for tenor in TENORS}
    monthly_on_rate_changes_bps: list[float] = []

    for idx_a, idx_b in zip(month_end_indices[:-1], month_end_indices[1:]):
        monthly_on_rate_changes_bps.append(
            (curve.rates["ON"][idx_b] - curve.rates["ON"][idx_a])
            * 10000.0
        )
        for tenor in TENORS:
            value_a = wealth_paths[tenor][idx_a]
            value_b = wealth_paths[tenor][idx_b]
            assert value_a is not None and value_b is not None
            monthly_returns[tenor].append(value_b / value_a - 1.0)

    on_monthly = np.array(monthly_returns["ON"])
    risk_rows: list[list[object]] = []

    for tenor in TENORS:
        returns = np.array(monthly_returns[tenor])
        annualized_return = float(returns.mean() * 12.0)
        annualized_volatility = float(
            returns.std(ddof=1) * math.sqrt(12.0)
        )

        if tenor == "ON":
            risk_rows.append(
                [
                    tenor,
                    annualized_return,
                    annualized_volatility,
                    0.0,
                    0.0,
                    None,
                    0.0,
                ]
            )
        else:
            excess = returns - on_monthly
            annualized_excess_bps = float(
                excess.mean() * 12.0 * 10000.0
            )
            tracking_error_bps = float(
                excess.std(ddof=1) * math.sqrt(12.0) * 10000.0
            )
            information_ratio = (
                annualized_excess_bps / tracking_error_bps
                if tracking_error_bps
                else None
            )
            risk_rows.append(
                [
                    tenor,
                    annualized_return,
                    annualized_volatility,
                    annualized_excess_bps,
                    tracking_error_bps,
                    information_ratio,
                    float(np.percentile(excess * 10000.0, 5)),
                ]
            )

    # ---------------------------------------------------------------
    # Efficient frontier
    # ---------------------------------------------------------------
    excess_matrix = np.column_stack(
        [
            np.array(monthly_returns[tenor]) - on_monthly
            for tenor in TERM_TENORS
        ]
    )
    overall_frontier = portfolio_frontier(excess_matrix)

    regime_indices = {
        "Hike / Rising": [],
        "Hold / Stable": [],
        "Cut / Falling": [],
    }
    for idx, rate_change_bps in enumerate(monthly_on_rate_changes_bps):
        if rate_change_bps > 5.0:
            regime_indices["Hike / Rising"].append(idx)
        elif rate_change_bps < -5.0:
            regime_indices["Cut / Falling"].append(idx)
        else:
            regime_indices["Hold / Stable"].append(idx)

    regime_frontiers = {
        regime: portfolio_frontier(excess_matrix[indices, :])
        for regime, indices in regime_indices.items()
    }

    # ---------------------------------------------------------------
    # Starting-period returns
    # ---------------------------------------------------------------
    start_rows: list[list[object]] = []
    for label, requested_start in START_SCENARIOS:
        idx = bisect_left(curve.dates, requested_start)
        values = {
            tenor: curve.rolling_result(idx, tenor) * 100.0
            for tenor in TENORS
        }
        best_tenor = max(TENORS, key=lambda tenor: values[tenor])
        horizon_days = (last_date - curve.dates[idx]).days
        factors = {tenor: values[tenor] / 100.0 for tenor in TENORS}
        cagrs = {
            tenor: factors[tenor] ** (365.25 / horizon_days) - 1.0
            for tenor in TENORS
        }

        start_rows.append(
            [
                label,
                curve.dates_dt[idx],
                values["ON"],
                values["1M"],
                values["2M"],
                values["3M"],
                values["6M"],
                best_tenor,
                values[best_tenor] - values["ON"],
                (cagrs[best_tenor] - cagrs["ON"]) * 10000.0,
            ]
        )

    # ---------------------------------------------------------------
    # FOMC historical patterns
    # ---------------------------------------------------------------
    reactions = {"Hike": [], "Hold": [], "Cut": []}
    pre_patterns = {"Hike": [], "Hold": [], "Cut": []}

    for meeting_date, decision, _action in FOMC_HISTORY:
        idx = bisect_left(curve.dates, meeting_date)
        pre_idx = idx - 10
        if pre_idx < start_idx or idx + 1 >= len(curve.dates):
            continue

        pre_spreads = [
            (
                curve.rates[tenor][pre_idx]
                - curve.rates["ON"][pre_idx]
            )
            * 10000.0
            for tenor in TERM_TENORS
        ]
        reaction = [
            (
                curve.rates[tenor][idx + 1]
                - curve.rates[tenor][idx - 1]
            )
            * 10000.0
            for tenor in TENORS
        ]
        pre_patterns[decision].append(pre_spreads)
        reactions[decision].append(reaction)

    pattern_rows: list[list[object]] = []
    for decision in ["Hike", "Hold", "Cut"]:
        matrix = np.array(pre_patterns[decision], dtype=float)
        averages = matrix.mean(axis=0)
        aggregate = matrix.mean(axis=1)
        pattern_rows.append(
            [
                decision,
                len(matrix),
                *averages.tolist(),
                float(aggregate.mean()),
                float(aggregate.std(ddof=1)),
            ]
        )

    current_rates = np.array(
        [curve.rates[tenor][-1] for tenor in TENORS]
    )
    scenario_rows: list[list[object]] = []
    for decision in ["Hold", "Hike", "Cut"]:
        average_reaction = np.array(reactions[decision], dtype=float).mean(
            axis=0
        )
        projected_rates = current_rates + average_reaction / 10000.0
        policy_action = (
            0 if decision == "Hold" else 25 if decision == "Hike" else -25
        )
        scenario_rows.append(
            [
                decision,
                policy_action,
                *average_reaction.tolist(),
                *projected_rates.tolist(),
            ]
        )

    current_spreads = np.array(
        [
            (curve.rates[tenor][-1] - curve.rates["ON"][-1])
            * 10000.0
            for tenor in TERM_TENORS
        ]
    )
    percentile_rows: list[list[object]] = []
    for tenor in TERM_TENORS:
        historical = np.array(
            [
                (
                    curve.rates[tenor][idx]
                    - curve.rates["ON"][idx]
                )
                * 10000.0
                for idx in range(start_idx, len(curve.dates))
            ]
        )
        current = (
            curve.rates[tenor][-1] - curve.rates["ON"][-1]
        ) * 10000.0
        percentile_rows.append(
            [tenor, current, float(100.0 * np.mean(historical <= current))]
        )

    # Pure-expectations illustration used in the original workbook.
    implied_post_meeting_on = (
        curve.rates["1M"][-1] * 31
        - curve.rates["ON"][-1] * 20
    ) / 11
    implied_change_bps = (
        implied_post_meeting_on - curve.rates["ON"][-1]
    ) * 10000.0

    # ---------------------------------------------------------------
    # Create workbook and sheets
    # ---------------------------------------------------------------
    workbook = Workbook.create()
    executive = workbook.worksheets.add("Executive")
    historical = workbook.worksheets.add("Historical Curve")
    premium = workbook.worksheets.add("Premium vs Realized")
    starting = workbook.worksheets.add("Starting Periods")
    risk = workbook.worksheets.add("Risk Return")
    frontier = workbook.worksheets.add("Efficient Frontier")
    future = workbook.worksheets.add("Next FOMC")
    sources = workbook.worksheets.add("Sources")

    # ---------------------------------------------------------------
    # Executive sheet
    # ---------------------------------------------------------------
    merged_title(
        executive,
        "A1:J1",
        "SOFR Curve, Returns and Risk Management",
    )
    executive.merge_cells("A2:J2")
    executive.get_range("A2").values = [
        [
            f"Analysis period: {curve.dates[start_idx].isoformat()} to "
            f"{last_date.isoformat()}. Future scenario is based on the "
            "10-Jul-2026 curve."
        ]
    ]
    executive.get_range("A2:J2").format = {
        "fill": LIGHT,
        "font": {"italic": True, "color": DARK},
    }

    merged_section(executive, "A4:J4", "Recommended Storyline")
    storyline = [
        [
            "1",
            "Historical curve",
            "Connect curve changes to FOMC meetings and show where repricing occurred.",
        ],
        [
            "2",
            "Premium priced vs realized",
            "Show whether quoted tenor premium actually beat rolled ON.",
        ],
        [
            "3",
            "Different starting periods",
            "Demonstrate entry-date sensitivity and the small absolute differences.",
        ],
        [
            "4",
            "Risk and return",
            "Compare volatility, tracking error, downside and information ratio.",
        ],
        [
            "5",
            "Efficient frontier",
            "Identify efficient tenor combinations in rising, stable and falling-rate regimes.",
        ],
        [
            "6",
            "Next meeting",
            "Use the current curve and historical FOMC patterns for hold, hike and cut scenarios.",
        ],
    ]
    executive.get_range("A5:J10").values = [
        [number, heading, explanation, None, None, None, None, None, None, None]
        for number, heading, explanation in storyline
    ]
    for row_number in range(5, 11):
        executive.merge_cells(f"C{row_number}:J{row_number}")
    executive.get_range("A5:A10").format = {
        "fill": DARK,
        "font": {"bold": True, "color": WHITE},
        "horizontal_alignment": "center",
    }
    executive.get_range("B5:B10").format.font = {"bold": True}
    executive.get_range("C5:J10").format.wrap_text = True
    executive.get_range("A5:J10").format.row_height = 34

    merged_section(executive, "A12:J12", "Core Conclusions")
    conclusions = [
        "2M produced the strongest historical risk-adjusted result: +1.06 bp annualized excess return versus ON with 1.97 bp tracking error.",
        "1M was the lowest-risk term extension, with only 0.78 bp annualized tracking error.",
        "3M and 6M were outside the historical efficient frontier since 2023.",
        "In falling-rate periods, 6M had the highest return but also the highest risk; the best risk-adjusted mix was 90% 2M / 10% 6M.",
        "The 10-Jul-2026 curve was unusually steep: all term premiums were above the 91st percentile since 2023.",
        "The curve shape was hike-like, but historical premium-to-realized-return correlations were weak. The presentation should use a hawkish-hold base case, not treat the curve as a certain hike forecast.",
    ]
    for row_number, text in enumerate(conclusions, start=13):
        executive.merge_cells(f"A{row_number}:J{row_number}")
        executive.get_range(f"A{row_number}").values = [[text]]
    executive.get_range("A13:J18").format = {
        "fill": LIGHT,
        "wrap_text": True,
        "row_height": 30,
    }

    merged_section(executive, "A20:J20", "Risk Management Approach")
    risk_management = [
        ["Liquidity core", "ON / 1M", "Minimize rollover and liquidity risk."],
        [

def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description="Build the SOFR presentation workbook."
    )
    group = parser.add_mutually_exclusive_group()
    group.add_argument(
        "--curve-file",
        help="Local curve.txt path. If omitted, the GitHub raw URL is used.",
    )
    group.add_argument(
        "--curve-url",
        default=DEFAULT_CURVE_URL,
        help=f"Curve URL. Default: {DEFAULT_CURVE_URL}",
    )
    parser.add_argument(
        "--output",
        default=DEFAULT_OUTPUT,
        help=f"Output XLSX path. Default: {DEFAULT_OUTPUT}",
    )
    return parser.parse_args()


def main() -> None:
    args = parse_args()
    curve_url = args.curve_url or DEFAULT_CURVE_URL
    curve = load_curve(args.curve_file, curve_url)
    build_workbook(curve, args.output)


if __name__ == "__main__":
    main()
