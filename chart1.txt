Private Const SHEET_DATA As String = "ChartData"
Private Const SHEET_DASH As String = "Dashboard"

Public Sub BuildSOFRExecutiveDashboard()
    Dim wb As Workbook
    Dim wsCurve As Worksheet, wsData As Worksheet, wsDash As Worksheet
    Dim lastRow As Long, outLastRow As Long
    Dim minRate As Double, maxRate As Double
    Dim startDate As Date, endDate As Date
    Dim oldCalc As XlCalculation

    On Error GoTo CleanFail

    Set wb = ThisWorkbook
    oldCalc = Application.Calculation
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    Set wsCurve = GetOrCreateSheet(wb, SHEET_CURVE)
    EnsureCurveHeaders wsCurve

    lastRow = LastUsedRow(wsCurve, "A")
    If lastRow < 2 Then
        MsgBox "Paste curve data into the Curve sheet first." & vbCrLf & _
               "Required columns: Date | ON | 1M | 2M | 3M | 6M", vbExclamation
        GoTo CleanExit
    End If

    ValidateCurveInput wsCurve, lastRow

    Set wsData = GetOrCreateSheet(wb, SHEET_DATA)
    Set wsDash = GetOrCreateSheet(wb, SHEET_DASH)

    ResetSheet wsData
    ResetSheet wsDash

    outLastRow = BuildNormalizedChartData(wsCurve, wsData, lastRow, DateSerial(2022, 1, 1))
    If outLastRow < 3 Then
        MsgBox "No valid curve observations were found on or after Jan-2022.", vbExclamation
        GoTo CleanExit
    End If

    startDate = wsData.Cells(2, 1).Value
    endDate = wsData.Cells(outLastRow, 1).Value
    minRate = WorksheetFunction.Min(wsData.Range("B2:F" & outLastRow))
    maxRate = WorksheetFunction.Max(wsData.Range("B2:F" & outLastRow))

    BuildDashboardLayout wsDash, startDate, endDate
    BuildHistoricalChart wsDash, wsData, outLastRow, minRate, maxRate, startDate, endDate
    BuildSlopeChart wsDash, wsData, outLastRow, startDate, endDate
    BuildCurrentSnapshot wsDash, wsData, outLastRow
    BuildPolicyWindowTable wsDash, wsData, outLastRow
    BuildExecutiveConclusion wsDash, wsData, outLastRow
    ApplyDashboardView wsDash

    wsData.Visible = xlSheetVeryHidden
    wsDash.Activate
    wsDash.Range("A1").Select

CleanExit:
    Application.Calculation = oldCalc
    Application.EnableEvents = True
    Application.ScreenUpdating = True
    Exit Sub

CleanFail:
    Application.Calculation = oldCalc
    Application.EnableEvents = True
    Application.ScreenUpdating = True
    MsgBox "Dashboard could not be built." & vbCrLf & Err.Description, vbCritical
End Sub

Private Sub EnsureCurveHeaders(ByVal ws As Worksheet)
    Dim headers As Variant
    headers = Array("Date", "ON", "1M", "2M", "3M", "6M")
    WriteHorizontalArray ws.Range("A1"), headers
    With ws.Range("A1:F1")
        .Font.Bold = True
        .Font.Color = vbWhite
        .Interior.Color = CLR_NAVY
        .HorizontalAlignment = xlCenter
    End With
    ws.Columns("A").NumberFormat = "mmm-dd-yyyy"
    ws.Columns("B:F").NumberFormat = "0.0000%"
End Sub

Private Sub ValidateCurveInput(ByVal ws As Worksheet, ByVal lastRow As Long)
    Dim r As Long, c As Long
    Dim v As Variant

    For r = 2 To lastRow
        If Len(ws.Cells(r, 1).Value) > 0 And Not IsDate(ws.Cells(r, 1).Value) Then
            Err.Raise vbObjectError + 101, , "Invalid date in Curve!A" & r
        End If
        For c = 2 To 6
            v = ws.Cells(r, c).Value
            If Len(v) > 0 And Not IsNumeric(v) Then
                Err.Raise vbObjectError + 102, , "Invalid rate in Curve!" & ws.Cells(r, c).Address(False, False)
            End If
        Next c
    Next r
End Sub

Private Function BuildNormalizedChartData(ByVal wsCurve As Worksheet, _
                                          ByVal wsData As Worksheet, _
                                          ByVal lastRow As Long, _
                                          ByVal analysisStart As Date) As Long
    Dim r As Long, o As Long, c As Long
    Dim d As Date, rateVal As Double
    Dim headers As Variant

    headers = Array("Date", "ON", "1M", "2M", "3M", "6M", "6M-ON (bp)")
    WriteHorizontalArray wsData.Range("A1"), headers

    o = 2
    For r = 2 To lastRow
        If IsDate(wsCurve.Cells(r, 1).Value) Then
            d = CDate(wsCurve.Cells(r, 1).Value)
            If d >= analysisStart Then
                If RowHasAllRates(wsCurve, r) Then
                    wsData.Cells(o, 1).Value = d
                    For c = 2 To 6
                        rateVal = CDbl(wsCurve.Cells(r, c).Value)
                        If Abs(rateVal) > 1 Then rateVal = rateVal / 100#
                        wsData.Cells(o, c).Value = rateVal
                    Next c
                    wsData.Cells(o, 7).FormulaR1C1 = "=(RC[-1]-RC[-5])*10000"
                    o = o + 1
                End If
            End If
        End If
    Next r

    If o > 2 Then
        wsData.Range("A2:G" & (o - 1)).Sort Key1:=wsData.Range("A2"), Order1:=xlAscending, Header:=xlNo
        wsData.Range("A2:A" & (o - 1)).NumberFormat = "mmm-dd-yyyy"
        wsData.Range("B2:F" & (o - 1)).NumberFormat = "0.0000%"
        wsData.Range("G2:G" & (o - 1)).NumberFormat = "0.0"
    End If

    BuildNormalizedChartData = o - 1
End Function

Private Function RowHasAllRates(ByVal ws As Worksheet, ByVal rowNum As Long) As Boolean
    Dim c As Long
    For c = 2 To 6
        If Not IsNumeric(ws.Cells(rowNum, c).Value) Or Len(ws.Cells(rowNum, c).Value) = 0 Then Exit Function
    Next c
    RowHasAllRates = True
End Function

Private Sub BuildHistoricalChart(ByVal wsDash As Worksheet, _
                                 ByVal wsData As Worksheet, _
                                 ByVal lastRow As Long, _
                                 ByVal minRate As Double, _
                                 ByVal maxRate As Double, _
                                 ByVal startDate As Date, _
                                 ByVal endDate As Date)
    Dim co As ChartObject, ch As Chart
    Dim i As Long
    Dim names As Variant, cols As Variant, widths As Variant
    Dim s As Series
    Dim yMin As Double, yMax As Double

    Set co = wsDash.ChartObjects.Add(Left:=wsDash.Range("A5").Left, _
                                     Top:=wsDash.Range("A5").Top, _
                                     Width:=wsDash.Range("A5:L24").Width, _
                                     Height:=wsDash.Range("A5:L24").Height)
    co.Name = "chtSOFRHistory"
    Set ch = co.Chart
    ch.ChartType = xlLine
    ch.HasLegend = True
    ch.Legend.Position = xlLegendPositionTop
    ch.HasTitle = True
    ch.ChartTitle.Text = "Historical SOFR curve by tenor"
    ch.ChartTitle.Format.TextFrame2.TextRange.Font.Size = 13
    ch.ChartTitle.Format.TextFrame2.TextRange.Font.Bold = True
    ch.ChartTitle.Format.TextFrame2.TextRange.Font.Fill.ForeColor.RGB = CLR_NAVY

    names = Array("ON", "1M", "2M", "3M", "6M")
    cols = Array(CLR_NAVY, CLR_BLUE, CLR_SKY, CLR_GREEN, CLR_ORANGE)
    widths = Array(2.75, 1.5, 1.5, 1.5, 1.75)

    For i = 0 To 4
        Set s = ch.SeriesCollection.NewSeries
        s.Name = names(i)
        s.XValues = wsData.Range("A2:A" & lastRow)
        s.Values = wsData.Range(wsData.Cells(2, i + 2), wsData.Cells(lastRow, i + 2))
        s.Format.Line.ForeColor.RGB = cols(i)
        s.Format.Line.Weight = widths(i)
        s.MarkerStyle = xlMarkerStyleNone
    Next i

    yMin = WorksheetFunction.Max(0, minRate - 0.004)
    yMax = maxRate + 0.004

    With ch.Axes(xlValue)
        .MinimumScale = yMin
        .MaximumScale = yMax
        .MajorUnit = 0.01
        .TickLabels.NumberFormat = "0.0%"
        .HasTitle = False
        .Format.Line.Visible = False
        .MajorGridlines.Format.Line.ForeColor.RGB = CLR_BORDER
        .MajorGridlines.Format.Line.transparency = 0.35
        .TickLabels.Font.Size = 9
        .TickLabels.Font.Color = CLR_MUTED
    End With

    With ch.Axes(xlCategory)
        .CategoryType = xlTimeScale
        .MinimumScale = CDbl(startDate)
        .MaximumScale = CDbl(endDate)
        .MajorUnit = 6
        .MajorUnitScale = xlMonths
        .TickLabels.NumberFormat = "mmm-yy"
        .TickLabels.Font.Size = 9
        .TickLabels.Font.Color = CLR_MUTED
        .Format.Line.ForeColor.RGB = CLR_BORDER
    End With

    ch.ChartArea.Format.Fill.ForeColor.RGB = vbWhite
    ch.ChartArea.Format.Line.Visible = False
    ch.PlotArea.Format.Fill.ForeColor.RGB = vbWhite
    ch.PlotArea.Format.Line.Visible = False
    ch.Legend.Format.TextFrame2.TextRange.Font.Size = 9

    AddPolicyBandsAndAnnotations wsDash, co, ch, wsData, lastRow, startDate, endDate, yMin, yMax
End Sub

Private Sub AddPolicyBandsAndAnnotations(ByVal ws As Worksheet, _
                                         ByVal co As ChartObject, _
                                         ByVal ch As Chart, _
                                         ByVal wsData As Worksheet, _
                                         ByVal lastRow As Long, _
                                         ByVal minDate As Date, _
                                         ByVal maxDate As Date, _
                                         ByVal yMin As Double, _
                                         ByVal yMax As Double)
    Dim plotLeft As Double, plotTop As Double, plotWidth As Double, plotHeight As Double
    Dim band As Shape

    co.Activate
    DoEvents

    plotLeft = co.Left + ch.PlotArea.InsideLeft
    plotTop = co.Top + ch.PlotArea.InsideTop
    plotWidth = ch.PlotArea.InsideWidth
    plotHeight = ch.PlotArea.InsideHeight

    ' Broad policy windows. These are presentation windows, not a substitute for event-level analysis.
    AddDateBand ws, "bandHike", DateSerial(2022, 3, 16), DateSerial(2023, 7, 26), _
                minDate, maxDate, plotLeft, plotTop, plotWidth, plotHeight, CLR_HIKE, 0.78, "HIKING CYCLE"
    AddDateBand ws, "bandHold", DateSerial(2023, 7, 27), DateSerial(2024, 9, 17), _
                minDate, maxDate, plotLeft, plotTop, plotWidth, plotHeight, CLR_HOLD, 0.82, "HIGHER FOR LONGER"
    AddDateBand ws, "bandEase", DateSerial(2024, 9, 18), DateSerial(2024, 12, 18), _
                minDate, maxDate, plotLeft, plotTop, plotWidth, plotHeight, CLR_EASE, 0.76, "EASING REPRICING"

    AddChartAnnotation ws, "annFirstHike", co, ch, DateSerial(2022, 3, 16), _
                       GetNearestRate(wsData, lastRow, DateSerial(2022, 3, 16), 2), _
                       minDate, maxDate, yMin, yMax, _
                       "Fed tightening begins" & vbCrLf & "ON reprices first", 12, -58

    AddChartAnnotation ws, "annPeak", co, ch, _
                       GetMaxDate(wsData, lastRow, 2), GetMaxValue(wsData, lastRow, 2), _
                       minDate, maxDate, yMin, yMax, _
                       "Peak policy rate" & vbCrLf & Format(GetMaxValue(wsData, lastRow, 2), "0.00%"), -95, -48

    AddChartAnnotation ws, "annEasing", co, ch, DateSerial(2024, 9, 18), _
                       GetNearestRate(wsData, lastRow, DateSerial(2024, 9, 18), 2), _
                       minDate, maxDate, yMin, yMax, _
                       "Easing cycle starts" & vbCrLf & "front end shifts lower", -85, 16

    AddChartAnnotation ws, "annLatest", co, ch, maxDate, wsData.Cells(lastRow, 6).Value, _
                       minDate, maxDate, yMin, yMax, _
                       "Current curve" & vbCrLf & "6M remains above ON", -125, 20

    On Error Resume Next
    ch.ChartArea.Select
    On Error GoTo 0
End Sub

Private Sub AddDateBand(ByVal ws As Worksheet, ByVal shapeName As String, _
                        ByVal bandStart As Date, ByVal bandEnd As Date, _
                        ByVal minDate As Date, ByVal maxDate As Date, _
                        ByVal plotLeft As Double, ByVal plotTop As Double, _
                        ByVal plotWidth As Double, ByVal plotHeight As Double, _
                        ByVal fillColor As Long, ByVal transparency As Double, _
                        ByVal labelText As String)
    Dim leftPos As Double, rightPos As Double
    Dim shp As Shape, lbl As Shape

    If bandEnd < minDate Or bandStart > maxDate Then Exit Sub
    If bandStart < minDate Then bandStart = minDate
    If bandEnd > maxDate Then bandEnd = maxDate

    leftPos = plotLeft + ((CDbl(bandStart) - CDbl(minDate)) / (CDbl(maxDate) - CDbl(minDate))) * plotWidth
    rightPos = plotLeft + ((CDbl(bandEnd) - CDbl(minDate)) / (CDbl(maxDate) - CDbl(minDate))) * plotWidth

    Set shp = ws.Shapes.AddShape(msoShapeRectangle, leftPos, plotTop, rightPos - leftPos, plotHeight)
    shp.Name = shapeName
    shp.Fill.ForeColor.RGB = fillColor
    shp.Fill.transparency = transparency
    shp.Line.Visible = False
    shp.ZOrder msoSendToBack

    Set lbl = ws.Shapes.AddTextbox(msoTextOrientationHorizontal, leftPos + 4, plotTop + 3, rightPos - leftPos - 8, 14)
    lbl.Name = shapeName & "Label"
    lbl.TextFrame2.TextRange.Text = labelText
    lbl.TextFrame2.TextRange.Font.Size = 7.5
    lbl.TextFrame2.TextRange.Font.Bold = True
    lbl.TextFrame2.TextRange.Font.Fill.ForeColor.RGB = CLR_MUTED
    lbl.Line.Visible = False
    lbl.Fill.Visible = False
    lbl.ZOrder msoBringToFront
End Sub

Private Sub AddChartAnnotation(ByVal ws As Worksheet, ByVal shapeName As String, _
                               ByVal co As ChartObject, ByVal ch As Chart, _
                               ByVal pointDate As Date, ByVal pointValue As Double, _
                               ByVal minDate As Date, ByVal maxDate As Date, _
                               ByVal yMin As Double, ByVal yMax As Double, _
                               ByVal annotationText As String, _
                               ByVal offsetX As Double, ByVal offsetY As Double)
    Dim x As Double, y As Double
    Dim plotLeft As Double, plotTop As Double, plotWidth As Double, plotHeight As Double
    Dim shp As Shape

    plotLeft = co.Left + ch.PlotArea.InsideLeft
    plotTop = co.Top + ch.PlotArea.InsideTop
    plotWidth = ch.PlotArea.InsideWidth
    plotHeight = ch.PlotArea.InsideHeight

    x = plotLeft + ((CDbl(pointDate) - CDbl(minDate)) / (CDbl(maxDate) - CDbl(minDate))) * plotWidth
    y = plotTop + (1 - ((pointValue - yMin) / (yMax - yMin))) * plotHeight

    Set shp = ws.Shapes.AddTextbox(msoTextOrientationHorizontal, x + offsetX, y + offsetY, 105, 34)
    shp.Name = shapeName
    With shp
        .Fill.ForeColor.RGB = vbWhite
        .Fill.transparency = 0.05
        .Line.ForeColor.RGB = CLR_BORDER
        .Line.Weight = 0.75
        .TextFrame2.MarginLeft = 5
        .TextFrame2.MarginRight = 5
        .TextFrame2.MarginTop = 3
        .TextFrame2.MarginBottom = 3
        .TextFrame2.TextRange.Text = annotationText
        .TextFrame2.TextRange.Font.Size = 8
        .TextFrame2.TextRange.Font.Fill.ForeColor.RGB = CLR_TEXT
        .TextFrame2.TextRange.ParagraphFormat.Alignment = msoAlignLeft
        .ZOrder msoBringToFront
    End With
End Sub

Private Sub BuildPolicyWindowTable(ByVal wsDash As Worksheet, ByVal wsData As Worksheet, ByVal lastRow As Long)
    Dim labels As Variant, starts As Variant, ends As Variant, messages As Variant
    Dim i As Long, startRow As Long, endRow As Long
    Dim onDelta As Double, avgSlope As Double

    labels = Array("Hiking cycle", "Higher for longer", "Easing repricing", "Current uncertainty")
    starts = Array(DateSerial(2022, 3, 16), DateSerial(2023, 7, 27), DateSerial(2024, 9, 18), DateSerial(2024, 12, 19))
    ends = Array(DateSerial(2023, 7, 26), DateSerial(2024, 9, 17), DateSerial(2024, 12, 18), wsData.Cells(lastRow, 1).Value)
    messages = Array( _
        "ON rose rapidly; the curve flattened as expected hikes became delivered hikes.", _
        "Policy stayed restrictive; longer tenors reflected uncertainty around the first cut.", _
        "Front-end rates moved lower as the market repriced the easing path.", _
        "The curve is upward sloping again, but the premium remains path-dependent.")

    With wsDash.Range("N21:Q21")
        .Merge
        .Value = "Policy windows: evidence behind the story"
        .Font.Size = 13
        .Font.Bold = True
        .Font.Color = CLR_NAVY
    End With

    WriteHorizontalArray wsDash.Range("N23"), Array("Window", "ON change", "Avg 6M-ON", "Interpretation")
    StyleTableHeader wsDash.Range("N23:Q23")

    For i = 0 To 3
        startRow = NearestRow(wsData, lastRow, starts(i))
        endRow = NearestRow(wsData, lastRow, ends(i))
        If endRow < startRow Then endRow = startRow

        onDelta = (wsData.Cells(endRow, 2).Value - wsData.Cells(startRow, 2).Value) * 10000
        avgSlope = WorksheetFunction.Average(wsData.Range(wsData.Cells(startRow, 7), wsData.Cells(endRow, 7)))

        wsDash.Cells(24 + i, 14).Value = labels(i)

Private Function GetNearestRate(ByVal ws As Worksheet, ByVal lastRow As Long, _
                                ByVal targetDate As Date, ByVal rateCol As Long) As Double
    GetNearestRate = ws.Cells(NearestRow(ws, lastRow, targetDate), rateCol).Value
End Function

Private Function NearestRow(ByVal ws As Worksheet, ByVal lastRow As Long, ByVal targetDate As Date) As Long
    Dim r As Long, bestRow As Long
    Dim bestDiff As Double, thisDiff As Double

    bestDiff = 1E+99
    bestRow = 2
    For r = 2 To lastRow
        thisDiff = Abs(CDbl(ws.Cells(r, 1).Value) - CDbl(targetDate))
        If thisDiff < bestDiff Then
            bestDiff = thisDiff
            bestRow = r
        End If
    Next r
    NearestRow = bestRow
End Function

Private Function GetMaxValue(ByVal ws As Worksheet, ByVal lastRow As Long, ByVal rateCol As Long) As Double
    GetMaxValue = WorksheetFunction.Max(ws.Range(ws.Cells(2, rateCol), ws.Cells(lastRow, rateCol)))
End Function

Private Function GetMaxDate(ByVal ws As Worksheet, ByVal lastRow As Long, ByVal rateCol As Long) As Date
    Dim maxVal As Double, r As Long
    maxVal = GetMaxValue(ws, lastRow, rateCol)
    For r = 2 To lastRow
        If ws.Cells(r, rateCol).Value = maxVal Then
            GetMaxDate = ws.Cells(r, 1).Value
            Exit Function
        End If
    Next r
End Function


Private Sub WriteHorizontalArray(ByVal firstCell As Range, ByVal items As Variant)
    Dim i As Long
    For i = LBound(items) To UBound(items)
        firstCell.Offset(0, i - LBound(items)).Value = items(i)
    Next i
End Sub

Private Function LastUsedRow(ByVal ws As Worksheet, ByVal columnLetter As String) As Long
    LastUsedRow = ws.Cells(ws.Rows.Count, columnLetter).End(xlUp).Row
End Function

Private Function GetOrCreateSheet(ByVal wb As Workbook, ByVal sheetName As String) As Worksheet
    On Error Resume Next
    Set GetOrCreateSheet = wb.Worksheets(sheetName)
    On Error GoTo 0

    If GetOrCreateSheet Is Nothing Then
        Set GetOrCreateSheet = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
        GetOrCreateSheet.Name = sheetName
    End If
End Function

Private Sub ResetSheet(ByVal ws As Worksheet)
    Dim i As Long
    On Error Resume Next
    ws.Visible = xlSheetVisible
    ws.Cells.UnMerge
    ws.Cells.Clear
    For i = ws.ChartObjects.Count To 1 Step -1
        ws.ChartObjects(i).Delete
    Next i
    For i = ws.Shapes.Count To 1 Step -1
        ws.Shapes(i).Delete
    Next i
    On Error GoTo 0
End Sub
