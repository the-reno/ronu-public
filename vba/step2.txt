Option Explicit

'===============================================================================
' ALLOCATION / RETURN / REGRET ANALYSIS | DETAILED CHART SOURCE VERSION
'
' REQUIRED INPUT
'   Scenario_Daily
'   Scenario_Assumptions
'
' NO CURVE SHEET REQUIRED.
'
' EXPECTED Scenario_Daily COLUMNS
'   A = Date
'   B = Scenario
'   D = ON
'   Q = 1M
'   R = 3M
'   S = 6M
'
' OUTPUT
'   Allocation_Strategies       named strategies + summary statistics
'   Allocation_Base             pure-tenor return engine (hidden)
'   Allocation_Return_By_Date   dates in rows, strategies in columns
'   Allocation_Regret_By_Date   dates in rows, strategies in columns
'   Allocation_Grid             EVERY feasible allocation + exact metrics
'   Allocation_Scenario_Summary named strategy x scenario analysis
'   Allocation_Best_Tenor       best pure-tenor frequency by scenario
'   Allocation_Extreme_Cases    worst realized regret dates by strategy
'   Allocation_Charts           expanded analytical charts
'   Allocation_Chart_Data       explicit scatter / frontier source mapping (hidden)
'
' RUN
'   Build_Allocation_Analysis
'
' KEY DESIGN CHANGE
'
'   1) RETURN / REGRET BY DATE
'
'      Each row = one scenario + one starting date.
'      Each strategy = one separate column.
'
'      This makes the calculation easy to audit:
'
'      Date | Best tenor | ON only | 1M only | 3M only | 6M only |
'             Short/Liquid | Equal Ladder | Barbell | Balanced | Term Heavy
'
'   2) EFFICIENT FRONTIER
'
'      Allocation_Grid contains one row for EVERY feasible allocation.
'
'      Allocation ID
'      Allocation description
'      Exact ON / 1M / 3M / 6M weights
'      Weighted tenor
'      Average return
'      Return volatility
'      Average regret
'      Worst regret
'      P95 regret
'      <=5bp hit rate
'      Efficient frontier flag
'
'      Average return vs worst regret therefore plots EVERY allocation.
'
'   3) COST OF BEING WRONG
'
'      Charts use Allocation_Regret_By_Date directly.
'      Dates are on the rows / x-axis.
'      Each named strategy is a separate series / column.
'
'===============================================================================

Private Const ALLOC_DEFAULT_HORIZON As Long = 180
Private Const ALLOC_DEFAULT_STEP As Double = 0.1
Private Const ALLOC_DEFAULT_MIN_SHORT As Double = 0.3
Private Const ALLOC_DEFAULT_MAX_6M As Double = 0.7
Private Const ALLOC_HIT_THRESHOLD_BP As Double = 5#

Private Const STRAT_FIRST_ROW As Long = 4
Private Const STRAT_LAST_ROW As Long = 12
Private Const STRAT_COUNT As Long = 9

'===============================================================================
' MAIN
'===============================================================================
Public Sub Build_Allocation_Analysis()

    Dim wb As Workbook
    Dim horizonDays As Long
    Dim gridStep As Double
    Dim minShort As Double
    Dim max6M As Double
    Dim oldCalc As XlCalculation

    Set wb = ThisWorkbook

    If Not Alloc_SheetExists(wb, "Scenario_Daily") Then
        MsgBox "Missing Scenario_Daily. Run the standalone scenario model first.", vbCritical
        Exit Sub
    End If

    If Not Alloc_SheetExists(wb, "Scenario_Assumptions") Then
        MsgBox "Missing Scenario_Assumptions. Run the standalone scenario model first.", vbCritical
        Exit Sub
    End If

    On Error GoTo Fail

    oldCalc = Application.Calculation

    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    Alloc_DeleteSheetIfExists wb, "Allocation_Strategies"
    Alloc_DeleteSheetIfExists wb, "Allocation_Base"
    Alloc_DeleteSheetIfExists wb, "Allocation_Return_By_Date"
    Alloc_DeleteSheetIfExists wb, "Allocation_Regret_By_Date"
    Alloc_DeleteSheetIfExists wb, "Allocation_Grid"
    Alloc_DeleteSheetIfExists wb, "Allocation_Scenario_Summary"
    Alloc_DeleteSheetIfExists wb, "Allocation_Best_Tenor"
    Alloc_DeleteSheetIfExists wb, "Allocation_Extreme_Cases"
    Alloc_DeleteSheetIfExists wb, "Allocation_Charts"
    Alloc_DeleteSheetIfExists wb, "Allocation_Chart_Data"

    Alloc_CreateStrategySheet wb

    horizonDays = CLng(wb.Worksheets("Allocation_Strategies").Range("B17").Value2)
    gridStep = CDbl(wb.Worksheets("Allocation_Strategies").Range("B18").Value2)
    minShort = CDbl(wb.Worksheets("Allocation_Strategies").Range("B19").Value2)
    max6M = CDbl(wb.Worksheets("Allocation_Strategies").Range("B20").Value2)

    If horizonDays < 30 Then Err.Raise vbObjectError + 201, , "Analysis horizon must be at least 30 days."
    If gridStep <= 0# Or gridStep > 1# Then Err.Raise vbObjectError + 202, , "Allocation step must be between 0 and 100%."
    If minShort < 0# Or minShort > 1# Then Err.Raise vbObjectError + 203, , "Minimum ON + 1M must be between 0 and 100%."
    If max6M < 0# Or max6M > 1# Then Err.Raise vbObjectError + 204, , "Maximum 6M must be between 0 and 100%."

    Alloc_BuildBaseReturns wb, horizonDays
    Alloc_BuildWideDateMatrices wb, horizonDays
    Alloc_SummarizeNamedStrategies wb
    Alloc_BuildScenarioSummary wb
    Alloc_BuildBestTenorSummary wb
    Alloc_BuildExtremeCases wb
    Alloc_BuildGridAnalysis wb, horizonDays, gridStep, minShort, max6M
    Alloc_MarkEfficientFrontier wb
    Alloc_BuildDetailedChartSource wb
    Alloc_BuildDetailedCharts wb
    Alloc_FormatOutputs wb

Done:
    On Error Resume Next
    Application.Calculation = oldCalc
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    On Error GoTo 0

    MsgBox "Detailed allocation return / regret analysis created.", vbInformation
    Exit Sub

Fail:
    MsgBox "Allocation analysis failed: " & Err.Number & " - " & Err.Description, vbCritical
    Resume Done

End Sub

Private Sub Alloc_SetStrategy(ByVal ws As Worksheet, ByVal r As Long, ByVal nm As String, ByVal wON As Double, ByVal w1 As Double, ByVal w3 As Double, ByVal w6 As Double, ByVal description As String)

    ws.Cells(r, 1).Value = nm
    ws.Cells(r, 2).Value = wON
    ws.Cells(r, 3).Value = w1
    ws.Cells(r, 4).Value = w3
    ws.Cells(r, 5).Value = w6
    ws.Cells(r, 6).Value = w1 + 3# * w3 + 6# * w6
    ws.Cells(r, 13).Value = description

End Sub

'===============================================================================
' PURE-TENOR BASE RETURN ENGINE
'
' One row = one scenario + one eligible daily starting date.
'
' This is the common calculation source for every strategy and every allocation.
'===============================================================================
Private Sub Alloc_BuildBaseReturns(ByVal wb As Workbook, ByVal horizonDays As Long)

    Dim src As Worksheet
    Dim ws As Worksheet
    Dim assumptions As Worksheet

    Dim scenarioHorizon As Long
    Dim s As Long
    Dim d As Long
    Dim firstRow As Long
    Dim startRow As Long
    Dim outRow As Long
    Dim maxStartDay As Long

    Dim retON As Double
    Dim ret1 As Double
    Dim ret3 As Double
    Dim ret6 As Double

    Dim annON As Double
    Dim ann1 As Double
    Dim ann3 As Double
    Dim ann6 As Double

    Dim bestAnn As Double
    Dim bestTenor As String

    Set src = wb.Worksheets("Scenario_Daily")
    Set assumptions = wb.Worksheets("Scenario_Assumptions")

    scenarioHorizon = CLng(assumptions.Range("B8").Value2)

    If horizonDays > scenarioHorizon Then
        Err.Raise vbObjectError + 210, , "Holding period exceeds scenario horizon."
    End If

    Set ws = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
    ws.Name = "Allocation_Base"

    ws.Range("A1:M1").Value = Array("Scenario", "Start date", "End date", "ON HP return", "1M HP return", "3M HP return", "6M HP return", "ON ann", "1M ann", "3M ann", "6M ann", "Best ann", "Best tenor")
    ws.Range("A1:M1").Font.Bold = True

    outRow = 2
    maxStartDay = scenarioHorizon - horizonDays

    For s = 1 To 5

        firstRow = 2 + (s - 1) * (scenarioHorizon + 1)

        For d = 0 To maxStartDay

            startRow = firstRow + d

            retON = Alloc_SleeveHoldingReturn(src, startRow, horizonDays, 1, 4)
            ret1 = Alloc_SleeveHoldingReturn(src, startRow, horizonDays, 30, 17)
            ret3 = Alloc_SleeveHoldingReturn(src, startRow, horizonDays, 90, 18)
            ret6 = Alloc_SleeveHoldingReturn(src, startRow, horizonDays, 180, 19)

            annON = Alloc_Annualize(retON, horizonDays)
            ann1 = Alloc_Annualize(ret1, horizonDays)
            ann3 = Alloc_Annualize(ret3, horizonDays)
            ann6 = Alloc_Annualize(ret6, horizonDays)

            bestAnn = annON
            bestTenor = "ON"

            If ann1 > bestAnn Then
                bestAnn = ann1
                bestTenor = "1M"
            End If

            If ann3 > bestAnn Then
                bestAnn = ann3
                bestTenor = "3M"
            End If

            If ann6 > bestAnn Then
                bestAnn = ann6
                bestTenor = "6M"
            End If

            ws.Cells(outRow, 1).Value = src.Cells(startRow, 2).Value
            ws.Cells(outRow, 2).Value = src.Cells(startRow, 1).Value2
            ws.Cells(outRow, 3).Value = src.Cells(startRow + horizonDays, 1).Value2

            ws.Cells(outRow, 4).Value = retON
            ws.Cells(outRow, 5).Value = ret1
            ws.Cells(outRow, 6).Value = ret3
            ws.Cells(outRow, 7).Value = ret6

            ws.Cells(outRow, 8).Value = annON
            ws.Cells(outRow, 9).Value = ann1
            ws.Cells(outRow, 10).Value = ann3
            ws.Cells(outRow, 11).Value = ann6

            ws.Cells(outRow, 12).Value = bestAnn
            ws.Cells(outRow, 13).Value = bestTenor

            outRow = outRow + 1

        Next d

    Next s

    ws.Columns("B:C").NumberFormat = "dd-mmm-yyyy"
    ws.Columns("D:L").NumberFormat = "0.000%"

End Sub

'===============================================================================
' SLEEVE HOLDING-PERIOD RETURN
'
' ON  : resets each day
' 1M  : trades at day 0, 30, 60, 90, 120, 150
' 3M  : trades at day 0 and 90
' 6M  : one trade at day 0
'
' The prevailing tenor rate on each reset date is locked until the next reset.
'===============================================================================
Private Function Alloc_SleeveHoldingReturn(ByVal ws As Worksheet, ByVal startRow As Long, ByVal holdingDays As Long, ByVal resetDays As Long, ByVal rateCol As Long) As Double

    Dim wealth As Double
    Dim elapsed As Long
    Dim daysThis As Long
    Dim lockedRate As Double

    wealth = 1#
    elapsed = 0

    Do While elapsed < holdingDays

        daysThis = resetDays

        If elapsed + daysThis > holdingDays Then
            daysThis = holdingDays - elapsed
        End If

        lockedRate = CDbl(ws.Cells(startRow + elapsed, rateCol).Value2)

        wealth = wealth * (1# + lockedRate * daysThis / 360#)

        elapsed = elapsed + daysThis

    Loop

    Alloc_SleeveHoldingReturn = wealth - 1#

End Function

Private Function Alloc_Annualize(ByVal holdingReturn As Double, ByVal holdingDays As Long) As Double

    If holdingDays <= 0 Then
        Alloc_Annualize = 0#
        Exit Function
    End If

    If 1# + holdingReturn <= 0# Then
        Alloc_Annualize = -1#
        Exit Function
    End If

    Alloc_Annualize = (1# + holdingReturn) ^ (360# / CDbl(holdingDays)) - 1#

End Function

'===============================================================================
' WIDE DATE MATRICES
'
' THIS IS THE MAIN AUDIT / CHART SOURCE REQUESTED.
'
' ROWS:
'   Scenario + Start Date
'
' COLUMNS:
'   Every named strategy
'
' Return sheet:
'   annualized return
'
' Regret sheet:
'   best pure-tenor annualized return - strategy annualized return, in bp
'===============================================================================
Private Sub Alloc_BuildWideDateMatrices(ByVal wb As Workbook, ByVal horizonDays As Long)

    Dim base As Worksheet
    Dim strat As Worksheet
    Dim retWS As Worksheet
    Dim regWS As Worksheet

    Dim baseLast As Long
    Dim r As Long
    Dim sr As Long
    Dim outCol As Long

    Dim wON As Double
    Dim w1 As Double
    Dim w3 As Double
    Dim w6 As Double

    Dim hpRet As Double
    Dim annRet As Double
    Dim bestAnn As Double
    Dim regret As Double

    Set base = wb.Worksheets("Allocation_Base")
    Set strat = wb.Worksheets("Allocation_Strategies")

    Set retWS = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
    retWS.Name = "Allocation_Return_By_Date"

    Set regWS = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
    regWS.Name = "Allocation_Regret_By_Date"

    ' Common descriptive columns
    retWS.Range("A1:E1").Value = Array("Scenario", "Start date", "End date", "Best tenor", "Best pure-tenor ann return")
    regWS.Range("A1:E1").Value = Array("Scenario", "Start date", "End date", "Best tenor", "Best pure-tenor ann return")

    ' One strategy per column
    outCol = 6

    For sr = STRAT_FIRST_ROW To STRAT_LAST_ROW

        retWS.Cells(1, outCol).Value = strat.Cells(sr, 1).Value
        regWS.Cells(1, outCol).Value = strat.Cells(sr, 1).Value

        outCol = outCol + 1

    Next sr

    baseLast = base.Cells(base.Rows.Count, 1).End(xlUp).Row

    For r = 2 To baseLast

        retWS.Cells(r, 1).Value = base.Cells(r, 1).Value
        retWS.Cells(r, 2).Value = base.Cells(r, 2).Value
        retWS.Cells(r, 3).Value = base.Cells(r, 3).Value
        retWS.Cells(r, 4).Value = base.Cells(r, 13).Value
        retWS.Cells(r, 5).Value = base.Cells(r, 12).Value

        regWS.Cells(r, 1).Value = base.Cells(r, 1).Value
        regWS.Cells(r, 2).Value = base.Cells(r, 2).Value
        regWS.Cells(r, 3).Value = base.Cells(r, 3).Value
        regWS.Cells(r, 4).Value = base.Cells(r, 13).Value
        regWS.Cells(r, 5).Value = base.Cells(r, 12).Value

        bestAnn = CDbl(base.Cells(r, 12).Value2)

        outCol = 6

        For sr = STRAT_FIRST_ROW To STRAT_LAST_ROW

            wON = CDbl(strat.Cells(sr, 2).Value2)
            w1 = CDbl(strat.Cells(sr, 3).Value2)
            w3 = CDbl(strat.Cells(sr, 4).Value2)
            w6 = CDbl(strat.Cells(sr, 5).Value2)

            Alloc_ValidateWeights wON, w1, w3, w6, CStr(strat.Cells(sr, 1).Value)

            hpRet = wON * CDbl(base.Cells(r, 4).Value2)
            hpRet = hpRet + w1 * CDbl(base.Cells(r, 5).Value2)
            hpRet = hpRet + w3 * CDbl(base.Cells(r, 6).Value2)
            hpRet = hpRet + w6 * CDbl(base.Cells(r, 7).Value2)

            annRet = Alloc_Annualize(hpRet, horizonDays)
            regret = (bestAnn - annRet) * 10000#

            If regret < 0# And Abs(regret) < 0.000001 Then regret = 0#

            retWS.Cells(r, outCol).Value = annRet
            regWS.Cells(r, outCol).Value = regret

            outCol = outCol + 1

        Next sr

    Next r

    retWS.Range("A1:N1").Font.Bold = True
    regWS.Range("A1:N1").Font.Bold = True

    retWS.Columns("B:C").NumberFormat = "dd-mmm-yyyy"
    regWS.Columns("B:C").NumberFormat = "dd-mmm-yyyy"

    retWS.Range("E2:N" & baseLast).NumberFormat = "0.000%"
    regWS.Range("E2:E" & baseLast).NumberFormat = "0.000%"
    regWS.Range("F2:N" & baseLast).NumberFormat = "0.0"

    ' Visual audit: low regret green, high regret red.
    With regWS.Range("F2:N" & baseLast)
        .FormatConditions.Delete
        .FormatConditions.AddColorScale ColorScaleType:=3
    End With

End Sub

'===============================================================================
' NAMED STRATEGY SUMMARY
'
' Summary values are calculated directly from the visible date-by-date matrices.
'===============================================================================
Private Sub Alloc_SummarizeNamedStrategies(ByVal wb As Workbook)

    Dim strat As Worksheet
    Dim retWS As Worksheet
    Dim regWS As Worksheet

    Dim lastRow As Long
    Dim sr As Long
    Dim dataCol As Long

    Dim avgRet As Double
    Dim volRet As Double
    Dim avgReg As Double
    Dim worstReg As Double
    Dim p95Reg As Double
    Dim hitRate As Double

    Set strat = wb.Worksheets("Allocation_Strategies")
    Set retWS = wb.Worksheets("Allocation_Return_By_Date")
    Set regWS = wb.Worksheets("Allocation_Regret_By_Date")

    lastRow = retWS.Cells(retWS.Rows.Count, 1).End(xlUp).Row

    dataCol = 6

    For sr = STRAT_FIRST_ROW To STRAT_LAST_ROW

        avgRet = Application.WorksheetFunction.Average(retWS.Range(retWS.Cells(2, dataCol), retWS.Cells(lastRow, dataCol)))
        volRet = Application.WorksheetFunction.StDev_P(retWS.Range(retWS.Cells(2, dataCol), retWS.Cells(lastRow, dataCol)))
        avgReg = Application.WorksheetFunction.Average(regWS.Range(regWS.Cells(2, dataCol), regWS.Cells(lastRow, dataCol)))
        worstReg = Application.WorksheetFunction.Max(regWS.Range(regWS.Cells(2, dataCol), regWS.Cells(lastRow, dataCol)))
        p95Reg = Alloc_RangePercentile(regWS.Range(regWS.Cells(2, dataCol), regWS.Cells(lastRow, dataCol)), 0.95)
        hitRate = Application.WorksheetFunction.CountIf(regWS.Range(regWS.Cells(2, dataCol), regWS.Cells(lastRow, dataCol)), "<=" & ALLOC_HIT_THRESHOLD_BP) / CDbl(lastRow - 1)

        strat.Cells(sr, 7).Value = avgRet
        strat.Cells(sr, 8).Value = volRet
        strat.Cells(sr, 9).Value = avgReg
        strat.Cells(sr, 10).Value = worstReg
        strat.Cells(sr, 11).Value = p95Reg
        strat.Cells(sr, 12).Value = hitRate

        dataCol = dataCol + 1

    Next sr

End Sub


'===============================================================================
' SCENARIO SUMMARY
'
' One row = one named strategy in one scenario.
'
' This table is intentionally visible and is the direct source for:
'   - Average return by scenario
'   - Average regret by scenario
'   - Near-best hit rate by scenario
'   - Scenario P95 / worst regret analysis
'===============================================================================
Private Sub Alloc_BuildScenarioSummary(ByVal wb As Workbook)

    Dim strat As Worksheet
    Dim retWS As Worksheet
    Dim regWS As Worksheet
    Dim ws As Worksheet

    Dim lastRow As Long
    Dim sr As Long
    Dim s As Long
    Dim r As Long
    Dim outRow As Long
    Dim dataCol As Long

    Dim scenarioName As String
    Dim n As Long
    Dim sumRet As Double
    Dim sumRetSq As Double
    Dim sumReg As Double
    Dim worstReg As Double
    Dim hitN As Long
    Dim retValue As Double
    Dim regValue As Double
    Dim avgRet As Double
    Dim volRet As Double
    Dim avgReg As Double
    Dim p95Reg As Double

    Dim tempRegrets() As Double

    Set strat = wb.Worksheets("Allocation_Strategies")
    Set retWS = wb.Worksheets("Allocation_Return_By_Date")
    Set regWS = wb.Worksheets("Allocation_Regret_By_Date")

    lastRow = retWS.Cells(retWS.Rows.Count, 1).End(xlUp).Row

    Set ws = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
    ws.Name = "Allocation_Scenario_Summary"

    ws.Range("A1:P1").Value = Array( _
        "Scenario", _
        "Strategy", _
        "ON", _
        "1M", _
        "3M", _
        "6M", _
        "Weighted tenor", _
        "Observations", _
        "Avg ann return", _
        "Return vol", _
        "Avg regret bp", _
        "P95 regret bp", _
        "Worst regret bp", _
        "<=5bp hit", _
        "Return rank", _
        "Regret rank")

    ws.Range("A1:P1").Font.Bold = True

    outRow = 2

    For s = 1 To 5

        scenarioName = Alloc_ScenarioName(s)

        For sr = STRAT_FIRST_ROW To STRAT_LAST_ROW

            dataCol = 6 + (sr - STRAT_FIRST_ROW)

            n = 0
            sumRet = 0#
            sumRetSq = 0#
            sumReg = 0#
            worstReg = 0#
            hitN = 0

            ReDim tempRegrets(1 To lastRow - 1)

            For r = 2 To lastRow

                If CStr(retWS.Cells(r, 1).Value) = scenarioName Then

                    n = n + 1

                    retValue = CDbl(retWS.Cells(r, dataCol).Value2)
                    regValue = CDbl(regWS.Cells(r, dataCol).Value2)

                    sumRet = sumRet + retValue
                    sumRetSq = sumRetSq + retValue * retValue
                    sumReg = sumReg + regValue

                    If regValue > worstReg Then worstReg = regValue
                    If regValue <= ALLOC_HIT_THRESHOLD_BP Then hitN = hitN + 1

                    tempRegrets(n) = regValue

                End If

            Next r

            If n > 0 Then

                avgRet = sumRet / n
                volRet = Sqr(Application.WorksheetFunction.Max(0#, sumRetSq / n - avgRet * avgRet))
                avgReg = sumReg / n
                p95Reg = Alloc_Percentile(tempRegrets, n, 0.95)

            Else

                avgRet = 0#
                volRet = 0#
                avgReg = 0#
                p95Reg = 0#

            End If

            ws.Cells(outRow, 1).Value = scenarioName
            ws.Cells(outRow, 2).Value = strat.Cells(sr, 1).Value

            ws.Cells(outRow, 3).Value = strat.Cells(sr, 2).Value
            ws.Cells(outRow, 4).Value = strat.Cells(sr, 3).Value
            ws.Cells(outRow, 5).Value = strat.Cells(sr, 4).Value
            ws.Cells(outRow, 6).Value = strat.Cells(sr, 5).Value
            ws.Cells(outRow, 7).Value = strat.Cells(sr, 6).Value

            ws.Cells(outRow, 8).Value = n
            ws.Cells(outRow, 9).Value = avgRet
            ws.Cells(outRow, 10).Value = volRet
            ws.Cells(outRow, 11).Value = avgReg
            ws.Cells(outRow, 12).Value = p95Reg
            ws.Cells(outRow, 13).Value = worstReg

            If n > 0 Then
                ws.Cells(outRow, 14).Value = hitN / CDbl(n)
            Else
                ws.Cells(outRow, 14).Value = 0#
            End If

            outRow = outRow + 1

        Next sr

    Next s

    ' Rank strategies inside each 9-row scenario block.
    Dim blockStart As Long
    Dim blockEnd As Long
    Dim rr As Long

    For s = 1 To 5

        blockStart = 2 + (s - 1) * STRAT_COUNT
        blockEnd = blockStart + STRAT_COUNT - 1

        For rr = blockStart To blockEnd

            ws.Cells(rr, 15).Formula = "=RANK.EQ(I" & rr & ",$I$" & blockStart & ":$I$" & blockEnd & ",0)"
            ws.Cells(rr, 16).Formula = "=RANK.EQ(K" & rr & ",$K$" & blockStart & ":$K$" & blockEnd & ",1)"

        Next rr

    Next s

    ws.Range("C2:F" & outRow - 1).NumberFormat = "0%"
    ws.Range("G2:G" & outRow - 1).NumberFormat = "0.0"
    ws.Range("I2:J" & outRow - 1).NumberFormat = "0.00%"
    ws.Range("K2:M" & outRow - 1).NumberFormat = "0.0"
    ws.Range("N2:N" & outRow - 1).NumberFormat = "0.0%"
    ws.Range("O2:P" & outRow - 1).NumberFormat = "0"

End Sub

'===============================================================================
' BEST PURE-TENOR FREQUENCY
'
' Shows which pure tenor wins ex-post for each daily starting date.
'
' Source:
'   Allocation_Return_By_Date column D = Best tenor
'
' This analysis is useful because it demonstrates whether one tenor dominates
' or whether the winning tenor changes materially with the path and start date.
'===============================================================================
Private Sub Alloc_BuildBestTenorSummary(ByVal wb As Workbook)

    Dim src As Worksheet
    Dim ws As Worksheet

    Dim s As Long
    Dim r As Long

'   strategy | scenario | start date | end date | return | best tenor | regret
'
' It makes the "cost of being wrong" tangible by showing the exact dates/paths
' that generate each strategy's largest opportunity cost.
'===============================================================================
Private Sub Alloc_BuildExtremeCases(ByVal wb As Workbook)

    Dim retWS As Worksheet
    Dim regWS As Worksheet
    Dim strat As Worksheet
    Dim ws As Worksheet

    Dim lastRow As Long
    Dim sr As Long
    Dim dataCol As Long
    Dim rankNo As Long
    Dim r As Long
    Dim bestRow As Long
    Dim outRow As Long

    Dim used() As Boolean

    Dim thisReg As Double

    Set retWS = wb.Worksheets("Allocation_Return_By_Date")

'===============================================================================
' FULL FEASIBLE ALLOCATION GRID
'
' EVERY ALLOCATION GETS ITS OWN ROW.
'
' This sheet is the direct source for the return-vs-regret scatter.
'===============================================================================
Private Sub Alloc_BuildGridAnalysis(ByVal wb As Workbook, ByVal horizonDays As Long, ByVal stepSize As Double, ByVal minShort As Double, ByVal max6M As Double)

    Dim base As Worksheet
    Dim ws As Worksheet

    Dim baseLast As Long
    Dim nSteps As Long

    Dim iON As Long
    Dim i1 As Long
    Dim i3 As Long
    Dim i6 As Long

    Dim wON As Double
    Dim w1 As Double
    Dim w3 As Double
    Dim w6 As Double

    Dim r As Long
    Dim outRow As Long
    Dim allocID As Long

    Dim hpRet As Double
    Dim annRet As Double
    Dim bestAnn As Double
    Dim regret As Double

    Dim sumRet As Double
    Dim sumRetSq As Double
    Dim sumReg As Double
    Dim worstReg As Double
    Dim hitN As Long
    Dim nObs As Long

    Dim regrets() As Double

    Dim avgRet As Double
    Dim volRet As Double
    Dim avgReg As Double
    Dim p95Reg As Double

    Set base = wb.Worksheets("Allocation_Base")

    baseLast = base.Cells(base.Rows.Count, 1).End(xlUp).Row
    nObs = baseLast - 1

    ReDim regrets(1 To nObs)

    nSteps = CLng(Round(1# / stepSize, 0))

    If Abs(nSteps * stepSize - 1#) > 0.000001 Then
        Err.Raise vbObjectError + 220, , "Allocation step must divide 100% exactly. Example: 10%, 5%, 20%."
    End If

    Set ws = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
    ws.Name = "Allocation_Grid"

    ws.Range("A1:N1").Value = Array("Allocation ID", "Allocation", "ON", "1M", "3M", "6M", "Weighted tenor", "Avg ann return", "Return vol", "Avg regret bp", "Worst regret bp", "P95 regret bp", "<=5bp hit", "Efficient")
    ws.Range("A1:N1").Font.Bold = True

    outRow = 2
    allocID = 1

    For iON = 0 To nSteps

        For i1 = 0 To nSteps - iON

            For i3 = 0 To nSteps - iON - i1

                i6 = nSteps - iON - i1 - i3

                wON = iON * stepSize
                w1 = i1 * stepSize
                w3 = i3 * stepSize
                w6 = i6 * stepSize

                If wON + w1 + 0.0000001 >= minShort And w6 <= max6M + 0.0000001 Then

                    sumRet = 0#
                    sumRetSq = 0#
                    sumReg = 0#
                    worstReg = 0#
                    hitN = 0

                    For r = 2 To baseLast

                        hpRet = wON * CDbl(base.Cells(r, 4).Value2)
                        hpRet = hpRet + w1 * CDbl(base.Cells(r, 5).Value2)
                        hpRet = hpRet + w3 * CDbl(base.Cells(r, 6).Value2)
                        hpRet = hpRet + w6 * CDbl(base.Cells(r, 7).Value2)

                        annRet = Alloc_Annualize(hpRet, horizonDays)
                        bestAnn = CDbl(base.Cells(r, 12).Value2)
                        regret = (bestAnn - annRet) * 10000#

                        If regret < 0# And Abs(regret) < 0.000001 Then regret = 0#

                        sumRet = sumRet + annRet
                        sumRetSq = sumRetSq + annRet * annRet
                        sumReg = sumReg + regret

                        If regret > worstReg Then worstReg = regret
                        If regret <= ALLOC_HIT_THRESHOLD_BP Then hitN = hitN + 1

                        regrets(r - 1) = regret

                    Next r

                    avgRet = sumRet / nObs
                    volRet = Sqr(Application.WorksheetFunction.Max(0#, sumRetSq / nObs - avgRet * avgRet))
                    avgReg = sumReg / nObs
                    p95Reg = Alloc_Percentile(regrets, nObs, 0.95)

                    ws.Cells(outRow, 1).Value = "A" & Format(allocID, "000")
                    ws.Cells(outRow, 2).Value = Alloc_AllocationLabel(wON, w1, w3, w6)

                    ws.Cells(outRow, 3).Value = wON
                    ws.Cells(outRow, 4).Value = w1
                    ws.Cells(outRow, 5).Value = w3
                    ws.Cells(outRow, 6).Value = w6

                    ws.Cells(outRow, 7).Value = w1 + 3# * w3 + 6# * w6
                    ws.Cells(outRow, 8).Value = avgRet
                    ws.Cells(outRow, 9).Value = volRet
                    ws.Cells(outRow, 10).Value = avgReg
                    ws.Cells(outRow, 11).Value = worstReg
                    ws.Cells(outRow, 12).Value = p95Reg
                    ws.Cells(outRow, 13).Value = hitN / CDbl(nObs)
                    ws.Cells(outRow, 14).Value = ""

                    outRow = outRow + 1
                    allocID = allocID + 1

                End If

            Next i3

        Next i1

    Next iON

    ws.Range("C2:F" & outRow - 1).NumberFormat = "0%"
    ws.Range("G2:G" & outRow - 1).NumberFormat = "0.0"
    ws.Range("H2:I" & outRow - 1).NumberFormat = "0.00%"
    ws.Range("J2:L" & outRow - 1).NumberFormat = "0.0"
    ws.Range("M2:M" & outRow - 1).NumberFormat = "0.0%"

End Sub

    Alloc_AllocationLabel = "ON " & Format(wON, "0%") & " | 1M " & Format(w1, "0%") & " | 3M " & Format(w3, "0%") & " | 6M " & Format(w6, "0%")

'===============================================================================
' RETURN / WORST-REGRET EFFICIENT FRONTIER
'
' A point is efficient when no other feasible allocation has BOTH:
'   higher/equal average return
'   lower/equal worst regret
'
' with at least one strict improvement.
'===============================================================================
Private Sub Alloc_MarkEfficientFrontier(ByVal wb As Workbook)

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long
    Dim j As Long

    Dim retI As Double
    Dim regI As Double
    Dim retJ As Double
    Dim regJ As Double

    Dim dominated As Boolean

    Set ws = wb.Worksheets("Allocation_Grid")
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row

    For i = 2 To lastRow

        retI = CDbl(ws.Cells(i, 8).Value2)
        regI = CDbl(ws.Cells(i, 11).Value2)

        dominated = False

        For j = 2 To lastRow

            If j <> i Then

                retJ = CDbl(ws.Cells(j, 8).Value2)
                regJ = CDbl(ws.Cells(j, 11).Value2)

                If retJ >= retI - 0.0000000001 And regJ <= regI + 0.0000000001 Then

                    If retJ > retI + 0.0000000001 Or regJ < regI - 0.0000000001 Then
                        dominated = True
                        Exit For
                    End If

                End If

            End If

        Next j

        If dominated Then
            ws.Cells(i, 14).Value = "No"
        Else
            ws.Cells(i, 14).Value = "Yes"
        End If

    Next i

End Sub

'===============================================================================
' PERCENTILE
'===============================================================================
Private Function Alloc_RangePercentile(ByVal rng As Range, ByVal p As Double) As Double

    Dim arr() As Double
    Dim n As Long
    Dim i As Long

    n = rng.Rows.Count
    ReDim arr(1 To n)

    For i = 1 To n
        arr(i) = CDbl(rng.Cells(i, 1).Value2)
    Next i

    Alloc_RangePercentile = Alloc_Percentile(arr, n, p)

End Function

Private Function Alloc_Percentile(ByRef values() As Double, ByVal n As Long, ByVal p As Double) As Double

    Dim temp() As Double
    Dim i As Long

    If n <= 0 Then
        Alloc_Percentile = 0#
        Exit Function
    End If

    ReDim temp(1 To n)

    For i = 1 To n
        temp(i) = values(i)
    Next i

    Alloc_QuickSort temp, 1, n

    Alloc_Percentile = Alloc_PercentileSorted(temp, n, p)

End Function

Private Function Alloc_PercentileSorted(ByRef values() As Double, ByVal n As Long, ByVal p As Double) As Double

    Dim pos As Double
    Dim lo As Long
    Dim hi As Long
    Dim frac As Double

    If p <= 0# Then
        Alloc_PercentileSorted = values(1)
        Exit Function
    End If

    If p >= 1# Then
        Alloc_PercentileSorted = values(n)
        Exit Function
    End If

    pos = 1# + (n - 1#) * p
    lo = Int(pos)
    hi = lo + 1

    If hi > n Then hi = n

    frac = pos - lo

    Alloc_PercentileSorted = values(lo) + frac * (values(hi) - values(lo))

End Function

Private Sub Alloc_QuickSort(ByRef arr() As Double, ByVal first As Long, ByVal last As Long)

    Dim i As Long
    Dim j As Long
    Dim pivot As Double
    Dim temp As Double

    i = first
    j = last
    pivot = arr((first + last) \ 2)

    Do While i <= j

        Do While arr(i) < pivot
            i = i + 1
        Loop

        Do While arr(j) > pivot
            j = j - 1
        Loop

        If i <= j Then

            temp = arr(i)
            arr(i) = arr(j)
            arr(j) = temp

            i = i + 1
            j = j - 1

        End If

    Loop

    If first < j Then Alloc_QuickSort arr, first, j
    If i < last Then Alloc_QuickSort arr, i, last

End Sub

Private Sub Alloc_ValidateWeights(ByVal wON As Double, ByVal w1 As Double, ByVal w3 As Double, ByVal w6 As Double, ByVal strategyName As String)

    Dim total As Double

    total = wON + w1 + w3 + w6

    If wON < 0# Or w1 < 0# Or w3 < 0# Or w6 < 0# Then
        Err.Raise vbObjectError + 230, , "Negative weight in strategy: " & strategyName
    End If

    If Abs(total - 1#) > 0.000001 Then
        Err.Raise vbObjectError + 231, , "Weights must total 100% for strategy: " & strategyName
    End If

End Sub

Private Function Alloc_SheetExists(ByVal wb As Workbook, ByVal sheetName As String) As Boolean

    Dim ws As Worksheet

    For Each ws In wb.Worksheets

        If LCase$(ws.Name) = LCase$(sheetName) Then
            Alloc_SheetExists = True
            Exit Function
        End If

    Next ws

End Function

Private Sub Alloc_DeleteSheetIfExists(ByVal wb As Workbook, ByVal sheetName As String)

    On Error Resume Next
    wb.Worksheets(sheetName).Delete
    On Error GoTo 0

End Sub
