'===============================================================================
' USD CASH ALLOCATION - CFO BOOKLET FROM CURVE TAB ONLY
'
' PURPOSE
'   Builds a portrait PowerPoint booklet directly from one Excel worksheet named
'   "curve". No PPT_* worksheets, object maps, pre-built analysis tabs, or named
'   charts are required.
'
' REQUIRED INPUT LAYOUT - worksheet "curve"
'   Row 1: headers
'       A1 = Date
'       B1 onward = curve series / tenors, e.g. ON, 1M, 3M, 6M, 9M, 12M ...
'   Row 2 onward: date + annualized rates.
'
' FLEXIBILITY
'   - Any number of curve lines can be added after the Date column.
'   - Recognized tenor labels include ON/O-N/OVERNIGHT/SOFR, D, W, M and Y.
'   - Rates may be stored as decimals (0.0355) or percentages (3.55).
'   - The historical chart plots every populated curve series.
'   - Quantitative pages automatically select an overnight/shortest series and
'     an anchor tenor. By default the anchor is the longest recognized tenor up
'     to MAX_ANALYSIS_MONTHS.
'   - Portfolio construction uses up to four representative tenors sampled from
'     the available curve, so the optimization remains readable even when the
'     input curve contains many lines.
'
' OUTPUT
'   - Portrait A4 PowerPoint booklet (.pptx)
'   - PDF copy when PowerPoint PDF export is available
'
' IMPORTANT
'   This is a risk framework, not a forecast engine. Forward states are explicit
'   stress assumptions designed to compare maturity choices under different rate
'   paths. Adjust the scenario constants below to fit the committee discussion.
'===============================================================================

Public Sub Build_USD_Cash_Booklet_From_Curve()

    Dim wsCurve As Worksheet
    Dim wsTmp As Worksheet
    Dim ppApp As Object
    Dim pres As Object
    Dim outFolder As String
    Dim pptPath As String
    Dim pdfPath As String

    On Error GoTo ErrHandler

    Application.ScreenUpdating = False
    Application.EnableEvents = False

    Set wsCurve = ThisWorkbook.Worksheets(CURVE_SHEET)

    LoadCurveData wsCurve

    If gN < 20 Then
        Err.Raise vbObjectError + 101, , "The curve tab does not contain enough valid dated observations."
    End If

    If gK < 2 Then
        Err.Raise vbObjectError + 102, , "At least two rate series are required after the Date column."
    End If

    Set wsTmp = PrepareTempSheet()

    WriteCleanCurveToTemp wsTmp
    BuildHistoricalSignal wsTmp
    BuildStressAnalysis wsTmp
    BuildScenarioAnalysis wsTmp
    BuildPortfolioAnalysis wsTmp
    BuildExcelCharts wsTmp

    Set ppApp = GetPowerPoint()
    If ppApp Is Nothing Then
        Err.Raise vbObjectError + 103, , "PowerPoint could not be started."
    End If

    ppApp.Visible = True
    Set pres = ppApp.Presentations.Add

    ' Portrait A4 booklet.
    pres.PageSetup.SlideWidth = 595.3
    pres.PageSetup.SlideHeight = 841.9

    Page01_Cover pres
    Page02_TheDecision pres
    Page03_TheCycle pres, wsTmp
    Page04_WhatTheCurvePrices pres, wsTmp
    Page05_HistoricalEvidence pres, wsTmp
    Page06_RiskFramework pres
    Page07_OppositePath pres, wsTmp
    Page08_ScenarioFramework pres, wsTmp
    Page09_PortfolioConstruction pres, wsTmp
    Page10_Governance pres
    Page11_HowWeCanHelp pres

    If Len(ThisWorkbook.Path) > 0 Then
        outFolder = ThisWorkbook.Path
    Else
        outFolder = Environ$("USERPROFILE") & "\Desktop"
    End If

    pptPath = outFolder & "\USD_Cash_Allocation_Booklet_From_Curve.pptx"
    pdfPath = outFolder & "\USD_Cash_Allocation_Booklet_From_Curve.pdf"

    pres.SaveAs pptPath, ppSaveAsOpenXMLPresentation

    On Error Resume Next
    pres.ExportAsFixedFormat pdfPath, ppFixedFormatTypePDF
    On Error GoTo ErrHandler

    If Not KEEP_TEMP_SHEET Then DeleteTempSheet

    Application.ScreenUpdating = True
    Application.EnableEvents = True

    MsgBox _
        "Booklet created from the curve tab only." & vbCrLf & vbCrLf & _
        "PowerPoint:" & vbCrLf & pptPath & vbCrLf & vbCrLf & _
        "PDF:" & vbCrLf & pdfPath, _
        vbInformation

    Exit Sub

ErrHandler:
    Application.ScreenUpdating = True
    Application.EnableEvents = True

    MsgBox _
        "Booklet generation stopped." & vbCrLf & vbCrLf & _
        "Error " & Err.Number & ": " & Err.Description, _
        vbCritical

End Sub

Private Sub LoadCurveData(ByVal ws As Worksheet)

    Dim lastCol As Long
    Dim lastRow As Long
    Dim c As Long
    Dim r As Long
    Dim s As Long
    Dim n As Long
    Dim v As Variant
    Dim hdr As String
    Dim validSeriesCount As Long
    Dim prevDate As Double

    gSourceDateCol = FindDateColumn(ws)
    If gSourceDateCol = 0 Then
        Err.Raise vbObjectError + 110, , "A Date header was not found in row 1 of the curve tab."
    End If

    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    lastRow = ws.Cells(ws.Rows.Count, gSourceDateCol).End(xlUp).Row

    ' Count populated rate-series headers containing at least one numeric value.
    validSeriesCount = 0
    For c = 1 To lastCol
        If c <> gSourceDateCol Then
            hdr = Trim$(CStr(ws.Cells(1, c).Value))
            If Len(hdr) > 0 Then
                If ColumnHasNumericData(ws, c, 2, lastRow) Then
                    validSeriesCount = validSeriesCount + 1
                End If
            End If
        End If
    Next c

    gK = validSeriesCount
    If gK = 0 Then Exit Sub

    ReDim gHeaders(1 To gK)
    ReDim gMonths(1 To gK)
    ReDim gSourceCols(1 To gK)

    s = 0
    For c = 1 To lastCol
        If c <> gSourceDateCol Then
            hdr = Trim$(CStr(ws.Cells(1, c).Value))
            If Len(hdr) > 0 Then
                If ColumnHasNumericData(ws, c, 2, lastRow) Then
                    s = s + 1
                    gHeaders(s) = hdr
                    gMonths(s) = ParseTenorMonths(hdr)
                    gSourceCols(s) = c
                End If
            End If
        End If
    Next c

    ' Count valid date rows.
    n = 0
    For r = 2 To lastRow
        v = ws.Cells(r, gSourceDateCol).Value
        If IsDateLike(v) Then n = n + 1
    Next r

    gN = n
    ReDim gDates(1 To gN)
    ReDim gRates(1 To gN, 1 To gK)
    ReDim gValid(1 To gN, 1 To gK)

    n = 0
    prevDate = 0

    For r = 2 To lastRow

        v = ws.Cells(r, gSourceDateCol).Value

        If IsDateLike(v) Then

            n = n + 1
            gDates(n) = CDbl(CDate(v))

            If n > 1 Then
                If gDates(n) <= prevDate Then
                    Err.Raise vbObjectError + 111, , _
                        "Dates in the curve tab must be strictly ascending. " & _
                        "Please sort the data by Date before running the macro."
                End If
            End If

            prevDate = gDates(n)

            For s = 1 To gK
                v = ws.Cells(r, gSourceCols(s)).Value
                If IsNumeric(v) And Len(CStr(v)) > 0 Then
                    gRates(n, s) = NormalizeRate(CDbl(v))
                    gValid(n, s) = True
                Else
                    gRates(n, s) = 0
                    gValid(n, s) = False
                End If
            Next s

        End If

    Next r

    gONIdx = FindOvernightSeries()
    gAnchorIdx = FindAnchorSeries()

    If gONIdx = 0 Then
        Err.Raise vbObjectError + 112, , _
            "No overnight/shortest recognizable tenor could be identified. " & _
            "Use a header such as ON, O/N, Overnight or SOFR, or a standard tenor label."
    End If

    If gAnchorIdx = 0 Or gAnchorIdx = gONIdx Then
        Err.Raise vbObjectError + 113, , _
            "No longer anchor tenor could be identified. Add at least one recognized tenor such as 1M, 3M or 6M."
    End If

End Sub

Private Function FindDateColumn(ByVal ws As Worksheet) As Long

    Dim lastCol As Long
    Dim c As Long
    Dim hdr As String

    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        hdr = UCase$(Trim$(CStr(ws.Cells(1, c).Value)))
        If hdr = "DATE" Or hdr = "DATA" Or hdr = "AS OF" Or hdr = "ASOF" Then
            FindDateColumn = c
            Exit Function
        End If
    Next c

    FindDateColumn = 0

End Function

Private Function ColumnHasNumericData( _
    ByVal ws As Worksheet, _
    ByVal colNo As Long, _
    ByVal startRow As Long, _
    ByVal endRow As Long) As Boolean

    If endRow < startRow Then Exit Function

    ColumnHasNumericData = _
        (Application.WorksheetFunction.Count( _
            ws.Range(ws.Cells(startRow, colNo), ws.Cells(endRow, colNo))) > 0)

End Function

Private Function IsDateLike(ByVal v As Variant) As Boolean

    On Error GoTo Nope

    If IsDate(v) Then
        IsDateLike = True
        Exit Function
    End If

    If IsNumeric(v) Then
        If CDbl(v) > 20000 And CDbl(v) < 100000 Then
            IsDateLike = True
            Exit Function
        End If
    End If

Nope:
End Function

Private Function NormalizeRate(ByVal v As Double) As Double
    If Abs(v) > 1# Then
        NormalizeRate = v / 100#
    Else
        NormalizeRate = v
    End If
End Function

Private Function ParseTenorMonths(ByVal rawHeader As String) As Double

    Dim h As String
    Dim i As Long
    Dim numText As String
    Dim unitText As String
    Dim n As Double

    h = UCase$(Trim$(rawHeader))
    h = Replace(h, " ", "")
    h = Replace(h, "-", "")
    h = Replace(h, "_", "")

    If h = "ON" Or h = "O/N" Or h = "OVERNIGHT" Or h = "SOFR" Or _
       h = "SOFRON" Or h = "ONSOFR" Then
        ParseTenorMonths = 1# / 30.4375
        Exit Function
    End If

    ' Keep only first numeric block plus trailing unit.
    numText = ""
    For i = 1 To Len(h)
        If Mid$(h, i, 1) Like "[0-9.]" Then
            numText = numText & Mid$(h, i, 1)
        ElseIf Len(numText) > 0 Then
            unitText = Mid$(h, i)
            Exit For
        End If
    Next i

    If Len(numText) = 0 Then
        ParseTenorMonths = -1
        Exit Function
    End If

    n = Val(numText)
    unitText = UCase$(unitText)

    If Left$(unitText, 1) = "D" Then
        ParseTenorMonths = n / 30.4375
    ElseIf Left$(unitText, 1) = "W" Then
        ParseTenorMonths = n * 7# / 30.4375
    ElseIf Left$(unitText, 1) = "M" Then
        ParseTenorMonths = n
    ElseIf Left$(unitText, 1) = "Y" Then
        ParseTenorMonths = n * 12#
    Else
        ' Common bare labels such as "1MO", "3MONTH", "1YR".
        If InStr(unitText, "MO") > 0 Or InStr(unitText, "MONTH") > 0 Then
            ParseTenorMonths = n
        ElseIf InStr(unitText, "YR") > 0 Or InStr(unitText, "YEAR") > 0 Then
            ParseTenorMonths = n * 12#
        Else
            ParseTenorMonths = -1
        End If
    End If

End Function

Private Function FindOvernightSeries() As Long

    Dim i As Long
    Dim h As String
    Dim bestMonths As Double
    Dim bestIdx As Long

    ' First use explicit overnight naming.
    For i = 1 To gK
        h = UCase$(Replace(Replace(Trim$(gHeaders(i)), " ", ""), "-", ""))
        If h = "ON" Or h = "O/N" Or h = "OVERNIGHT" Or h = "SOFR" Or _
           h = "SOFRON" Or h = "ONSOFR" Then
            FindOvernightSeries = i
            Exit Function
        End If
    Next i

    ' Otherwise use the shortest recognized tenor.
    bestMonths = 1E+30
    For i = 1 To gK
        If gMonths(i) > 0 Then
            If gMonths(i) < bestMonths Then
                bestMonths = gMonths(i)
                bestIdx = i
            End If
        End If
    Next i

    FindOvernightSeries = bestIdx

End Function

Private Function FindAnchorSeries() As Long

    Dim i As Long
    Dim bestMonths As Double
    Dim bestIdx As Long

    ' Prefer the longest recognizable point no longer than MAX_ANALYSIS_MONTHS.
    bestMonths = -1
    For i = 1 To gK
        If gMonths(i) > bestMonths And gMonths(i) <= MAX_ANALYSIS_MONTHS Then
            If i <> gONIdx Then
                bestMonths = gMonths(i)
                bestIdx = i
            End If
        End If
    Next i

    ' If all recognized points are longer, take the longest recognized point.
    If bestIdx = 0 Then
        bestMonths = -1
        For i = 1 To gK
            If gMonths(i) > bestMonths And i <> gONIdx Then
                bestMonths = gMonths(i)
                bestIdx = i
            End If
        Next i
    End If

    FindAnchorSeries = bestIdx

End Function

    DeleteTempSheet

Private Sub DeleteTempSheet()

    Dim ws As Worksheet

    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(TEMP_SHEET)
    On Error GoTo 0

    If Not ws Is Nothing Then
        Application.DisplayAlerts = False
        ws.Delete
        Application.DisplayAlerts = True
    End If

End Sub

Private Sub WriteCleanCurveToTemp(ByVal ws As Worksheet)

    Dim arr() As Variant
    Dim r As Long
    Dim s As Long

    ReDim arr(1 To gN + 1, 1 To gK + 1)

    arr(1, 1) = "Date"
    For s = 1 To gK
        arr(1, s + 1) = gHeaders(s)
    Next s

    For r = 1 To gN
        arr(r + 1, 1) = CDate(gDates(r))
        For s = 1 To gK
            If gValid(r, s) Then
                arr(r + 1, s + 1) = gRates(r, s)
            Else
                arr(r + 1, s + 1) = Empty
            End If
        Next s
    Next r

    ws.Range("A1").Resize(gN + 1, gK + 1).Value = arr
    ws.Range("A2:A" & gN + 1).NumberFormat = "dd-mmm-yyyy"
    ws.Range(ws.Cells(2, 2), ws.Cells(gN + 1, gK + 1)).NumberFormat = "0.000%"

End Sub

Private Sub BuildHistoricalSignal(ByVal ws As Worksheet)

    Dim out() As Variant
    Dim x() As Double
    Dim y() As Double
    Dim z() As Double
    Dim i As Long
    Dim n As Long
    Dim targetDate As Double
    Dim onRealized As Double
    Dim anchorRate As Double
    Dim premiumBp As Double
    Dim excessBp As Double
    Dim futureMoveBp As Double
    Dim endOn As Double
    Dim hasEndOn As Boolean
    Dim sumPremium As Double
    Dim sumExcess As Double
    Dim winCount As Long

    ReDim out(1 To gN, 1 To 4)
    ReDim x(1 To gN)
    ReDim y(1 To gN)
    ReDim z(1 To gN)

    n = 0

    For i = 1 To gN - 1

        If gValid(i, gONIdx) And gValid(i, gAnchorIdx) Then

            targetDate = TenorTargetDate(gDates(i), gMonths(gAnchorIdx))

            If targetDate > gDates(gN) Then Exit For

            onRealized = CalcRollingONAnnualized(i, targetDate)

            If onRealized > -100 Then

                anchorRate = gRates(i, gAnchorIdx)
                premiumBp = (anchorRate - gRates(i, gONIdx)) * 10000#
                excessBp = (anchorRate - onRealized) * 10000#

                hasEndOn = GetRateAtOrBeforeDate(gONIdx, targetDate, endOn)
                If hasEndOn Then
                    futureMoveBp = (endOn - gRates(i, gONIdx)) * 10000#
                Else
                    futureMoveBp = 0
                End If

                n = n + 1
                out(n, 1) = CDate(gDates(i))
                out(n, 2) = premiumBp
                out(n, 3) = excessBp
                out(n, 4) = futureMoveBp

                x(n) = premiumBp
                y(n) = excessBp
                z(n) = futureMoveBp

                sumPremium = sumPremium + premiumBp
                sumExcess = sumExcess + excessBp
                If excessBp > 0 Then winCount = winCount + 1

            End If

        End If

    Next i

    gSignalN = n

    If n > 2 Then
        gSignalR2 = CalcR2(x, y, n)
        gFutureMoveR2 = CalcR2(z, y, n)
        gWinRate = winCount / n
        gAvgPremiumBp = sumPremium / n
        gAvgExcessBp = sumExcess / n
    End If

    ws.Range("J1:M1").Value = Array("StartDate", "StartingPremiumBp", "RealizedExcessBp", "FutureONMoveBp")
    If n > 0 Then
        ' Write the full preallocated array; rows after n remain blank.
        ws.Range("J2").Resize(gN, 4).Value = out
        ws.Range("J2:J" & n + 1).NumberFormat = "dd-mmm-yyyy"
        ws.Range("K2:M" & n + 1).NumberFormat = "0.0"
    End If

End Sub

Private Function CalcRollingONAnnualized( _
    ByVal startIdx As Long, _
    ByVal targetDate As Double) As Double

    Dim factor As Double
    Dim k As Long
    Dim intervalStart As Double
    Dim intervalEnd As Double
    Dim useEnd As Double
    Dim days As Double
    Dim totalDays As Double

    factor = 1#
    totalDays = targetDate - gDates(startIdx)

    If totalDays <= 0 Then
        CalcRollingONAnnualized = -999
        Exit Function
    End If

    For k = startIdx To gN - 1

        intervalStart = gDates(k)
        intervalEnd = gDates(k + 1)

        If intervalStart >= targetDate Then Exit For

        useEnd = intervalEnd
        If useEnd > targetDate Then useEnd = targetDate

        days = useEnd - intervalStart

        If days > 0 Then
            If gValid(k, gONIdx) Then
                factor = factor * (1# + gRates(k, gONIdx) * days / 360#)
            Else
                CalcRollingONAnnualized = -999
                Exit Function
            End If
        End If

        If intervalEnd >= targetDate Then Exit For

    Next k

    CalcRollingONAnnualized = (factor - 1#) * 360# / totalDays

End Function

Private Function GetRateAtOrBeforeDate( _
    ByVal seriesIdx As Long, _
    ByVal targetDate As Double, _
    ByRef resultRate As Double) As Boolean

    Dim i As Long

    For i = gN To 1 Step -1
        If gDates(i) <= targetDate Then
            If gValid(i, seriesIdx) Then
                resultRate = gRates(i, seriesIdx)
                GetRateAtOrBeforeDate = True
                Exit Function
            End If
        End If
    Next i

End Function

Private Function TenorTargetDate( _
    ByVal startDate As Double, _
    ByVal monthsValue As Double) As Double

    Dim roundedMonths As Long

    roundedMonths = CLng(Round(monthsValue, 0))

    If monthsValue >= 0.75 And Abs(monthsValue - roundedMonths) < 0.05 Then
        TenorTargetDate = CDbl(DateAdd("m", roundedMonths, CDate(startDate)))
    Else
        TenorTargetDate = startDate + Application.WorksheetFunction.Max(1, Round(monthsValue * 30.4375, 0))
    End If

End Function

Private Function CalcR2( _
    ByRef x() As Double, _
    ByRef y() As Double, _
    ByVal n As Long) As Double

    Dim i As Long
    Dim meanX As Double
    Dim meanY As Double
    Dim cov As Double
    Dim varX As Double
    Dim varY As Double
    Dim corr As Double

    If n < 3 Then Exit Function

    For i = 1 To n
        meanX = meanX + x(i)
        meanY = meanY + y(i)
    Next i

    meanX = meanX / n
    meanY = meanY / n

    For i = 1 To n
        cov = cov + (x(i) - meanX) * (y(i) - meanY)
        varX = varX + (x(i) - meanX) ^ 2
        varY = varY + (y(i) - meanY) ^ 2
    Next i

    If varX <= 0 Or varY <= 0 Then Exit Function

    corr = cov / Sqr(varX * varY)
    CalcR2 = corr ^ 2

End Function

Private Sub BuildStressAnalysis(ByVal ws As Worksheet)

    Dim horizonDays As Long
    Dim currentON As Double
    Dim anchorRate As Double
    Dim riseON As Double
    Dim holdON As Double
    Dim fallON As Double
    Dim arr(1 To 3, 1 To 3) As Variant

    currentON = LatestValidRate(gONIdx)
    anchorRate = LatestValidRate(gAnchorIdx)

    horizonDays = CLng(Application.WorksheetFunction.Max(30, Round(gMonths(gAnchorIdx) * 30.4375, 0)))

    riseON = SimulatedRollingONLinear(currentON, currentON + STRESS_UP_BP / 10000#, horizonDays)
    holdON = SimulatedRollingONLinear(currentON, currentON, horizonDays)
    fallON = SimulatedRollingONLinear(currentON, Application.WorksheetFunction.Max(0, currentON + STRESS_DOWN_BP / 10000#), horizonDays)

    gStressRiseExcessBp = (anchorRate - riseON) * 10000#
    gStressHoldExcessBp = (anchorRate - holdON) * 10000#
    gStressFallExcessBp = (anchorRate - fallON) * 10000#
    gStressSwingBp = gStressFallExcessBp - gStressRiseExcessBp

    arr(1, 1) = "Rates +100bp"
    arr(1, 2) = riseON
    arr(1, 3) = gStressRiseExcessBp

    arr(2, 1) = "Rates hold"
    arr(2, 2) = holdON
    arr(2, 3) = gStressHoldExcessBp

    arr(3, 1) = "Rates -100bp"
    arr(3, 2) = fallON
    arr(3, 3) = gStressFallExcessBp

    ws.Range("P1:R1").Value = Array("Stress", "RollingONAnnualized", "AnchorExcessBp")
    ws.Range("P2:R4").Value = arr
    ws.Range("Q2:Q4").NumberFormat = "0.00%"
    ws.Range("R2:R4").NumberFormat = "0.0"

End Sub

Private Function SimulatedRollingONLinear( _
    ByVal startRate As Double, _
    ByVal endRate As Double, _
    ByVal horizonDays As Long) As Double

    Dim d As Long
    Dim r As Double
    Dim factor As Double

    factor = 1#

    For d = 0 To horizonDays - 1
        If horizonDays <= 1 Then
            r = startRate
        Else
            r = startRate + (endRate - startRate) * d / (horizonDays - 1)
        End If
        factor = factor * (1# + r / 360#)
    Next d

    SimulatedRollingONLinear = (factor - 1#) * 360# / horizonDays

End Function

Private Sub BuildScenarioAnalysis(ByVal ws As Worksheet)

    Dim currentON As Double
    Dim months As Long
    Dim m As Long
    Dim s As Long
    Dim rep As Long
    Dim arr() As Variant

    currentON = LatestValidRate(gONIdx)

    gScenarioCount = 5
    ReDim gScenarioNames(1 To gScenarioCount)
    ReDim gScenarioShiftBp(1 To gScenarioCount)

    gScenarioNames(1) = "Higher"
    gScenarioShiftBp(1) = SCEN_HIGHER_BP

    gScenarioNames(2) = "Still elevated"
    gScenarioShiftBp(2) = SCEN_ELEVATED_BP

    gScenarioNames(3) = "Hold"
    gScenarioShiftBp(3) = SCEN_HOLD_BP

    gScenarioNames(4) = "Gradual easing"
    gScenarioShiftBp(4) = SCEN_EASING_BP

    gScenarioNames(5) = "Faster easing"
    gScenarioShiftBp(5) = SCEN_FAST_EASING_BP

    SelectRepresentativeSeries

    ReDim gScenarioReturns(1 To gScenarioCount, 1 To gRepCount)

    months = SCENARIO_HORIZON_MONTHS
    ReDim arr(1 To months + 2, 1 To gScenarioCount + 1)

    arr(1, 1) = "Month"
    For s = 1 To gScenarioCount
        arr(1, s + 1) = gScenarioNames(s)
    Next s

    For m = 0 To months
        arr(m + 2, 1) = m
        For s = 1 To gScenarioCount
            arr(m + 2, s + 1) = ScenarioONRate(currentON, gScenarioShiftBp(s), m, months)
        Next s
    Next m

    ws.Range("T1").Resize(months + 2, gScenarioCount + 1).Value = arr
    ws.Range("U2:Y" & months + 2).NumberFormat = "0.00%"

    ' Strategy return by representative tenor and scenario.
    ws.Range("AA1").Value = "Scenario"
    For rep = 1 To gRepCount
        ws.Cells(1, 27 + rep).Value = gHeaders(gRepIdx(rep))
    Next rep

    For s = 1 To gScenarioCount
        ws.Cells(s + 1, 27).Value = gScenarioNames(s)
        For rep = 1 To gRepCount
            gScenarioReturns(s, rep) = SimulateTenorStrategy(gRepIdx(rep), gScenarioShiftBp(s), months)
            ws.Cells(s + 1, 27 + rep).Value = gScenarioReturns(s, rep)
        Next rep
    Next s

    ws.Range(ws.Cells(2, 28), ws.Cells(gScenarioCount + 1, 27 + gRepCount)).NumberFormat = "0.00%"

End Sub

Private Function ScenarioONRate( _
    ByVal currentON As Double, _
    ByVal terminalShiftBp As Double, _
    ByVal monthNo As Double, _
    ByVal totalMonths As Double) As Double

    Dim terminalRate As Double
    Dim result As Double

    terminalRate = currentON + terminalShiftBp / 10000#
    If terminalRate < 0 Then terminalRate = 0

    If totalMonths <= 0 Then
        result = currentON
    Else
        result = currentON + (terminalRate - currentON) * monthNo / totalMonths
    End If

    If result < 0 Then result = 0
    ScenarioONRate = result

End Function

Private Function SimulateTenorStrategy( _
    ByVal seriesIdx As Long, _
    ByVal terminalShiftBp As Double, _
    ByVal horizonMonths As Long) As Double

    Dim currentON As Double
    Dim spread As Double
    Dim totalDays As Long
    Dim tenorDays As Long
    Dim elapsed As Long
    Dim segmentDays As Long
    Dim monthAtRoll As Double
    Dim onAtRoll As Double
    Dim rateAtRoll As Double
    Dim factor As Double

    currentON = LatestValidRate(gONIdx)
    spread = LatestValidRate(seriesIdx) - currentON

    totalDays = CLng(Round(horizonMonths * 365.25 / 12#, 0))
    tenorDays = CLng(Application.WorksheetFunction.Max(1, Round(gMonths(seriesIdx) * 30.4375, 0)))

    factor = 1#
    elapsed = 0

    Do While elapsed < totalDays

        segmentDays = tenorDays
        If elapsed + segmentDays > totalDays Then segmentDays = totalDays - elapsed

        monthAtRoll = elapsed * 12# / 365.25
        onAtRoll = ScenarioONRate(currentON, terminalShiftBp, monthAtRoll, horizonMonths)
        rateAtRoll = onAtRoll + spread
        If rateAtRoll < 0 Then rateAtRoll = 0

        factor = factor * (1# + rateAtRoll * segmentDays / 360#)
        elapsed = elapsed + segmentDays

    Loop

    SimulateTenorStrategy = (factor - 1#) * 360# / totalDays

End Function

Private Sub SelectRepresentativeSeries()

    Dim idx() As Long
    Dim cnt As Long
    Dim i As Long

    GetRecognizedSortedSeries idx, cnt

    If cnt = 0 Then
        Err.Raise vbObjectError + 120, , "No recognizable tenors are available for scenario analysis."
    End If

    If cnt <= 4 Then
        gRepCount = cnt
        ReDim gRepIdx(1 To gRepCount)
        For i = 1 To cnt
            gRepIdx(i) = idx(i)
        Next i
    Else
        gRepCount = 4
        ReDim gRepIdx(1 To 4)
        gRepIdx(1) = idx(1)
        gRepIdx(2) = idx(1 + CLng(Round((cnt - 1) / 3#, 0)))
        gRepIdx(3) = idx(1 + CLng(Round(2# * (cnt - 1) / 3#, 0)))
        gRepIdx(4) = idx(cnt)

        ' Protect against duplicate rounded selections.
        EnsureUniqueRepresentativeIndexes idx, cnt
    End If

End Sub

Private Sub EnsureUniqueRepresentativeIndexes( _
    ByRef sortedIdx() As Long, _
    ByVal cnt As Long)

    Dim used As Object
    Dim i As Long
    Dim j As Long

    Set used = CreateObject("Scripting.Dictionary")

    For i = 1 To gRepCount

        If Not used.Exists(CStr(gRepIdx(i))) Then
            used.Add CStr(gRepIdx(i)), True
        Else
            For j = 1 To cnt
                If Not used.Exists(CStr(sortedIdx(j))) Then
                    gRepIdx(i) = sortedIdx(j)
                    used.Add CStr(gRepIdx(i)), True
                    Exit For
                End If
            Next j
        End If

    Next i

    SortRepIndexesByMonths

End Sub

Private Sub SortRepIndexesByMonths()

    Dim i As Long
    Dim j As Long
    Dim tmp As Long

    For i = 1 To gRepCount - 1
        For j = i + 1 To gRepCount
            If gMonths(gRepIdx(j)) < gMonths(gRepIdx(i)) Then
                tmp = gRepIdx(i)
                gRepIdx(i) = gRepIdx(j)
                gRepIdx(j) = tmp
            End If
        Next j
    Next i

End Sub

Private Sub GetRecognizedSortedSeries( _
    ByRef idx() As Long, _
    ByRef cnt As Long)

    Dim i As Long
    Dim j As Long
    Dim tmp As Long

    cnt = 0
    ReDim idx(1 To gK)

    For i = 1 To gK
        If gMonths(i) > 0 Then
            cnt = cnt + 1
            idx(cnt) = i
        End If
    Next i

    If cnt = 0 Then Exit Sub
    ReDim Preserve idx(1 To cnt)

    For i = 1 To cnt - 1
        For j = i + 1 To cnt
            If gMonths(idx(j)) < gMonths(idx(i)) Then
                tmp = idx(i)
                idx(i) = idx(j)
                idx(j) = tmp
            End If
        Next j
    Next i

End Sub

Private Sub BuildPortfolioAnalysis(ByVal ws As Worksheet)

    Dim unitsTotal As Long
    Dim weights() As Long
    Dim rowOut As Long
    Dim headers() As Variant
    Dim i As Long

    unitsTotal = 100 \ PORTFOLIO_WEIGHT_STEP_PCT
    ReDim weights(1 To gRepCount)

    ' Output starts at AA10.
    ws.Cells(10, 27).Value = "WeightedTenorMonths"
    ws.Cells(10, 28).Value = "WorstRegretBp"
    ws.Cells(10, 29).Value = "Weights"

    For i = 1 To gRepCount
        ws.Cells(10, 29 + i).Value = gHeaders(gRepIdx(i)) & " Weight"
    Next i

    rowOut = 11
    gPortfolioN = 0
    gMinRegretBp = 1E+30
    gBestWeightsLabel = ""
    gBestWeightedTenor = 0

    EnumeratePortfolioWeights ws, 1, unitsTotal, weights, rowOut

    gPortfolioN = rowOut - 11

    FindRobustPortfolioBand ws, 11, rowOut - 1

    ws.Range(ws.Cells(11, 27), ws.Cells(rowOut - 1, 28)).NumberFormat = "0.00"
    If gRepCount > 0 Then
        ws.Range(ws.Cells(11, 30), ws.Cells(rowOut - 1, 29 + gRepCount)).NumberFormat = "0%"
    End If

End Sub

Private Sub EnumeratePortfolioWeights( _
    ByVal ws As Worksheet, _
    ByVal position As Long, _
    ByVal unitsRemaining As Long, _
    ByRef weights() As Long, _
    ByRef rowOut As Long)

    Dim u As Long

    If position = gRepCount Then
        weights(position) = unitsRemaining
        EvaluatePortfolio ws, weights, rowOut
        rowOut = rowOut + 1
        Exit Sub
    End If

    For u = 0 To unitsRemaining
        weights(position) = u
        EnumeratePortfolioWeights ws, position + 1, unitsRemaining - u, weights, rowOut
    Next u

End Sub

Private Sub EvaluatePortfolio( _
    ByVal ws As Worksheet, _
    ByRef weights() As Long, _
    ByVal rowOut As Long)

    Dim s As Long
    Dim r As Long
    Dim unitsTotal As Long
    Dim w As Double
    Dim portReturn As Double
    Dim bestReturn As Double
    Dim regretBp As Double
    Dim worstRegretBp As Double
    Dim weightedTenor As Double
    Dim label As String

    unitsTotal = 100 \ PORTFOLIO_WEIGHT_STEP_PCT

    For r = 1 To gRepCount
        w = weights(r) / unitsTotal
        weightedTenor = weightedTenor + w * gMonths(gRepIdx(r))
        If weights(r) > 0 Then
            If Len(label) > 0 Then label = label & " / "
            label = label & Format(w, "0%") & " " & gHeaders(gRepIdx(r))
        End If
    Next r

    worstRegretBp = 0

    For s = 1 To gScenarioCount

        portReturn = 0
        bestReturn = -1E+30

        For r = 1 To gRepCount
            w = weights(r) / unitsTotal
            portReturn = portReturn + w * gScenarioReturns(s, r)
            If gScenarioReturns(s, r) > bestReturn Then bestReturn = gScenarioReturns(s, r)
        Next r

        regretBp = (bestReturn - portReturn) * 10000#
        If regretBp > worstRegretBp Then worstRegretBp = regretBp

    Next s

    ws.Cells(rowOut, 27).Value = weightedTenor
    ws.Cells(rowOut, 28).Value = worstRegretBp
    ws.Cells(rowOut, 29).Value = label

    For r = 1 To gRepCount
        ws.Cells(rowOut, 29 + r).Value = weights(r) / unitsTotal
    Next r

    If worstRegretBp < gMinRegretBp Then
        gMinRegretBp = worstRegretBp
        gBestWeightsLabel = label
        gBestWeightedTenor = weightedTenor
    End If

End Sub

Private Sub FindRobustPortfolioBand( _
    ByVal ws As Worksheet, _
    ByVal startRow As Long, _
    ByVal endRow As Long)

    Dim r As Long
    Dim regretBp As Double
    Dim tenor As Double

    gRobustCount = 0
    gRobustMinTenor = 1E+30
    gRobustMaxTenor = -1E+30

    For r = startRow To endRow

        regretBp = CDbl(ws.Cells(r, 28).Value)
        tenor = CDbl(ws.Cells(r, 27).Value)

        If regretBp <= gMinRegretBp + ROBUST_TOLERANCE_BP Then
            gRobustCount = gRobustCount + 1
            If tenor < gRobustMinTenor Then gRobustMinTenor = tenor
            If tenor > gRobustMaxTenor Then gRobustMaxTenor = tenor
        End If

    Next r

    If gRobustCount = 0 Then
        gRobustMinTenor = gBestWeightedTenor
        gRobustMaxTenor = gBestWeightedTenor
    End If

End Sub

Private Function LatestValidRate(ByVal seriesIdx As Long) As Double

    Dim i As Long

    For i = gN To 1 Step -1
        If gValid(i, seriesIdx) Then
            LatestValidRate = gRates(i, seriesIdx)
            Exit Function
        End If
    Next i

    Err.Raise vbObjectError + 121, , "No valid rate found for " & gHeaders(seriesIdx) & "."

End Function

Private Function AddText( _
    ByVal s As Object, _
    ByVal txt As String, _
    ByVal L As Single, _
    ByVal T As Single, _
    ByVal W As Single, _
    ByVal H As Single, _
    ByVal fontSize As Single, _
    ByVal fontColor As Long, _
    Optional ByVal isBold As Boolean = False, _
    Optional ByVal align As Long = ppAlignLeft, _
    Optional ByVal fontName As String = FONT_MAIN) As Object

    Dim shp As Object

    Set shp = s.Shapes.AddTextbox(msoTextOrientationHorizontal, L, T, W, H)

    With shp
        .Line.Visible = msoFalse
        .Fill.Visible = msoFalse
        .TextFrame.WordWrap = msoTrue
        .TextFrame.MarginLeft = 0
        .TextFrame.MarginRight = 0
        .TextFrame.MarginTop = 0
        .TextFrame.MarginBottom = 0

        With .TextFrame.TextRange
            .Text = txt
            .Font.Name = fontName
            .Font.Size = fontSize
            .Font.Color.RGB = fontColor
            .Font.Bold = isBold
            .ParagraphFormat.Alignment = align
        End With
    End With

    Set AddText = shp

End Function

Private Sub AddDynamicTenorSpectrum(ByVal s As Object)

    Dim idx() As Long
    Dim cnt As Long
    Dim showCount As Long
    Dim pick() As Long
    Dim i As Long
    Dim xStart As Single
    Dim xEnd As Single
    Dim x As Single
    Dim pos As Double

    GetRecognizedSortedSeries idx, cnt
    If cnt = 0 Then Exit Sub

    showCount = Application.WorksheetFunction.Min(5, cnt)
    ReDim pick(1 To showCount)

    If cnt <= 5 Then
        For i = 1 To cnt
            pick(i) = idx(i)
        Next i
    Else
        For i = 1 To showCount
            pos = 1 + (i - 1) * (cnt - 1) / (showCount - 1)
            pick(i) = idx(CLng(Round(pos, 0)))
        Next i
    End If

    xStart = 65
    xEnd = 500
    AddLineShape s, xStart + 15, 270, xEnd - 15, 270, C_Line(), 2

    For i = 1 To showCount
        If showCount = 1 Then
            x = (xStart + xEnd) / 2
        Else
            x = xStart + (i - 1) * (xEnd - xStart) / (showCount - 1)
        End If
        AddTenorNode s, x, 270, gHeaders(pick(i))
    Next i

    AddText s, "More flexibility / faster reset", 48, 320, 200, 20, 9.5, C_Gray()
    AddText s, "More rate certainty / slower reset", 335, 320, 212, 20, 9.5, C_Gray(), False, ppAlignRight

End Sub

Private Sub PasteExcelChart( _
    ByVal s As Object, _
    ByVal ws As Worksheet, _
    ByVal chartName As String, _
    ByVal L As Single, _
    ByVal T As Single, _
    ByVal W As Single, _
    ByVal H As Single)

    Dim co As ChartObject
    Dim sr As Object
    Dim shp As Object
    Dim ph As Object

    On Error GoTo ChartMissing

    Set co = ws.ChartObjects(chartName)
    co.Chart.CopyPicture Appearance:=xlScreen, Format:=xlPicture
    DoEvents

    Set sr = s.Shapes.PasteSpecial(ppPasteEnhancedMetafile)
    Set shp = sr(1)
    FitShape shp, L, T, W, H

    Exit Sub

ChartMissing:

    On Error GoTo 0

    Set ph = s.Shapes.AddShape(msoShapeRoundedRectangle, L, T, W, H)
    With ph
        .Fill.ForeColor.RGB = RGB(248, 249, 251)
        .Line.ForeColor.RGB = C_Line()
    End With

    AddText s, "Chart could not be created: " & chartName, L + 20, T + H / 2 - 20, W - 40, 40, 12, C_Gray(), True, ppAlignCenter

End Sub

Private Sub FitShape(ByVal shp As Object, ByVal L As Single, ByVal T As Single, ByVal W As Single, ByVal H As Single)

    Dim scaleRatio As Double

    shp.LockAspectRatio = msoTrue
    scaleRatio = W / shp.Width
    If shp.Height * scaleRatio > H Then scaleRatio = H / shp.Height

    shp.Width = shp.Width * scaleRatio
    shp.Height = shp.Height * scaleRatio
    shp.Left = L + (W - shp.Width) / 2
    shp.Top = T + (H - shp.Height) / 2

End Sub

Private Function FormatSignedBp(ByVal bp As Double) As String
    If bp > 0 Then
        FormatSignedBp = "+" & Format(bp, "0.0") & " bp"
    ElseIf bp < 0 Then
        FormatSignedBp = Format(bp, "0.0") & " bp"
    Else
        FormatSignedBp = "0.0 bp"
    End If
End Function
