Option Explicit

'===============================================================================
' CFO CASH ALLOCATION | SINGLE-MODULE SLIDE BUILDER
' Version 3.0
'
' PURPOSE
'   Builds the complete Slide 4 allocation analysis from the five 3-year
'   scenario curves created in Curve_Scenarios.
'
'   The slide is designed for a CFO / corporate treasury audience.
'
' CORE MESSAGE
'   Diversification can be optimized, not guessed.
'
'   The model:
'       1. Reads five 3-year market-rate scenarios.
'       2. Calculates the return from rolling ON, 1M, 3M and 6M.
'       3. Tests every feasible maturity mix.
'       4. Measures return and the cost of being wrong (maximum regret).
'       5. Builds an efficient frontier.
'       6. Selects the highest-return portfolio within a small tolerance of
'          the minimum-regret solution.
'       7. Creates a slide-ready Excel layout with:
'             - Efficient-frontier chart
'             - Model-selected allocation
'             - KPI cards
'             - Regret-by-scenario comparison
'             - Methodology strip
'             - CFO conclusion
'
' REQUIRED INPUT SHEET
'   Curve_Scenarios
'
' EXPECTED CURVE_SCENARIOS STRUCTURE
'   Scenario block 1 begins at row 4.
'   Each block is 11 rows.
'
'       Row +0 : Scenario name
'       Row +1 : Headers
'       Row +2 : Today
'       Row +3 : 6M
'       Row +4 : 12M
'       Row +5 : 18M
'       Row +6 : 24M
'       Row +7 : 30M
'       Row +8 : 36M
'
'   Columns:
'       A Horizon
'       B Month
'       C ON
'       D 1M
'       E 3M
'       F 6M
'
' FIRST RUN
'   Run:
'       BuildCFOAllocationSlide
'
'   No allocation input is required before first run.
'
' OUTPUT SHEETS
'   Allocation_Inputs
'   Strategy_Returns
'   Allocation_Model
'   Allocation_ChartData
'   Allocation_Slide
'
' IMPORTANT RATE CONVENTION
'   Curve_Scenarios rates are percentage points:
'       4.00 means 4.00%
'
'===============================================================================

Private Const SCENARIO_COUNT As Long = 5
Private Const FIRST_SCENARIO_ROW As Long = 4
Private Const SCENARIO_BLOCK_HEIGHT As Long = 11
Private Const HORIZON_MONTHS As Long = 36

' Slide palette
Private Const CLR_BLUE As Long = 10441984
Private Const CLR_DARKBLUE As Long = 7945735
Private Const CLR_LIGHTBLUE As Long = 15790320
Private Const CLR_PALEBLUE As Long = 16448250
Private Const CLR_DARKTEXT As Long = 3158064
Private Const CLR_MIDTEXT As Long = 6842472
Private Const CLR_RED As Long = 192
Private Const CLR_GREEN As Long = 5287936
Private Const CLR_WHITE As Long = 16777215

Public Sub BuildCFOAllocationSlide()

    Dim wsCurves As Worksheet
    Dim wsInputs As Worksheet
    Dim wsReturns As Worksheet
    Dim wsModel As Worksheet
    Dim wsChartData As Worksheet
    Dim wsSlide As Worksheet

    Dim modelLastRow As Long
    Dim recommendedRow As Long
    Dim minRegretRow As Long
    Dim stage As String

    On Error GoTo CleanFail

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    Application.Calculation = xlCalculationManual

    stage = "Checking Curve_Scenarios"
    Set wsCurves = GetSheetIfExists("Curve_Scenarios")

    If wsCurves Is Nothing Then
        MsgBox "Required sheet 'Curve_Scenarios' was not found." & vbCrLf & _
               "Run the data-based curve scenario builder first.", vbExclamation
        GoTo CleanExit
    End If

    stage = "Creating output sheets"
    Set wsInputs = GetOrCreateSheet("Allocation_Inputs")
    Set wsReturns = GetOrCreateSheet("Strategy_Returns")
    Set wsModel = GetOrCreateSheet("Allocation_Model")
    Set wsChartData = GetOrCreateSheet("Allocation_ChartData")
    Set wsSlide = GetOrCreateSheet("Allocation_Slide")

    stage = "Setting default inputs"
    SetupAllocationInputs wsInputs

    stage = "Clearing previous outputs"
    wsReturns.Cells.Clear
    wsModel.Cells.Clear
    wsChartData.Cells.Clear
    ClearSlide wsSlide

    stage = "Calculating strategy returns"
    BuildStrategyReturns wsCurves, wsReturns

    stage = "Building portfolio universe"
    modelLastRow = BuildAllocationModel(wsInputs, wsReturns, wsModel)

    stage = "Building efficient frontier"
    MarkEfficientFrontier wsModel, modelLastRow

    stage = "Finding minimum regret"
    minRegretRow = FindMinimumRegretRow(wsModel, modelLastRow)

    stage = "Selecting robust allocation"
    recommendedRow = SelectRecommendedPortfolio(wsInputs, wsModel, modelLastRow, minRegretRow)

    If recommendedRow < 2 Then
        Err.Raise vbObjectError + 500, , "No valid recommended allocation was found."
    End If

    stage = "Preparing chart data"
    BuildChartData wsReturns, wsModel, wsChartData, modelLastRow, recommendedRow, minRegretRow

    stage = "Creating slide layout"
    BuildSelectedSlideLayout _
        wsInputs, wsReturns, wsModel, wsChartData, wsSlide, _
        recommendedRow, minRegretRow

    stage = "Formatting audit sheets"
    FormatInputs wsInputs
    FormatStrategyReturns wsReturns
    FormatModel wsModel
    FormatChartData wsChartData

    stage = "Finalizing slide"
    wsSlide.Activate

    On Error Resume Next
    ActiveWindow.DisplayGridlines = False
    ActiveWindow.Zoom = 80
    On Error GoTo CleanFail

CleanExit:
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True
    Exit Sub

CleanFail:
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox "CFO allocation slide builder error." & vbCrLf & vbCrLf & _
           "Stage: " & stage & vbCrLf & _
           "Error " & Err.Number & ": " & Err.Description, vbExclamation

End Sub

Private Sub BuildStrategyReturns(ByVal wsCurves As Worksheet, ByVal wsReturns As Worksheet)

    Dim i As Long
    Dim scenarioStartRow As Long
    Dim outRow As Long
    Dim scenarioName As String

    Dim rON As Double
    Dim r1M As Double
    Dim r3M As Double
    Dim r6M As Double
    Dim bestReturn As Double
    Dim bestStrategy As String

    wsReturns.Range("A1:G1").Merge
    wsReturns.Range("A1").Value = "Pure Strategy Returns by Scenario"

    wsReturns.Range("A3").Value = "Scenario"
    wsReturns.Range("B3").Value = "ON"
    wsReturns.Range("C3").Value = "1M"
    wsReturns.Range("D3").Value = "3M"
    wsReturns.Range("E3").Value = "6M"
    wsReturns.Range("F3").Value = "Best ex-post"
    wsReturns.Range("G3").Value = "Best pure strategy"

    For i = 0 To SCENARIO_COUNT - 1

        scenarioStartRow = FIRST_SCENARIO_ROW + i * SCENARIO_BLOCK_HEIGHT
        outRow = 4 + i

        scenarioName = Trim$(CStr(wsCurves.Cells(scenarioStartRow, 1).Value))

        If Len(scenarioName) = 0 Then
            Err.Raise vbObjectError + 501, , _
                "Missing scenario name in Curve_Scenarios row " & scenarioStartRow & "."
        End If

        ValidateScenarioBlock wsCurves, scenarioStartRow

        rON = AnnualizedStrategyReturn(wsCurves, scenarioStartRow, 3, 1)
        r1M = AnnualizedStrategyReturn(wsCurves, scenarioStartRow, 4, 1)
        r3M = AnnualizedStrategyReturn(wsCurves, scenarioStartRow, 5, 3)
        r6M = AnnualizedStrategyReturn(wsCurves, scenarioStartRow, 6, 6)

        bestReturn = rON
        bestStrategy = "ON"

        If r1M > bestReturn Then
            bestReturn = r1M
            bestStrategy = "1M"
        End If

        If r3M > bestReturn Then
            bestReturn = r3M
            bestStrategy = "3M"
        End If

        If r6M > bestReturn Then
            bestReturn = r6M
            bestStrategy = "6M"
        End If

        wsReturns.Cells(outRow, 1).Value = scenarioName
        wsReturns.Cells(outRow, 2).Value = rON
        wsReturns.Cells(outRow, 3).Value = r1M
        wsReturns.Cells(outRow, 4).Value = r3M
        wsReturns.Cells(outRow, 5).Value = r6M
        wsReturns.Cells(outRow, 6).Value = bestReturn
        wsReturns.Cells(outRow, 7).Value = bestStrategy

    Next i

End Sub

Private Sub ValidateScenarioBlock(ByVal ws As Worksheet, ByVal scenarioStartRow As Long)

    Dim dataStart As Long
    Dim r As Long
    Dim c As Long

    dataStart = scenarioStartRow + 2

    For r = dataStart To dataStart + 6

        If Not IsNumeric(ws.Cells(r, 2).Value) Then
            Err.Raise vbObjectError + 502, , _
                "Non-numeric month in Curve_Scenarios row " & r & "."
        End If

        For c = 3 To 6
            If Not IsNumeric(ws.Cells(r, c).Value) Then
                Err.Raise vbObjectError + 503, , _
                    "Non-numeric rate in Curve_Scenarios cell " & _
                    ws.Cells(r, c).Address(False, False) & "."
            End If
        Next c

    Next r

End Sub

Private Function AnnualizedStrategyReturn(ByVal wsCurves As Worksheet, ByVal scenarioStartRow As Long, ByVal tenorCol As Long, ByVal rolloverMonths As Long) As Double

    Dim m As Long
    Dim periodMonths As Long
    Dim ratePct As Double
    Dim wealth As Double
    Dim years As Double

    If rolloverMonths <= 0 Then rolloverMonths = 1

    wealth = 1#
    m = 0

    Do While m < HORIZON_MONTHS

        periodMonths = rolloverMonths

        If m + periodMonths > HORIZON_MONTHS Then
            periodMonths = HORIZON_MONTHS - m
        End If

        ratePct = RateAtMonth(wsCurves, scenarioStartRow, tenorCol, CDbl(m))

        wealth = wealth * (1# + (ratePct / 100#) * (periodMonths / 12#))

        m = m + periodMonths

    Loop

    years = HORIZON_MONTHS / 12#

    If wealth <= 0 Then
        AnnualizedStrategyReturn = -100
    Else
        AnnualizedStrategyReturn = (wealth ^ (1# / years) - 1#) * 100#
    End If

End Function

Private Function RateAtMonth(ByVal wsCurves As Worksheet, ByVal scenarioStartRow As Long, ByVal tenorCol As Long, ByVal targetMonth As Double) As Double

    Dim firstDataRow As Long
    Dim r As Long

    Dim m1 As Double
    Dim m2 As Double
    Dim y1 As Double
    Dim y2 As Double

    firstDataRow = scenarioStartRow + 2

    If targetMonth <= 0 Then
        RateAtMonth = CDbl(wsCurves.Cells(firstDataRow, tenorCol).Value)
        Exit Function
    End If

    If targetMonth >= 36 Then
        RateAtMonth = CDbl(wsCurves.Cells(firstDataRow + 6, tenorCol).Value)
        Exit Function
    End If

    For r = firstDataRow To firstDataRow + 5

        m1 = CDbl(wsCurves.Cells(r, 2).Value)
        m2 = CDbl(wsCurves.Cells(r + 1, 2).Value)

        If targetMonth >= m1 And targetMonth <= m2 Then

            y1 = CDbl(wsCurves.Cells(r, tenorCol).Value)
            y2 = CDbl(wsCurves.Cells(r + 1, tenorCol).Value)

            RateAtMonth = LinearInterpolate(targetMonth, m1, y1, m2, y2)
            Exit Function

        End If

    Next r

    Err.Raise vbObjectError + 504, , _
        "Could not interpolate scenario rate at month " & targetMonth & "."

End Function

Private Function BuildAllocationModel(ByVal wsInputs As Worksheet, ByVal wsReturns As Worksheet, ByVal wsModel As Worksheet) As Long

    Dim stepPct As Double
    Dim steps As Long

    Dim a As Long
    Dim b As Long
    Dim c As Long
    Dim d As Long

    Dim wON As Double
    Dim w1M As Double
    Dim w3M As Double
    Dim w6M As Double

    Dim rowOut As Long
    Dim idNum As Long

    Dim minON As Double
    Dim max6M As Double

    stepPct = CDbl(wsInputs.Range("B4").Value)

    If stepPct <= 0 Then stepPct = 5
    If stepPct > 25 Then stepPct = 25

    steps = CLng(100# / stepPct)

    If Abs(steps * stepPct - 100#) > 0.0001 Then
        Err.Raise vbObjectError + 505, , _
            "Allocation step must divide 100 exactly. Use 5, 10, 20 or 25."
    End If

    minON = CDbl(wsInputs.Range("B6").Value) / 100#
    max6M = CDbl(wsInputs.Range("B7").Value) / 100#

    WriteModelHeaders wsModel

    rowOut = 2
    idNum = 1

    For a = 0 To steps

        For b = 0 To steps - a

            For c = 0 To steps - a - b

                d = steps - a - b - c

                wON = a / steps
                w1M = b / steps
                w3M = c / steps
                w6M = d / steps

                WritePortfolioRow _
                    wsReturns, wsModel, rowOut, idNum, _
                    wON, w1M, w3M, w6M, _
                    minON, max6M

                rowOut = rowOut + 1
                idNum = idNum + 1

            Next c

        Next b

    Next a

    BuildAllocationModel = rowOut - 1

End Function

Private Sub WritePortfolioRow(ByVal wsReturns As Worksheet, ByVal wsModel As Worksheet, ByVal rowOut As Long, ByVal idNum As Long, ByVal wON As Double, ByVal w1M As Double, ByVal w3M As Double, ByVal w6M As Double, ByVal minON As Double, ByVal max6M As Double)

    Dim s As Long
    Dim retRow As Long

    Dim pureON As Double
    Dim pure1M As Double
    Dim pure3M As Double
    Dim pure6M As Double
    Dim bestReturn As Double

    Dim wealthON As Double
    Dim wealth1M As Double
    Dim wealth3M As Double
    Dim wealth6M As Double
    Dim portfolioWealth As Double
    Dim portfolioReturn As Double

    Dim regretBP As Double
    Dim excessBP As Double

    Dim sumExcessBP As Double
    Dim maxRegretBP As Double
    Dim worstExcessBP As Double
    Dim hitCount As Long
    Dim validFlag As Long

    wsModel.Cells(rowOut, 1).Value = idNum
    wsModel.Cells(rowOut, 2).Value = wON
    wsModel.Cells(rowOut, 3).Value = w1M
    wsModel.Cells(rowOut, 4).Value = w3M
    wsModel.Cells(rowOut, 5).Value = w6M

    maxRegretBP = -1
    worstExcessBP = 1E+30
    sumExcessBP = 0
    hitCount = 0

    For s = 1 To SCENARIO_COUNT

        retRow = 3 + s

        pureON = CDbl(wsReturns.Cells(retRow, 2).Value)
        pure1M = CDbl(wsReturns.Cells(retRow, 3).Value)
        pure3M = CDbl(wsReturns.Cells(retRow, 4).Value)
        pure6M = CDbl(wsReturns.Cells(retRow, 5).Value)
        bestReturn = CDbl(wsReturns.Cells(retRow, 6).Value)

        wealthON = (1# + pureON / 100#) ^ 3#
        wealth1M = (1# + pure1M / 100#) ^ 3#
        wealth3M = (1# + pure3M / 100#) ^ 3#
        wealth6M = (1# + pure6M / 100#) ^ 3#

        portfolioWealth = _
            wON * wealthON + _
            w1M * wealth1M + _
            w3M * wealth3M + _
            w6M * wealth6M

        portfolioReturn = (portfolioWealth ^ (1# / 3#) - 1#) * 100#

        regretBP = (bestReturn - portfolioReturn) * 100#
        excessBP = (portfolioReturn - pureON) * 100#

        wsModel.Cells(rowOut, 5 + s).Value = portfolioReturn
        wsModel.Cells(rowOut, 10 + s).Value = regretBP

        sumExcessBP = sumExcessBP + excessBP

        If regretBP > maxRegretBP Then maxRegretBP = regretBP
        If excessBP < worstExcessBP Then worstExcessBP = excessBP

        If excessBP > 0.0000001 Then
            hitCount = hitCount + 1
        End If

    Next s

    validFlag = 1

    If wON + 0.0000001 < minON Then validFlag = 0
    If w6M - 0.0000001 > max6M Then validFlag = 0

    wsModel.Cells(rowOut, 16).Value = sumExcessBP / SCENARIO_COUNT
    wsModel.Cells(rowOut, 17).Value = maxRegretBP
    wsModel.Cells(rowOut, 18).Value = worstExcessBP
    wsModel.Cells(rowOut, 19).Value = hitCount / SCENARIO_COUNT
    wsModel.Cells(rowOut, 20).Value = validFlag
    wsModel.Cells(rowOut, 21).Value = 0
    wsModel.Cells(rowOut, 22).Value = 0

End Sub

Private Sub MarkEfficientFrontier(ByVal wsModel As Worksheet, ByVal lastRow As Long)

    Dim r As Long
    Dim j As Long

    Dim riskR As Double
    Dim returnR As Double
    Dim riskJ As Double
    Dim returnJ As Double

    Dim dominated As Boolean

    For r = 2 To lastRow

        If CLng(wsModel.Cells(r, 20).Value) = 1 Then

            riskR = CDbl(wsModel.Cells(r, 17).Value)
            returnR = CDbl(wsModel.Cells(r, 16).Value)
            dominated = False

            For j = 2 To lastRow

                If j <> r Then

                    If CLng(wsModel.Cells(j, 20).Value) = 1 Then

                        riskJ = CDbl(wsModel.Cells(j, 17).Value)
                        returnJ = CDbl(wsModel.Cells(j, 16).Value)

                        If riskJ <= riskR + 0.0000001 _
                           And returnJ >= returnR - 0.0000001 Then

                            If riskJ < riskR - 0.0000001 _
                               Or returnJ > returnR + 0.0000001 Then

                                dominated = True
                                Exit For

                            End If

                        End If

                    End If

                End If

            Next j

            If dominated Then
                wsModel.Cells(r, 21).Value = 0
            Else
                wsModel.Cells(r, 21).Value = 1
            End If

        End If

    Next r

End Sub

Private Function FindMinimumRegretRow(ByVal wsModel As Worksheet, ByVal lastRow As Long) As Long

    Dim r As Long
    Dim bestRow As Long
    Dim bestRegret As Double
    Dim thisRegret As Double
    Dim thisReturn As Double
    Dim bestReturn As Double

    bestRegret = 1E+30
    bestReturn = -1E+30
    bestRow = 0

    For r = 2 To lastRow

        If CLng(wsModel.Cells(r, 20).Value) = 1 Then

            thisRegret = CDbl(wsModel.Cells(r, 17).Value)
            thisReturn = CDbl(wsModel.Cells(r, 16).Value)

            If thisRegret < bestRegret - 0.0000001 Then

                bestRegret = thisRegret
                bestReturn = thisReturn
                bestRow = r

            ElseIf Abs(thisRegret - bestRegret) < 0.0000001 Then

                If thisReturn > bestReturn Then
                    bestReturn = thisReturn
                    bestRow = r
                End If

            End If

        End If

    Next r

    FindMinimumRegretRow = bestRow

End Function

Private Function SelectRecommendedPortfolio(ByVal wsInputs As Worksheet, ByVal wsModel As Worksheet, ByVal lastRow As Long, ByVal minRegretRow As Long) As Long

    Dim r As Long

    Dim toleranceBP As Double
    Dim allowedRegret As Double

    Dim thisRegret As Double
    Dim thisReturn As Double

    Dim bestReturn As Double
    Dim bestRow As Long

    If minRegretRow < 2 Then
        SelectRecommendedPortfolio = 0
        Exit Function
    End If

    toleranceBP = CDbl(wsInputs.Range("B5").Value)
    If toleranceBP < 0 Then toleranceBP = 0

    allowedRegret = CDbl(wsModel.Cells(minRegretRow, 17).Value) + toleranceBP

    bestReturn = -1E+30
    bestRow = 0

    For r = 2 To lastRow

        If CLng(wsModel.Cells(r, 20).Value) = 1 Then

            thisRegret = CDbl(wsModel.Cells(r, 17).Value)
            thisReturn = CDbl(wsModel.Cells(r, 16).Value)

            If thisRegret <= allowedRegret + 0.0000001 Then

                If thisReturn > bestReturn + 0.0000001 Then

                    bestReturn = thisReturn
                    bestRow = r

                ElseIf Abs(thisReturn - bestReturn) < 0.0000001 Then

                    If bestRow > 0 Then
                        If thisRegret < CDbl(wsModel.Cells(bestRow, 17).Value) Then
                            bestRow = r
                        End If
                    End If

                End If

            End If

        End If

    Next r

    If bestRow > 0 Then
        wsModel.Cells(bestRow, 22).Value = 1
    End If

    SelectRecommendedPortfolio = bestRow

End Function

Private Sub BuildChartData(ByVal wsReturns As Worksheet, ByVal wsModel As Worksheet, ByVal wsData As Worksheet, ByVal modelLastRow As Long, ByVal recommendedRow As Long, ByVal minRegretRow As Long)

    Dim recRet As Double

    wsData.Range("A2").Value = "Max regret bp"
    wsData.Range("B2").Value = "Avg excess bp"

    For r = 2 To modelLastRow

        If CLng(wsModel.Cells(r, 20).Value) = 1 Then

            wsData.Cells(outRow, 1).Value = wsModel.Cells(r, 17).Value
            wsData.Cells(outRow, 2).Value = wsModel.Cells(r, 16).Value

    wsData.Range("D2").Value = "Max regret bp"
    wsData.Range("E2").Value = "Avg excess bp"

            wsData.Cells(outRow, 5).Value = wsModel.Cells(r, 16).Value

    wsData.Range("K2").Value = "Avg excess bp"
    wsData.Range("J3").Value = wsModel.Cells(minRegretRow, 17).Value
    wsData.Range("K3").Value = wsModel.Cells(minRegretRow, 16).Value

    wsData.Range("P2").Value = "3M"
    wsData.Range("Q2").Value = "6M"

    wsData.Range("N3").Value = wsModel.Cells(recommendedRow, 2).Value
    wsData.Range("O3").Value = wsModel.Cells(recommendedRow, 3).Value
    wsData.Range("P3").Value = wsModel.Cells(recommendedRow, 4).Value
    wsData.Range("Q3").Value = wsModel.Cells(recommendedRow, 5).Value

    wsData.Range("M6").Value = "Scenario"
    wsData.Range("N6").Value = "Selected"
    wsData.Range("O6").Value = "100% ON"
    wsData.Range("P6").Value = "100% 6M"

    For s = 1 To SCENARIO_COUNT

        retRow = 3 + s
        scenarioOutRow = 6 + s

        bestRet = CDbl(wsReturns.Cells(retRow, 6).Value)
        recRet = CDbl(wsModel.Cells(recommendedRow, 5 + s).Value)
        onRet = CDbl(wsReturns.Cells(retRow, 2).Value)
        m6Ret = CDbl(wsReturns.Cells(retRow, 5).Value)

        wsData.Cells(scenarioOutRow, 13).Value = wsReturns.Cells(retRow, 1).Value
        wsData.Cells(scenarioOutRow, 14).Value = (bestRet - recRet) * 100#
        wsData.Cells(scenarioOutRow, 15).Value = (bestRet - onRet) * 100#
        wsData.Cells(scenarioOutRow, 16).Value = (bestRet - m6Ret) * 100#

    Next s

End Sub

Private Sub AddXYSeries(ByVal ch As Chart, ByVal ws As Worksheet, ByVal seriesName As String, ByVal xRange As Range, ByVal yRange As Range, ByVal seriesColor As Long, ByVal markerStyle As XlMarkerStyle, ByVal markerSize As Long, ByVal connectLine As Boolean)

    Dim s As Series

    Set s = ch.SeriesCollection.NewSeries

    s.Name = seriesName
    s.XValues = xRange
    s.Values = yRange

    If connectLine Then
        s.ChartType = xlXYScatterLines
    Else
        s.ChartType = xlXYScatter
    End If

    On Error Resume Next

    s.MarkerStyle = markerStyle
    s.MarkerSize = markerSize
    s.MarkerForegroundColor = seriesColor
    s.MarkerBackgroundColor = seriesColor

    If connectLine Then
        s.Format.Line.Visible = msoTrue
        s.Format.Line.ForeColor.RGB = seriesColor
        s.Format.Line.Weight = 1.8
    Else
        s.Format.Line.Visible = msoFalse
    End If

    On Error GoTo 0

End Sub

Private Sub AddTextBox(ByVal ws As Worksheet, ByVal leftPos As Double, ByVal topPos As Double, ByVal boxWidth As Double, ByVal boxHeight As Double, ByVal textValue As String, ByVal fontSize As Double, ByVal isBold As Boolean, ByVal fontColor As Long, ByVal hasFill As Boolean, ByVal paragraphAlignment As Long)

    Dim shp As Shape

    Set shp = ws.Shapes.AddTextbox( _
                    Orientation:=msoTextOrientationHorizontal, _
                    Left:=leftPos, _
                    Top:=topPos, _
                    Width:=boxWidth, _
                    Height:=boxHeight)

    shp.Line.Visible = msoFalse

    If hasFill Then
        shp.Fill.Visible = msoTrue
        shp.Fill.ForeColor.RGB = RGB(255, 255, 255)
    Else
        shp.Fill.Visible = msoFalse
    End If

    With shp.TextFrame2

        .MarginLeft = 1
        .MarginRight = 1
        .MarginTop = 1
        .MarginBottom = 1
        .WordWrap = msoTrue

        .TextRange.Text = textValue
        .TextRange.Font.Name = "Arial"
        .TextRange.Font.Size = fontSize
        .TextRange.Font.Fill.ForeColor.RGB = fontColor

        If isBold Then
            .TextRange.Font.Bold = msoTrue
        Else
            .TextRange.Font.Bold = msoFalse
        End If

        .TextRange.ParagraphFormat.Alignment = paragraphAlignment

    End With

End Sub

Private Function FormatPercent0(ByVal v As Double) As String

    FormatPercent0 = Format(v, "0%")

End Function

Private Function FormatBP(ByVal v As Double) As String

    FormatBP = Format(v, "0.0") & " bp"

End Function

Private Function FormatSignedBP(ByVal v As Double) As String

    If v > 0.0000001 Then
        FormatSignedBP = "+" & Format(v, "0.0") & " bp"
    ElseIf v < -0.0000001 Then
        FormatSignedBP = Format(v, "0.0") & " bp"
    Else
        FormatSignedBP = "0.0 bp"
    End If

End Function

Private Function LinearInterpolate(ByVal x As Double, ByVal x1 As Double, ByVal y1 As Double, ByVal x2 As Double, ByVal y2 As Double) As Double

    If x2 = x1 Then
        LinearInterpolate = y2
    Else
        LinearInterpolate = y1 + (y2 - y1) * (x - x1) / (x2 - x1)
    End If

End Function

Private Function GetOrCreateSheet(ByVal sheetName As String) As Worksheet

    On Error Resume Next
    Set GetOrCreateSheet = ThisWorkbook.Worksheets(sheetName)
    On Error GoTo 0

    If GetOrCreateSheet Is Nothing Then

        Set GetOrCreateSheet = _
            ThisWorkbook.Worksheets.Add( _
                After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))

        GetOrCreateSheet.Name = sheetName

    End If

End Function
