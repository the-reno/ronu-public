'
' RUN
'   Build_CFO_Cash_Allocation
'
' REQUIRED SOURCE SHEET
'   curve
'   A:E = Date | ON | 1M | 3M | 6M
'
' OUTPUT
'   CFO_1_Decision
'   CFO_2_Context
'   CFO_3_Evidence
'   CFO_4_Portfolio
'   CFO_5_Partnership
'   CFO_Data_Hist   (hidden)
'   CFO_Data_Scen   (hidden)
'
' IMPORTANT
'   This version does NOT depend on any PPT_* sheets.

Private Const NAVY As Long = 5067283
Private Const BLUE As Long = 10185519
Private Const TEAL As Long = 8682558
Private Const TXT As Long = 4006438
Private Const MUTED As Long = 8817775
Private Const LIGHT As Long = 16316664
Private Const BORDER As Long = 15191257
Private Const WHITE As Long = 16777215

Private gStage As String

Public Sub Build_CFO_Cash_Allocation()
    Dim wb As Workbook
    Dim oldCalc As XlCalculation
    Set wb = ThisWorkbook

    If Not HasSheet(wb, "curve") Then
        MsgBox "Missing required sheet: curve" & vbCrLf & _
               "Expected columns A:E = Date | ON | 1M | 3M | 6M", vbCritical
        Exit Sub
    End If

    If Not ValidateCurve(wb.Worksheets("curve")) Then Exit Sub

    On Error GoTo Fail

    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.EnableEvents = False
    oldCalc = Application.Calculation
    Application.Calculation = xlCalculationManual

    gStage = "Delete prior output sheets"
    DeleteOutputs wb

    gStage = "Build historical six-month evidence"
    BuildHistoricalData wb

    gStage = "Build forward scenario engine"
    BuildScenarioData wb

    gStage = "Build page 1"
    Page1 wb

    gStage = "Build page 2"
    Page2 wb

    gStage = "Build page 3"
    Page3 wb

    gStage = "Build page 4"
    Page4 wb

    gStage = "Build page 5"
    Page5 wb

    gStage = "Final formatting"
    wb.Worksheets("CFO_Data_Hist").Visible = xlSheetHidden
    wb.Worksheets("CFO_Data_Scen").Visible = xlSheetHidden
    wb.Worksheets("CFO_1_Decision").Activate
    ActiveWindow.DisplayGridlines = False
    ActiveWindow.Zoom = 85

Done:
    On Error Resume Next
    Application.Calculation = oldCalc
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    Application.StatusBar = False
    On Error GoTo 0
    Exit Sub

Fail:
    MsgBox "Build failed." & vbCrLf & vbCrLf & _
           "Stage: " & gStage & vbCrLf & _
           "Error: " & Err.Number & " - " & Err.Description, vbCritical
    Resume Done
End Sub

'====================================================================================
' HISTORICAL ENGINE
'====================================================================================
Private Sub BuildHistoricalData(ByVal wb As Workbook)
    Dim src As Worksheet, out As Worksheet
    Dim lastRow As Long, i As Long, j As Long, k As Long, outRow As Long
    Dim d0 As Date, targetDate As Date, dm As Date
    Dim rON As Double, r6 As Double, rolled As Double, locked As Double
    Dim dtDays As Long, daysHeld As Long

    Set src = wb.Worksheets("curve")
    Set out = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
    out.Name = "CFO_Data_Hist"

    PutRow out.Range("A1"), Array("Decision date", "Maturity date", "ON", "6M", _
        "6M-ON spread bp", "Rolled ON return", "Locked 6M return", _
        "6M excess bp", "Winner", "ON at maturity", "ON change bp")

    lastRow = src.Cells(src.Rows.Count, 1).End(xlUp).Row
    outRow = 2

    For i = 2 To lastRow
        If CellHasDate(src.Cells(i, 1)) Then
            d0 = CDate(src.Cells(i, 1).Value)
            targetDate = DateAdd("m", 6, d0)

            If CellHasDate(src.Cells(lastRow, 1)) Then
                If targetDate <= CDate(src.Cells(lastRow, 1).Value) Then
                    j = FindFloorDateRow(src, targetDate, i, lastRow)
                Else
                    j = 0
                End If
            End If

            If j > i Then
                dm = CDate(src.Cells(j, 1).Value)
                rON = SafeRate(src.Cells(i, 2).Value)
                r6 = SafeRate(src.Cells(i, 5).Value)
                rolled = 1#

                For k = i To j - 1
                    If CellHasDate(src.Cells(k, 1)) And CellHasDate(src.Cells(k + 1, 1)) Then
                        dtDays = DateDiff("d", CDate(src.Cells(k, 1).Value), CDate(src.Cells(k + 1, 1).Value))
                        If dtDays > 0 Then
                            rolled = rolled * (1# + SafeRate(src.Cells(k, 2).Value) * dtDays / 360#)
                        End If
                    End If
                Next k

                rolled = rolled - 1#
                daysHeld = DateDiff("d", d0, dm)
                locked = r6 * daysHeld / 360#

                out.Cells(outRow, 1).Value = d0
                out.Cells(outRow, 2).Value = dm
                out.Cells(outRow, 3).Value = rON
                out.Cells(outRow, 4).Value = r6
                out.Cells(outRow, 5).Value = (r6 - rON) * 10000#
                out.Cells(outRow, 6).Value = rolled
                out.Cells(outRow, 7).Value = locked
                out.Cells(outRow, 8).Value = (locked - rolled) * 10000#
                If locked > rolled Then out.Cells(outRow, 9).Value = "6M" Else out.Cells(outRow, 9).Value = "ON"
                out.Cells(outRow, 10).Value = SafeRate(src.Cells(j, 2).Value)
                out.Cells(outRow, 11).Value = (SafeRate(src.Cells(j, 2).Value) - rON) * 10000#
                outRow = outRow + 1
            End If
        End If
    Next i

    If outRow <= 2 Then Err.Raise vbObjectError + 101, , "No complete six-month historical decisions could be built."

    out.Columns("A:B").NumberFormat = "dd-mmm-yyyy"
    out.Columns("C:D,F:G,J").NumberFormat = "0.000%"
    out.Columns("E,H,K").NumberFormat = "0.0"
End Sub

'====================================================================================
' SCENARIO ENGINE - WORKSHEET BACKED; NO MULTIDIMENSIONAL ARRAY PASSING
'====================================================================================
Private Sub BuildScenarioData(ByVal wb As Workbook)
    Dim ws As Worksheet, src As Worksheet
    Dim lr As Long, s As Long, m As Long, r As Long
    Dim baseON As Double, p1 As Double, p3 As Double, p6 As Double
    Dim scenName As Variant, terminal As Variant, hump As Variant
    Dim rowOut As Long, wON As Long, w1 As Long, w3 As Long, w6 As Long
    Dim ret As Double, bestRet As Double, regret As Double
    Dim worstReg As Double, avgReg As Double
    Dim endGrid As Long, bestRow As Long

    Set src = wb.Worksheets("curve")
    lr = src.Cells(src.Rows.Count, 1).End(xlUp).Row

    baseON = SafeRate(src.Cells(lr, 2).Value)
    p1 = SafeRate(src.Cells(lr, 3).Value) - baseON
    p3 = SafeRate(src.Cells(lr, 4).Value) - baseON
    p6 = SafeRate(src.Cells(lr, 5).Value) - baseON

    Set ws = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
    ws.Name = "CFO_Data_Scen"

    scenName = Array("Gradual normalization", "Higher for longer", "Renewed inflation", _
                     "Faster normalization", "Recession / rapid cuts")
    terminal = Array(0.0315, 0.0375, 0.034, 0.029, 0.025)
    hump = Array(0.002, 0.007, 0.012, -0.005, -0.02)

    ' Scenario monthly paths in A:AL
    ws.Cells(1, 1).Value = "Scenario"
    For m = 1 To 36
        ws.Cells(1, m + 1).Value = "M" & m
    Next m
    ws.Cells(1, 38).Value = "Economic state"

    For s = 1 To 5
        ws.Cells(s + 1, 1).Value = CStr(scenName(s - 1))
        For m = 1 To 36
            ws.Cells(s + 1, m + 1).Value = ScenarioRate(baseON, CDbl(terminal(s - 1)), CDbl(hump(s - 1)), s, m)
        Next m
        ws.Cells(s + 1, 38).Value = ScenarioDescription(s)
    Next s
    ws.Range("B2:AK6").NumberFormat = "0.00%"

    ' Sleeve returns at A9:G14
    PutRow ws.Range("A9"), Array("Scenario", "ON", "1M", "3M", "6M", "Best tenor", "Best return")

    For s = 1 To 5
        ws.Cells(s + 9, 1).Value = CStr(scenName(s - 1))
        ws.Cells(s + 9, 2).Value = SleeveReturnFromSheet(ws, s + 1, 0#, 1)
        ws.Cells(s + 9, 3).Value = SleeveReturnFromSheet(ws, s + 1, p1, 1)
        ws.Cells(s + 9, 4).Value = SleeveReturnFromSheet(ws, s + 1, p3, 3)
        ws.Cells(s + 9, 5).Value = SleeveReturnFromSheet(ws, s + 1, p6, 6)
        ws.Cells(s + 9, 6).Value = BestTenorFromCells(ws, s + 9)
        ws.Cells(s + 9, 7).Value = Max4( _
            CDbl(ws.Cells(s + 9, 2).Value), _
            CDbl(ws.Cells(s + 9, 3).Value), _
            CDbl(ws.Cells(s + 9, 4).Value), _
            CDbl(ws.Cells(s + 9, 5).Value))
    Next s
    ws.Range("B10:E14,G10:G14").NumberFormat = "0.00%"

    ' Allocation grid at J:P
    PutRow ws.Range("J1"), Array("ON", "1M", "3M", "6M", "Weighted months", "Worst regret bp", "Average regret bp")
    rowOut = 2

    For wON = 0 To 10
        For w1 = 0 To 10 - wON
            For w3 = 0 To 10 - wON - w1
                w6 = 10 - wON - w1 - w3
                worstReg = 0#
                avgReg = 0#

                For s = 1 To 5
                    ret = (wON * CDbl(ws.Cells(s + 9, 2).Value) + _
                           w1 * CDbl(ws.Cells(s + 9, 3).Value) + _
                           w3 * CDbl(ws.Cells(s + 9, 4).Value) + _
                           w6 * CDbl(ws.Cells(s + 9, 5).Value)) / 10#

                    bestRet = CDbl(ws.Cells(s + 9, 7).Value)
                    regret = (bestRet - ret) * 10000#
                    If regret > worstReg Then worstReg = regret
                    avgReg = avgReg + regret / 5#
                Next s

                ws.Cells(rowOut, 10).Value = wON / 10#
                ws.Cells(rowOut, 11).Value = w1 / 10#
                ws.Cells(rowOut, 12).Value = w3 / 10#
                ws.Cells(rowOut, 13).Value = w6 / 10#
                ws.Cells(rowOut, 14).Value = (w1 + 3# * w3 + 6# * w6) / 10#
                ws.Cells(rowOut, 15).Value = worstReg
                ws.Cells(rowOut, 16).Value = avgReg
                rowOut = rowOut + 1
            Next w3
        Next w1
    Next wON

    endGrid = rowOut - 1
    bestRow = 2
    For r = 3 To endGrid
        If CDbl(ws.Cells(r, 15).Value) < CDbl(ws.Cells(bestRow, 15).Value) Then bestRow = r
    Next r

    PutRow ws.Range("J" & endGrid + 3), Array("REFERENCE", "ON", "1M", "3M", "6M", "Weighted months", "Worst regret bp", "Average regret bp")
    ws.Cells(endGrid + 4, 10).Value = "Selected"
    ws.Cells(endGrid + 4, 11).Value = ws.Cells(bestRow, 10).Value
    ws.Cells(endGrid + 4, 12).Value = ws.Cells(bestRow, 11).Value
    ws.Cells(endGrid + 4, 13).Value = ws.Cells(bestRow, 12).Value
    ws.Cells(endGrid + 4, 14).Value = ws.Cells(bestRow, 13).Value
    ws.Cells(endGrid + 4, 15).Value = ws.Cells(bestRow, 14).Value
    ws.Cells(endGrid + 4, 16).Value = ws.Cells(bestRow, 15).Value
    ws.Cells(endGrid + 4, 17).Value = ws.Cells(bestRow, 16).Value

    ws.Range("J2:M" & endGrid).NumberFormat = "0%"
End Sub

Private Function SleeveReturnFromSheet(ByVal ws As Worksheet, ByVal pathRow As Long, _
                                       ByVal premium As Double, ByVal resetEvery As Long) As Double
    Dim wealth As Double, rr As Double
    Dim m As Long

    wealth = 1#
    rr = 0#

    For m = 1 To 36
        If m = 1 Or ((m - 1) Mod resetEvery = 0) Then
            rr = CDbl(ws.Cells(pathRow, m + 1).Value) + premium
        End If
        wealth = wealth * (1# + rr / 12#)
    Next m

    SleeveReturnFromSheet = wealth ^ (1# / 3#) - 1#
End Function

Private Function BestTenorFromCells(ByVal ws As Worksheet, ByVal rowNum As Long) As String
    Dim bestCol As Long, c As Long
    Dim bestVal As Double, x As Double

    bestCol = 2
    bestVal = CDbl(ws.Cells(rowNum, 2).Value)

    For c = 3 To 5
        x = CDbl(ws.Cells(rowNum, c).Value)
        If x > bestVal Then
            bestVal = x
            bestCol = c
        End If
    Next c

    Select Case bestCol
        Case 2: BestTenorFromCells = "ON"
        Case 3: BestTenorFromCells = "1M"
        Case 4: BestTenorFromCells = "3M"
        Case 5: BestTenorFromCells = "6M"
    End Select
End Function

Private Function Max4(ByVal a As Double, ByVal b As Double, ByVal c As Double, ByVal d As Double) As Double
    Dim x As Double
    x = a
    If b > x Then x = b
    If c > x Then x = c
    If d > x Then x = d
    Max4 = x
End Function

Private Function ScenarioRate(ByVal startRate As Double, ByVal endRate As Double, _
                              ByVal hump As Double, ByVal scen As Long, ByVal m As Long) As Double
    Dim t As Double, x As Double
    t = (m - 1) / 35#

    Select Case scen
        Case 1, 3
            x = startRate + hump * Sin(3.14159265358979# * t) + (endRate - startRate) * t
        Case 2
            If t < 0.6 Then
                x = startRate + hump * (t / 0.6)
            Else
                x = startRate + hump + (endRate - startRate - hump) * ((t - 0.6) / 0.4)
            End If
        Case 4
            x = startRate + hump * (1# - Exp(-5# * t)) + (endRate - startRate - hump) * t
        Case 5
            x = startRate + hump * (1# - Exp(-8# * t)) + (endRate - startRate - hump) * t
    End Select

    If x < 0.0001 Then x = 0.0001
    ScenarioRate = x
End Function

Private Function ScenarioDescription(ByVal scen As Long) As String
    Select Case scen
        Case 1: ScenarioDescription = "Mild near-term firmness, then measured easing."
        Case 2: ScenarioDescription = "Sticky inflation keeps policy restrictive."
        Case 3: ScenarioDescription = "Reacceleration triggers additional tightening before normalization."
        Case 4: ScenarioDescription = "Benign disinflation allows faster normalization."
        Case 5: ScenarioDescription = "Growth shock forces rapid cuts, followed by recovery."
    End Select
End Function

    wr = CDbl(sc.Cells(refRow, 16).Value)

    minWorst = CDbl(sc.Cells(2, 15).Value)
    For r = 3 To gridEnd
        If CDbl(sc.Cells(r, 15).Value) < minWorst Then minWorst = CDbl(sc.Cells(r, 15).Value)
    Next r

    minTen = 1E+99
    maxTen = -1E+99
    nearN = 0

    For r = 2 To gridEnd
        If CDbl(sc.Cells(r, 15).Value) <= minWorst + 0.25 Then
            nearN = nearN + 1
            If CDbl(sc.Cells(r, 14).Value) < minTen Then minTen = CDbl(sc.Cells(r, 14).Value)
            If CDbl(sc.Cells(r, 14).Value) > maxTen Then maxTen = CDbl(sc.Cells(r, 14).Value)
        End If
    Next r

    Head ws, "When direction is uncertain, focus on the cost of being wrong - not on one forecast", _
        "Alternative rate paths expose the trade-off and help identify a robust allocation region rather than a single 'correct' tenor."

    Box ws, "B6:F10", "IF RATES RISE", _
        "Shorter resets retain more ability to participate in higher market rates." & vbLf & vbLf & _

    Box ws, "B11:F15", "IF RATES FALL", _
        "Longer locking protects against reinvestment at lower rates." & vbLf & vbLf & _
        "Economic risk: reinvestment opportunity cost.", RGB(234, 243, 243)

    ws.Range("B17:H22").Value = sc.Range("A9:G14").Value
    Tbl ws.Range("B17:H22")
    ws.Range("C18:F22").NumberFormat = "0.00%"
    ws.Range("H18:H22").NumberFormat = "0.00%"

    Set co = ws.ChartObjects.Add(ws.Range("J6").Left, ws.Range("J6").Top, 455, 275)
    With co.Chart
        .ChartType = xlXYScatter
        .SeriesCollection.NewSeries
        .SeriesCollection(1).XValues = sc.Range("N2:N" & gridEnd)
        .SeriesCollection(1).Values = sc.Range("O2:O" & gridEnd)
        .SeriesCollection(1).MarkerSize = 4
        .SeriesCollection(1).MarkerForegroundColor = BLUE
        .HasTitle = True
        .ChartTitle.Text = "Robust allocation region"
        .Axes(xlCategory).HasTitle = True
        .Axes(xlValue).HasTitle = True
        .Axes(xlCategory).AxisTitle.Text = "Weighted tenor (months)"
        .Axes(xlValue).AxisTitle.Text = "Worst regret (bp/year)"
        .HasLegend = False
        .ChartArea.Format.Line.Visible = msoFalse
    End With

    Kpi ws, "J17:L20", "Reference mix", MixLabel(wON, w1, w3, w6), "Illustrative minimax-regret anchor."
    Kpi ws, "M17:O20", "Weighted tenor", Format(wt, "0.0") & " months", "Position on the curve."
    Kpi ws, "P17:R20", "Worst regret", Format(wr, "0.0") & " bp/yr", "Maximum model-state shortfall."

    TextBox ws, "J22:R24", _
        "The model identifies a risk-balanced region rather than one precise answer. " & nearN & _
        " portfolios sit within 0.25 bp/year of the minimum, spanning approximately " & _
        Format(minTen, "0.0") & " to " & Format(maxTen, "0.0") & " months weighted tenor.", _
        10, False, TXT, LIGHT, xlLeft

    Bottom ws, "The objective is to reduce dependence on one rate view while preserving flexibility to adapt as conditions change.", _
        "Source: curve tab. Forward paths are transparent sensitivities, not forecasts or return guarantees."
End Sub

Private Sub PutRow(ByVal firstCell As Range, ByVal values As Variant)
    Dim i As Long
    For i = LBound(values) To UBound(values)
        firstCell.Offset(0, i - LBound(values)).Value = values(i)

Private Function SafeRate(ByVal x As Variant) As Double
    If IsError(x) Or IsEmpty(x) Or Len(Trim$(CStr(x))) = 0 Then
        SafeRate = 0#
    ElseIf IsNumeric(x) Then
        SafeRate = CDbl(x)
        If Abs(SafeRate) > 1# Then SafeRate = SafeRate / 100#
    Else
        Err.Raise vbObjectError + 301, , "Non-numeric rate value encountered: " & CStr(x)
    End If
End Function

Private Function CellHasDate(ByVal c As Range) As Boolean
    On Error GoTo Nope
    If IsError(c.Value) Or IsEmpty(c.Value) Then GoTo Nope
    CellHasDate = IsDate(c.Value)
    Exit Function
Nope:
    CellHasDate = False
End Function

Private Function FormatSignedBp(ByVal x As Double) As String
    If x > 0 Then
        FormatSignedBp = "+" & Format(x, "0.0") & " bp"
    Else
        FormatSignedBp = Format(x, "0.0") & " bp"
    End If
End Function

Private Function MixLabel(ByVal wON As Double, ByVal w1 As Double, ByVal w3 As Double, ByVal w6 As Double) As String
    Dim s As String
    s = ""
    If wON > 0 Then s = s & Format(wON, "0%") & " ON / "
    If w1 > 0 Then s = s & Format(w1, "0%") & " 1M / "
    If w3 > 0 Then s = s & Format(w3, "0%") & " 3M / "
    If w6 > 0 Then s = s & Format(w6, "0%") & " 6M / "
    If Len(s) >= 3 Then If Right$(s, 3) = " / " Then s = Left$(s, Len(s) - 3)
    MixLabel = s
End Function

Private Function FindFloorDateRow(ByVal ws As Worksheet, ByVal targetDate As Date, _
                                  ByVal startRow As Long, ByVal lastRow As Long) As Long
    Dim lo As Long, hi As Long, mid As Long, ans As Long
    lo = startRow
    hi = lastRow
    ans = 0

    Do While lo <= hi
        mid = (lo + hi) \ 2
        If Not CellHasDate(ws.Cells(mid, 1)) Then
            Err.Raise vbObjectError + 302, , "Invalid date in curve row " & mid
        End If

        If CDate(ws.Cells(mid, 1).Value) <= targetDate Then
            ans = mid
            lo = mid + 1
        Else
            hi = mid - 1
        End If
    Loop

    FindFloorDateRow = ans
End Function

Private Function AverageIfGE(ByVal criteriaRng As Range, ByVal avgRng As Range, ByVal threshold As Double) As Double
    Dim i As Long, n As Long
    Dim total As Double

    For i = 1 To criteriaRng.Rows.Count
        If IsNumeric(criteriaRng.Cells(i, 1).Value) And IsNumeric(avgRng.Cells(i, 1).Value) Then
            If CDbl(criteriaRng.Cells(i, 1).Value) >= threshold Then
                total = total + CDbl(avgRng.Cells(i, 1).Value)
                n = n + 1
            End If
        End If
    Next i

    If n > 0 Then AverageIfGE = total / n Else AverageIfGE = 0#
End Function

Private Function PositiveHitIfGE(ByVal criteriaRng As Range, ByVal valueRng As Range, ByVal threshold As Double) As Double
    Dim i As Long, n As Long, positive As Long

    For i = 1 To criteriaRng.Rows.Count
        If IsNumeric(criteriaRng.Cells(i, 1).Value) And IsNumeric(valueRng.Cells(i, 1).Value) Then
            If CDbl(criteriaRng.Cells(i, 1).Value) >= threshold Then
                n = n + 1
                If CDbl(valueRng.Cells(i, 1).Value) > 0 Then positive = positive + 1
            End If
        End If
    Next i

    If n > 0 Then PositiveHitIfGE = positive / n Else PositiveHitIfGE = 0#
End Function

Private Function ValidateCurve(ByVal ws As Worksheet) As Boolean
    Dim lr As Long, i As Long
    Dim prevDate As Date, thisDate As Date

    lr = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    If lr < 10 Then
        MsgBox "The curve sheet does not contain enough observations.", vbCritical
        Exit Function
    End If

    If Not CellHasDate(ws.Cells(2, 1)) Then
        MsgBox "curve!A2 must contain a valid date.", vbCritical
        Exit Function
    End If

    For i = 2 To 5
        If Not IsNumeric(ws.Cells(2, i).Value) Then
            MsgBox "Expected numeric rates in curve columns B:E." & vbCrLf & _
                   "Layout must be Date | ON | 1M | 3M | 6M", vbCritical
            Exit Function
        End If
    Next i

    prevDate = CDate(ws.Cells(2, 1).Value)
    For i = 3 To lr
        If Not CellHasDate(ws.Cells(i, 1)) Then
            MsgBox "Invalid date in curve row " & i & ".", vbCritical
            Exit Function
        End If
        thisDate = CDate(ws.Cells(i, 1).Value)
        If thisDate < prevDate Then
            MsgBox "The curve dates must be sorted ascending. Problem at row " & i & ".", vbCritical
            Exit Function
        End If
        prevDate = thisDate
    Next i

    ValidateCurve = True
End Function
