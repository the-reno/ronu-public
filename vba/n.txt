'====================================================================================
' USD CASH ALLOCATION | CFO EXCEL PRESENTATION | CURVE-ONLY VERSION
'
' RUN
'   Build_CFO_Cash_Allocation
'
' REQUIRED SOURCE SHEET
'   curve
'   Columns A:E = Date | ON | 1M | 3M | 6M
'
' THIS VERSION HAS NO DEPENDENCY ON PPT_* SHEETS.
' It rebuilds the historical analysis, charts, scenarios and allocation grid
' directly from the curve tab.

Private Const NAVY As Long = 5067283
Private Const BLUE As Long = 10185519
Private Const TEAL As Long = 8682558
Private Const TXT As Long = 4006438
Private Const MUTED As Long = 8817775
Private Const LIGHT As Long = 16316664
Private Const BORDER As Long = 15191257
Private Const WHITE As Long = 16777215

Public Sub Build_CFO_Cash_Allocation()
    Dim wb As Workbook
    Dim oldCalc As XlCalculation
    Set wb = ThisWorkbook

    If Not HasSheet(wb, "curve") Then
        MsgBox "Missing required sheet: curve" & vbCrLf & _
               "Expected A:E = Date | ON | 1M | 3M | 6M", vbCritical
        Exit Sub
    End If
    If Not ValidateCurve(wb.Worksheets("curve")) Then Exit Sub

    On Error GoTo Fail
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.EnableEvents = False
    oldCalc = Application.Calculation
    Application.Calculation = xlCalculationManual

    DeleteOutputs wb
    BuildHistoricalData wb
    BuildScenarioData wb
    Page1 wb
    Page2 wb
    Page3 wb
    Page4 wb
    Page5 wb

    wb.Worksheets("CFO_Data_Hist").Visible = xlSheetHidden
    wb.Worksheets("CFO_Data_Scen").Visible = xlSheetHidden
    wb.Worksheets("CFO_1_Decision").Activate
    ActiveWindow.DisplayGridlines = False
    ActiveWindow.Zoom = 85

Done:
    Application.Calculation = oldCalc
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    Application.StatusBar = False
    Exit Sub
Fail:
    MsgBox "Build failed: " & Err.Number & " - " & Err.Description, vbCritical
    Resume Done
End Sub

Private Sub BuildHistoricalData(ByVal wb As Workbook)
    Dim src As Worksheet, out As Worksheet
    Dim lastRow As Long, i As Long, j As Long, k As Long, outRow As Long
    Dim d0 As Date, targetDate As Date, dm As Date
    Dim rON As Double, r6 As Double, rolled As Double, locked As Double
    Dim days As Long, dtDays As Long

    Set src = wb.Worksheets("curve")
    Set out = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
    out.Name = "CFO_Data_Hist"

    PutRow out.Range("A1"), Array("Decision date", "Maturity date", "ON", "6M", "6M-ON spread bp", _
        "Rolled ON return", "Locked 6M return", "6M excess bp", "Winner", "ON at maturity", "ON change bp")

    lastRow = src.Cells(src.Rows.Count, 1).End(xlUp).Row
    outRow = 2

    For i = 2 To lastRow
        If IsDate(src.Cells(i, 1).Value) Then
            d0 = CDate(src.Cells(i, 1).Value)
            targetDate = DateAdd("m", 6, d0)
            If targetDate <= CDate(src.Cells(lastRow, 1).Value) Then
                j = FindFloorDateRow(src, targetDate, i, lastRow)
            Else
                j = 0
            End If

            If j > i Then
                dm = CDate(src.Cells(j, 1).Value)
                rON = Rate(src.Cells(i, 2).Value)
                r6 = Rate(src.Cells(i, 5).Value)
                rolled = 1#

                For k = i To j - 1
                    dtDays = DateDiff("d", CDate(src.Cells(k, 1).Value), CDate(src.Cells(k + 1, 1).Value))
                    If dtDays > 0 Then rolled = rolled * (1# + Rate(src.Cells(k, 2).Value) * dtDays / 360#)
                Next k
                rolled = rolled - 1#

                days = DateDiff("d", d0, dm)
                locked = r6 * days / 360#

                out.Cells(outRow, 1).Value = d0
                out.Cells(outRow, 2).Value = dm
                out.Cells(outRow, 3).Value = rON
                out.Cells(outRow, 4).Value = r6
                out.Cells(outRow, 5).Value = (r6 - rON) * 10000#
                out.Cells(outRow, 6).Value = rolled
                out.Cells(outRow, 7).Value = locked
                out.Cells(outRow, 8).Value = (locked - rolled) * 10000#
                out.Cells(outRow, 9).Value = IIf(locked > rolled, "6M", "ON")
                out.Cells(outRow, 10).Value = Rate(src.Cells(j, 2).Value)
                out.Cells(outRow, 11).Value = (Rate(src.Cells(j, 2).Value) - rON) * 10000#
                outRow = outRow + 1
            End If
        End If
    Next i

    out.Columns("A:B").NumberFormat = "dd-mmm-yyyy"
    out.Columns("C:D,F:G,J").NumberFormat = "0.000%"
    out.Columns("E,H,K").NumberFormat = "0.0"
End Sub

Private Sub BuildScenarioData(ByVal wb As Workbook)
    Dim ws As Worksheet, src As Worksheet
    Dim lr As Long, s As Long, m As Long, r As Long
    Dim baseON As Double, p1 As Double, p3 As Double, p6 As Double
    Dim nm As Variant, terminal As Variant, hump As Variant
    Dim path(1 To 5, 1 To 36) As Double, sleeve(1 To 5, 1 To 4) As Double
    Dim wON As Long, w1 As Long, w3 As Long, w6 As Long, rowOut As Long
    Dim ret As Double, bestRet As Double, reg As Double, worstReg As Double, avgReg As Double
    Dim endRow As Long, bestRow As Long

    Set src = wb.Worksheets("curve")
    lr = src.Cells(src.Rows.Count, 1).End(xlUp).Row
    baseON = Rate(src.Cells(lr, 2).Value)
    p1 = Rate(src.Cells(lr, 3).Value) - baseON
    p3 = Rate(src.Cells(lr, 4).Value) - baseON
    p6 = Rate(src.Cells(lr, 5).Value) - baseON

    Set ws = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
    ws.Name = "CFO_Data_Scen"

    nm = Array("Gradual normalization", "Higher for longer", "Renewed inflation", "Faster normalization", "Recession / rapid cuts")
    terminal = Array(0.0315, 0.0375, 0.034, 0.029, 0.025)
    hump = Array(0.002, 0.007, 0.012, -0.005, -0.02)

    PutRow ws.Range("A1"), Array("Scenario", "M1", "M7", "M17", "M29", "M36", "Economic state")
    For s = 1 To 5
        For m = 1 To 36
            path(s, m) = ScenarioRate(baseON, CDbl(terminal(s - 1)), CDbl(hump(s - 1)), s, m)
        Next m
        ws.Cells(s + 1, 1).Value = nm(s - 1)
        ws.Cells(s + 1, 2).Value = path(s, 1)
        ws.Cells(s + 1, 3).Value = path(s, 7)
        ws.Cells(s + 1, 4).Value = path(s, 17)
        ws.Cells(s + 1, 5).Value = path(s, 29)
        ws.Cells(s + 1, 6).Value = path(s, 36)
        ws.Cells(s + 1, 7).Value = ScenarioDescription(s)
    Next s

    PutRow ws.Range("A9"), Array("Scenario", "ON", "1M", "3M", "6M", "Best tenor", "Best return")
    For s = 1 To 5
        sleeve(s, 1) = SleeveReturn(path, s, 0#, 1)
        sleeve(s, 2) = SleeveReturn(path, s, p1, 1)
        sleeve(s, 3) = SleeveReturn(path, s, p3, 3)
        sleeve(s, 4) = SleeveReturn(path, s, p6, 6)
        ws.Cells(s + 9, 1).Value = nm(s - 1)
        For r = 1 To 4: ws.Cells(s + 9, r + 1).Value = sleeve(s, r): Next r
        ws.Cells(s + 9, 6).Value = BestTenor(sleeve, s)
        ws.Cells(s + 9, 7).Value = Application.WorksheetFunction.Max(sleeve(s, 1), sleeve(s, 2), sleeve(s, 3), sleeve(s, 4))
    Next s

    PutRow ws.Range("J1"), Array("ON", "1M", "3M", "6M", "Weighted months", "Worst regret bp", "Average regret bp")
    rowOut = 2
    For wON = 0 To 10
        For w1 = 0 To 10 - wON
            For w3 = 0 To 10 - wON - w1
                w6 = 10 - wON - w1 - w3
                worstReg = 0#: avgReg = 0#
                For s = 1 To 5
                    ret = (wON * sleeve(s, 1) + w1 * sleeve(s, 2) + w3 * sleeve(s, 3) + w6 * sleeve(s, 4)) / 10#
                    bestRet = Application.WorksheetFunction.Max(sleeve(s, 1), sleeve(s, 2), sleeve(s, 3), sleeve(s, 4))
                    reg = (bestRet - ret) * 10000#
                    If reg > worstReg Then worstReg = reg
                    avgReg = avgReg + reg / 5#
                Next s
                ws.Cells(rowOut, 10).Value = wON / 10#
                ws.Cells(rowOut, 11).Value = w1 / 10#
                ws.Cells(rowOut, 12).Value = w3 / 10#
                ws.Cells(rowOut, 13).Value = w6 / 10#
                ws.Cells(rowOut, 14).Value = (w1 + 3# * w3 + 6# * w6) / 10#
                ws.Cells(rowOut, 15).Value = worstReg
                ws.Cells(rowOut, 16).Value = avgReg
                rowOut = rowOut + 1
            Next w3
        Next w1
    Next wON

    endRow = rowOut - 1
    bestRow = 2
    For r = 3 To endRow
        If ws.Cells(r, 15).Value < ws.Cells(bestRow, 15).Value Then bestRow = r
    Next r

    PutRow ws.Range("J" & endRow + 3), Array("REFERENCE", "ON", "1M", "3M", "6M", "Weighted months", "Worst regret bp", "Average regret bp")
    ws.Cells(endRow + 4, 10).Value = "Selected"
    ws.Cells(endRow + 4, 11).Value = ws.Cells(bestRow, 10).Value
    ws.Cells(endRow + 4, 12).Value = ws.Cells(bestRow, 11).Value
    ws.Cells(endRow + 4, 13).Value = ws.Cells(bestRow, 12).Value
    ws.Cells(endRow + 4, 14).Value = ws.Cells(bestRow, 13).Value
    ws.Cells(endRow + 4, 15).Value = ws.Cells(bestRow, 14).Value
    ws.Cells(endRow + 4, 16).Value = ws.Cells(bestRow, 15).Value
    ws.Cells(endRow + 4, 17).Value = ws.Cells(bestRow, 16).Value

    ws.Range("B2:F6").NumberFormat = "0.00%"
    ws.Range("B10:E14,G10:G14").NumberFormat = "0.00%"
    ws.Range("J2:M" & endRow).NumberFormat = "0%"
End Sub

Private Function ScenarioRate(ByVal startRate As Double, ByVal endRate As Double, ByVal hump As Double, ByVal scen As Long, ByVal m As Long) As Double
    Dim t As Double, x As Double
    t = (m - 1) / 35#
    Select Case scen
        Case 1, 3
            x = startRate + hump * Sin(Application.WorksheetFunction.Pi() * t) + (endRate - startRate) * t
        Case 2
            If t < 0.6 Then
                x = startRate + hump * (t / 0.6)
            Else
                x = startRate + hump + (endRate - startRate - hump) * ((t - 0.6) / 0.4)
            End If
        Case 4
            x = startRate + hump * (1 - Exp(-5# * t)) + (endRate - startRate - hump) * t
        Case 5
            x = startRate + hump * (1 - Exp(-8# * t)) + (endRate - startRate - hump) * t
    End Select
    If x < 0.0001 Then x = 0.0001
    ScenarioRate = x
End Function

Private Function SleeveReturn(ByRef path() As Double, ByVal scen As Long, ByVal premium As Double, ByVal resetEvery As Long) As Double
    Dim wealth As Double, m As Long, rr As Double
    wealth = 1#
    For m = 1 To 36
        If m = 1 Or ((m - 1) Mod resetEvery = 0) Then rr = path(scen, m) + premium
        wealth = wealth * (1# + rr / 12#)
    Next m
    SleeveReturn = wealth ^ (1# / 3#) - 1#
End Function

Private Function BestTenor(ByRef sleeve() As Double, ByVal scen As Long) As String
    Dim i As Long, bi As Long, bv As Double
    bi = 1: bv = sleeve(scen, 1)
    For i = 2 To 4
        If sleeve(scen, i) > bv Then bv = sleeve(scen, i): bi = i
    Next i
    BestTenor = Choose(bi, "ON", "1M", "3M", "6M")
End Function

Private Function ScenarioDescription(ByVal scen As Long) As String
    Select Case scen
        Case 1: ScenarioDescription = "Mild near-term firmness, then measured easing."
        Case 2: ScenarioDescription = "Sticky inflation keeps policy restrictive."
        Case 3: ScenarioDescription = "Reacceleration triggers additional tightening before normalization."
        Case 4: ScenarioDescription = "Benign disinflation allows faster normalization."
        Case 5: ScenarioDescription = "Growth shock forces rapid cuts, followed by recovery."
    End Select
End Function

        .ChartArea.Format.Line.Visible = msoFalse
    End With
    Bottom ws, "We are choosing an exposure profile - not making a binary call that rates must rise or fall.", "Source: curve tab."

Private Sub Page4(ByVal wb As Workbook)
    Dim ws As Worksheet, sc As Worksheet, co As ChartObject
    Dim refRow As Long, gridEnd As Long, r As Long
    Dim wON As Double, w1 As Double, w3 As Double, w6 As Double, wt As Double, wr As Double
    Dim minWorst As Double, nearN As Long, minTen As Double, maxTen As Double

    Set ws = NewPage(wb, "CFO_4_Portfolio")
    Set sc = wb.Worksheets("CFO_Data_Scen")
    gridEnd = sc.Cells(sc.Rows.Count, 10).End(xlUp).Row
    refRow = gridEnd
    Do While refRow > 1 And sc.Cells(refRow, 10).Value <> "Selected": refRow = refRow - 1: Loop

    wON = sc.Cells(refRow,11).Value: w1 = sc.Cells(refRow,12).Value
    w3 = sc.Cells(refRow,13).Value: w6 = sc.Cells(refRow,14).Value
    wt = sc.Cells(refRow,15).Value: wr = sc.Cells(refRow,16).Value

    minWorst = Application.WorksheetFunction.Min(sc.Range("O2:O" & refRow-4))
    minTen = 1E+99: maxTen = -1E+99
    For r = 2 To refRow-4
        If sc.Cells(r,15).Value <= minWorst + 0.25 Then
            nearN = nearN + 1
            If sc.Cells(r,14).Value < minTen Then minTen = sc.Cells(r,14).Value
            If sc.Cells(r,14).Value > maxTen Then maxTen = sc.Cells(r,14).Value
        End If
    Next r

    Head ws, "When direction is uncertain, focus on the cost of being wrong - not on one forecast", _
        "Alternative rate paths expose the trade-off and help identify a robust allocation region rather than a single 'correct' tenor."
    Box ws, "B6:F10", "IF RATES RISE", "Shorter resets retain more ability to participate in higher market rates." & vbLf & vbLf & "Economic risk: lock-in opportunity cost.", RGB(245,241,231)
    Box ws, "B11:F15", "IF RATES FALL", "Longer locking protects against reinvestment at lower rates." & vbLf & vbLf & "Economic risk: reinvestment opportunity cost.", RGB(234,243,243)

    ws.Range("B17:H22").Value = sc.Range("A9:G14").Value: Tbl ws.Range("B17:H22")
    ws.Range("C18:F22").NumberFormat = "0.00%": ws.Range("H18:H22").NumberFormat = "0.00%"

    Set co = ws.ChartObjects.Add(ws.Range("J6").Left, ws.Range("J6").Top, 455, 275)
    With co.Chart
        .ChartType = xlXYScatter: .SeriesCollection.NewSeries
        .SeriesCollection(1).XValues = sc.Range("N2:N" & refRow-4): .SeriesCollection(1).Values = sc.Range("O2:O" & refRow-4)
        .SeriesCollection(1).MarkerSize = 4: .SeriesCollection(1).MarkerForegroundColor = BLUE
        .HasTitle = True: .ChartTitle.Text = "Robust allocation region"
        .Axes(xlCategory).HasTitle = True: .Axes(xlValue).HasTitle = True
        .Axes(xlCategory).AxisTitle.Text = "Weighted tenor (months)": .Axes(xlValue).AxisTitle.Text = "Worst regret (bp/year)"
        .HasLegend = False: .ChartArea.Format.Line.Visible = msoFalse
    End With

    Kpi ws, "J17:L20", "Reference mix", MixLabel(wON,w1,w3,w6), "Illustrative minimax-regret anchor."
    Kpi ws, "M17:O20", "Weighted tenor", Format(wt,"0.0") & " months", "Position on the curve."
    Kpi ws, "P17:R20", "Worst regret", Bp(wr) & "/yr", "Maximum model-state shortfall."
    TextBox ws, "J22:R24", "The model identifies a risk-balanced region rather than one precise answer. " & nearN & " portfolios sit within 0.25 bp/year of the minimum, spanning approximately " & Format(minTen,"0.0") & " to " & Format(maxTen,"0.0") & " months weighted tenor.", 10, False, TXT, LIGHT, xlLeft
    Bottom ws, "The objective is to reduce dependence on one rate view while preserving flexibility to adapt as conditions change.", "Source: curve tab. Forward paths are transparent sensitivities, not forecasts or return guarantees."
End Sub

Private Function MixLabel(ByVal wON As Double, ByVal w1 As Double, ByVal w3 As Double, ByVal w6 As Double) As String
    Dim s As String
    If wON > 0 Then s = s & Format(wON,"0%") & " ON / "
    If w1 > 0 Then s = s & Format(w1,"0%") & " 1M / "
    If w3 > 0 Then s = s & Format(w3,"0%") & " 3M / "
    If w6 > 0 Then s = s & Format(w6,"0%") & " 6M / "
    If Right$(s,3) = " / " Then s = Left$(s,Len(s)-3)
    MixLabel = s
End Function

Private Function FindFloorDateRow(ByVal ws As Worksheet, ByVal targetDate As Date, ByVal startRow As Long, ByVal lastRow As Long) As Long
    Dim lo As Long, hi As Long, mid As Long, ans As Long
    lo = startRow: hi = lastRow
    Do While lo <= hi
        mid = (lo + hi) \ 2
        If CDate(ws.Cells(mid,1).Value) <= targetDate Then ans = mid: lo = mid + 1 Else hi = mid - 1
    Loop
    FindFloorDateRow = ans
End Function

Private Function SafeCorrel(ByVal x As Range, ByVal y As Range) As Double
    On Error GoTo Nope
    SafeCorrel = Application.WorksheetFunction.Correl(x,y): Exit Function
Nope:
    SafeCorrel = 0#
End Function

Private Function AverageIfGE(ByVal criteriaRng As Range, ByVal avgRng As Range, ByVal threshold As Double) As Double
    Dim i As Long, n As Long, s As Double
    For i = 1 To criteriaRng.Rows.Count
        If CDbl(criteriaRng.Cells(i,1).Value) >= threshold Then s = s + CDbl(avgRng.Cells(i,1).Value): n = n + 1
    Next i
    If n > 0 Then AverageIfGE = s / n
End Function

Private Function PositiveHitIfGE(ByVal criteriaRng As Range, ByVal valueRng As Range, ByVal threshold As Double) As Double
    Dim i As Long, n As Long, p As Long
    For i = 1 To criteriaRng.Rows.Count
        If CDbl(criteriaRng.Cells(i,1).Value) >= threshold Then n = n + 1: If CDbl(valueRng.Cells(i,1).Value) > 0 Then p = p + 1
    Next i
    If n > 0 Then PositiveHitIfGE = p / n
End Function

Private Function HasSheet(ByVal wb As Workbook, ByVal nm As String) As Boolean
    Dim ws As Worksheet
    For Each ws In wb.Worksheets
        If LCase$(ws.Name) = LCase$(nm) Then HasSheet = True: Exit Function
    Next ws
End Function

Private Function ValidateCurve(ByVal ws As Worksheet) As Boolean
    Dim lr As Long: lr = ws.Cells(ws.Rows.Count,1).End(xlUp).Row
    If lr < 10 Then MsgBox "The curve sheet does not contain enough observations.", vbCritical: Exit Function
    If Not IsDate(ws.Cells(2,1).Value) Then MsgBox "curve!A2 must contain a valid date.", vbCritical: Exit Function
    If Not IsNumeric(ws.Cells(2,2).Value) Or Not IsNumeric(ws.Cells(2,5).Value) Then MsgBox "Expected A:E = Date | ON | 1M | 3M | 6M", vbCritical: Exit Function
    ValidateCurve = True
End Function
