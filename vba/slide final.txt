Option Explicit

'===============================================================================
' CASH ALLOCATION | FINAL REPLACEMENT SLIDE
'
' Purpose
'   Replaces the slide titled "No single tenor wins in every scenario" with a
'   cleaner final decision slide.
'
' How to use
'   1. Open the PowerPoint deck you want to update.
'   2. Open Excel.
'   3. Paste this full module into Excel VBA.
'   4. Run: ReplaceNoSingleTenorSlideWithFinal
'
' Behavior
'   - If a slide title contains "No single tenor wins", the macro deletes that
'     slide and inserts the new slide in the same position.
'   - If it cannot find that slide, it appends the new slide at the end.
'   - Uses classic Office-safe formatting: Arial, native shapes, no external files.
'===============================================================================

Public Sub ReplaceNoSingleTenorSlideWithFinal()

    Const ppLayoutBlank As Long = 12

    Dim ppApp As Object
    Dim ppPres As Object
    Dim sld As Object
    Dim slideIndex As Long
    Dim replaceIndex As Long

    On Error Resume Next
    Set ppApp = GetObject(, "PowerPoint.Application")
    On Error GoTo 0

    If ppApp Is Nothing Then
        Set ppApp = CreateObject("PowerPoint.Application")
    End If

    ppApp.Visible = True

    If ppApp.Presentations.Count = 0 Then
        Set ppPres = ppApp.Presentations.Add
    Else
        Set ppPres = ppApp.ActivePresentation
    End If

    With ppPres.PageSetup
        .SlideWidth = 13.333333 * 72
        .SlideHeight = 7.5 * 72
    End With

    replaceIndex = FindSlideByTitle(ppPres, "No single tenor wins")

    If replaceIndex > 0 Then
        slideIndex = replaceIndex
        ppPres.Slides(replaceIndex).Delete
    Else
        slideIndex = ppPres.Slides.Count + 1
    End If

    Set sld = ppPres.Slides.Add(slideIndex, ppLayoutBlank)
    BuildFinalCashAllocationSlide sld

End Sub

Private Function FindSlideByTitle(ByVal ppPres As Object, ByVal titleFragment As String) As Long

    Dim sld As Object
    Dim shp As Object
    Dim txt As String

    For Each sld In ppPres.Slides

        For Each shp In sld.Shapes

            txt = ""

            On Error Resume Next
            If shp.HasTextFrame Then
                If shp.TextFrame.HasText Then
                    txt = shp.TextFrame.TextRange.Text
                End If
            End If
            On Error GoTo 0

            If InStr(1, txt, titleFragment, vbTextCompare) > 0 Then
                FindSlideByTitle = CLng(sld.SlideIndex)
                Exit Function
            End If

        Next shp

    Next sld

    FindSlideByTitle = 0

End Function

Private Sub AddArrow( _

    ByVal y1 As Double, _

    ByVal y2 As Double, _

    Const msoLineDash As Long = 4
    Const msoArrowheadTriangle As Long = 3

    Dim ln As Object

    Set ln = sld.Shapes.AddLine(x1 * 72, y1 * 72, x2 * 72, y2 * 72)

    With ln.Line
        .ForeColor.RGB = lineColor

Private Sub AddRect( _
    ByVal sld As Object, _
    ByVal x As Double, _
    ByVal y As Double, _
    ByVal w As Double, _
    ByVal h As Double, _
    ByVal fillColor As Long, _
    ByVal lineColor As Long, _
    ByVal radius As Double)

    Const msoShapeRectangle As Long = 1
    Const msoShapeRoundedRectangle As Long = 5

    Dim shp As Object
    Dim shapeType As Long

    If radius > 0 Then
        shapeType = msoShapeRoundedRectangle
    Else
        shapeType = msoShapeRectangle
    End If

    Set shp = sld.Shapes.AddShape(shapeType, x * 72, y * 72, w * 72, h * 72)

    With shp
        .Fill.ForeColor.RGB = fillColor
        .Line.ForeColor.RGB = lineColor
        .Line.Weight = 0.7
        If radius > 0 Then
            On Error Resume Next
            .Adjustments.Item(1) = radius
            On Error GoTo 0
        End If
    End With

End Sub

Private Function AddText( _
    ByVal sld As Object, _
    ByVal txt As String, _
    ByVal x As Double, _
    ByVal y As Double, _
    ByVal w As Double, _
    ByVal h As Double, _
    ByVal fontSize As Double, _
    ByVal fontColor As Long, _
    ByVal isBold As Boolean, _
    ByVal alignMode As Long) As Object

    Const msoTextOrientationHorizontal As Long = 1
    Const msoAlignLeft As Long = 1
    Const msoAlignCenter As Long = 2

    Dim shp As Object

    Set shp = sld.Shapes.AddTextbox(msoTextOrientationHorizontal, x * 72, y * 72, w * 72, h * 72)

    With shp.TextFrame
        .MarginLeft = 0
        .MarginRight = 0
        .MarginTop = 0
        .MarginBottom = 0
        .WordWrap = True

        With .TextRange
            .Text = txt
            .Font.Name = "Arial"
            .Font.Size = fontSize
            .Font.Color.RGB = fontColor
            .Font.Bold = isBold
            If alignMode = 1 Then
                .ParagraphFormat.Alignment = msoAlignCenter
            Else
                .ParagraphFormat.Alignment = msoAlignLeft
            End If
        End With
    End With

    Set AddText = shp

End Function
