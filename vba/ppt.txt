Option Explicit

'====================================================================================
' RM DECK -> POWERPOINT EXPORTER (v2)
'
' Reproduces the five RM_* Excel sheets as PowerPoint slides using the EXACT design
' already reviewed and approved (RM_CFO_Slide1.pptx .. RM_CFO_Slide5.pptx): same
' colors, same fonts, same geometry - hardcoded here rather than introspected from
' whatever presentation happens to be open, which is what made v1 fragile.
'
' RUN
'   Export_RM_Deck_To_PowerPoint
'   - PowerPoint must already be open with a presentation active. New slides are
'     appended after whatever is already there. Nothing is saved - review and save
'     yourself when ready.
'
' REQUIRED
'   Workbook already built by RM_Full_Deck (sheets RM_1_History .. RM_5_Partnership).
'
' DESIGN SYSTEM (fixed, matches the approved pptx mockups - not read from anywhere)
'   Slide size assumed widescreen 13.333 x 7.5 in (960 x 540 pt); if the open
'   presentation is a different size the exporter scales all geometry to fit.
'   Colors : Navy 1E2761 (titles/hero) | Slate 5B6770 (body) | Gray 8A94A0 (muted)
'            Amber E49B31 (accent)     | Ice EEF2F8 (card fill) | Line D8DEE6 (borders)
'   Fonts  : Cambria for titles and the takeaway band; Arial for everything else.
'   Slide 5 only: dark navy background, amber closing band (mirrors the approved
'   mockup's deliberate light/dark contrast for the closing slide).
'====================================================================================

Private Const S1 As String = "RM_1_History"
Private Const S2 As String = "RM_2_Return"
Private Const S3 As String = "RM_3_Inversion"
Private Const S4 As String = "RM_4_Regret"
Private Const S5 As String = "RM_5_Partnership"

' ---- design system, fixed hex, matches the approved pptx mockups ----
Private Const HEX_NAVY As String = "1E2761"
Private Const HEX_SLATE As String = "5B6770"
Private Const HEX_GRAY As String = "8A94A0"
Private Const HEX_AMBER As String = "E49B31"
Private Const HEX_ICE As String = "EEF2F8"
Private Const HEX_CARD2 As String = "F7F9FC"
Private Const HEX_LINE As String = "D8DEE6"
Private Const HEX_WHITE As String = "FFFFFF"
Private Const FONT_TITLE As String = "Cambria"
Private Const FONT_BODY As String = "Arial"

' ---- base geometry, INCHES, matches the approved pptx mockups (13.333 x 7.5in) ----
Private Const BASE_W As Single = 13.333
Private Const BASE_H As Single = 7.5

Private Const msoTextOrientationHorizontal As Long = 1
Private Const msoShapeRectangle As Long = 1
Private Const msoTrue As Long = -1
Private Const msoFalse As Long = 0
Private Const ppAlignCenter As Long = 2
Private Const ppAlignLeft As Long = 1
Private Const ppPasteEnhancedMetafile As Long = 2
Private Const ppLayoutBlank As Long = 12

Private gStage As String
Private gScaleX As Single, gScaleY As Single   ' open presentation's size / BASE_W,BASE_H

'====================================================================================
' ENTRY
'====================================================================================
Public Sub Export_RM_Deck_To_PowerPoint()
    Dim wb As Workbook
    Dim pptApp As Object, pptPres As Object
    Dim startCount As Long

    Set wb = ThisWorkbook
    Dim missing As String, s As Variant
    For Each s In Array(S1, S2, S3, S4, S5)
        If Not SheetExists(wb, CStr(s)) Then missing = missing & CStr(s) & vbCrLf
    Next s
    If Len(missing) > 0 Then
        MsgBox "Run Build_RM_Full_Deck first - missing sheet(s):" & vbCrLf & missing, vbCritical
        Exit Sub
    End If

    On Error GoTo Fail
    gStage = "Connect to open PowerPoint"
    On Error Resume Next
    Set pptApp = GetObject(, "PowerPoint.Application")
    On Error GoTo Fail
    If pptApp Is Nothing Then
        MsgBox "PowerPoint isn't open. Open your presentation and run this again.", vbExclamation
        Exit Sub
    End If
    If pptApp.Presentations.Count = 0 Then
        MsgBox "No presentation is open in PowerPoint.", vbExclamation
        Exit Sub
    End If
    Set pptPres = pptApp.ActivePresentation

    gStage = "Compute scale"
    gScaleX = pptPres.PageSetup.SlideWidth / (BASE_W * 72#)
    gScaleY = pptPres.PageSetup.SlideHeight / (BASE_H * 72#)
    startCount = pptPres.Slides.Count

    gStage = "Slide 1": BuildSlide1PPT wb, pptPres, S1, startCount + 1
    gStage = "Slide 2": BuildSlide2PPT wb, pptPres, S2, startCount + 2
    gStage = "Slide 3": BuildSlide3PPT wb, pptPres, S3, startCount + 3
    gStage = "Slide 4": BuildSlide4PPT wb, pptPres, S4, startCount + 4
    gStage = "Slide 5": BuildSlide5PPT wb, pptPres, S5, startCount + 5

    MsgBox "Added 5 slides (" & CStr(startCount + 1) & "-" & CStr(startCount + 5) & _
           ") to """ & pptPres.Name & """. Not saved - review and save when ready.", vbInformation
    Exit Sub

Fail:
    MsgBox "Export failed." & vbCrLf & "Stage: " & gStage & vbCrLf & _
           "Error: " & Err.Number & " - " & Err.Description, vbCritical
End Sub

'====================================================================================
' GEOMETRY HELPERS - inches (matching the approved mockups) scaled to the open deck
'====================================================================================
Private Function Pt(ByVal xIn As Single) As Single
    Pt = xIn * 72# * gScaleX
End Function
Private Function PtY(ByVal yIn As Single) As Single
    PtY = yIn * 72# * gScaleY
End Function
Private Function PtW(ByVal wIn As Single) As Single
    PtW = wIn * 72# * gScaleX
End Function
Private Function PtH(ByVal hIn As Single) As Single
    PtH = hIn * 72# * gScaleY
End Function

Private Function HexColor(ByVal hx As String) As Long
    HexColor = RGB(CLng("&H" & Mid$(hx, 1, 2)), CLng("&H" & Mid$(hx, 3, 2)), CLng("&H" & Mid$(hx, 5, 2)))
End Function

'====================================================================================
' SHARED SLIDE SCAFFOLD: new blank slide + kicker + title, reading straight off the sheet
'====================================================================================
Private Function NewBlankSlide(ByVal pptPres As Object, ByVal atIndex As Long) As Object
    Dim lay As Object, sld As Object
    On Error Resume Next
    Set lay = pptPres.SlideMaster.CustomLayouts(ppLayoutBlank)
    On Error GoTo 0
    If lay Is Nothing Then
        Set sld = pptPres.Slides.Add(atIndex, ppLayoutBlank)
    Else
        Set sld = pptPres.Slides.AddSlide(atIndex, lay)
    End If
    Set NewBlankSlide = sld
End Function

Private Sub AddKickerAndTitle(ByVal sld As Object, ByVal ws As Worksheet, ByVal isDark As Boolean)
    Dim kicker As String, titleText As String, shp As Object
    kicker = SafeCellText(ws, "B2")
    titleText = SafeCellText(ws, "B3")

    Set shp = sld.Shapes.AddTextbox(msoTextOrientationHorizontal, Pt(0.55), PtY(0.3), PtW(9.5), PtH(0.3))
    With shp.TextFrame.TextRange
        .Text = UCase$(kicker)
        .Font.Name = FONT_BODY
        .Font.Size = 10
        .Font.Bold = msoTrue
        .Font.Color.RGB = HexColor(IIf(isDark, HEX_WHITE, HEX_GRAY))
    End With

    Set shp = sld.Shapes.AddTextbox(msoTextOrientationHorizontal, Pt(0.55), PtY(0.62), PtW(12.25), PtH(0.92))
    With shp.TextFrame
        .WordWrap = msoTrue
        .TextRange.Text = titleText
        .TextRange.Font.Name = FONT_TITLE
        .TextRange.Font.Bold = msoTrue
        .TextRange.Font.Color.RGB = HexColor(IIf(isDark, HEX_WHITE, HEX_NAVY))
    End With
    AutoFitTitleFont shp, titleText
End Sub

Private Sub AddBand(ByVal sld As Object, ByVal ws As Worksheet, ByVal bandAddr As String, _
                    ByVal bx As Single, ByVal by_ As Single, ByVal bw As Single, ByVal bh As Single, _
                    ByVal fillHex As String, ByVal textHex As String)
    Dim bandText As String, shp As Object
    bandText = SafeCellText(ws, bandAddr)
    If Len(bandText) = 0 Then Exit Sub
    Set shp = sld.Shapes.AddShape(msoShapeRectangle, Pt(bx), PtY(by_), PtW(bw), PtH(bh))
    shp.Fill.Solid
    shp.Fill.ForeColor.RGB = HexColor(fillHex)
    shp.Line.Visible = msoFalse
    With shp.TextFrame
        .WordWrap = msoTrue
        .MarginLeft = 10: .MarginRight = 10
        .TextRange.Text = bandText
        .TextRange.Font.Name = FONT_TITLE
        .TextRange.Font.Bold = msoTrue
        .TextRange.Font.Size = 13
        .TextRange.Font.Color.RGB = HexColor(textHex)
        .TextRange.ParagraphFormat.Alignment = ppAlignCenter
    End With
    AutoFitBoxFont shp, bandText, 13, 9
End Sub

Private Sub AddFooter(ByVal sld As Object, ByVal ws As Worksheet, ByVal footAddr As String, _
                      ByVal slideNo As Long, ByVal fy As Single, ByVal isDark As Boolean)
    Dim footText As String, shp As Object
    footText = SafeCellText(ws, footAddr)
    If Len(footText) > 0 Then
        Set shp = sld.Shapes.AddTextbox(msoTextOrientationHorizontal, Pt(0.55), PtY(fy), PtW(11.6), PtH(0.3))
        With shp.TextFrame.TextRange
            .Text = footText
            .Font.Name = FONT_BODY
            .Font.Size = 7.5
            .Font.Color.RGB = HexColor(HEX_GRAY)
        End With
    End If
    Set shp = sld.Shapes.AddTextbox(msoTextOrientationHorizontal, Pt(12.55), PtY(fy), PtW(0.35), PtH(0.3))
    With shp.TextFrame.TextRange
        .Text = CStr(slideNo)
        .Font.Name = FONT_BODY
        .Font.Size = 9
        .Font.Color.RGB = HexColor(HEX_GRAY)
        .ParagraphFormat.Alignment = ppAlignLeft
    End With
End Sub

' Paste the Nth ChartObject on the sheet into the given rectangle (inches).
Private Sub PasteChart(ByVal ws As Worksheet, ByVal chartIndex As Long, ByVal sld As Object, _
                       ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single)
    On Error Resume Next
    If ws.ChartObjects.Count < chartIndex Then Exit Sub
    ws.ChartObjects(chartIndex).Copy
    sld.Shapes.PasteSpecial DataType:=ppPasteEnhancedMetafile
    Dim shp As Object
    Set shp = sld.Shapes(sld.Shapes.Count)
    If Not shp Is Nothing Then
        shp.LockAspectRatio = msoFalse
        shp.Left = Pt(x): shp.Top = PtY(y)
        shp.Width = PtW(w): shp.Height = PtH(h)
    End If
    On Error GoTo 0
End Sub

' One stat box: LABEL / big value / note - matches the pptx KPI card design.
Private Sub AddStatBox(ByVal sld As Object, ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single, _
                       ByVal rawText As String, ByVal fillHex As String, ByVal valueHex As String)
    Dim shp As Object, lbl As String, val_ As String, note As String
    ParseKpiText rawText, lbl, val_, note

    Set shp = sld.Shapes.AddShape(msoShapeRectangle, Pt(x), PtY(y), PtW(w), PtH(h))
    shp.Fill.Solid
    shp.Fill.ForeColor.RGB = HexColor(fillHex)
    shp.Line.ForeColor.RGB = HexColor(HEX_LINE)
    shp.Line.Weight = 0.75

    With shp.TextFrame
        .WordWrap = msoTrue
        .MarginLeft = 10: .MarginRight = 10: .MarginTop = 6: .MarginBottom = 6
        .TextRange.Text = UCase$(lbl) & Chr(11) & val_ & Chr(11) & note
    End With
    With shp.TextFrame.TextRange.Lines(1)
        .Font.Name = FONT_BODY: .Font.Size = 8.5: .Font.Bold = msoTrue
        .Font.Color.RGB = HexColor(HEX_SLATE)
    End With
    With shp.TextFrame.TextRange.Lines(2)
        .Font.Name = FONT_TITLE: .Font.Size = 17: .Font.Bold = msoTrue
        .Font.Color.RGB = HexColor(valueHex)
    End With
    On Error Resume Next
    With shp.TextFrame.TextRange.Lines(3)
        .Font.Name = FONT_BODY: .Font.Size = 8: .Font.Color.RGB = HexColor(HEX_SLATE)
    End With
    On Error GoTo 0
End Sub

' Narrative card: "N  Title   date" header line, then body text - matches the chained-card design.
Private Sub AddNarrativeCard(ByVal sld As Object, ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single, _
                             ByVal rawText As String, ByVal fillHex As String)
    Dim shp As Object, headLine As String, bodyText As String, nl As Long
    nl = InStr(rawText, Chr(10))
    If nl > 0 Then
        headLine = Left$(rawText, nl - 1)
        bodyText = Mid$(rawText, nl + 1)
    Else
        headLine = rawText: bodyText = ""
    End If

    Set shp = sld.Shapes.AddShape(msoShapeRectangle, Pt(x), PtY(y), PtW(w), PtH(h))
    shp.Fill.Solid
    shp.Fill.ForeColor.RGB = HexColor(fillHex)
    shp.Line.ForeColor.RGB = HexColor(HEX_LINE)
    shp.Line.Weight = 0.75

    With shp.TextFrame
        .WordWrap = msoTrue
        .MarginLeft = 8: .MarginRight = 8: .MarginTop = 6: .MarginBottom = 6
        .TextRange.Text = headLine & Chr(11) & bodyText
    End With
    With shp.TextFrame.TextRange.Lines(1)
        .Font.Name = FONT_TITLE: .Font.Size = 11: .Font.Bold = msoTrue
        .Font.Color.RGB = HexColor(HEX_NAVY)
    End With
    If Len(bodyText) > 0 Then
        On Error Resume Next
        With shp.TextFrame.TextRange.Lines(2)
            .Font.Name = FONT_BODY: .Font.Size = 9: .Font.Color.RGB = HexColor(HEX_SLATE)
        End With
        On Error GoTo 0
    End If
    AutoFitBoxFont shp, rawText, 9, 7
End Sub

' A single wrapped box of prose with a bold lead-in label (e.g. "CFO LENS").
Private Sub AddLensBox(ByVal sld As Object, ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single, _
                       ByVal fullText As String, ByVal leadIn As String)
    Dim shp As Object
    If Len(fullText) = 0 Then Exit Sub
    Set shp = sld.Shapes.AddShape(msoShapeRectangle, Pt(x), PtY(y), PtW(w), PtH(h))
    shp.Fill.Solid
    shp.Fill.ForeColor.RGB = HexColor(HEX_ICE)
    shp.Line.ForeColor.RGB = HexColor(HEX_LINE)
    shp.Line.Weight = 0.75
    With shp.TextFrame
        .WordWrap = msoTrue
        .MarginLeft = 10: .MarginRight = 10: .MarginTop = 6: .MarginBottom = 6
        .TextRange.Text = fullText
        .TextRange.Font.Name = FONT_BODY
        .TextRange.Font.Size = 10.5
        .TextRange.Font.Color.RGB = HexColor(HEX_SLATE)
    End With
    On Error Resume Next
    shp.TextFrame.TextRange.Characters(1, Len(leadIn)).Font.Bold = msoTrue
    shp.TextFrame.TextRange.Characters(1, Len(leadIn)).Font.Color.RGB = HexColor(HEX_NAVY)
    On Error GoTo 0
End Sub

' ---- Slide 1: full-width chart, four chained cards, band, footer ----
Private Sub BuildSlide1PPT(ByVal wb As Workbook, ByVal pptPres As Object, ByVal sheetName As String, ByVal slideNo As Long)
    Dim ws As Worksheet, sld As Object, i As Long
    Set ws = wb.Worksheets(sheetName)
    Set sld = NewBlankSlide(pptPres, slideNo)
    AddKickerAndTitle sld, ws, False
    PasteChart ws, 1, sld, 0.55, 1.7, 12.25, 3.15

    Dim addrs As Variant, cw As Single, gap As Single, cx As Single, t As String
    addrs = Array("B22", "F22", "J22", "N22")
    cw = 2.98: gap = 0.11: cx = 0.55
    For i = 0 To UBound(addrs)
        t = SafeCellText(ws, CStr(addrs(i)))
        If Len(Trim$(t)) > 0 Then
            AddNarrativeCard sld, cx, 5.02, cw, 1.34, t, IIf(i Mod 2 = 0, HEX_ICE, HEX_CARD2)
        End If
        cx = cx + cw + gap
    Next i

    AddBand sld, ws, "B28", 0.55, 6.55, 12.25, 0.5, HEX_NAVY, HEX_WHITE
    AddFooter sld, ws, "B31", slideNo, 7.12, False
End Sub

' ---- Slide 2: two charts side by side, four KPI stat boxes, band, footer ----
Private Sub BuildSlide2PPT(ByVal wb As Workbook, ByVal pptPres As Object, ByVal sheetName As String, ByVal slideNo As Long)
    Dim ws As Worksheet, sld As Object, i As Long
    Set ws = wb.Worksheets(sheetName)
    Set sld = NewBlankSlide(pptPres, slideNo)
    AddKickerAndTitle sld, ws, False
    PasteChart ws, 1, sld, 0.55, 1.68, 6.05, 3.05
    PasteChart ws, 2, sld, 6.85, 1.68, 5.95, 3.05

    Dim addrs As Variant, cw As Single, gap As Single, cx As Single, t As String
    addrs = Array("B24", "F24", "J24", "N24")
    cw = 2.98: gap = 0.11: cx = 0.55
    For i = 0 To UBound(addrs)
        t = SafeCellText(ws, CStr(addrs(i)))
        If Len(Trim$(t)) > 0 Then AddStatBox sld, cx, 5#, cw, 1.15, t, HEX_ICE, HEX_NAVY
        cx = cx + cw + gap
    Next i

    AddBand sld, ws, "B29", 0.55, 6.55, 12.25, 0.5, HEX_NAVY, HEX_WHITE
    AddFooter sld, ws, "B32", slideNo, 7.12, False
End Sub

' ---- Slide 4: two stacked charts, return table, two KPI boxes, CFO lens, band, footer ----
Private Sub BuildSlide4PPT(ByVal wb As Workbook, ByVal pptPres As Object, ByVal sheetName As String, ByVal slideNo As Long)
    Dim ws As Worksheet, sld As Object
    Set ws = wb.Worksheets(sheetName)
    Set sld = NewBlankSlide(pptPres, slideNo)
    AddKickerAndTitle sld, ws, False
    PasteChart ws, 1, sld, 0.55, 1.68, 6.05, 2.55
    PasteChart ws, 2, sld, 0.55, 4.4, 6.05, 1.85

    ' return table: copy the Excel range as a picture, keeps its exact formatting
    On Error Resume Next
    ws.Range("B23:G28").CopyPicture xlScreen, xlPicture
    sld.Shapes.PasteSpecial DataType:=ppPasteEnhancedMetafile
    Dim tshp As Object
    Set tshp = sld.Shapes(sld.Shapes.Count)
    If Not tshp Is Nothing Then
        tshp.LockAspectRatio = msoTrue
        tshp.Left = Pt(6.85): tshp.Top = PtY(1.68)
        tshp.Width = PtW(5.95)
    End If
    On Error GoTo 0

    Dim t1 As String, t2 As String
    t1 = SafeCellText(ws, "I23"): t2 = SafeCellText(ws, "N23")
    If Len(Trim$(t1)) > 0 Then AddStatBox sld, 6.85, 4.4, 2.9, 1.1, t1, HEX_ICE, HEX_NAVY
    If Len(Trim$(t2)) > 0 Then AddStatBox sld, 9.9, 4.4, 2.9, 1.1, t2, HEX_ICE, HEX_AMBER

    AddLensBox sld, 6.85, 5.65, 5.95, 0.85, SafeCellText(ws, "I26"), "CFO LENS"
    AddBand sld, ws, "B30", 0.55, 6.62, 12.25, 0.5, HEX_NAVY, HEX_WHITE
    AddFooter sld, ws, "B33", slideNo, 7.25, False
End Sub

' ---- Slide 5: dark close, three pillars, optional premium chart + 2 KPIs, amber band, footer ----
Private Sub BuildSlide5PPT(ByVal wb As Workbook, ByVal pptPres As Object, ByVal sheetName As String, ByVal slideNo As Long)
    Dim ws As Worksheet, sld As Object, i As Long
    Set ws = wb.Worksheets(sheetName)
    Set sld = NewBlankSlide(pptPres, slideNo)

    sld.Background.Fill.Solid
    sld.Background.Fill.ForeColor.RGB = HexColor(HEX_NAVY)

    AddKickerAndTitle sld, ws, True

    Dim addrs As Variant, pw As Single, pgap As Single, px As Single, t As String
    addrs = Array("B6", "H6", "N6")
    pw = 3.92: pgap = 0.24: px = 0.55
    For i = 0 To UBound(addrs)
        t = SafeCellText(ws, CStr(addrs(i)))
        If Len(Trim$(t)) > 0 Then AddDarkPillar sld, px, 1.62, pw, 1.62, t
        px = px + pw + pgap
    Next i

    If ws.ChartObjects.Count > 0 Then
        PasteChart ws, 1, sld, 0.8, 3.65, 6.5, 1.8
        Dim k1 As String, k2 As String
        k1 = SafeCellText(ws, "N17"): k2 = SafeCellText(ws, "N20")
        If Len(Trim$(k1)) > 0 Then AddStatBox sld, 9.1, 3.68, 3.45, 0.86, k1, HEX_WHITE, HEX_AMBER
        If Len(Trim$(k2)) > 0 Then AddStatBox sld, 9.1, 4.62, 3.45, 0.86, k2, HEX_WHITE, HEX_AMBER
    End If

    AddBand sld, ws, "B25", 0.55, 6.28, 12.25, 0.62, HEX_AMBER, HEX_NAVY
    AddFooter sld, ws, "B28", slideNo, 7.06, True
End Sub

Private Sub AddDarkPillar(ByVal sld As Object, ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single, ByVal rawText As String)
    Dim shp As Object, headLine As String, bodyText As String, nl As Long
    nl = InStr(rawText, Chr(10))
    If nl > 0 Then
        headLine = Left$(rawText, nl - 1)
        bodyText = Mid$(rawText, nl + 1)
    Else
        headLine = rawText: bodyText = ""
    End If
    Set shp = sld.Shapes.AddShape(msoShapeRectangle, Pt(x), PtY(y), PtW(w), PtH(h))
    shp.Fill.Solid
    shp.Fill.ForeColor.RGB = RGB(39, 53, 114)
    shp.Line.ForeColor.RGB = RGB(59, 74, 143)
    shp.Line.Weight = 0.75
    With shp.TextFrame
        .WordWrap = msoTrue
        .MarginLeft = 10: .MarginRight = 10: .MarginTop = 8: .MarginBottom = 8
        .TextRange.Text = headLine & Chr(11) & bodyText
    End With
    With shp.TextFrame.TextRange.Lines(1)
        .Font.Name = FONT_TITLE: .Font.Size = 13.5: .Font.Bold = msoTrue
        .Font.Color.RGB = HexColor(HEX_WHITE)
    End With
    If Len(bodyText) > 0 Then
        On Error Resume Next
        With shp.TextFrame.TextRange.Lines(2)
            .Font.Name = FONT_BODY: .Font.Size = 9.5
            .Font.Color.RGB = RGB(174, 184, 224)
        End With
        On Error GoTo 0
    End If
    AutoFitBoxFont shp, rawText, 9.5, 7.5
End Sub

'====================================================================================
' TEXT PARSING - our Excel cells store "LABEL<LF>VALUE<LF>note" (Kpi) or
' "N  Title   date<LF>body" (Card) - these split on the line-feed we wrote there.
'====================================================================================
Private Sub ParseKpiText(ByVal rawText As String, ByRef lbl As String, ByRef val_ As String, ByRef note As String)
    Dim parts() As String
    parts = Split(rawText, Chr(10))
    lbl = "": val_ = "": note = ""
    If UBound(parts) >= 0 Then lbl = parts(0)
    If UBound(parts) >= 1 Then val_ = parts(1)
    If UBound(parts) >= 2 Then note = parts(2)
End Sub

Private Function SafeCellText(ByVal ws As Worksheet, ByVal adr As String) As String
    On Error Resume Next
    SafeCellText = CStr(ws.Range(adr).Value)
    On Error GoTo 0
End Function

Private Function SheetExists(ByVal wb As Workbook, ByVal nm As String) As Boolean
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = wb.Worksheets(nm)
    SheetExists = Not ws Is Nothing
    On Error GoTo 0
End Function

'====================================================================================
' AUTO-FIT - step the font down rather than a fixed size, based on content length
'====================================================================================
Private Sub AutoFitTitleFont(ByVal shp As Object, ByVal titleText As String)
    On Error Resume Next
    Select Case Len(titleText)
        Case Is > 160: shp.TextFrame.TextRange.Font.Size = 20
        Case Is > 120: shp.TextFrame.TextRange.Font.Size = 24
        Case Is > 80: shp.TextFrame.TextRange.Font.Size = 28
        Case Else: shp.TextFrame.TextRange.Font.Size = 32
    End Select
    On Error GoTo 0
End Sub

Private Sub AutoFitBoxFont(ByVal shp As Object, ByVal fullText As String, ByVal normalSize As Single, ByVal smallSize As Single)
    On Error Resume Next
    If Len(fullText) > 220 Then
        Dim ln As Object
        For Each ln In shp.TextFrame.TextRange.Lines
            If ln.Font.Size > smallSize Then ln.Font.Size = ln.Font.Size - (normalSize - smallSize)
        Next ln
    End If
    On Error GoTo 0
End Sub
