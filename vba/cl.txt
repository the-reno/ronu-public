' USD CASH ALLOCATION | FULL 5-SLIDE RISK-MANAGEMENT DECK | EXCEL GENERATOR
' CURVE-ONLY SOURCE + EDITABLE CONFIG | NEUTRAL COLOR THEME
'
' RUN
'   Build_RM_Full_Deck
'
' REQUIRED SOURCE SHEET
'   Curve        A:E = Date | ON | 1M | 3M | 6M   (daily; any number of rows)
'
' CREATED IF MISSING (then USER-EDITABLE, never overwritten once present)
'   Deck_Config  A:B  CPI table   : Month (yyyy-mm) | CPI YoY %
'                D:E  THEME table : Element | Hex   (neutral palette, edit freely)
'   Premium      A:C  OPTIONAL. Date | Prem3M_bp | Prem6M_bp (desk premium vs market)
'                If absent, Slide 5's premium panel shows a placeholder note.
'
' OUTPUT (rebuilt on every run)
'   RM_1_History      historical curve, macro events, term premium
'   RM_2_Return        ON vs 6M realized return; regime breakdown; signal quality
'   RM_3_Inversion     inverted-curve mirror stress test
'   RM_4_Regret        five forward scenarios; regret cost; robust allocation
'   RM_5_Partnership   credentials + premium evidence
'   RM_Data            (hidden) all chart-source data for the five slides
'
' METHODOLOGY (identical to the analysis reviewed and approved slide-by-slide)
'   Historical returns   simple ACT/360; ON compounds daily; 6M is a single lock
'                         to maturity; six calendar months, floored to latest Curve date
'   Lift-off/plateau/cut  derived by rule on monthly-average ON (see Slide 1 facts)
'   Mirror convention     5.50% ON anchor; realized changes and their exact inverse
'   Forward engine        36 monthly periods; 5 scenario states (assumptions below);
'                          ON/1M roll monthly, 3M quarterly, 6M semiannually;
'                          286 weight mixes in 10% steps; minimax worst-case regret
'
' CONTROL
'   Curve is never modified. All RM_* / Deck_Config sheets are user- or run-owned
'   as documented above; Deck_Config and Premium are created once, then left alone.
'====================================================================================

'====================================================================================

Public Sub Build_RM_Full_Deck()
    Dim wb As Workbook
    Dim oldCalc As XlCalculation
    Set wb = ThisWorkbook

    If Not SheetExists(wb, SH_CURVE) Then
        MsgBox "Missing required sheet: " & SH_CURVE & vbCrLf & _
               "Expected columns A:E = Date | ON | 1M | 3M | 6M", vbCritical
        Exit Sub
    End If

    On Error GoTo Fail
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    oldCalc = Application.Calculation
    Application.Calculation = xlCalculationManual

    gStage = "Ensure Deck_Config":        EnsureConfig wb
    gStage = "Load Curve":                LoadCurve wb
    gStage = "Load forward assumptions":  LoadForwardAssumptions
    gStage = "Monthly averages":          BuildMonthly
    gStage = "Load CPI":                  LoadCPI wb
    gStage = "Slide 1 facts":             ComputeSlide1Facts
    gStage = "Daily evidence":            CalcDailyEvidence
    gStage = "Wealth series":             CalcWealthSeries
    gStage = "Mirror stress":             CalcMirrorStress
    gStage = "Forward allocation":        CalcForwardAllocation
    gStage = "Data sheet":                BuildDataSheet wb
    gStage = "Slide 1":                   BuildSlide1 wb
    gStage = "Slide 2":                   BuildSlide2 wb
    gStage = "Slide 3":                   BuildSlide3 wb
    gStage = "Slide 4":                   BuildSlide4 wb
    gStage = "Slide 5":                   BuildSlide5 wb

    wb.Worksheets(SH_DATA).Visible = xlSheetHidden
    Application.CalculateFull
    wb.Worksheets(S1).Activate
    ActiveWindow.DisplayGridlines = False

Done:
    On Error Resume Next
    Application.Calculation = oldCalc
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    On Error GoTo 0
    Exit Sub

Fail:
    MsgBox "Build failed." & vbCrLf & "Stage: " & gStage & vbCrLf & _
           "Error: " & Err.Number & " - " & Err.Description, vbCritical
    Resume Done
End Sub

Private Function ThemeColor(ByVal wb As Workbook, ByVal elementName As String, ByVal fallbackHex As String) As Long
    Dim ws As Worksheet, r As Long, hx As String
    hx = fallbackHex
    If SheetExists(wb, SH_CFG) Then
        Set ws = wb.Worksheets(SH_CFG)
        r = 4
        Do While Len(ws.Cells(r, 4).Value) > 0
            If StrComp(CStr(ws.Cells(r, 4).Value), elementName, vbTextCompare) = 0 Then
                hx = CStr(ws.Cells(r, 5).Value): Exit Do
            End If
            r = r + 1
        Loop
    End If
    hx = Replace(hx, "#", "")
    If Len(hx) <> 6 Then hx = fallbackHex
    ThemeColor = RGB(CLng("&H" & Mid$(hx, 1, 2)), CLng("&H" & Mid$(hx, 3, 2)), CLng("&H" & Mid$(hx, 5, 2)))
End Function

'====================================================================================
' LOAD CURVE (Date, ON, 1M, 3M, 6M)
'====================================================================================
Private Sub LoadCurve(ByVal wb As Workbook)
    Dim ws As Worksheet, lastRow As Long, v As Variant, i As Long, j As Long, k As Long
    Dim colMax(1 To 4) As Double, colScale(1 To 4) As Double, rawVal As Double
    Set ws = wb.Worksheets(SH_CURVE)
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    If lastRow < 30 Then Err.Raise vbObjectError + 5001, , "Curve needs at least ~30 daily rows."
    v = ws.Range("A2:E" & lastRow).Value2

    ReDim mDate(1 To UBound(v, 1))
    ReDim mRate(1 To 4, 1 To UBound(v, 1))

    ' Pass 1: find the largest magnitude in each column across the WHOLE curve.
    ' Scale is decided once per column (not per cell) so a near-zero-rate era
    ' (e.g. ON printed as 0.0100 meaning 0.01%) is never mistaken for a value
    ' already in decimal form - the column's own peak (e.g. 5.34 at the hiking
    ' cycle high) is what proves the column is in percent units.
    For j = 1 To 4: colMax(j) = 0#: Next j
    k = 0
    For i = 1 To UBound(v, 1)
        If IsNumeric(v(i, 1)) And IsNumeric(v(i, 2)) Then
            If CDbl(v(i, 1)) > 0 Then
                k = k + 1
                For j = 1 To 4
                    If IsNumeric(v(i, j + 1)) And Len(CStr(v(i, j + 1))) > 0 Then
                        rawVal = Abs(CDbl(v(i, j + 1)))
                        If rawVal > colMax(j) Then colMax(j) = rawVal
                    End If
                Next j
            End If
        End If
    Next i
    If k = 0 Then Err.Raise vbObjectError + 5003, , "No valid Date/ON rows found in Curve."
    For j = 1 To 4
        If colMax(j) > 0.5 Then colScale(j) = 100# Else colScale(j) = 1#
    Next j

    ' Pass 2: load using the per-column scale decided above.
    k = 0
    For i = 1 To UBound(v, 1)
        If IsNumeric(v(i, 1)) And IsNumeric(v(i, 2)) Then
            If CDbl(v(i, 1)) > 0 Then
                k = k + 1
                mDate(k) = CDbl(v(i, 1))
                For j = 1 To 4
                    If IsNumeric(v(i, j + 1)) And Len(CStr(v(i, j + 1))) > 0 Then
                        mRate(j, k) = CDbl(v(i, j + 1)) / colScale(j)
                        If j = 4 Then mHas6M = True
                    Else
                        mRate(j, k) = -1#
                    End If
                Next j
                If k > 1 Then
                    If mDate(k) <= mDate(k - 1) Then _
                        Err.Raise vbObjectError + 5002, , "Curve dates must be strictly increasing (row " & CStr(i + 1) & ")."
                End If
            End If
        End If
    Next i
    mCurveN = k
    ReDim Preserve mDate(1 To mCurveN)
    Dim tmp() As Double
    ReDim tmp(1 To 4, 1 To mCurveN)
    For i = 1 To mCurveN
        For j = 1 To 4
            tmp(j, i) = mRate(j, i)
        Next j
    Next i
    mRate = tmp
End Sub

Private Sub LoadForwardAssumptions()
    Dim scenario As Long, anchorNo As Long, tenor As Long
    Dim defaultAnchors As Variant, defaultPremiums As Variant, rowValues As Variant

    defaultAnchors = Array( _
        Array(0.0359, 0.0372, 0.0355, 0.0335, 0.0315), _
        Array(0.0425, 0.0425, 0.0425, 0.04, 0.0375), _
        Array(0.0425, 0.0475, 0.045, 0.0375, 0.034), _
        Array(0.03, 0.0275, 0.029, 0.0275, 0.029), _
        Array(0.015, 0.015, 0.02, 0.02, 0.025))
    defaultPremiums = Array( _
        Array(2#, 5#, 8#), Array(-3#, -5#, -10#), Array(-3#, -5#, -10#), _
        Array(2#, 4#, 7#), Array(2#, 5#, 10#))

    For scenario = 1 To 5
        rowValues = defaultAnchors(scenario - 1)
        For anchorNo = 1 To 5: mFwdAnchor(scenario, anchorNo) = CDbl(rowValues(anchorNo - 1)): Next anchorNo
        rowValues = defaultPremiums(scenario - 1)
        For tenor = 2 To 4: mFwdPremium(scenario, tenor) = CDbl(rowValues(tenor - 2)): Next tenor
        mFwdFade(scenario) = 6#
    Next scenario
End Sub

'====================================================================================
' SLIDE 1 FACTS: monthly ON averages, lift-off / plateau / cut, CPI
'====================================================================================
Private Sub BuildMonthly()
    Dim i As Long, key As Long, curKey As Long
    Dim tot As Double, cnt As Long, tot6 As Double, cnt6 As Long

    ReDim mMonKey(1 To mCurveN)
    ReDim mMonAvg(1 To mCurveN)
    ReDim mMon6(1 To mCurveN)
    mMonN = 0: curKey = 0

    For i = 1 To mCurveN
        key = Year(CDate(mDate(i))) * 100 + Month(CDate(mDate(i)))
        If key <> curKey Then
            If curKey <> 0 Then
                mMonN = mMonN + 1
                mMonKey(mMonN) = curKey
                mMonAvg(mMonN) = 100# * tot / cnt
                If cnt6 > 0 Then mMon6(mMonN) = 100# * tot6 / cnt6 Else mMon6(mMonN) = -1#
            End If
            curKey = key: tot = 0#: cnt = 0: tot6 = 0#: cnt6 = 0
        End If
        tot = tot + mRate(1, i): cnt = cnt + 1
        If mRate(4, i) >= 0# Then tot6 = tot6 + mRate(4, i): cnt6 = cnt6 + 1
    Next i
    mMonN = mMonN + 1
    mMonKey(mMonN) = curKey
    mMonAvg(mMonN) = 100# * tot / cnt
    If cnt6 > 0 Then mMon6(mMonN) = 100# * tot6 / cnt6 Else mMon6(mMonN) = -1#
End Sub

Private Sub LoadCPI(ByVal wb As Workbook)
    Dim ws As Worksheet, r As Long, sv As String, key As Long, n As Long, lastRow As Long
    Set ws = wb.Worksheets(SH_CFG)
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    ReDim mCpiKey(1 To lastRow)
    ReDim mCpiVal(1 To lastRow)
    n = 0
    r = 4
    Do While Len(ws.Cells(r, 1).Value) > 0
        sv = CStr(ws.Cells(r, 1).Value)
        If Len(sv) >= 7 And IsNumeric(ws.Cells(r, 2).Value) Then
            key = CLng(Left$(sv, 4)) * 100 + CLng(Mid$(sv, 6, 2))
            n = n + 1
            mCpiKey(n) = key
            mCpiVal(n) = CDbl(ws.Cells(r, 2).Value)
        End If
        r = r + 1
    Loop
    mCpiN = n
    If mCpiN = 0 Then Err.Raise vbObjectError + 5004, , "Deck_Config CPI table is empty."
End Sub

' Linear lookup helpers replacing Scripting.Dictionary (not available on Mac Excel).
Private Function CpiExists(ByVal key As Long) As Boolean
    Dim i As Long
    For i = 1 To mCpiN
        If mCpiKey(i) = key Then CpiExists = True: Exit Function
    Next i
    CpiExists = False
End Function

Private Function CpiValue(ByVal key As Long) As Double
    Dim i As Long
    For i = 1 To mCpiN
        If mCpiKey(i) = key Then CpiValue = mCpiVal(i): Exit Function
    Next i
    CpiValue = 0#
End Function

Private Function KeyLbl(ByVal key As Long) As String
    Dim mon As Variant
    mon = Array("Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec")
    KeyLbl = mon((key Mod 100) - 1) & "-" & Right$(CStr(key \ 100), 2)
End Function

Private Function MonthsBetween(ByVal k1 As Long, ByVal k2 As Long) As Long
    MonthsBetween = ((k2 \ 100) * 12 + (k2 Mod 100)) - ((k1 \ 100) * 12 + (k1 Mod 100))
End Function

Private Sub ComputeSlide1Facts()
    Dim i As Long, runMin As Double, best As Double, drop As Double, a As Long, b As Long

    fCur = mMonAvg(mMonN)
    fPeak = mMonAvg(1)
    For i = 2 To mMonN
        If mMonAvg(i) > fPeak Then fPeak = mMonAvg(i)
    Next i

    runMin = mMonAvg(1): fLiftIdx = 0
    For i = 1 To mMonN
        If mMonAvg(i) < runMin Then runMin = mMonAvg(i)
        If fLiftIdx = 0 And mMonAvg(i) > runMin + 0.1 Then fLiftIdx = i
    Next i
    If fLiftIdx = 0 Then fLiftIdx = 1

    ' fBase = the pre-hike floor (lowest monthly average up to and including
    ' lift-off) - not literally the curve's first observation, which may sit
    ' well above zero if Curve extends back before the low-rate era.
    fBase = mMonAvg(1)
    For i = 1 To fLiftIdx
        If mMonAvg(i) < fBase Then fBase = mMonAvg(i)
    Next i

    fPlatA = 0: fPlatB = 0
    For i = 1 To mMonN
        If mMonAvg(i) >= fPeak - 0.15 Then
            If fPlatA = 0 Then fPlatA = i
            fPlatB = i
        End If
    Next i
    fPlatMonths = fPlatB - fPlatA + 1

    fCutIdx = 0
    For i = fPlatA + 1 To mMonN
        If mMonAvg(i - 1) - mMonAvg(i) > 0.15 Then fCutIdx = i: Exit For
    Next i
    If fCutIdx = 0 Then fCutIdx = mMonN

    a = 0: b = 0
    For i = 1 To mCurveN
        If Year(CDate(mDate(i))) * 100 + Month(CDate(mDate(i))) = mMonKey(fCutIdx) Then
            If a = 0 Then a = i
            b = i
        End If
    Next i
    If a > 10 Then a = a - 10
    If b < mCurveN - 10 Then b = b + 10
    best = 0#
    For i = a To b - 5
        drop = mRate(1, i) - mRate(1, i + 5)
        If drop > best Then best = drop
    Next i
    fCutDailyBp = CLng(best * 10000#)

    fHikeBp = CLng((fPeak - fBase) * 100#)
    fHikeMonths = MonthsBetween(mMonKey(fLiftIdx), mMonKey(fPlatA))
    fCutsBp = CLng((fPeak - fCur) * 100#)

    If CpiExists(mMonKey(1)) Then fCpiStart = CpiValue(mMonKey(1))
    If CpiExists(mMonKey(fLiftIdx)) Then fCpiLift = CpiValue(mMonKey(fLiftIdx))
    If CpiExists(mMonKey(fCutIdx)) Then fCpiCut = CpiValue(mMonKey(fCutIdx))
    fCpiPeak = -999
    Dim ci As Long
    For ci = 1 To mCpiN
        If mCpiVal(ci) > fCpiPeak Then fCpiPeak = mCpiVal(ci): fCpiPeakKey = mCpiKey(ci)
        If mCpiKey(ci) > fCpiLastKey Then fCpiLastKey = mCpiKey(ci)
    Next ci
    fCpiLast = CpiValue(fCpiLastKey)

    If mHas6M Then
        If mMon6(mMonN) >= 0# Then
            f6Cur = mMon6(mMonN)
            fPremCurBp = CLng((mMon6(mMonN) - mMonAvg(mMonN)) * 100#)
        End If
        If mMon6(fLiftIdx) >= 0# Then fPremLiftBp = CLng((mMon6(fLiftIdx) - mMonAvg(fLiftIdx)) * 100#)
    End If
End Sub

'====================================================================================
' ENGINE HELPERS (shared floor/rate/path logic)
'====================================================================================
Private Function FloorIndex(ByVal targetDate As Double) As Long
    Dim lo As Long, hi As Long, mid As Long
    If targetDate <= mDate(1) Then FloorIndex = 1: Exit Function
    If targetDate >= mDate(mCurveN) Then FloorIndex = mCurveN: Exit Function
    lo = 1: hi = mCurveN
    Do While lo <= hi
        mid = (lo + hi) \ 2
        If mDate(mid) = targetDate Then FloorIndex = mid: Exit Function
        If mDate(mid) < targetDate Then lo = mid + 1 Else hi = mid - 1
    Loop
    FloorIndex = hi
End Function

Private Function TenorMonths(ByVal t As Long) As Long
    Select Case t
        Case 1: TenorMonths = 0
        Case 2: TenorMonths = 1
        Case 3: TenorMonths = 3
        Case 4: TenorMonths = 6
    End Select
End Function

Private Function TenorName(ByVal t As Long) As String
    Select Case t
        Case 1: TenorName = "ON"
        Case 2: TenorName = "1M"
        Case 3: TenorName = "3M"
        Case 4: TenorName = "6M"
    End Select
End Function

Private Function PathRate(ByVal tenorIdx As Long, ByVal obs As Long, ByVal pathCode As Long) As Double
    Dim r As Double, r0 As Double
    r = mRate(tenorIdx, obs)
    r0 = mRate(1, 1)
    Select Case pathCode
        Case 0: PathRate = r
        Case 1: PathRate = STRESS_ON + r - r0
        Case 2: PathRate = STRESS_ON + 2# * mRate(tenorIdx, 1) - r0 - r
    End Select
End Function

Private Function PathReturn(ByVal startIdx As Long, ByVal tenorIdx As Long, _
                            ByVal tenorMonths As Long, ByVal pathCode As Long) As Variant
    Dim startDate As Date, maturityDate As Date, endIdx As Long
    Dim curIdx As Long, nxtIdx As Long, nxtTarget As Date, factor As Double, k As Long, rr As Double

    startDate = CDate(mDate(startIdx))
    maturityDate = DateAdd("m", 6, startDate)
    If maturityDate > CDate(mDate(mCurveN)) Then PathReturn = Null: Exit Function
    endIdx = FloorIndex(CDbl(maturityDate))
    If endIdx <= startIdx Then PathReturn = Null: Exit Function
    factor = 1#

    If tenorMonths = 0 Then
        For curIdx = startIdx To endIdx - 1
            rr = PathRate(tenorIdx, curIdx, pathCode)
            factor = factor * (1# + rr * (mDate(curIdx + 1) - mDate(curIdx)) / 360#)
        Next curIdx
    Else
        rr = PathRate(tenorIdx, startIdx, pathCode)
        factor = factor * (1# + rr * (mDate(endIdx) - mDate(startIdx)) / 360#)
    End If
    PathReturn = factor - 1#
End Function

'====================================================================================
' SLIDE 2: DAILY EVIDENCE (realized ON vs 6M) + FULL-CURVE WEALTH SERIES
'====================================================================================
Private Sub CalcDailyEvidence()
    Dim i As Long, n As Long, lastD As Double, onR As Variant, sixR As Variant
    Dim mx As Double, my As Double, sxx As Double, syy As Double, sxy As Double
    Dim dx As Double, dy As Double, hits As Long

    If Not mHas6M Then Exit Sub
    lastD = mDate(mCurveN)
    For i = 1 To mCurveN
        If mRate(4, i) >= 0# Then
            If CDbl(DateAdd("m", 6, CDate(mDate(i)))) <= lastD Then n = n + 1
        End If
    Next i
    If n = 0 Then Exit Sub

    mDailyN = n
    ReDim mDailyIdx(1 To n): ReDim mDailySpread(1 To n): ReDim mDailyExcess(1 To n)
    ReDim mDailyOnRet(1 To n): ReDim mDaily6Ret(1 To n)

    n = 0
    For i = 1 To mCurveN
        If mRate(4, i) >= 0# Then
            If CDbl(DateAdd("m", 6, CDate(mDate(i)))) <= lastD Then
                onR = PathReturn(i, 1, 0, 0)
                sixR = PathReturn(i, 4, 6, 0)
                If Not IsNull(onR) And Not IsNull(sixR) Then
                    n = n + 1
                    mDailyIdx(n) = i
                    mDailySpread(n) = (mRate(4, i) - mRate(1, i)) * 10000#
                    mDailyOnRet(n) = onR
                    mDaily6Ret(n) = sixR
                    mDailyExcess(n) = (sixR - onR) * 10000#
                    mx = mx + mDailySpread(n): my = my + mDailyExcess(n)
                    If mDailyExcess(n) > 0# Then hits = hits + 1
                End If
            End If
        End If
    Next i
    mDailyN = n
    If mDailyN = 0 Then Exit Sub
    ReDim Preserve mDailyIdx(1 To mDailyN): ReDim Preserve mDailySpread(1 To mDailyN)
    ReDim Preserve mDailyExcess(1 To mDailyN): ReDim Preserve mDailyOnRet(1 To mDailyN)
    ReDim Preserve mDaily6Ret(1 To mDailyN)

    mx = mx / mDailyN: my = my / mDailyN
    For i = 1 To mDailyN
        dx = mDailySpread(i) - mx: dy = mDailyExcess(i) - my
        sxx = sxx + dx * dx: syy = syy + dy * dy: sxy = sxy + dx * dy
    Next i
    If sxx > 0# And syy > 0# Then mR2 = (sxy * sxy) / (sxx * syy)
    mHitRate = hits / mDailyN
    mMeanExcess = my

    mBestI = 1: mWorstI = 1
    For i = 2 To mDailyN
        If mDailyExcess(i) > mDailyExcess(mBestI) Then mBestI = i
        If mDailyExcess(i) < mDailyExcess(mWorstI) Then mWorstI = i
    Next i
End Sub

Private Sub CalcWealthSeries()
    Dim i As Long, lockRate As Double, lockEndI As Long
    If Not mHas6M Then Exit Sub
    ReDim mWealthOn(1 To mCurveN)
    ReDim mWealth6(1 To mCurveN)
    mWealthOn(1) = 1#: mWealth6(1) = 1#
    lockRate = mRate(4, 1)
    lockEndI = FloorIndex(CDbl(DateAdd("m", 6, CDate(mDate(1)))))
    For i = 2 To mCurveN
        mWealthOn(i) = mWealthOn(i - 1) * (1# + mRate(1, i - 1) * (mDate(i) - mDate(i - 1)) / 360#)
        mWealth6(i) = mWealth6(i - 1) * (1# + lockRate * (mDate(i) - mDate(i - 1)) / 360#)
        If i >= lockEndI And mRate(4, i) >= 0# Then
            lockRate = mRate(4, i)
            lockEndI = FloorIndex(CDbl(DateAdd("m", 6, CDate(mDate(i)))))
        End If
    Next i
End Sub

'====================================================================================
' SLIDE 3: MIRROR / INVERTED STRESS TEST
'====================================================================================
Private Sub CalcMirrorStress()
    Dim i As Long, hits As Long, tot As Double, onM As Variant, sixM As Variant

    ReDim mReanchor(1 To mCurveN)
    ReDim mMirrorLevel(1 To mCurveN)
    For i = 1 To mCurveN
        mReanchor(i) = PathRate(1, i, 1)
        mMirrorLevel(i) = PathRate(1, i, 2)
    Next i

    If Not mHas6M Then Exit Sub
    ReDim mMirrorExcess(1 To mDailyN)
    For i = 1 To mDailyN
        onM = PathReturn(mDailyIdx(i), 1, 0, 2)
        sixM = PathReturn(mDailyIdx(i), 4, 6, 2)
        If IsNull(onM) Or IsNull(sixM) Then
            mMirrorExcess(i) = 0#
        Else
            mMirrorExcess(i) = (sixM - onM) * 10000#
            tot = tot + mMirrorExcess(i)
            If mMirrorExcess(i) > 0# Then hits = hits + 1
        End If
    Next i
    mMirrorHit = hits / mDailyN
    mMirrorMean = tot / mDailyN
End Sub

' SLIDE 4: FORWARD SCENARIOS + MINIMAX-REGRET ALLOCATION

Private Function Max2(ByVal a As Double, ByVal b As Double) As Double
    If a > b Then Max2 = a Else Max2 = b
End Function

Private Function FwdONRate(ByVal s As Long, ByVal m As Long, ByVal startON As Double) As Double
    Dim a1 As Double, a2 As Double, a3 As Double, a4 As Double, a5 As Double
    a1 = mFwdAnchor(s, 1): a2 = mFwdAnchor(s, 2): a3 = mFwdAnchor(s, 3)
    a4 = mFwdAnchor(s, 4): a5 = mFwdAnchor(s, 5)
    If m <= 1 Then
        FwdONRate = startON + (a1 - startON) * m
    ElseIf m <= 7 Then
        FwdONRate = a1 + (a2 - a1) * (m - 1) / 6#
    ElseIf m <= 17 Then
        FwdONRate = a2 + (a3 - a2) * (m - 7) / 10#
    ElseIf m <= 29 Then
        FwdONRate = a3 + (a4 - a3) * (m - 17) / 12#
    Else
        FwdONRate = a4 + (a5 - a4) * (m - 29) / 7#
    End If
End Function

Private Function FwdAvgON(ByVal s As Long, ByVal startM As Long, ByVal ahead As Long) As Double
    Dim i As Long, fin As Long, tot As Double, cnt As Long
    fin = startM + ahead: If fin > 36 Then fin = 36
    For i = startM To fin: tot = tot + mFwdPath(s, 1, i): cnt = cnt + 1: Next i
    FwdAvgON = tot / cnt
End Function

Private Function RollMonths(ByVal t As Long) As Long
    Select Case t
        Case 1, 2: RollMonths = 1
        Case 3: RollMonths = 3
        Case 4: RollMonths = 6
    End Select
End Function

Private Function ScenarioName(ByVal s As Long) As String
    Select Case s
        Case 1: ScenarioName = "Gradual normalization"
        Case 2: ScenarioName = "Higher for longer"
        Case 3: ScenarioName = "Renewed inflation"
        Case 4: ScenarioName = "Faster normalization"
        Case 5: ScenarioName = "Recession / rapid cuts"
    End Select
End Function

Private Sub CalcForwardAllocation()
    Dim s As Long, t As Long, m As Long, mixNo As Long
    Dim u0 As Long, u1 As Long, u3 As Long, u6 As Long
    Dim fade As Double, baseV As Double, curV As Double, wealth As Double, stepM As Long
    Dim w(1 To 4) As Double, cumW As Double, ret As Double, regret As Double, worst As Double, avg As Double
    Dim bestWorst As Double, robustW(1 To 4) As Double, robustRegret(1 To 5) As Double
    Dim robustWorst As Double, robustAvg As Double, sRegret(1 To 5) As Double
    Dim onLast As Double, m1Last As Double, m3Last As Double, m6Last As Double

    onLast = mRate(1, mCurveN): m1Last = mRate(2, mCurveN)
    m3Last = mRate(3, mCurveN): m6Last = mRate(4, mCurveN)
    If m1Last < 0 Then m1Last = onLast
    If m3Last < 0 Then m3Last = onLast
    If m6Last < 0 Then m6Last = onLast

    For s = 1 To 5
        fade = mFwdFade(s)
        For m = 0 To 36: mFwdPath(s, 1, m) = FwdONRate(s, m, onLast): Next m

        baseV = mFwdPath(s, 1, 0) + mFwdPremium(s, 2) / 10000#
        For m = 0 To 36
            curV = mFwdPath(s, 1, m) + mFwdPremium(s, 2) / 10000#
            mFwdPath(s, 2, m) = Max2(0#, curV + (m1Last - baseV) * Max2(0#, 1# - m / fade))
        Next m

        baseV = 0.65 * mFwdPath(s, 1, 0) + 0.35 * FwdAvgON(s, 0, 2) + mFwdPremium(s, 3) / 10000#
        For m = 0 To 36
            curV = 0.65 * mFwdPath(s, 1, m) + 0.35 * FwdAvgON(s, m, 2) + mFwdPremium(s, 3) / 10000#
            mFwdPath(s, 3, m) = Max2(0#, curV + (m3Last - baseV) * Max2(0#, 1# - m / fade))
        Next m

        baseV = 0.5 * mFwdPath(s, 1, 0) + 0.5 * FwdAvgON(s, 0, 5) + mFwdPremium(s, 4) / 10000#
        For m = 0 To 36
            curV = 0.5 * mFwdPath(s, 1, m) + 0.5 * FwdAvgON(s, m, 5) + mFwdPremium(s, 4) / 10000#
            mFwdPath(s, 4, m) = Max2(0#, curV + (m6Last - baseV) * Max2(0#, 1# - m / fade))
        Next m
    Next s

    For s = 1 To 5
        mFwdBest(s) = -1E+99
        For t = 1 To 4
            wealth = 1#: stepM = RollMonths(t)
            For m = 0 To 35 Step stepM
                wealth = wealth * (1# + mFwdPath(s, t, m) * stepM / 12#)
            Next m
            mFwdReturn(s, t) = wealth ^ (1# / 3#) - 1#
            If mFwdReturn(s, t) > mFwdBest(s) Then mFwdBest(s) = mFwdReturn(s, t)
        Next t
    Next s

    bestWorst = 1E+99: mixNo = 0
    For u0 = 0 To 10
        For u1 = 0 To 10 - u0
            For u3 = 0 To 10 - u0 - u1
                u6 = 10 - u0 - u1 - u3
                mixNo = mixNo + 1
                w(1) = u0 / 10#: w(2) = u1 / 10#: w(3) = u3 / 10#: w(4) = u6 / 10#
                worst = -1E+99: avg = 0#
                For s = 1 To 5
                    cumW = 0#
                    For t = 1 To 4: cumW = cumW + w(t) * ((1# + mFwdReturn(s, t)) ^ 3# - 1#): Next t
                    ret = (1# + cumW) ^ (1# / 3#) - 1#
                    regret = (mFwdBest(s) - ret) * 10000#
                    sRegret(s) = regret
                    avg = avg + regret / 5#
                    If regret > worst Then worst = regret
                Next s
                If worst < bestWorst Then
                    bestWorst = worst: robustWorst = worst: robustAvg = avg
                    For t = 1 To 4: robustW(t) = w(t): Next t
                    For s = 1 To 5: robustRegret(s) = sRegret(s): Next s
                End If
                If u0 = 10 Then For s = 1 To 5: mON100Regret(s) = sRegret(s): Next s
                If u6 = 10 Then For s = 1 To 5: m6M100Regret(s) = sRegret(s): Next s
            Next u3
        Next u1
    Next u0

    For t = 1 To 4: mMix(t) = robustW(t): Next t
    For s = 1 To 5: mMixRegret(s) = robustRegret(s): Next s
    mMixWorstRegret = robustWorst: mMixAvgRegret = robustAvg
    mWeightedMonths = mMix(2) + 3# * mMix(3) + 6# * mMix(4)
End Sub

Private Function BestTenorOf(ByVal s As Long) As Long
    Dim t As Long, b As Long
    b = 1
    For t = 2 To 4: If mFwdReturn(s, t) > mFwdReturn(s, b) Then b = t
    Next t
    BestTenorOf = b
End Function

        .Merge
        .Value = num & "   " & ttl & Chr(10) & Chr(10) & body
        .WrapText = True: .VerticalAlignment = xlTop
        .Interior.Color = fill: .Borders.Color = borderC
        .Font.Name = "Arial": .Font.Size = 10: .Font.Color = accentC
        Dim hdr As String: hdr = num & "   " & ttl
        .Cells(1, 1).Characters(1, Len(hdr)).Font.Bold = True
        .Cells(1, 1).Characters(1, Len(hdr)).Font.Size = 13
        .Cells(1, 1).Characters(1, Len(hdr)).Font.Color = RGB(255, 255, 255)
        .Cells(1, 1).Characters(1, Len(hdr)).Font.Name = "Cambria"
    End With
End Sub

Private Function SheetExists(ByVal wb As Workbook, ByVal nm As String) As Boolean
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = wb.Worksheets(nm)
    SheetExists = Not ws Is Nothing
    On Error GoTo 0
End Function

Private Function NewSlideSheet(ByVal wb As Workbook, ByVal nm As String) As Worksheet
    Dim ws As Worksheet
    If SheetExists(wb, nm) Then wb.Worksheets(nm).Delete
    Set ws = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
    ws.Name = nm
    With ws
        .Columns("A:R").ColumnWidth = 7.2
        .Rows("1:40").RowHeight = 18
        .Rows(3).RowHeight = 30: .Rows(4).RowHeight = 30
        .Range("A1:R40").Font.Name = "Arial"
        With .PageSetup
            .Orientation = xlLandscape: .Zoom = False
            .FitToPagesWide = 1: .FitToPagesTall = 1
            .PrintArea = "$A$1:$R$40"
        End With
    End With
    Set NewSlideSheet = ws
End Function

Private Sub TxtBlock(ByVal ws As Worksheet, ByVal adr As String, ByVal sv As String, ByVal fs As Double, _
                     ByVal b As Boolean, ByVal fc As Long, ByVal fill As Long, ByVal ha As XlHAlign)
    With ws.Range(adr)
        .Merge: .Value = sv: .WrapText = True
        .Font.Name = "Arial": .Font.Size = fs: .Font.Bold = b: .Font.Color = fc
        .Interior.Color = fill: .HorizontalAlignment = ha: .VerticalAlignment = xlCenter
    End With
End Sub

Private Sub Card(ByVal ws As Worksheet, ByVal adr As String, ByVal num As String, ByVal ttl As String, _
                 ByVal dts As String, ByVal body As String, ByVal fill As Long, ByVal titleC As Long, _
                 ByVal bodyC As Long, ByVal borderC As Long, ByVal accC As Long, ByVal chevron As Boolean)
    Dim hdr As String
    hdr = num & "  " & ttl & "   " & dts
    With ws.Range(adr)
        .Merge
        .Value = hdr & Chr(10) & body & IIf(chevron, "  >", "")
        .WrapText = True: .VerticalAlignment = xlTop
        .Interior.Color = fill: .Borders.Color = borderC
        .Font.Name = "Arial": .Font.Size = 9: .Font.Color = bodyC
        .Cells(1, 1).Characters(1, Len(hdr)).Font.Bold = True
        .Cells(1, 1).Characters(1, Len(hdr)).Font.Size = 10.5
        .Cells(1, 1).Characters(1, Len(hdr)).Font.Color = titleC
        .Cells(1, 1).Characters(1, Len(num)).Font.Color = accC
    End With
End Sub

Private Sub Kpi(ByVal ws As Worksheet, ByVal adr As String, ByVal lbl As String, ByVal val As String, _
                ByVal note As String, ByVal fill As Long, ByVal titleC As Long, ByVal noteC As Long, ByVal valC As Long)
    Dim valueStart As Long
    valueStart = Len(UCase$(lbl)) + 2
    With ws.Range(adr)
        .Merge
        .Value = UCase$(lbl) & Chr(10) & val & Chr(10) & note
        .WrapText = True: .VerticalAlignment = xlCenter
        .Interior.Color = fill: .Borders.Color = noteC
        .Font.Name = "Arial": .Font.Size = 8.5: .Font.Color = noteC
        .Cells(1, 1).Characters(1, Len(lbl)).Font.Bold = True
        .Cells(1, 1).Characters(1, Len(lbl)).Font.Color = titleC
        .Cells(1, 1).Characters(valueStart, Len(val)).Font.Bold = True
        .Cells(1, 1).Characters(valueStart, Len(val)).Font.Size = 15
        .Cells(1, 1).Characters(valueStart, Len(val)).Font.Color = valC
    End With
End Sub

Private Sub PutRow(ByVal firstCell As Range, ByVal v As Variant)
    Dim i As Long
    For i = 0 To UBound(v): firstCell.Offset(0, i).Value = v(i): Next i
End Sub

Private Sub Tbl(ByVal r As Range, ByVal headC As Long, ByVal borderC As Long)
    With r
        .Borders.Color = borderC
        .Font.Name = "Arial": .Font.Size = 9: .VerticalAlignment = xlCenter
    End With
    With r.Rows(1)
        .Font.Bold = True: .Font.Color = RGB(255, 255, 255): .Interior.Color = headC
    End With
End Sub
