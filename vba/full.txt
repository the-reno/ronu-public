Option Explicit

'===============================================================================
' CFO CASH ALLOCATION | SINGLE MODULE
' Version 4.0
'
' ONE-CLICK WORKFLOW
'
'   Required source sheet only:
'       Curve
'
'   Required headers in row 1:
'       Date | ON | 1M | 3M | 6M
'
'   Run:
'       BuildCFOAllocationSlide
'
'   The macro automatically:
'       1. Reads the historical Curve sheet
'       2. Calibrates term / uncertainty behavior from historical data
'       3. Creates five 3-year market scenarios
'       4. Creates Curve_Scenarios
'       5. Calculates ON / 1M / 3M / 6M strategy returns
'       6. Tests all feasible maturity allocations
'       7. Calculates maximum regret and efficient frontier
'       8. Selects the robust allocation
'       9. Builds the CFO slide-ready layout
'
' OUTPUT SHEETS
'       Scenario_Inputs
'       Curve_Calibration
'       Curve_Scenarios
'       Allocation_Inputs
'       Strategy_Returns
'       Allocation_Model
'       Allocation_ChartData
'       Allocation_Slide
'
' IMPORTANT RATE CONVENTION
'   Curve may contain either:
'       4.00   = 4.00%
'   or
'       0.0400 = 4.00%
'
'===============================================================================

Private Const FIRST_DATA_ROW As Long = 2
Private Const SCENARIO_COUNT As Long = 5
Private Const FIRST_SCENARIO_ROW As Long = 4
Private Const SCENARIO_BLOCK_HEIGHT As Long = 11
Private Const HORIZON_MONTHS As Long = 36
Private Const MAX_PREMIUM_BP As Double = 100#

'===============================================================================
' MAIN
'===============================================================================

Public Sub BuildCFOAllocationSlide()

    Dim wsCurve As Worksheet
    Dim wsScenarioInputs As Worksheet
    Dim wsCal As Worksheet
    Dim wsCurves As Worksheet

    Dim wsInputs As Worksheet
    Dim wsReturns As Worksheet
    Dim wsModel As Worksheet
    Dim wsChartData As Worksheet
    Dim wsSlide As Worksheet

    Dim dateCol As Long
    Dim onCol As Long
    Dim m1Col As Long
    Dim m3Col As Long
    Dim m6Col As Long
    Dim lastRow As Long

    Dim modelLastRow As Long
    Dim recommendedRow As Long
    Dim minRegretRow As Long

    Dim stage As String

    On Error GoTo CleanFail

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    Application.Calculation = xlCalculationManual

    '---------------------------------------------------------------------------
    ' SOURCE
    '---------------------------------------------------------------------------

    stage = "Checking Curve source sheet"
    Set wsCurve = GetSheetIfExists("Curve")

    If wsCurve Is Nothing Then
        MsgBox "Required source sheet 'Curve' was not found." & vbCrLf & vbCrLf & _
               "Required headers in row 1:" & vbCrLf & _
               "Date | ON | 1M | 3M | 6M", vbExclamation
        GoTo CleanExit
    End If

    stage = "Reading Curve headers"

    dateCol = FindHeaderColumn(wsCurve, "Date")
    onCol = FindHeaderColumn(wsCurve, "ON")
    m1Col = FindHeaderColumn(wsCurve, "1M")
    m3Col = FindHeaderColumn(wsCurve, "3M")
    m6Col = FindHeaderColumn(wsCurve, "6M")

    If dateCol = 0 Or onCol = 0 Or m1Col = 0 Or m3Col = 0 Or m6Col = 0 Then
        MsgBox "Curve must contain these headers in row 1:" & vbCrLf & _
               "Date | ON | 1M | 3M | 6M", vbExclamation
        GoTo CleanExit
    End If

    lastRow = wsCurve.Cells(wsCurve.Rows.Count, dateCol).End(xlUp).Row

    If lastRow < 30 Then
        MsgBox "Curve does not contain enough historical observations." & vbCrLf & _
               "At least 30 data rows are recommended.", vbExclamation
        GoTo CleanExit
    End If

    '---------------------------------------------------------------------------
    ' CREATE OUTPUT SHEETS
    '---------------------------------------------------------------------------

    stage = "Creating scenario sheets"

    Set wsScenarioInputs = GetOrCreateSheet("Scenario_Inputs")
    Set wsCal = GetOrCreateSheet("Curve_Calibration")
    Set wsCurves = GetOrCreateSheet("Curve_Scenarios")

    stage = "Creating allocation sheets"

    Set wsInputs = GetOrCreateSheet("Allocation_Inputs")
    Set wsReturns = GetOrCreateSheet("Strategy_Returns")
    Set wsModel = GetOrCreateSheet("Allocation_Model")
    Set wsChartData = GetOrCreateSheet("Allocation_ChartData")
    Set wsSlide = GetOrCreateSheet("Allocation_Slide")

    '---------------------------------------------------------------------------
    ' BUILD SCENARIOS
    '---------------------------------------------------------------------------

    stage = "Setting scenario inputs"
    SetupScenarioInputs wsScenarioInputs

    stage = "Calibrating term premium"
    wsCal.Cells.Clear

    BuildCalibration _
        wsCurve, wsCal, _
        dateCol, onCol, m1Col, m3Col, m6Col, lastRow

    stage = "Building 3-year scenario curves"
    wsCurves.Cells.Clear

    BuildScenarioCurves _
        wsCurve, wsScenarioInputs, wsCal, wsCurves, _
        dateCol, onCol, m1Col, m3Col, m6Col, lastRow

    '---------------------------------------------------------------------------
    ' BUILD ALLOCATION
    '---------------------------------------------------------------------------

    stage = "Setting allocation inputs"
    SetupAllocationInputs wsInputs

    stage = "Clearing previous allocation outputs"
    wsReturns.Cells.Clear
    wsModel.Cells.Clear
    wsChartData.Cells.Clear
    ClearSlide wsSlide

    stage = "Calculating pure strategy returns"
    BuildStrategyReturns wsCurves, wsReturns

    stage = "Building portfolio universe"
    modelLastRow = BuildAllocationModel(wsInputs, wsReturns, wsModel)

    stage = "Building efficient frontier"
    MarkEfficientFrontier wsModel, modelLastRow

    stage = "Finding minimum-regret portfolio"
    minRegretRow = FindMinimumRegretRow(wsModel, modelLastRow)

    stage = "Selecting robust allocation"
    recommendedRow = SelectRecommendedPortfolio( _
                        wsInputs, wsModel, modelLastRow, minRegretRow)

    If recommendedRow < 2 Then
        Err.Raise vbObjectError + 900, , _
            "No valid recommended allocation was found."
    End If

    stage = "Preparing chart data"
    BuildChartData _
        wsReturns, wsModel, wsChartData, _
        modelLastRow, recommendedRow, minRegretRow

    stage = "Building CFO slide"
    BuildSelectedSlideLayout _
        wsInputs, wsReturns, wsModel, wsChartData, wsSlide, _
        recommendedRow, minRegretRow

    '---------------------------------------------------------------------------
    ' FORMAT SUPPORT SHEETS
    '---------------------------------------------------------------------------

    stage = "Formatting output sheets"

    FormatScenarioInputs wsScenarioInputs
    FormatCalibration wsCal
    FormatScenarioCurves wsCurves
    FormatAllocationInputs wsInputs
    FormatStrategyReturns wsReturns
    FormatModel wsModel
    FormatChartData wsChartData

    wsSlide.Activate

    On Error Resume Next
    ActiveWindow.DisplayGridlines = False
    ActiveWindow.Zoom = 80
    On Error GoTo CleanFail

CleanExit:

    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    Exit Sub

CleanFail:

    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox "CFO allocation builder error." & vbCrLf & vbCrLf & _
           "Stage: " & stage & vbCrLf & _
           "Error " & Err.Number & ": " & Err.Description, vbExclamation

End Sub

'===============================================================================
' PART 2 - HISTORICAL TERM / UNCERTAINTY CALIBRATION
'
' Historical residual:
'
'   observed term rate
'   - subsequently realized average ON over the tenor
'
' Regression:
'
'   residual bp = Alpha + Beta x Abs(future ON move bp)
'
' Alpha and Beta are floored at zero for the scenario engine.
'===============================================================================

Private Sub BuildCalibration( _
    ByVal wsCurve As Worksheet, _
    ByVal wsCal As Worksheet, _
    ByVal dateCol As Long, _
    ByVal onCol As Long, _
    ByVal m1Col As Long, _
    ByVal m3Col As Long, _
    ByVal m6Col As Long, _
    ByVal lastRow As Long)

    wsCal.Range("A1:H1").Merge
    wsCal.Range("A1").Value = "Historical Term / Uncertainty Calibration"

    wsCal.Range("A3").Value = "Tenor"
    wsCal.Range("B3").Value = "Days"
    wsCal.Range("C3").Value = "Raw Alpha bp"
    wsCal.Range("D3").Value = "Raw Beta"
    wsCal.Range("E3").Value = "Used Alpha bp"
    wsCal.Range("F3").Value = "Used Beta"
    wsCal.Range("G3").Value = "Observations"
    wsCal.Range("H3").Value = "Interpretation"

    CalibrateOneTenor _
        wsCurve, wsCal, 4, "1M", 30, _
        dateCol, onCol, m1Col, lastRow

    CalibrateOneTenor _
        wsCurve, wsCal, 5, "3M", 90, _
        dateCol, onCol, m3Col, lastRow

    CalibrateOneTenor _
        wsCurve, wsCal, 6, "6M", 180, _
        dateCol, onCol, m6Col, lastRow

    wsCal.Range("A8:H8").Merge
    wsCal.Range("A8").Value = _
        "Scenario term rate = expected average ON + historically calibrated compensation."

End Sub

Private Sub CalibrateOneTenor( _
    ByVal wsCurve As Worksheet, _
    ByVal wsCal As Worksheet, _
    ByVal outputRow As Long, _
    ByVal tenorName As String, _
    ByVal tenorDays As Long, _
    ByVal dateCol As Long, _
    ByVal onCol As Long, _
    ByVal termCol As Long, _
    ByVal lastRow As Long)

    Dim r As Long
    Dim obsDate As Date

    Dim currentON As Double
    Dim termRate As Double
    Dim futureAvgON As Double
    Dim futureON As Double

    Dim x As Double
    Dim y As Double

    Dim n As Long
    Dim sumX As Double
    Dim sumY As Double
    Dim sumXX As Double
    Dim sumXY As Double

    Dim meanX As Double
    Dim meanY As Double
    Dim denom As Double

    Dim rawAlpha As Double
    Dim rawBeta As Double
    Dim usedAlpha As Double
    Dim usedBeta As Double

    For r = FIRST_DATA_ROW To lastRow

        If IsDate(wsCurve.Cells(r, dateCol).Value) _
           And IsNumeric(wsCurve.Cells(r, onCol).Value) _
           And IsNumeric(wsCurve.Cells(r, termCol).Value) Then

            obsDate = CDate(wsCurve.Cells(r, dateCol).Value)

            If obsDate + tenorDays <= CDate(wsCurve.Cells(lastRow, dateCol).Value) Then

                currentON = NormalizeRate(wsCurve.Cells(r, onCol).Value)
                termRate = NormalizeRate(wsCurve.Cells(r, termCol).Value)

                futureAvgON = AverageONBetweenDates( _
                                wsCurve, _
                                obsDate, _
                                obsDate + tenorDays, _
                                dateCol, onCol, lastRow)

                futureON = ONAtOrAfterDate( _
                                wsCurve, _
                                obsDate + tenorDays, _
                                dateCol, onCol, lastRow)

                If futureAvgON > -900 And futureON > -900 Then

                    x = Abs(futureON - currentON) * 100#
                    y = (termRate - futureAvgON) * 100#

                    n = n + 1
                    sumX = sumX + x
                    sumY = sumY + y
                    sumXX = sumXX + x * x
                    sumXY = sumXY + x * y

                End If

            End If

        End If

    Next r

    If n >= 10 Then

        meanX = sumX / n
        meanY = sumY / n

        denom = sumXX - n * meanX * meanX

        If Abs(denom) > 0.0000001 Then
            rawBeta = (sumXY - n * meanX * meanY) / denom
        Else
            rawBeta = 0
        End If

        rawAlpha = meanY - rawBeta * meanX

    Else

        rawAlpha = 0
        rawBeta = 0

    End If

    usedAlpha = rawAlpha
    usedBeta = rawBeta

    If usedAlpha < 0 Then usedAlpha = 0
    If usedBeta < 0 Then usedBeta = 0

    wsCal.Cells(outputRow, 1).Value = tenorName
    wsCal.Cells(outputRow, 2).Value = tenorDays
    wsCal.Cells(outputRow, 3).Value = rawAlpha
    wsCal.Cells(outputRow, 4).Value = rawBeta
    wsCal.Cells(outputRow, 5).Value = usedAlpha
    wsCal.Cells(outputRow, 6).Value = usedBeta
    wsCal.Cells(outputRow, 7).Value = n

    wsCal.Cells(outputRow, 8).Value = _
        "Premium = " & Format(usedAlpha, "0.0") & _
        " bp + " & Format(usedBeta * 100, "0.0") & _
        " bp per 100 bp expected ON move"

End Sub

'===============================================================================
' PART 3 - CREATE FIVE 3-YEAR CURVES
'===============================================================================

Private Sub BuildScenarioCurves( _
    ByVal wsCurve As Worksheet, _
    ByVal wsIn As Worksheet, _
    ByVal wsCal As Worksheet, _
    ByVal wsOut As Worksheet, _
    ByVal dateCol As Long, _
    ByVal onCol As Long, _
    ByVal m1Col As Long, _
    ByVal m3Col As Long, _
    ByVal m6Col As Long, _
    ByVal lastRow As Long)

    Dim currentON As Double
    Dim current1M As Double
    Dim current3M As Double
    Dim current6M As Double

    Dim alpha1 As Double
    Dim beta1 As Double
    Dim alpha3 As Double
    Dim beta3 As Double
    Dim alpha6 As Double
    Dim beta6 As Double

    Dim scenarioRow As Long
    Dim outputStartRow As Long

    currentON = NormalizeRate(wsCurve.Cells(lastRow, onCol).Value)
    current1M = NormalizeRate(wsCurve.Cells(lastRow, m1Col).Value)
    current3M = NormalizeRate(wsCurve.Cells(lastRow, m3Col).Value)
    current6M = NormalizeRate(wsCurve.Cells(lastRow, m6Col).Value)

    alpha1 = CDbl(wsCal.Range("E4").Value)
    beta1 = CDbl(wsCal.Range("F4").Value)

    alpha3 = CDbl(wsCal.Range("E5").Value)
    beta3 = CDbl(wsCal.Range("F5").Value)

    alpha6 = CDbl(wsCal.Range("E6").Value)
    beta6 = CDbl(wsCal.Range("F6").Value)

    wsOut.Range("A1:F1").Merge
    wsOut.Range("A1").Value = "3-Year Scenario Curves"

    wsOut.Range("A2:F2").Merge
    wsOut.Range("A2").Value = _
        "Term rates reflect expected ON over the tenor plus historical uncertainty compensation."

    outputStartRow = FIRST_SCENARIO_ROW

    For scenarioRow = 6 To 10

        WriteOneScenario _
            wsIn, wsOut, scenarioRow, outputStartRow, _
            currentON, current1M, current3M, current6M, _
            alpha1, beta1, alpha3, beta3, alpha6, beta6

        outputStartRow = outputStartRow + SCENARIO_BLOCK_HEIGHT

    Next scenarioRow

End Sub

Private Sub WriteOneScenario( _
    ByVal wsIn As Worksheet, _
    ByVal wsOut As Worksheet, _
    ByVal scenarioRow As Long, _
    ByVal startRow As Long, _
    ByVal currentON As Double, _
    ByVal current1M As Double, _
    ByVal current3M As Double, _
    ByVal current6M As Double, _
    ByVal alpha1 As Double, _
    ByVal beta1 As Double, _
    ByVal alpha3 As Double, _
    ByVal beta3 As Double, _
    ByVal alpha6 As Double, _
    ByVal beta6 As Double)

    Dim scenarioName As String

    Dim move6 As Double
    Dim move12 As Double
    Dim move24 As Double
    Dim move36 As Double

    Dim i As Long
    Dim monthNode As Double

    Dim onRate As Double
    Dim r1 As Double
    Dim r3 As Double
    Dim r6 As Double

    scenarioName = CStr(wsIn.Cells(scenarioRow, 1).Value)

    move6 = CDbl(wsIn.Cells(scenarioRow, 2).Value)
    move12 = CDbl(wsIn.Cells(scenarioRow, 3).Value)
    move24 = CDbl(wsIn.Cells(scenarioRow, 4).Value)
    move36 = CDbl(wsIn.Cells(scenarioRow, 5).Value)

    wsOut.Cells(startRow, 1).Value = scenarioName

    wsOut.Cells(startRow + 1, 1).Value = "Horizon"
    wsOut.Cells(startRow + 1, 2).Value = "Month"
    wsOut.Cells(startRow + 1, 3).Value = "ON"
    wsOut.Cells(startRow + 1, 4).Value = "1M"
    wsOut.Cells(startRow + 1, 5).Value = "3M"
    wsOut.Cells(startRow + 1, 6).Value = "6M"

    For i = 0 To 6

        monthNode = i * 6

        wsOut.Cells(startRow + 2 + i, 1).Value = HorizonLabel(monthNode)
        wsOut.Cells(startRow + 2 + i, 2).Value = monthNode

        onRate = ScenarioExpectedON( _
                    currentON, _
                    move6, move12, move24, move36, _
                    monthNode)

        If monthNode = 0 Then

            r1 = current1M
            r3 = current3M
            r6 = current6M

        Else

            r1 = ScenarioTermRate( _
                    currentON, _
                    move6, move12, move24, move36, _
                    monthNode, 1, alpha1, beta1)

            r3 = ScenarioTermRate( _
                    currentON, _
                    move6, move12, move24, move36, _
                    monthNode, 3, alpha3, beta3)

            r6 = ScenarioTermRate( _
                    currentON, _
                    move6, move12, move24, move36, _
                    monthNode, 6, alpha6, beta6)

        End If

        wsOut.Cells(startRow + 2 + i, 3).Value = onRate
        wsOut.Cells(startRow + 2 + i, 4).Value = r1
        wsOut.Cells(startRow + 2 + i, 5).Value = r3
        wsOut.Cells(startRow + 2 + i, 6).Value = r6

    Next i

End Sub

Private Function ScenarioTermRate( _
    ByVal currentON As Double, _
    ByVal move6 As Double, _
    ByVal move12 As Double, _
    ByVal move24 As Double, _
    ByVal move36 As Double, _
    ByVal currentMonth As Double, _
    ByVal tenorMonths As Long, _
    ByVal alphaBP As Double, _
    ByVal beta As Double) As Double

    Dim expectedAverage As Double
    Dim expectedEndON As Double
    Dim onAtStart As Double

    Dim expectedMoveBP As Double
    Dim premiumBP As Double

    expectedAverage = ScenarioExpectedAverageON( _
                        currentON, _
                        move6, move12, move24, move36, _
                        currentMonth, tenorMonths)

    onAtStart = ScenarioExpectedON( _
                    currentON, _
                    move6, move12, move24, move36, _
                    currentMonth)

    expectedEndON = ScenarioExpectedON( _
                        currentON, _
                        move6, move12, move24, move36, _
                        currentMonth + tenorMonths)

    expectedMoveBP = Abs(expectedEndON - onAtStart) * 100#

    premiumBP = alphaBP + beta * expectedMoveBP

    If premiumBP < 0 Then premiumBP = 0
    If premiumBP > MAX_PREMIUM_BP Then premiumBP = MAX_PREMIUM_BP

    ScenarioTermRate = expectedAverage + premiumBP / 100#

End Function

Private Function ScenarioExpectedAverageON( _
    ByVal currentON As Double, _
    ByVal move6 As Double, _
    ByVal move12 As Double, _
    ByVal move24 As Double, _
    ByVal move36 As Double, _
    ByVal startMonth As Double, _
    ByVal tenorMonths As Long) As Double

    Dim k As Long
    Dim sampleMonth As Double
    Dim totalRate As Double

    If tenorMonths <= 0 Then

        ScenarioExpectedAverageON = ScenarioExpectedON( _
                                        currentON, _
                                        move6, move12, move24, move36, _
                                        startMonth)
        Exit Function

    End If

    For k = 1 To tenorMonths

        sampleMonth = startMonth + k - 0.5

        totalRate = totalRate + ScenarioExpectedON( _
                                    currentON, _
                                    move6, move12, move24, move36, _
                                    sampleMonth)

    Next k

    ScenarioExpectedAverageON = totalRate / tenorMonths

End Function

Private Function ScenarioExpectedON( _
    ByVal currentON As Double, _
    ByVal move6 As Double, _
    ByVal move12 As Double, _
    ByVal move24 As Double, _
    ByVal move36 As Double, _
    ByVal monthNumber As Double) As Double

    Dim cumulativeMoveBP As Double

    If monthNumber <= 0 Then

        cumulativeMoveBP = 0

    ElseIf monthNumber <= 6 Then

        cumulativeMoveBP = LinearInterpolate( _
                                monthNumber, _
                                0, 0, _
                                6, move6)

    ElseIf monthNumber <= 12 Then

        cumulativeMoveBP = LinearInterpolate( _
                                monthNumber, _
                                6, move6, _
                                12, move12)

    ElseIf monthNumber <= 24 Then

        cumulativeMoveBP = LinearInterpolate( _
                                monthNumber, _
                                12, move12, _
                                24, move24)

    ElseIf monthNumber <= 36 Then

        cumulativeMoveBP = LinearInterpolate( _
                                monthNumber, _
                                24, move24, _
                                36, move36)

    Else

        cumulativeMoveBP = move36

    End If

    ScenarioExpectedON = currentON + cumulativeMoveBP / 100#

End Function

'===============================================================================
' PART 5 - PURE STRATEGY RETURNS
'===============================================================================

Private Sub BuildStrategyReturns( _
    ByVal wsCurves As Worksheet, _
    ByVal wsReturns As Worksheet)

    Dim i As Long
    Dim scenarioStartRow As Long
    Dim outRow As Long
    Dim scenarioName As String

    Dim rON As Double
    Dim r1M As Double
    Dim r3M As Double
    Dim r6M As Double

    Dim bestReturn As Double
    Dim bestStrategy As String

    wsReturns.Range("A1:G1").Merge
    wsReturns.Range("A1").Value = "Pure Strategy Returns by Scenario"

    wsReturns.Range("A3").Value = "Scenario"
    wsReturns.Range("B3").Value = "ON"
    wsReturns.Range("C3").Value = "1M"
    wsReturns.Range("D3").Value = "3M"
    wsReturns.Range("E3").Value = "6M"
    wsReturns.Range("F3").Value = "Best ex-post"
    wsReturns.Range("G3").Value = "Best pure strategy"

    For i = 0 To SCENARIO_COUNT - 1

        scenarioStartRow = FIRST_SCENARIO_ROW + i * SCENARIO_BLOCK_HEIGHT
        outRow = 4 + i

        scenarioName = Trim$(CStr(wsCurves.Cells(scenarioStartRow, 1).Value))

        If Len(scenarioName) = 0 Then
            Err.Raise vbObjectError + 901, , _
                "Missing scenario name at row " & scenarioStartRow & "."
        End If

        ValidateScenarioBlock wsCurves, scenarioStartRow

        rON = AnnualizedStrategyReturn(wsCurves, scenarioStartRow, 3, 1)
        r1M = AnnualizedStrategyReturn(wsCurves, scenarioStartRow, 4, 1)
        r3M = AnnualizedStrategyReturn(wsCurves, scenarioStartRow, 5, 3)
        r6M = AnnualizedStrategyReturn(wsCurves, scenarioStartRow, 6, 6)

        bestReturn = rON
        bestStrategy = "ON"

        If r1M > bestReturn Then
            bestReturn = r1M
            bestStrategy = "1M"
        End If

        If r3M > bestReturn Then
            bestReturn = r3M
            bestStrategy = "3M"
        End If

        If r6M > bestReturn Then
            bestReturn = r6M
            bestStrategy = "6M"
        End If

        wsReturns.Cells(outRow, 1).Value = scenarioName
        wsReturns.Cells(outRow, 2).Value = rON
        wsReturns.Cells(outRow, 3).Value = r1M
        wsReturns.Cells(outRow, 4).Value = r3M
        wsReturns.Cells(outRow, 5).Value = r6M
        wsReturns.Cells(outRow, 6).Value = bestReturn

Private Sub ValidateScenarioBlock( _
    ByVal ws As Worksheet, _
    ByVal scenarioStartRow As Long)

    Dim dataStart As Long
    Dim r As Long
    Dim c As Long

    dataStart = scenarioStartRow + 2

    For r = dataStart To dataStart + 6

        If Not IsNumeric(ws.Cells(r, 2).Value) Then
            Err.Raise vbObjectError + 902, , _
                "Non-numeric month in Curve_Scenarios row " & r & "."
        End If

        For c = 3 To 6

            If Not IsNumeric(ws.Cells(r, c).Value) Then
                Err.Raise vbObjectError + 903, , _
                    "Non-numeric rate in Curve_Scenarios cell " & _
                    ws.Cells(r, c).Address(False, False) & "."
            End If

        Next c

    Next r

End Sub

Private Function AnnualizedStrategyReturn( _
    ByVal wsCurves As Worksheet, _
    ByVal scenarioStartRow As Long, _
    ByVal tenorCol As Long, _
    ByVal rolloverMonths As Long) As Double

    Dim m As Long
    Dim periodMonths As Long
    Dim ratePct As Double

    Dim wealth As Double
    Dim years As Double

    If rolloverMonths <= 0 Then rolloverMonths = 1

    wealth = 1#
    m = 0

    Do While m < HORIZON_MONTHS

        periodMonths = rolloverMonths

        If m + periodMonths > HORIZON_MONTHS Then
            periodMonths = HORIZON_MONTHS - m
        End If

        ratePct = RateAtMonth( _
                    wsCurves, scenarioStartRow, tenorCol, CDbl(m))

        wealth = wealth * _
            (1# + (ratePct / 100#) * (periodMonths / 12#))

        m = m + periodMonths

    Loop

    years = HORIZON_MONTHS / 12#

    If wealth <= 0 Then
        AnnualizedStrategyReturn = -100
    Else
        AnnualizedStrategyReturn = _
            (wealth ^ (1# / years) - 1#) * 100#
    End If

End Function

Private Function RateAtMonth( _
    ByVal wsCurves As Worksheet, _
    ByVal scenarioStartRow As Long, _
    ByVal tenorCol As Long, _
    ByVal targetMonth As Double) As Double

    Dim firstDataRow As Long
    Dim r As Long

    Dim m1 As Double
    Dim m2 As Double
    Dim y1 As Double
    Dim y2 As Double

    firstDataRow = scenarioStartRow + 2

    If targetMonth <= 0 Then

        RateAtMonth = CDbl(wsCurves.Cells(firstDataRow, tenorCol).Value)
        Exit Function

    End If

    If targetMonth >= 36 Then

        RateAtMonth = CDbl(wsCurves.Cells(firstDataRow + 6, tenorCol).Value)
        Exit Function

    End If

    For r = firstDataRow To firstDataRow + 5

        m1 = CDbl(wsCurves.Cells(r, 2).Value)
        m2 = CDbl(wsCurves.Cells(r + 1, 2).Value)

        If targetMonth >= m1 And targetMonth <= m2 Then

            y1 = CDbl(wsCurves.Cells(r, tenorCol).Value)
            y2 = CDbl(wsCurves.Cells(r + 1, tenorCol).Value)

            RateAtMonth = LinearInterpolate( _
                            targetMonth, m1, y1, m2, y2)

            Exit Function

        End If

    Next r

    Err.Raise vbObjectError + 904, , _
        "Unable to interpolate scenario rate at month " & targetMonth & "."

End Function

Private Function BuildAllocationModel( _
    ByVal wsInputs As Worksheet, _
    ByVal wsReturns As Worksheet, _
    ByVal wsModel As Worksheet) As Long

    Dim stepPct As Double
    Dim steps As Long

    Dim a As Long
    Dim b As Long
    Dim c As Long
    Dim d As Long

    Dim wON As Double
    Dim w1M As Double
    Dim w3M As Double
    Dim w6M As Double

    Dim rowOut As Long
    Dim idNum As Long

    Dim minON As Double
    Dim max6M As Double

    stepPct = CDbl(wsInputs.Range("B4").Value)

    If stepPct <= 0 Then stepPct = 5
    If stepPct > 25 Then stepPct = 25

    steps = CLng(100# / stepPct)

    If Abs(steps * stepPct - 100#) > 0.0001 Then

        Err.Raise vbObjectError + 905, , _
            "Allocation step must divide 100 exactly. Use 5, 10, 20 or 25."

    End If

    minON = CDbl(wsInputs.Range("B6").Value) / 100#
    max6M = CDbl(wsInputs.Range("B7").Value) / 100#

    WriteModelHeaders wsModel

    rowOut = 2
    idNum = 1

    For a = 0 To steps

        For b = 0 To steps - a

            For c = 0 To steps - a - b

                d = steps - a - b - c

                wON = a / steps
                w1M = b / steps
                w3M = c / steps
                w6M = d / steps

                WritePortfolioRow _
                    wsReturns, wsModel, rowOut, idNum, _
                    wON, w1M, w3M, w6M, minON, max6M

                rowOut = rowOut + 1
                idNum = idNum + 1

            Next c

        Next b

    Next a

    BuildAllocationModel = rowOut - 1

End Function

Private Sub WritePortfolioRow( _
    ByVal wsReturns As Worksheet, _
    ByVal wsModel As Worksheet, _
    ByVal rowOut As Long, _
    ByVal idNum As Long, _
    ByVal wON As Double, _
    ByVal w1M As Double, _
    ByVal w3M As Double, _
    ByVal w6M As Double, _
    ByVal minON As Double, _
    ByVal max6M As Double)

    Dim s As Long
    Dim retRow As Long

    Dim pureON As Double
    Dim pure1M As Double
    Dim pure3M As Double
    Dim pure6M As Double
    Dim bestReturn As Double

    Dim wealthON As Double
    Dim wealth1M As Double
    Dim wealth3M As Double
    Dim wealth6M As Double

    Dim portfolioWealth As Double
    Dim portfolioReturn As Double

    Dim regretBP As Double
    Dim excessBP As Double

    Dim sumExcessBP As Double
    Dim maxRegretBP As Double
    Dim worstExcessBP As Double

    Dim hitCount As Long
    Dim validFlag As Long

    wsModel.Cells(rowOut, 1).Value = idNum
    wsModel.Cells(rowOut, 2).Value = wON
    wsModel.Cells(rowOut, 3).Value = w1M
    wsModel.Cells(rowOut, 4).Value = w3M
    wsModel.Cells(rowOut, 5).Value = w6M

    maxRegretBP = -1
    worstExcessBP = 1E+30
    sumExcessBP = 0
    hitCount = 0

    For s = 1 To SCENARIO_COUNT

        retRow = 3 + s

        pureON = CDbl(wsReturns.Cells(retRow, 2).Value)
        pure1M = CDbl(wsReturns.Cells(retRow, 3).Value)
        pure3M = CDbl(wsReturns.Cells(retRow, 4).Value)
        pure6M = CDbl(wsReturns.Cells(retRow, 5).Value)

        bestReturn = CDbl(wsReturns.Cells(retRow, 6).Value)

        wealthON = (1# + pureON / 100#) ^ 3#
        wealth1M = (1# + pure1M / 100#) ^ 3#
        wealth3M = (1# + pure3M / 100#) ^ 3#
        wealth6M = (1# + pure6M / 100#) ^ 3#

        portfolioWealth = _
            wON * wealthON + _
            w1M * wealth1M + _
            w3M * wealth3M + _
            w6M * wealth6M

        portfolioReturn = _
            (portfolioWealth ^ (1# / 3#) - 1#) * 100#

        regretBP = (bestReturn - portfolioReturn) * 100#
        excessBP = (portfolioReturn - pureON) * 100#

        wsModel.Cells(rowOut, 5 + s).Value = portfolioReturn
        wsModel.Cells(rowOut, 10 + s).Value = regretBP

        sumExcessBP = sumExcessBP + excessBP

        If regretBP > maxRegretBP Then
            maxRegretBP = regretBP
        End If

        If excessBP < worstExcessBP Then
            worstExcessBP = excessBP
        End If

        If excessBP > 0.0000001 Then
            hitCount = hitCount + 1
        End If

    Next s

    validFlag = 1

    If wON + 0.0000001 < minON Then validFlag = 0
    If w6M - 0.0000001 > max6M Then validFlag = 0

    wsModel.Cells(rowOut, 16).Value = _
        sumExcessBP / SCENARIO_COUNT

    wsModel.Cells(rowOut, 17).Value = maxRegretBP
    wsModel.Cells(rowOut, 18).Value = worstExcessBP
    wsModel.Cells(rowOut, 19).Value = _
        hitCount / SCENARIO_COUNT

    wsModel.Cells(rowOut, 20).Value = validFlag
    wsModel.Cells(rowOut, 21).Value = 0
    wsModel.Cells(rowOut, 22).Value = 0

End Sub

Private Sub MarkEfficientFrontier( _
    ByVal wsModel As Worksheet, _
    ByVal lastRow As Long)

    Dim r As Long
    Dim j As Long

    Dim riskR As Double
    Dim returnR As Double
    Dim riskJ As Double
    Dim returnJ As Double

    Dim dominated As Boolean

    For r = 2 To lastRow

        If CLng(wsModel.Cells(r, 20).Value) = 1 Then

            riskR = CDbl(wsModel.Cells(r, 17).Value)
            returnR = CDbl(wsModel.Cells(r, 16).Value)

            dominated = False

            For j = 2 To lastRow

                If j <> r Then

                    If CLng(wsModel.Cells(j, 20).Value) = 1 Then

                        riskJ = CDbl(wsModel.Cells(j, 17).Value)
                        returnJ = CDbl(wsModel.Cells(j, 16).Value)

                        If riskJ <= riskR + 0.0000001 _
                           And returnJ >= returnR - 0.0000001 Then

                            If riskJ < riskR - 0.0000001 _
                               Or returnJ > returnR + 0.0000001 Then

                                dominated = True
                                Exit For

                            End If

                        End If

                    End If

                End If

            Next j

            If dominated Then
                wsModel.Cells(r, 21).Value = 0
            Else
                wsModel.Cells(r, 21).Value = 1
            End If

        End If

    Next r

End Sub

Private Function FindMinimumRegretRow( _
    ByVal wsModel As Worksheet, _
    ByVal lastRow As Long) As Long

    Dim r As Long

    Dim bestRow As Long
    Dim bestRegret As Double
    Dim thisRegret As Double

    Dim thisReturn As Double
    Dim bestReturn As Double

    bestRegret = 1E+30
    bestReturn = -1E+30
    bestRow = 0

    For r = 2 To lastRow

        If CLng(wsModel.Cells(r, 20).Value) = 1 Then

            thisRegret = CDbl(wsModel.Cells(r, 17).Value)
            thisReturn = CDbl(wsModel.Cells(r, 16).Value)

            If thisRegret < bestRegret - 0.0000001 Then

                bestRegret = thisRegret
                bestReturn = thisReturn
                bestRow = r

            ElseIf Abs(thisRegret - bestRegret) < 0.0000001 Then

                If thisReturn > bestReturn Then
                    bestReturn = thisReturn
                    bestRow = r
                End If

            End If

        End If

    Next r

    FindMinimumRegretRow = bestRow

End Function

Private Function SelectRecommendedPortfolio( _
    ByVal wsInputs As Worksheet, _
    ByVal wsModel As Worksheet, _
    ByVal lastRow As Long, _
    ByVal minRegretRow As Long) As Long

    Dim r As Long

    Dim toleranceBP As Double
    Dim allowedRegret As Double

    Dim thisRegret As Double
    Dim thisReturn As Double

    Dim bestReturn As Double
    Dim bestRow As Long

    If minRegretRow < 2 Then

        SelectRecommendedPortfolio = 0
        Exit Function

    End If

    toleranceBP = CDbl(wsInputs.Range("B5").Value)

    If toleranceBP < 0 Then toleranceBP = 0

    allowedRegret = _
        CDbl(wsModel.Cells(minRegretRow, 17).Value) + toleranceBP

    bestReturn = -1E+30
    bestRow = 0

    For r = 2 To lastRow

        If CLng(wsModel.Cells(r, 20).Value) = 1 Then

            thisRegret = CDbl(wsModel.Cells(r, 17).Value)
            thisReturn = CDbl(wsModel.Cells(r, 16).Value)

            If thisRegret <= allowedRegret + 0.0000001 Then

                If thisReturn > bestReturn + 0.0000001 Then

                    bestReturn = thisReturn
                    bestRow = r

                ElseIf Abs(thisReturn - bestReturn) < 0.0000001 Then

                    If bestRow > 0 Then

                        If thisRegret < _
                           CDbl(wsModel.Cells(bestRow, 17).Value) Then

                            bestRow = r

                        End If

                    End If

                End If

            End If

        End If

    Next r

    If bestRow > 0 Then
        wsModel.Cells(bestRow, 22).Value = 1
    End If

    SelectRecommendedPortfolio = bestRow

End Function

Private Function AverageONBetweenDates( _
    ByVal ws As Worksheet, _
    ByVal startDate As Date, _
    ByVal endDate As Date, _
    ByVal dateCol As Long, _
    ByVal onCol As Long, _
    ByVal lastRow As Long) As Double

    Dim r As Long
    Dim d As Date

    Dim totalRate As Double
    Dim n As Long

    For r = FIRST_DATA_ROW To lastRow

        If IsDate(ws.Cells(r, dateCol).Value) _
           And IsNumeric(ws.Cells(r, onCol).Value) Then

            d = CDate(ws.Cells(r, dateCol).Value)

            If d >= startDate And d <= endDate Then

                totalRate = totalRate + _
                    NormalizeRate(ws.Cells(r, onCol).Value)

                n = n + 1

            ElseIf d > endDate Then

                Exit For

            End If

        End If

    Next r

    If n = 0 Then
        AverageONBetweenDates = -999
    Else
        AverageONBetweenDates = totalRate / n
    End If

End Function

Private Function ONAtOrAfterDate( _
    ByVal ws As Worksheet, _
    ByVal targetDate As Date, _
    ByVal dateCol As Long, _
    ByVal onCol As Long, _
    ByVal lastRow As Long) As Double

    Dim r As Long

    For r = FIRST_DATA_ROW To lastRow

        If IsDate(ws.Cells(r, dateCol).Value) _
           And IsNumeric(ws.Cells(r, onCol).Value) Then

            If CDate(ws.Cells(r, dateCol).Value) >= targetDate Then

                ONAtOrAfterDate = _
                    NormalizeRate(ws.Cells(r, onCol).Value)

                Exit Function

            End If

        End If

    Next r

    ONAtOrAfterDate = -999

End Function

Private Function NormalizeRate(ByVal rawValue As Variant) As Double

    Dim v As Double

    v = CDbl(rawValue)

    If Abs(v) < 1 Then
        NormalizeRate = v * 100#
    Else
        NormalizeRate = v
    End If

End Function

Private Function FindHeaderColumn( _
    ByVal ws As Worksheet, _
    ByVal headerText As String) As Long

    Dim lastCol As Long
    Dim c As Long

    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol

        If UCase$(Trim$(CStr(ws.Cells(1, c).Value))) = _
           UCase$(headerText) Then

            FindHeaderColumn = c
            Exit Function

        End If

    Next c

    FindHeaderColumn = 0

End Function

Private Function LinearInterpolate( _
    ByVal x As Double, _
    ByVal x1 As Double, _
    ByVal y1 As Double, _
    ByVal x2 As Double, _
    ByVal y2 As Double) As Double

    If x2 = x1 Then
        LinearInterpolate = y2
    Else
        LinearInterpolate = _
            y1 + (y2 - y1) * (x - x1) / (x2 - x1)
    End If

End Function

Private Function GetOrCreateSheet( _
    ByVal sheetName As String) As Worksheet

    On Error Resume Next
    Set GetOrCreateSheet = ThisWorkbook.Worksheets(sheetName)
    On Error GoTo 0

    If GetOrCreateSheet Is Nothing Then

        Set GetOrCreateSheet = _
            ThisWorkbook.Worksheets.Add( _
                After:=ThisWorkbook.Worksheets( _
                    ThisWorkbook.Worksheets.Count))

        GetOrCreateSheet.Name = sheetName

    End If

End Function
