2) Add this procedure after USD_ValidateSourceWorkbook:

Private Sub USD_DeleteGeneratedSheets()

    Dim outputSheets As Variant
    Dim i As Long
    Dim sheetName As String
    Dim oldAlerts As Boolean

    outputSheets = Array(S_INDEX, S_OBJECTS, S1, S2, S3, S4, S5, S_AUDIT)
    oldAlerts = Application.DisplayAlerts

    On Error GoTo clean_fail

    ThisWorkbook.Worksheets("Curve").Activate
    Application.DisplayAlerts = False

    For i = LBound(outputSheets) To UBound(outputSheets)
        sheetName = CStr(outputSheets(i))
        If USD_SheetExists(sheetName) Then
            ThisWorkbook.Worksheets(sheetName).Delete
        End If
    Next i

clean_exit:
    Application.DisplayAlerts = oldAlerts
    Exit Sub

clean_fail:
    Application.DisplayAlerts = oldAlerts
    Err.Raise Err.Number, Err.Source, Err.Description

End Sub


3) Replace USD_LoadForwardAssumptions completely with:

Private Sub USD_LoadForwardAssumptions()

    Dim scenario As Long
    Dim anchorNo As Long
    Dim tenor As Long
    Dim defaultAnchors As Variant
    Dim defaultPremiums As Variant
    Dim rowValues As Variant

    'Forward states are governed VBA assumptions, never worksheet inputs.
    defaultAnchors = Array( _
        Array(0.0359, 0.0372, 0.0355, 0.0335, 0.0315), _
        Array(0.0425, 0.0425, 0.0425, 0.04, 0.0375), _
        Array(0.0425, 0.0475, 0.045, 0.0375, 0.034), _
        Array(0.03, 0.0275, 0.029, 0.0275, 0.029), _
        Array(0.015, 0.015, 0.02, 0.02, 0.025))

    defaultPremiums = Array( _
        Array(2#, 5#, 8#), _
        Array(-3#, -5#, -10#), _
        Array(-3#, -5#, -10#), _
        Array(2#, 4#, 7#), _
        Array(2#, 5#, 10#))

    For scenario = 1 To 5
        rowValues = defaultAnchors(scenario - 1)
        For anchorNo = 1 To 5
            mForwardAnchorInput(scenario, anchorNo) = CDbl(rowValues(anchorNo - 1))
        Next anchorNo

        rowValues = defaultPremiums(scenario - 1)
        For tenor = 2 To 4
            mForwardPremiumInput(scenario, tenor) = CDbl(rowValues(tenor - 2))
        Next tenor

        mForwardFadeInput(scenario) = 6#
    Next scenario

End Sub


4) Strengthen USD_ValidateSourceWorkbook so Curve is the only required source:

Private Sub USD_ValidateSourceWorkbook()

    Dim ws As Worksheet
    Dim lastRow As Long

    If Not USD_SheetExists("Curve") Then _
        Err.Raise vbObjectError + 3100, , "Required source sheet 'Curve' was not found."

    Set ws = ThisWorkbook.Worksheets("Curve")

    If UCase$(Trim$(CStr(ws.Cells(1, 1).Value))) <> "DATE" Then _
        Err.Raise vbObjectError + 3101, , "Curve!A1 must contain Date."
    If UCase$(Trim$(CStr(ws.Cells(1, 2).Value))) <> "ON" Then _
        Err.Raise vbObjectError + 3102, , "Curve!B1 must contain ON."
    If UCase$(Trim$(CStr(ws.Cells(1, 3).Value))) <> "1M" Then _
        Err.Raise vbObjectError + 3103, , "Curve!C1 must contain 1M."
    If UCase$(Trim$(CStr(ws.Cells(1, 4).Value))) <> "3M" Then _
        Err.Raise vbObjectError + 3104, , "Curve!D1 must contain 3M."
    If UCase$(Trim$(CStr(ws.Cells(1, 5).Value))) <> "6M" Then _
        Err.Raise vbObjectError + 3105, , "Curve!E1 must contain 6M."

    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    If lastRow < 3 Then _
        Err.Raise vbObjectError + 3106, , "Curve must contain at least two observations."

End Sub
