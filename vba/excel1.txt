'
' PURPOSE
'   Creates the complete slide-ready data, charts, committee comments and audit trail
'   for slides 1-5 inside Excel. It does NOT open or automate PowerPoint.
'
' SOURCE WORKBOOK
'   Curve is the ONLY required workbook sheet and the ONLY worksheet used as a
'   source by the model:
'
'       Curve          Date | ON | 1M | 3M | 6M
'
'   Historical evidence, inverted-path stress and the forward allocation engine are
'   rebuilt from Curve. Forward economic scenarios are explicit VBA assumptions.
'   Existing PPT_* sheets are outputs only: they are deleted and rebuilt on each run
'   and are never read back as model inputs.
'
' OUTPUT SHEETS
'   PPT_Index          manual PowerPoint workflow and source map
'   PPT_Object_Map     object-level source, range, formula and decision logic
'   PPT_S1_Curve       historical curve, macro events and realized examples
'   PPT_S2_Signal      priced premium versus realized return and regression
'   PPT_S3_Stress      normalized realized/inverted paths and return effect
'   PPT_S4_Allocation  five forward scenarios and minimax-regret allocation
'   PPT_S5_Decision    risk controls, execution inputs and evidence ownership
'   PPT_Audit          model checks and reference-value reconciliation
'
' USE
'   1. Start from a workbook containing the Curve tab shown above.
'   2. Save it as an Excel Macro-Enabled Workbook (*.xlsm).
'   3. Press Alt+F11 and insert a standard Module.
'   4. Paste this entire TXT file into the Module.
'   5. Run USD_CreateCashAllocationExcelPack.
'   6. Copy the Excel charts and nearby comments into PowerPoint manually.
'
' IMPORTANT
'   - Curve is never deleted or modified by this module.
'   - All PPT_* sheets are disposable generated outputs.
'   - Forward assumptions are reset from this VBA file on every run.
'====================================================================================

Public Sub USD_CreateCashAllocationExcelPack()

    Dim oldCalc As Long
    Dim oldScreen As Boolean
    Dim oldEvents As Boolean
    Dim oldAlerts As Boolean
    Dim errText As String

    On Error GoTo fail

    oldCalc = Application.Calculation
    oldScreen = Application.ScreenUpdating
    oldEvents = Application.EnableEvents
    oldAlerts = Application.DisplayAlerts

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    Application.Calculation = xlCalculationManual

    If ThisWorkbook.ProtectStructure Then _
        Err.Raise vbObjectError + 3099, , "Workbook structure is protected. Unprotect it before running the module."

    Application.StatusBar = "Building the USD cash-allocation Excel slide pack from Curve..."

    USD_ResetState
    USD_ValidateSourceWorkbook

    'Curve is the only workbook input. All generated sheets are disposable outputs.
    USD_DeleteGeneratedSheets

    USD_LoadCurve
    USD_LoadForwardAssumptions
    USD_CalculateDailyEvidence
    USD_CalculateDirectionalStress
    USD_CalculateForwardAllocation

    USD_BuildObjectMap
    USD_BuildIndex
    USD_BuildSlide1Sheet
    USD_BuildSlide2Sheet
    USD_BuildSlide3Sheet
    USD_BuildSlide4Sheet
    USD_BuildSlide5Sheet
    USD_BuildAuditSheet

    Application.CalculateFull
    ThisWorkbook.Worksheets(S_INDEX).Activate
    ThisWorkbook.Worksheets(S_INDEX).Range("A1").Select

clean_exit:
    Application.StatusBar = False
    Application.DisplayAlerts = oldAlerts
    Application.Calculation = oldCalc
    Application.EnableEvents = oldEvents
    Application.ScreenUpdating = oldScreen

    If Len(errText) = 0 Then
        MsgBox "Excel slide pack created from Curve." & vbCrLf & vbCrLf & _
               "No PowerPoint file was created." & vbCrLf & _
               "Use PPT_Index to copy charts and comments manually.", _
               vbInformation, "USD cash allocation"
    End If
    Exit Sub

fail:
    errText = Err.Description
    MsgBox "USD_CreateCashAllocationExcelPack stopped:" & vbCrLf & vbCrLf & errText, _
           vbCritical, "USD cash allocation"
    Resume clean_exit

End Sub

Private Sub USD_ValidateSourceWorkbook()

    Dim ws As Worksheet
    Dim lastRow As Long

    If Not USD_SheetExists("Curve") Then _
        Err.Raise vbObjectError + 3100, , "Required source sheet 'Curve' was not found."

    Set ws = ThisWorkbook.Worksheets("Curve")

    If UCase$(Trim$(CStr(ws.Cells(1, 1).Value))) <> "DATE" Then _
        Err.Raise vbObjectError + 3101, , "Curve!A1 must contain Date."
    If UCase$(Trim$(CStr(ws.Cells(1, 2).Value))) <> "ON" Then _
        Err.Raise vbObjectError + 3102, , "Curve!B1 must contain ON."
    If UCase$(Trim$(CStr(ws.Cells(1, 3).Value))) <> "1M" Then _
        Err.Raise vbObjectError + 3103, , "Curve!C1 must contain 1M."
    If UCase$(Trim$(CStr(ws.Cells(1, 4).Value))) <> "3M" Then _
        Err.Raise vbObjectError + 3104, , "Curve!D1 must contain 3M."
    If UCase$(Trim$(CStr(ws.Cells(1, 5).Value))) <> "6M" Then _
        Err.Raise vbObjectError + 3105, , "Curve!E1 must contain 6M."

    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    If lastRow < 3 Then _
        Err.Raise vbObjectError + 3106, , "Curve must contain at least two observations."

End Sub

Private Sub USD_DeleteGeneratedSheets()

    Dim outputSheets As Variant
    Dim i As Long
    Dim sheetName As String

    outputSheets = Array(S_INDEX, S_OBJECTS, S1, S2, S3, S4, S5, S_AUDIT)

    'Keep Curve active so Excel always retains the source workbook sheet.
    ThisWorkbook.Worksheets("Curve").Activate

    For i = LBound(outputSheets) To UBound(outputSheets)
        sheetName = CStr(outputSheets(i))
        If USD_SheetExists(sheetName) Then
            ThisWorkbook.Worksheets(sheetName).Delete
        End If
    Next i

End Sub

Private Sub USD_LoadForwardAssumptions()

    Dim scenario As Long
    Dim anchorNo As Long
    Dim tenor As Long
    Dim defaultAnchors As Variant
    Dim defaultPremiums As Variant
    Dim rowValues As Variant

    'Forward states are governed VBA assumptions, never worksheet inputs.
    defaultAnchors = Array( _
        Array(0.0359, 0.0372, 0.0355, 0.0335, 0.0315), _
        Array(0.0425, 0.0425, 0.0425, 0.04, 0.0375), _
        Array(0.0425, 0.0475, 0.045, 0.0375, 0.034), _
        Array(0.03, 0.0275, 0.029, 0.0275, 0.029), _
        Array(0.015, 0.015, 0.02, 0.02, 0.025))

    defaultPremiums = Array( _
        Array(2#, 5#, 8#), _
        Array(-3#, -5#, -10#), _
        Array(-3#, -5#, -10#), _
        Array(2#, 4#, 7#), _
        Array(2#, 5#, 10#))

    For scenario = 1 To 5
        rowValues = defaultAnchors(scenario - 1)
        For anchorNo = 1 To 5
            mForwardAnchorInput(scenario, anchorNo) = CDbl(rowValues(anchorNo - 1))
        Next anchorNo

        rowValues = defaultPremiums(scenario - 1)
        For tenor = 2 To 4
            mForwardPremiumInput(scenario, tenor) = CDbl(rowValues(tenor - 2))
        Next tenor

        mForwardFadeInput(scenario) = 6#
    Next scenario

End Sub

Private Function USD_SheetExists(ByVal sheetName As String) As Boolean
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(sheetName)
    USD_SheetExists = Not ws Is Nothing
    On Error GoTo 0
End Function
