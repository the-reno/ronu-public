Option Explicit

'====================================================================
' DETERMINISTIC EVENT-DRIVEN CASH SIMULATION
'
' DASHBOARD DESIGN
' ----------------
' User-facing sheet:
'     Sim_Control
'
' Events run LEFT TO RIGHT across columns.
' Economic variables run TOP TO BOTTOM across rows.
'
' Daily calculation sheet:
'     Sim_Curve
'
' Core commitments
' ----------------
' 1. CPI / Payroll shocks change EXPECTATIONS, not actual ON.
' 2. Actual ON changes only when an actual FOMC policy move occurs.
' 3. Existing locked investments do not reprice when sentiment changes.
' 4. New market rates affect a strategy only when it rolls / reinvests.
' 5. Term rates are generated from the expected future policy path.
' 6. Strategy returns continue to use the existing R_CURVE engine.
' 7. Dashboard shows event dates only; Sim_Curve retains daily data.
'
' Dependency
' ----------
' Existing workbook function:
'     R_CURVE(start, table, tenor, adj, type, n, maturity)
'
' Workflow
' --------
' From the Curve tab controller, click:
'     Simulation Model
'
' Or run once:
'     Create_Simple_Simulation_Model
'
' Thereafter edit Sim_Control and click:
'     RUN SIMULATION
'====================================================================


'====================================================================
' COMPATIBLE ENTRY POINT USED BY THE CURVE-TAB CONTROLLER
'====================================================================

Sub Create_Simple_Simulation_Model()

    Dim ws As Worksheet
    Dim rebuild As Boolean

    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets("Sim_Control")
    On Error GoTo 0

    rebuild = False

    If ws Is Nothing Then
        rebuild = True
    ElseIf UCase(Trim(ws.Range("A13").Value)) <> "EVENT TIMELINE" Then
        'Automatically migrate the older vertical dashboard.
        rebuild = True
    End If

    If rebuild Then
        SIM_CreateDashboardLayout
        SIM_LoadFalseHawkishInputs
    End If

    Run_Simulation

End Sub


'====================================================================
' OPTIONAL SCENARIO LOADER
'====================================================================

Sub Run_False_Hawkish_Scenario()

    SIM_CreateDashboardLayout
    SIM_LoadFalseHawkishInputs
    Run_Simulation

End Sub


'====================================================================
' MAIN ONE-CLICK SIMULATION
'====================================================================

Sub Run_Simulation()

    Dim ws As Worksheet
    Dim wsCurve As Worksheet

    Dim startDate As Date
    Dim endDate As Date

    Dim capital As Double
    Dim actualON As Double

    Dim premium1M As Double
    Dim premium3M As Double
    Dim premium6M As Double

    Dim lastCol As Long
    Dim nFomc As Long

    Dim fomcDates() As Date
    Dim expectedMoves() As Double
    Dim actualMoves() As Double

    Dim d As Date

    Dim c As Long
    Dim i As Long
    Dim f As Long

    Dim outRow As Long
    Dim lastCurveRow As Long

    Dim eventText As String

    Dim rate1M As Double
    Dim rate3M As Double
    Dim rate6M As Double

    Dim next1 As Double
    Dim next2 As Double
    Dim next3 As Double

    Dim factorON As Double
    Dim factor1M As Double
    Dim factor3M As Double
    Dim factor6M As Double

    Dim tableRange As Range
    Dim ch As ChartObject
    Dim s As Series


    Set ws = ThisWorkbook.Worksheets("Sim_Control")


    '----------------------------------------------------------------
    ' General assumptions
    '----------------------------------------------------------------

    If Not IsDate(ws.Range("B3").Value) _
       Or Not IsDate(ws.Range("B4").Value) Then

        MsgBox "Start Date and End Date must be valid dates.", vbCritical
        Exit Sub

    End If

    startDate = ws.Range("B3").Value
    endDate = ws.Range("B4").Value

    If endDate <= startDate Then
        MsgBox "End Date must be after Start Date.", vbCritical
        Exit Sub
    End If

    capital = ws.Range("B5").Value
    actualON = ws.Range("B6").Value

    premium1M = ws.Range("B8").Value
    premium3M = ws.Range("B9").Value
    premium6M = ws.Range("B10").Value


    '----------------------------------------------------------------
    ' Timeline extends horizontally from column B.
    ' Row 14 contains event dates.
    '----------------------------------------------------------------

    lastCol = ws.Cells(14, ws.Columns.Count).End(xlToLeft).Column

    If lastCol < 2 Then
        MsgBox "No simulation events found on the timeline.", vbCritical
        Exit Sub
    End If


    '----------------------------------------------------------------
    ' Validate active event dates.
    '----------------------------------------------------------------

    For c = 2 To lastCol

        If SIM_IsActive(ws.Cells(16, c).Value) Then

            If Not IsDate(ws.Cells(14, c).Value) Then
                MsgBox "Invalid event date in column " & c & ".", vbCritical
                Exit Sub
            End If

            If Weekday(CDate(ws.Cells(14, c).Value), vbMonday) > 5 Then
                MsgBox _
                    "Active events must be on business days." & vbCrLf & _
                    "Check " & ws.Cells(14, c).Address(False, False) & ".", _
                    vbCritical
                Exit Sub
            End If

        End If

    Next c


    '----------------------------------------------------------------
    ' Count active FOMC meetings.
    '----------------------------------------------------------------

    nFomc = 0

    For c = 2 To lastCol

        If UCase(Trim(ws.Cells(15, c).Value)) = "FOMC" _
           And SIM_IsActive(ws.Cells(16, c).Value) Then

            nFomc = nFomc + 1

        End If

    Next c

    If nFomc = 0 Then
        MsgBox "No active FOMC meetings found on the timeline.", vbCritical
        Exit Sub
    End If


    '----------------------------------------------------------------
    ' Load FOMC arrays.
    '
    ' Row 17 = Initial Expected Move (bp)
    ' Row 18 = Actual Policy Move (bp)
    '----------------------------------------------------------------

    ReDim fomcDates(1 To nFomc)
    ReDim expectedMoves(1 To nFomc)
    ReDim actualMoves(1 To nFomc)

    f = 0

    For c = 2 To lastCol

        If UCase(Trim(ws.Cells(15, c).Value)) = "FOMC" _
           And SIM_IsActive(ws.Cells(16, c).Value) Then

            f = f + 1

            fomcDates(f) = ws.Cells(14, c).Value
            expectedMoves(f) = SIM_Num(ws.Cells(17, c).Value) / 10000
            actualMoves(f) = SIM_Num(ws.Cells(18, c).Value) / 10000

        End If

    Next c


    'FOMC dates must be ascending for path generation.
    SIM_SortFomcArrays fomcDates, expectedMoves, actualMoves, nFomc


    '----------------------------------------------------------------
    ' Create / reset daily calculation layer.
    '----------------------------------------------------------------

    Set wsCurve = SIM_GetOrCreateSheet("Sim_Curve")

    wsCurve.Cells.Clear

    For Each ch In wsCurve.ChartObjects
        ch.Delete
    Next ch


    'A:E deliberately maintain R_CURVE input format.

    wsCurve.Range("A1:J1").Value = Array( _
        "Date", _
        "ON", _
        "1M", _
        "3M", _
        "6M", _
        "Event", _
        "Next FOMC", _
        "2nd FOMC", _
        "3rd FOMC", _
        "Cum Next 3")

    outRow = 2


    '================================================================
    ' DAILY SIMULATION
    '================================================================

    For d = startDate To endDate

        If Weekday(d, vbMonday) <= 5 Then


            '--------------------------------------------------------
            ' STEP 1 - ACTUAL POLICY DECISION
            '
            ' Actual ON changes only when the Fed actually changes
            ' policy. The expectation for today's meeting is then
            ' resolved and removed from the forward expectation set.
            '--------------------------------------------------------

            For i = 1 To nFomc

                If fomcDates(i) = d Then

                    actualON = actualON + actualMoves(i)
                    expectedMoves(i) = 0

                End If

            Next i


            '--------------------------------------------------------
            ' STEP 2 - INFORMATION EVENTS
            '
            ' Rows 19:21 change expected moves at the next three
            ' future FOMC meetings. They do not mechanically change ON.
            '--------------------------------------------------------

            eventText = ""

            For c = 2 To lastCol

                If SIM_IsActive(ws.Cells(16, c).Value) _
                   And IsDate(ws.Cells(14, c).Value) Then

                    If CDate(ws.Cells(14, c).Value) = d Then

                        If eventText <> "" Then
                            eventText = eventText & " / "
                        End If

                        eventText = eventText & CStr(ws.Cells(15, c).Value)

                        SIM_ApplyExpectationShock _
                            d, _
                            SIM_Num(ws.Cells(19, c).Value), _
                            SIM_Num(ws.Cells(20, c).Value), _
                            SIM_Num(ws.Cells(21, c).Value), _
                            fomcDates, _
                            expectedMoves, _
                            nFomc

                    End If

                End If

            Next c


            '--------------------------------------------------------
            ' STEP 3 - NEWLY AVAILABLE TERM CURVE
            '
            ' Term rates are generated from expected average future ON
            ' plus optional tenor premiums from the dashboard.
            '--------------------------------------------------------

            rate1M = _
                SIM_ExpectedAverageRate( _
                    d, _
                    DateAdd("m", 1, d), _
                    actualON, _
                    fomcDates, _
                    expectedMoves, _
                    nFomc) _
                + premium1M

            rate3M = _
                SIM_ExpectedAverageRate( _
                    d, _
                    DateAdd("m", 3, d), _
                    actualON, _
                    fomcDates, _
                    expectedMoves, _
                    nFomc) _
                + premium3M

            rate6M = _
                SIM_ExpectedAverageRate( _
                    d, _
                    DateAdd("m", 6, d), _
                    actualON, _
                    fomcDates, _
                    expectedMoves, _
                    nFomc) _
                + premium6M


            SIM_GetNextThreeExpectedMoves _
                d, _
                fomcDates, _
                expectedMoves, _
                nFomc, _
                next1, _
                next2, _
                next3


            '--------------------------------------------------------
            ' Daily calculation output
            '--------------------------------------------------------

            wsCurve.Cells(outRow, 1).Value = d
            wsCurve.Cells(outRow, 2).Value = actualON
            wsCurve.Cells(outRow, 3).Value = rate1M
            wsCurve.Cells(outRow, 4).Value = rate3M
            wsCurve.Cells(outRow, 5).Value = rate6M
            wsCurve.Cells(outRow, 6).Value = eventText

            wsCurve.Cells(outRow, 7).Value = next1 * 10000
            wsCurve.Cells(outRow, 8).Value = next2 * 10000
            wsCurve.Cells(outRow, 9).Value = next3 * 10000
            wsCurve.Cells(outRow, 10).Value = _
                (next1 + next2 + next3) * 10000

            outRow = outRow + 1

        End If

    Next d


    lastCurveRow = outRow - 1

    wsCurve.Columns("A").NumberFormat = "dd-mmm-yyyy"
    wsCurve.Columns("B:E").NumberFormat = "0.000%"
    wsCurve.Columns("G:J").NumberFormat = "0.0"


    '================================================================
    ' TERMINAL STRATEGY RETURNS - EXISTING R_CURVE ENGINE
    '================================================================

    Set tableRange = wsCurve.Range("A1:E" & lastCurveRow)

    factorON = R_CURVE( _
        startDate, tableRange, "ON", 0, "FACTOR", 0, endDate)

    factor1M = R_CURVE( _
        startDate, tableRange, "1M", 0, "FACTOR", 0, endDate)

    factor3M = R_CURVE( _
        startDate, tableRange, "3M", 0, "FACTOR", 0, endDate)

    factor6M = R_CURVE( _
        startDate, tableRange, "6M", 0, "FACTOR", 0, endDate)


    '================================================================
    ' DAILY CUMULATIVE WEALTH PATHS
    '================================================================

    SIM_BuildWealthPath _
        wsCurve, _
        startDate, _
        capital, _
        tableRange, _
        lastCurveRow


    '================================================================
    ' POPULATE EVENT-TIMELINE OUTPUT ROWS
    '================================================================

    SIM_PopulateEventDashboard ws, wsCurve, lastCurveRow, lastCol


    '================================================================
    ' TERMINAL RESULT TABLE
    '================================================================

    ws.Range("A41:E46").ClearContents

    ws.Range("A41:E41").Value = Array( _
        "Strategy", _
        "Initial Capital", _
        "Final Wealth", _
        "1Y Return", _
        "Excess vs ON (bp)")

    SIM_WriteTerminalResult ws, 42, "ON", capital, factorON, factorON
    SIM_WriteTerminalResult ws, 43, "1M", capital, factor1M, factorON
    SIM_WriteTerminalResult ws, 44, "3M", capital, factor3M, factorON
    SIM_WriteTerminalResult ws, 45, "6M", capital, factor6M, factorON

    ws.Range("B42:C45").NumberFormat = "0.0000"
    ws.Range("D42:D45").NumberFormat = "0.000%"
    ws.Range("E42:E45").NumberFormat = "0.0"


    '================================================================
    ' DASHBOARD CHARTS
    '================================================================

    SIM_CreateRateChart ws, wsCurve, lastCurveRow
    SIM_CreateWealthChart ws, wsCurve, lastCurveRow


    SIM_FormatDashboard ws, lastCol
    wsCurve.Columns("A:N").AutoFit

    MsgBox _
        "Simulation complete." & vbCrLf & _
        "Read the scenario from left to right on Sim_Control.", _
        vbInformation

End Sub

'====================================================================
' LOAD FALSE HAWKISH EXAMPLE INTO HORIZONTAL TIMELINE
'
' Jan CPI        : market prices three +25bp hikes.
' Feb FOMC       : expected hike is not delivered.
' Mar Payroll    : remaining hikes are removed.
' Actual ON stays at 4.50% throughout.
'====================================================================

Private Sub SIM_LoadFalseHawkishInputs()

    Dim ws As Worksheet

    Dim fomcDates(1 To 8) As Date

    Dim c As Long
    Dim i As Long
    Dim m As Long
    Dim d As Date
    Dim lastCol As Long

    Set ws = ThisWorkbook.Worksheets("Sim_Control")


    '----------------------------------------------------------------
    ' Scenario assumptions
    '----------------------------------------------------------------

    ws.Range("B2").Value = "False Hawkish"
    ws.Range("B3").Value = DateSerial(2027, 1, 4)
    ws.Range("B4").Value = DateSerial(2027, 12, 31)
    ws.Range("B5").Value = 100
    ws.Range("B6").Value = 0.045

    ws.Range("B8").Value = 0
    ws.Range("B9").Value = 0
    ws.Range("B10").Value = 0

    ws.Range("B3:B4").NumberFormat = "dd-mmm-yyyy"
    ws.Range("B6:B10").NumberFormat = "0.000%"


    'Clear previous timeline, leaving row labels intact.
    ws.Range("B14:XFD38").ClearContents


    c = 2


    '----------------------------------------------------------------
    ' Start marker
    '----------------------------------------------------------------

    SIM_WriteEventColumn _
        ws, c, _
        DateSerial(2027, 1, 4), _
        "Start", _
        True, _
        0, 0, 0, 0, 0, _
        "Initial investment"

    c = c + 1


    '----------------------------------------------------------------
    ' FOMC meetings - illustrative, not historical.
    ' No hikes are ultimately delivered.
    '----------------------------------------------------------------

    fomcDates(1) = DateSerial(2027, 2, 17)
    fomcDates(2) = DateSerial(2027, 3, 31)
    fomcDates(3) = DateSerial(2027, 5, 12)
    fomcDates(4) = DateSerial(2027, 6, 23)
    fomcDates(5) = DateSerial(2027, 8, 4)
    fomcDates(6) = DateSerial(2027, 9, 15)
    fomcDates(7) = DateSerial(2027, 10, 27)
    fomcDates(8) = DateSerial(2027, 12, 8)

    For i = 1 To 8

        SIM_WriteEventColumn _
            ws, c, _
            fomcDates(i), _
            "FOMC", _
            True, _
            0, _
            0, _
            0, 0, 0, _
            "Fed does not deliver expected hike"

        c = c + 1

    Next i


    '----------------------------------------------------------------
    ' Monthly CPI
    ' January CPI prices three future hikes.
    '----------------------------------------------------------------

    For m = 1 To 12

        d = SIM_NextBusinessDay(DateSerial(2027, m, 15))

        If m = 1 Then

            SIM_WriteEventColumn _
                ws, c, _
                d, _
                "CPI", _
                True, _
                0, 0, _
                25, 25, 25, _
                "Strong CPI: market prices three hikes"

        Else

            SIM_WriteEventColumn _
                ws, c, _
                d, _
                "CPI", _
                True, _
                0, 0, _
                0, 0, 0, _
                "No shock"

        End If

        c = c + 1

    Next m


    '----------------------------------------------------------------
    ' Monthly Payroll
    ' March Payroll removes the two remaining expected hikes.
    '----------------------------------------------------------------

    For m = 1 To 12

        d = SIM_FirstFriday(2027, m)

        If m = 3 Then

            SIM_WriteEventColumn _
                ws, c, _
                d, _
                "Payroll", _
                True, _
                0, 0, _
                -25, -25, 0, _
                "Weak Payroll: remaining hikes removed"

        Else

            SIM_WriteEventColumn _
                ws, c, _
                d, _
                "Payroll", _
                True, _
                0, 0, _
                0, 0, 0, _
                "No shock"

        End If

        c = c + 1

    Next m


    '----------------------------------------------------------------
    ' End marker
    '----------------------------------------------------------------

    SIM_WriteEventColumn _
        ws, c, _
        DateSerial(2027, 12, 31), _
        "End", _
        True, _
        0, 0, 0, 0, 0, _
        "Terminal valuation"


    lastCol = c


    'Sort all timeline columns chronologically while preserving rows.

    ws.Range( _
        ws.Cells(14, 2), _
        ws.Cells(38, lastCol)).Sort _
            Key1:=ws.Cells(14, 2), _
            Order1:=xlAscending, _
            Header:=xlNo, _
            Orientation:=xlLeftToRight


    SIM_FormatDashboard ws, lastCol

End Sub


'====================================================================
' WRITE ONE EVENT COLUMN
'====================================================================

Private Sub SIM_WriteEventColumn( _
    ByVal ws As Worksheet, _
    ByVal colNum As Long, _
    ByVal eventDate As Date, _
    ByVal eventName As String, _
    ByVal active As Boolean, _
    ByVal initialExpectedBp As Double, _
    ByVal actualMoveBp As Double, _
    ByVal shock1Bp As Double, _
    ByVal shock2Bp As Double, _
    ByVal shock3Bp As Double, _
    ByVal noteText As String)

    ws.Cells(14, colNum).Value = eventDate
    ws.Cells(15, colNum).Value = eventName
    ws.Cells(16, colNum).Value = active
    ws.Cells(17, colNum).Value = initialExpectedBp
    ws.Cells(18, colNum).Value = actualMoveBp
    ws.Cells(19, colNum).Value = shock1Bp
    ws.Cells(20, colNum).Value = shock2Bp
    ws.Cells(21, colNum).Value = shock3Bp
    ws.Cells(22, colNum).Value = noteText

End Sub


'====================================================================
' POPULATE CALCULATED MARKET STATE + WEALTH ON EVENT DATES
'====================================================================

Private Sub SIM_PopulateEventDashboard( _
    ByVal ws As Worksheet, _
    ByVal wsCurve As Worksheet, _
    ByVal lastCurveRow As Long, _
    ByVal lastCol As Long)

    Dim c As Long
    Dim curveRow As Variant
    Dim eventDate As Date

    ws.Range(ws.Cells(25, 2), ws.Cells(38, lastCol)).ClearContents

    For c = 2 To lastCol

        If SIM_IsActive(ws.Cells(16, c).Value) _
           And IsDate(ws.Cells(14, c).Value) Then

            eventDate = ws.Cells(14, c).Value

            curveRow = Application.Match( _
                CDbl(eventDate), _
                wsCurve.Range("A1:A" & lastCurveRow), _
                0)

            If Not IsError(curveRow) Then

                'Expected FOMC state
                ws.Cells(25, c).Value = wsCurve.Cells(curveRow, 7).Value
                ws.Cells(26, c).Value = wsCurve.Cells(curveRow, 8).Value
                ws.Cells(27, c).Value = wsCurve.Cells(curveRow, 9).Value
                ws.Cells(28, c).Value = wsCurve.Cells(curveRow, 10).Value

                'Available market rates
                ws.Cells(29, c).Value = wsCurve.Cells(curveRow, 2).Value
                ws.Cells(30, c).Value = wsCurve.Cells(curveRow, 3).Value
                ws.Cells(31, c).Value = wsCurve.Cells(curveRow, 4).Value
                ws.Cells(32, c).Value = wsCurve.Cells(curveRow, 5).Value

                'Strategy wealth
                ws.Cells(35, c).Value = wsCurve.Cells(curveRow, 11).Value
                ws.Cells(36, c).Value = wsCurve.Cells(curveRow, 12).Value
                ws.Cells(37, c).Value = wsCurve.Cells(curveRow, 13).Value
                ws.Cells(38, c).Value = wsCurve.Cells(curveRow, 14).Value

            End If

        End If

    Next c

End Sub


'====================================================================
' BUILD DAILY CUMULATIVE WEALTH PATH
'
' R_CURVE remains the return engine.
' Market rates can reprice immediately, but an existing locked
' investment changes strategy earnings only at its scheduled rollover.
'====================================================================

Private Sub SIM_BuildWealthPath( _
    ByVal wsCurve As Worksheet, _
    ByVal startDate As Date, _
    ByVal capital As Double, _
    ByVal tableRange As Range, _
    ByVal lastCurveRow As Long)

    Dim i As Long
    Dim currentDate As Date

    Dim fON As Double
    Dim f1M As Double
    Dim f3M As Double
    Dim f6M As Double

    wsCurve.Range("K:N").ClearContents

    wsCurve.Range("K1:N1").Value = Array( _
        "ON Wealth", _
        "1M Wealth", _
        "3M Wealth", _
        "6M Wealth")

    Application.ScreenUpdating = False

    For i = 2 To lastCurveRow

        currentDate = wsCurve.Cells(i, 1).Value

        If currentDate <= startDate Then

            fON = 1
            f1M = 1
            f3M = 1
            f6M = 1

        Else

            fON = R_CURVE( _
                startDate, tableRange, "ON", 0, "FACTOR", 0, currentDate)

            f1M = R_CURVE( _
                startDate, tableRange, "1M", 0, "FACTOR", 0, currentDate)

            f3M = R_CURVE( _
                startDate, tableRange, "3M", 0, "FACTOR", 0, currentDate)

            f6M = R_CURVE( _
                startDate, tableRange, "6M", 0, "FACTOR", 0, currentDate)

        End If

        wsCurve.Cells(i, 11).Value = capital * fON
        wsCurve.Cells(i, 12).Value = capital * f1M
        wsCurve.Cells(i, 13).Value = capital * f3M
        wsCurve.Cells(i, 14).Value = capital * f6M

    Next i

    Application.ScreenUpdating = True

    wsCurve.Columns("K:N").NumberFormat = "0.0000"

End Sub


'====================================================================
' RATE CHART - DAILY MARKET PERCEPTION / AVAILABLE RATES
'====================================================================

Private Sub SIM_CreateRateChart( _
    ByVal ws As Worksheet, _
    ByVal wsCurve As Worksheet, _
    ByVal lastCurveRow As Long)

    Dim ch As ChartObject
    Dim s As Series
    Dim i As Long

    On Error Resume Next
    ws.ChartObjects("SimulationRateChart").Delete
    On Error GoTo 0

    Set ch = ws.ChartObjects.Add( _
        Left:=ws.Range("A48").Left, _
        Top:=ws.Range("A48").Top, _
        Width:=900, _
        Height:=420)

    ch.Name = "SimulationRateChart"

    With ch.Chart

        .ChartType = xlLine

        For i = 2 To 5

            Set s = .SeriesCollection.NewSeries

            s.Name = wsCurve.Cells(1, i).Value
            s.XValues = wsCurve.Range("A2:A" & lastCurveRow)
            s.Values = wsCurve.Range( _
                wsCurve.Cells(2, i), _
                wsCurve.Cells(lastCurveRow, i))
            s.MarkerStyle = xlMarkerStyleNone

        Next i

        .HasTitle = True
        .ChartTitle.Text = _
            "Market Rates - " & ws.Range("B2").Value

        .HasLegend = True
        .Legend.Position = xlLegendPositionBottom

        With .Axes(xlCategory)
            .TickLabels.NumberFormat = "mmm-yy"
        End With

        With .Axes(xlValue)
            .HasTitle = True
            .AxisTitle.Text = "Rate"
            .TickLabels.NumberFormat = "0.00%"
        End With

    End With

End Sub

'====================================================================
' APPLY INFORMATION SHOCK TO NEXT THREE FUTURE FOMC MEETINGS
'====================================================================

Private Sub SIM_ApplyExpectationShock( _
    ByVal eventDate As Date, _
    ByVal shock1Bp As Double, _
    ByVal shock2Bp As Double, _
    ByVal shock3Bp As Double, _
    ByRef fomcDates() As Date, _
    ByRef expectedMoves() As Double, _
    ByVal nFomc As Long)

    Dim i As Long
    Dim n As Long

    n = 0

    For i = 1 To nFomc

        If fomcDates(i) > eventDate Then

            n = n + 1

            Select Case n

                Case 1
                    expectedMoves(i) = _
                        expectedMoves(i) + shock1Bp / 10000

                Case 2
                    expectedMoves(i) = _
                        expectedMoves(i) + shock2Bp / 10000

                Case 3
                    expectedMoves(i) = _
                        expectedMoves(i) + shock3Bp / 10000
                    Exit For

            End Select

        End If

    Next i

End Sub


'====================================================================
' GENERATE TERM RATE FROM EXPECTED POLICY PATH
'
' Version 1 assumption:
' quoted term rate = expected average ON over tenor + optional premium.
'====================================================================

Private Function SIM_ExpectedAverageRate( _
    ByVal asOfDate As Date, _
    ByVal maturityDate As Date, _
    ByVal actualON As Double, _
    ByRef fomcDates() As Date, _
    ByRef expectedMoves() As Double, _
    ByVal nFomc As Long) As Double

    Dim d As Date
    Dim i As Long

    Dim expectedRate As Double
    Dim total As Double
    Dim days As Long

    total = 0
    days = 0

    For d = asOfDate To maturityDate - 1

        expectedRate = actualON

        For i = 1 To nFomc

            If fomcDates(i) > asOfDate _
               And fomcDates(i) <= d Then

                expectedRate = expectedRate + expectedMoves(i)

            End If

        Next i

        total = total + expectedRate
        days = days + 1

    Next d

    If days > 0 Then
        SIM_ExpectedAverageRate = total / days
    Else
        SIM_ExpectedAverageRate = actualON
    End If

End Function


'====================================================================
' GET NEXT THREE EXPECTED FOMC MOVES
'====================================================================

Private Sub SIM_GetNextThreeExpectedMoves( _
    ByVal asOfDate As Date, _
    ByRef fomcDates() As Date, _
    ByRef expectedMoves() As Double, _
    ByVal nFomc As Long, _
    ByRef move1 As Double, _
    ByRef move2 As Double, _
    ByRef move3 As Double)

    Dim i As Long
    Dim n As Long

    move1 = 0
    move2 = 0
    move3 = 0

    n = 0

    For i = 1 To nFomc

        If fomcDates(i) > asOfDate Then

            n = n + 1

            Select Case n
                Case 1
                    move1 = expectedMoves(i)
                Case 2
                    move2 = expectedMoves(i)
                Case 3
                    move3 = expectedMoves(i)
                    Exit For
            End Select

        End If

    Next i

End Sub


'====================================================================
' SORT FOMC ARRAYS BY DATE
'====================================================================

Private Sub SIM_SortFomcArrays( _
    ByRef fomcDates() As Date, _
    ByRef expectedMoves() As Double, _
    ByRef actualMoves() As Double, _
    ByVal nFomc As Long)

    Dim i As Long
    Dim j As Long

    Dim tmpDate As Date
    Dim tmpExp As Double
    Dim tmpActual As Double

    For i = 1 To nFomc - 1

        For j = i + 1 To nFomc

            If fomcDates(j) < fomcDates(i) Then

                tmpDate = fomcDates(i)
                fomcDates(i) = fomcDates(j)
                fomcDates(j) = tmpDate

                tmpExp = expectedMoves(i)
                expectedMoves(i) = expectedMoves(j)
                expectedMoves(j) = tmpExp

                tmpActual = actualMoves(i)
                actualMoves(i) = actualMoves(j)
                actualMoves(j) = tmpActual

            End If

        Next j

    Next i

End Sub


'====================================================================
' TERMINAL RESULT TABLE
'====================================================================

Private Sub SIM_WriteTerminalResult( _
    ByVal ws As Worksheet, _
    ByVal rowNumber As Long, _
    ByVal strategy As String, _
    ByVal capital As Double, _
    ByVal factor As Double, _
    ByVal onFactor As Double)

    ws.Cells(rowNumber, 1).Value = strategy
    ws.Cells(rowNumber, 2).Value = capital
    ws.Cells(rowNumber, 3).Value = capital * factor
    ws.Cells(rowNumber, 4).Value = factor - 1
    ws.Cells(rowNumber, 5).Value = (factor - onFactor) * 10000

End Sub


'====================================================================
' DASHBOARD FORMATTING
'====================================================================

Private Sub SIM_FormatDashboard( _
    ByVal ws As Worksheet, _
    ByVal lastCol As Long)

    Dim c As Long

    ws.Range("A13:A38").Font.Bold = True

    ws.Range("A13:" & ws.Cells(13, lastCol).Address).Font.Bold = True
    ws.Range("A24:" & ws.Cells(24, lastCol).Address).Font.Bold = True
    ws.Range("A34:" & ws.Cells(34, lastCol).Address).Font.Bold = True

    ws.Range(ws.Cells(14, 2), ws.Cells(14, lastCol)).NumberFormat = _
        "dd-mmm"

    ws.Range(ws.Cells(17, 2), ws.Cells(21, lastCol)).NumberFormat = _
        "0.0"

    ws.Range(ws.Cells(25, 2), ws.Cells(28, lastCol)).NumberFormat = _
        "0.0"

    ws.Range(ws.Cells(29, 2), ws.Cells(32, lastCol)).NumberFormat = _
        "0.000%"

    ws.Range(ws.Cells(35, 2), ws.Cells(38, lastCol)).NumberFormat = _
        "$0.0000"

    ws.Range(ws.Cells(14, 2), ws.Cells(38, lastCol)).HorizontalAlignment = _
        xlCenter

    ws.Range(ws.Cells(22, 2), ws.Cells(22, lastCol)).WrapText = True

    For c = 2 To lastCol
        ws.Columns(c).ColumnWidth = 12
    Next c

    ws.Columns("A").ColumnWidth = 30
    ws.Rows(22).RowHeight = 45

End Sub


'====================================================================
' ACTIVE FLAG PARSER
'====================================================================

Private Function SIM_IsActive(ByVal v As Variant) As Boolean

    If VarType(v) = vbBoolean Then
        SIM_IsActive = CBool(v)
        Exit Function
    End If

    Select Case UCase(Trim(CStr(v)))
        Case "TRUE", "YES", "Y", "1"
            SIM_IsActive = True
        Case Else
            SIM_IsActive = False
    End Select

End Function


'====================================================================
' SAFE NUMERIC PARSER
'====================================================================

Private Function SIM_Num(ByVal v As Variant) As Double

    If IsNumeric(v) Then
        SIM_Num = CDbl(v)
    Else
        SIM_Num = 0
    End If

End Function


'====================================================================
' FIRST FRIDAY OF MONTH
'====================================================================

Private Function SIM_FirstFriday( _
    ByVal yr As Long, _
    ByVal mn As Long) As Date

    Dim d As Date

    d = DateSerial(yr, mn, 1)

    Do While Weekday(d, vbMonday) <> 5
        d = d + 1
    Loop

    SIM_FirstFriday = d

End Function


'====================================================================
' WEEKEND DATE -> NEXT MONDAY
'====================================================================

Private Function SIM_NextBusinessDay( _
    ByVal d As Date) As Date

    Do While Weekday(d, vbMonday) > 5
        d = d + 1
    Loop

    SIM_NextBusinessDay = d

End Function


'====================================================================
' GET OR CREATE WORKSHEET
'====================================================================

Private Function SIM_GetOrCreateSheet( _
    ByVal sheetName As String) As Worksheet

    On Error Resume Next
    Set SIM_GetOrCreateSheet = ThisWorkbook.Worksheets(sheetName)
    On Error GoTo 0

    If SIM_GetOrCreateSheet Is Nothing Then

        Set SIM_GetOrCreateSheet = _
            ThisWorkbook.Worksheets.Add( _
                After:=ThisWorkbook.Worksheets( _
                    ThisWorkbook.Worksheets.Count))

        SIM_GetOrCreateSheet.Name = sheetName

    End If

End Function
