Option Explicit

'====================================================================
' INITIAL 6M CURVE SPREAD VS REALIZED 6M EXCESS RETURN
'
' Sources:
'   curve
'   Analysis_2
'
' Output:
'   Analysis_3
'
' IMPORTANT:
'   This macro does NOT call R_CURVE.
'
' Purpose:
'   Compare an ex-ante observable curve signal at the investment date
'   with the ex-post realized relative return of locking 6M vs ON.
'
' X-axis:
'   Initial 6M rate - ON rate (bp)
'
' Y-axis:
'   Realized 6M return - ON return (bp)
'
' Note:
'   This is a simple curve-shape diagnostic, not a full measure of
'   market forecast accuracy. The expected future ON path is richer
'   than the spot 6M-minus-ON spread.
'====================================================================

Sub Build_Curve_vs_Outcome()

    Dim wsCurve As Worksheet
    Dim wsSource As Worksheet
    Dim wsOut As Worksheet
    
    Dim lastCurveRow As Long
    Dim lastSourceRow As Long
    Dim i As Long
    Dim outRow As Long
    
    Dim startDate As Date
    Dim curveRow As Variant
    
    Dim rON As Double
    Dim r1M As Double
    Dim r3M As Double
    Dim r6M As Double
    
    Dim spread1M As Double
    Dim spread3M As Double
    Dim spread6M As Double
    
    Dim realized1M As Double
    Dim realized3M As Double
    Dim realized6M As Double
    
    Dim ch As ChartObject
    Dim s As Series
    
    Set wsCurve = ThisWorkbook.Worksheets("curve")
    Set wsSource = ThisWorkbook.Worksheets("Analysis_2")
    
    lastCurveRow = wsCurve.Cells(wsCurve.Rows.Count, "A").End(xlUp).Row
    lastSourceRow = wsSource.Cells(wsSource.Rows.Count, "A").End(xlUp).Row
    
    On Error Resume Next
    Set wsOut = ThisWorkbook.Worksheets("Analysis_3")
    On Error GoTo 0
    
    If wsOut Is Nothing Then
        Set wsOut = ThisWorkbook.Worksheets.Add(After:=wsSource)
        wsOut.Name = "Analysis_3"
    Else
        wsOut.Cells.Clear
        For Each ch In wsOut.ChartObjects
            ch.Delete
        Next ch
    End If
    
    wsOut.Range("A1:K1").Value = Array( _
        "Start", _
        "ON Spot", _
        "1M Spot", _
        "3M Spot", _
        "6M Spot", _
        "1M - ON Curve (bp)", _
        "3M - ON Curve (bp)", _
        "6M - ON Curve (bp)", _
        "Realized 1M - ON (bp)", _
        "Realized 3M - ON (bp)", _
        "Realized 6M - ON (bp)")
    
    outRow = 2
    
    For i = 2 To lastSourceRow
        
        If IsDate(wsSource.Cells(i, 1).Value) Then
            
            startDate = wsSource.Cells(i, 1).Value
            
            curveRow = Application.Match( _
                        CDbl(startDate), _
                        wsCurve.Range("A1:A" & lastCurveRow), _
                        0)
            
            If Not IsError(curveRow) Then
                
                rON = wsCurve.Cells(curveRow, 2).Value
                r1M = wsCurve.Cells(curveRow, 3).Value
                r3M = wsCurve.Cells(curveRow, 4).Value
                r6M = wsCurve.Cells(curveRow, 5).Value
                
                spread1M = (r1M - rON) * 10000
                spread3M = (r3M - rON) * 10000
                spread6M = (r6M - rON) * 10000
                
                realized1M = wsSource.Cells(i, 2).Value
                realized3M = wsSource.Cells(i, 3).Value
                realized6M = wsSource.Cells(i, 4).Value
                
                wsOut.Cells(outRow, 1).Value = startDate
                wsOut.Cells(outRow, 2).Value = rON
                wsOut.Cells(outRow, 3).Value = r1M
                wsOut.Cells(outRow, 4).Value = r3M
                wsOut.Cells(outRow, 5).Value = r6M
                wsOut.Cells(outRow, 6).Value = spread1M
                wsOut.Cells(outRow, 7).Value = spread3M
                wsOut.Cells(outRow, 8).Value = spread6M
                wsOut.Cells(outRow, 9).Value = realized1M
                wsOut.Cells(outRow, 10).Value = realized3M
                wsOut.Cells(outRow, 11).Value = realized6M
                
                outRow = outRow + 1
                
            End If
            
        End If
        
    Next i
    
    wsOut.Columns("A").NumberFormat = "dd-mmm-yyyy"
    wsOut.Columns("B:E").NumberFormat = "0.000%"
    wsOut.Columns("F:K").NumberFormat = "0.0"
    wsOut.Columns.AutoFit
    
    Set ch = wsOut.ChartObjects.Add( _
        Left:=20, _
        Top:=40, _
        Width:=760, _
        Height:=500)
    
    With ch.Chart
        .ChartType = xlXYScatter
        
        Set s = .SeriesCollection.NewSeries
        
        With s
            .Name = "6M"
            .XValues = wsOut.Range("H2:H" & outRow - 1)
            .Values = wsOut.Range("K2:K" & outRow - 1)
            .MarkerStyle = xlMarkerStyleCircle
            .MarkerSize = 5
        End With
        
        .HasTitle = True
        .ChartTitle.Text = _
            "Initial 6M Curve Spread vs Realized 6M Excess Return"
        
        With .Axes(xlCategory)
            .HasTitle = True
            .AxisTitle.Text = _
                "6M Rate - ON Rate at Investment Date (bp)"
            .CrossesAt = 0
        End With
        
        With .Axes(xlValue)
            .HasTitle = True
            .AxisTitle.Text = _
                "Realized 6M Return - ON Return (bp)"
            .CrossesAt = 0
        End With
        
        .HasLegend = False
    End With
    
    MsgBox (outRow - 2) & _
           " observations created in Analysis_3.", _
           vbInformation

End Sub
