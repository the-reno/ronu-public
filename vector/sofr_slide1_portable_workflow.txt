SOFR SLIDE 1 - PORTABLE EXCEL VBA + PYTHON WORKFLOW
Version: 2026-08-05

PURPOSE
-------
1. Build the Slide 1 analysis dataset from an existing Excel curve containing:
   Date | ON | 1M | 3M | 6M
2. Calculate the realized 6M excess return versus compounded ON using R_CURVE.
3. Export the dataset to UTF-8 CSV.
4. Create the minimalist two-panel PNG on any local machine.

PREREQUISITES
-------------
Excel:
- Microsoft Excel with macros enabled.
- The workbook must already contain the R_CURVE VBA function.
- The source curve must be strictly ascending by date with no blank data rows.

Python:
- Python 3.10 or later.
- Install the plotting packages once:

  Windows:
    py -m pip install pandas matplotlib numpy

  macOS/Linux:
    python3 -m pip install pandas matplotlib numpy

RUN THE WORKFLOW
----------------
A. Excel
1. Save the workbook as .xlsm.
2. Press Alt+F11.
3. Choose Insert > Module.
4. Paste the complete VBA section below into the new standard module.
5. Run Build_And_Export_Analysis_6M.
6. Select the curve table including headers.
7. Select the top-left output cell.
8. Choose the CSV destination when prompted.

B. Python
1. Save the Python section below as sofr_slide1.py.
2. Open a terminal in the folder containing the exported CSV.
3. Run:

   Windows:
     py sofr_slide1.py analysis_6m.csv

   macOS/Linux:
     python3 sofr_slide1.py analysis_6m.csv

Optional output path:
  py sofr_slide1.py analysis_6m.csv --output slide_01_sofr_preview.png

Optional resolution:
  py sofr_slide1.py analysis_6m.csv --dpi 300

The Python script accepts either Excel serial dates or ordinary date strings and
accepts comma- or semicolon-delimited CSV files. Rates may be stored as decimals
(0.0525) or percentage points (5.25).

Option Explicit

'===============================================================================
' PORTABLE SLIDE 1 DATASET BUILDER AND CSV EXPORTER
'
' Required existing dependency:
'   R_CURVE(start, table, tenor, adj, [type], [n], [maturity])
'
' Source range headers:
'   Date | ON | 1M | 3M | 6M
'
' Default behavior is portable: Excel asks the user to select the source curve,
' the output cell, and the CSV destination. Set USE_SELECTION_DIALOGS = False
' to use the manual sheet/range variables instead.
'===============================================================================

Public Sub Build_And_Export_Analysis_6M()

    '---------------------------------------------------------------------------
    ' USER SETTINGS
    '---------------------------------------------------------------------------
    Const USE_SELECTION_DIALOGS As Boolean = True

    Const SOURCE_SHEET As String = "Curve"
    Const SOURCE_RANGE As String = "A1:E1128"

    Const OUTPUT_SHEET As String = "Analysis"
    Const OUTPUT_CELL As String = "A1"

    Const ROLL_ADJ As Long = 0       '0 = previous curve date; 1 = next date
    Const CSV_FILE_NAME As String = "analysis_6m.csv"

    '---------------------------------------------------------------------------
    ' VARIABLES
    '---------------------------------------------------------------------------
    Dim tbl As Range
    Dim outCell As Range
    Dim targetRange As Range
    Dim bodyRange As Range
    Dim wsOutput As Worksheet

    Dim v As Variant
    Dim result() As Variant
    Dim headers As Variant

    Dim nRows As Long
    Dim i As Long
    Dim j As Long

    Dim colDate As Long
    Dim colON As Long
    Dim col1M As Long
    Dim col3M As Long
    Dim col6M As Long

    Dim startDate As Double
    Dim previousDate As Double
    Dim lastCurveDate As Double
    Dim targetMaturity As Double
    Dim maturity6M As Variant
    Dim maturitySerial As Double
    Dim factor6M As Variant
    Dim factorON As Variant

    Dim defaultCsvPath As String
    Dim selectedCsvPath As Variant

    On Error GoTo Fail

    If ROLL_ADJ <> 0 And ROLL_ADJ <> 1 Then
        Err.Raise vbObjectError + 1000, , "ROLL_ADJ must be 0 or 1."
    End If

    '---------------------------------------------------------------------------
    ' SELECT OR RESOLVE INPUT AND OUTPUT
    '---------------------------------------------------------------------------
    If USE_SELECTION_DIALOGS Then

        On Error Resume Next
        Set tbl = Application.InputBox( _
            Prompt:="Select the curve table INCLUDING headers." & vbCrLf & _
                    "Required headers: Date, ON, 1M, 3M, 6M", _
            Title:="Select SOFR Curve Data", _
            Type:=8)
        On Error GoTo Fail

        If tbl Is Nothing Then Exit Sub

        On Error Resume Next
        Set outCell = Application.InputBox( _
            Prompt:="Select the TOP-LEFT output cell for the analysis dataset.", _
            Title:="Select Analysis Output", _
            Type:=8)
        On Error GoTo Fail

        If outCell Is Nothing Then Exit Sub
        Set outCell = outCell.Cells(1, 1)

    Else

        Set tbl = ThisWorkbook.Worksheets(SOURCE_SHEET).Range(SOURCE_RANGE)
        Set wsOutput = GetOrCreateWorksheet(ThisWorkbook, OUTPUT_SHEET)
        Set outCell = wsOutput.Range(OUTPUT_CELL)

    End If

    If Not (tbl.Parent.Parent Is ThisWorkbook) Then
        Err.Raise vbObjectError + 1001, , _
            "Select source data in the workbook containing this VBA code."
    End If

    If Not (outCell.Parent.Parent Is ThisWorkbook) Then
        Err.Raise vbObjectError + 1002, , _
            "Select an output cell in the workbook containing this VBA code."
    End If

    Set wsOutput = outCell.Worksheet
    v = tbl.Value2

    If Not IsArray(v) Then
        Err.Raise vbObjectError + 1003, , "The selected source range is invalid."
    End If

    nRows = UBound(v, 1)
    If nRows < 2 Then
        Err.Raise vbObjectError + 1004, , "The selected curve has no data rows."
    End If

    '---------------------------------------------------------------------------
    ' FIND AND VALIDATE REQUIRED COLUMNS
    '---------------------------------------------------------------------------
    colDate = FindHeader(v, "Date")
    colON = FindHeader(v, "ON")
    col1M = FindHeader(v, "1M")
    col3M = FindHeader(v, "3M")
    col6M = FindHeader(v, "6M")

    If colDate = 0 Or colON = 0 Or col1M = 0 Or col3M = 0 Or col6M = 0 Then
        Err.Raise vbObjectError + 1005, , _
            "Could not find Date, ON, 1M, 3M, and 6M headers."
    End If

    previousDate = 0
    For i = 2 To nRows
        startDate = CoerceDateSerial(v(i, colDate))

        If startDate <= 0 Then
            Err.Raise vbObjectError + 1006, , _
                "Invalid date in source row " & i & "."
        End If

        If previousDate > 0 And startDate <= previousDate Then
            Err.Raise vbObjectError + 1007, , _
                "Curve dates must be strictly increasing. Check source row " & i & "."
        End If

        previousDate = startDate
    Next i

    lastCurveDate = previousDate

    'Prevent the analysis output from overwriting the selected source table.
    If tbl.Parent Is wsOutput Then
        If Not Intersect(tbl, outCell.Resize(nRows, 15)) Is Nothing Then
            Err.Raise vbObjectError + 1008, , _
                "The output range would overlap the selected curve data."
        End If
    End If

    '---------------------------------------------------------------------------
    ' OUTPUT HEADERS
    '---------------------------------------------------------------------------
    headers = Array( _
        "StartDate", "ON", "1M", "3M", "6M", "6M_ON_Spread_bp", _
        "Maturity6M", "Days6M", "Factor6M", "FactorON6M", _
        "Return6M", "ReturnON6M", "Excess6M_bp", _
        "RelativeExcess6M_bp", "Valid6M")

    ReDim result(1 To nRows, 1 To 15)

    For j = 0 To UBound(headers)
        result(1, j + 1) = headers(j)
    Next j

    '---------------------------------------------------------------------------
    ' BUILD THE ANALYSIS DATASET
    '---------------------------------------------------------------------------
    For i = 2 To nRows

        startDate = CoerceDateSerial(v(i, colDate))

        result(i, 1) = CDate(startDate)
        result(i, 2) = v(i, colON)
        result(i, 3) = v(i, col1M)
        result(i, 4) = v(i, col3M)
        result(i, 5) = v(i, col6M)
        result(i, 15) = 0

        If IsNumeric(v(i, col6M)) And IsNumeric(v(i, colON)) Then
            result(i, 6) = _
                (CDbl(v(i, col6M)) - CDbl(v(i, colON))) * 10000#
        End If

        targetMaturity = CDbl(DateAdd("m", 6, CDate(startDate)))

        If targetMaturity <= lastCurveDate Then

            maturity6M = R_CURVE( _
                startDate, tbl, "6M", ROLL_ADJ, _
                "DATE", 0, targetMaturity)

            maturitySerial = CoerceDateSerial(maturity6M)

            If maturitySerial > startDate Then

                factor6M = R_CURVE( _
                    startDate, tbl, "6M", ROLL_ADJ, _
                    "FACTOR", 0, maturitySerial)

                factorON = R_CURVE( _
                    startDate, tbl, "ON", ROLL_ADJ, _
                    "FACTOR", 0, maturitySerial)

                If IsNumeric(factor6M) And IsNumeric(factorON) Then

                    result(i, 7) = CDate(maturitySerial)
                    result(i, 8) = maturitySerial - startDate

                    result(i, 9) = CDbl(factor6M)
                    result(i, 10) = CDbl(factorON)

                    result(i, 11) = CDbl(factor6M) - 1#
                    result(i, 12) = CDbl(factorON) - 1#

                    result(i, 13) = _
                        (CDbl(factor6M) - CDbl(factorON)) * 10000#

                    result(i, 14) = _
                        (CDbl(factor6M) / CDbl(factorON) - 1#) * 10000#

                    result(i, 15) = 1

                End If
            End If
        End If
    Next i

    '---------------------------------------------------------------------------
    ' WRITE AND FORMAT OUTPUT
    '---------------------------------------------------------------------------
    Set targetRange = outCell.Resize(nRows, 15)
    targetRange.Value = result

    With targetRange.Rows(1)
        .Font.Bold = True
        .Font.Color = RGB(255, 255, 255)
        .Interior.Color = RGB(23, 54, 93)
    End With

    Set bodyRange = targetRange.Offset(1, 0).Resize(nRows - 1, 15)

    bodyRange.Columns(1).NumberFormat = "yyyy-mm-dd"
    bodyRange.Columns(7).NumberFormat = "yyyy-mm-dd"
    bodyRange.Columns(2).Resize(, 4).NumberFormat = "0.0000%"
    bodyRange.Columns(9).Resize(, 2).NumberFormat = "0.000000"
    bodyRange.Columns(11).Resize(, 2).NumberFormat = "0.0000%"
    bodyRange.Columns(6).NumberFormat = "0.0"
    bodyRange.Columns(13).Resize(, 2).NumberFormat = "0.0"

    targetRange.Columns.AutoFit

    If Not wsOutput.AutoFilterMode Then targetRange.AutoFilter

    '---------------------------------------------------------------------------
    ' CHOOSE CSV DESTINATION AND EXPORT
    '---------------------------------------------------------------------------
    defaultCsvPath = DefaultOutputPath(CSV_FILE_NAME)

    selectedCsvPath = Application.GetSaveAsFilename( _
        InitialFileName:=defaultCsvPath, _
        FileFilter:="CSV UTF-8 (*.csv), *.csv", _
        Title:="Save the Slide 1 analysis dataset")

    If VarType(selectedCsvPath) = vbBoolean Then
        If selectedCsvPath = False Then
            MsgBox "Analysis dataset created. CSV export was cancelled.", vbInformation
            Exit Sub
        End If
    End If

    ExportRangeToCsvUtf8 targetRange, EnsureCsvExtension(CStr(selectedCsvPath))

    MsgBox _
        "Analysis dataset created and exported:" & vbCrLf & _
        EnsureCsvExtension(CStr(selectedCsvPath)), _
        vbInformation

    Exit Sub

Fail:
    MsgBox "Build/export failed:" & vbCrLf & Err.Description, vbExclamation

End Sub


Private Function FindHeader(ByVal values As Variant, ByVal headerName As String) As Long

    Dim columnIndex As Long

    FindHeader = 0

    For columnIndex = 1 To UBound(values, 2)
        If UCase$(Trim$(CStr(values(1, columnIndex)))) = _
           UCase$(Trim$(headerName)) Then

            FindHeader = columnIndex
            Exit Function
        End If
    Next columnIndex

End Function


Private Function CoerceDateSerial(ByVal value As Variant) As Double

    CoerceDateSerial = 0

    If IsError(value) Or IsEmpty(value) Then Exit Function

    If IsNumeric(value) Then
        CoerceDateSerial = CDbl(value)
    ElseIf IsDate(value) Then
        CoerceDateSerial = CDbl(CDate(value))
    End If

End Function


Private Function GetOrCreateWorksheet( _
    ByVal workbookObject As Workbook, _
    ByVal sheetName As String) As Worksheet

    On Error Resume Next
    Set GetOrCreateWorksheet = workbookObject.Worksheets(sheetName)
    On Error GoTo 0

    If GetOrCreateWorksheet Is Nothing Then
        Set GetOrCreateWorksheet = workbookObject.Worksheets.Add( _
            After:=workbookObject.Worksheets(workbookObject.Worksheets.Count))
        GetOrCreateWorksheet.Name = sheetName
    End If

End Function


Private Function DefaultOutputPath(ByVal fileName As String) As String

    Dim folderPath As String

    folderPath = ThisWorkbook.Path

    If Len(folderPath) = 0 Then
        folderPath = Environ$("USERPROFILE") & Application.PathSeparator & "Desktop"
    End If

    DefaultOutputPath = _
        folderPath & Application.PathSeparator & fileName

End Function


Private Function EnsureCsvExtension(ByVal filePath As String) As String

    If LCase$(Right$(filePath, 4)) = ".csv" Then
        EnsureCsvExtension = filePath
    Else
        EnsureCsvExtension = filePath & ".csv"
    End If

End Function


Private Sub ExportRangeToCsvUtf8( _
    ByVal sourceRange As Range, _
    ByVal csvPath As String)

    Dim csvWorkbook As Workbook
    Dim csvWorksheet As Worksheet
    Dim destination As Range

    Dim priorDisplayAlerts As Boolean
    Dim errorNumber As Long
    Dim errorDescription As String

    On Error GoTo ExportFail

    priorDisplayAlerts = Application.DisplayAlerts

    Set csvWorkbook = Workbooks.Add(xlWBATWorksheet)
    Set csvWorksheet = csvWorkbook.Worksheets(1)
    Set destination = csvWorksheet.Range("A1").Resize( _
        sourceRange.Rows.Count, sourceRange.Columns.Count)

    sourceRange.Copy
    destination.PasteSpecial xlPasteValuesAndNumberFormats
    Application.CutCopyMode = False

    'Force machine-independent date text in the CSV.
    destination.Columns(1).NumberFormat = "yyyy-mm-dd"
    destination.Columns(7).NumberFormat = "yyyy-mm-dd"

    Application.DisplayAlerts = False

    csvWorkbook.SaveAs _
        Filename:=csvPath, _
        FileFormat:=xlCSVUTF8, _
        CreateBackup:=False, _
        Local:=False

    csvWorkbook.Close SaveChanges:=False
    Application.DisplayAlerts = priorDisplayAlerts

    Exit Sub

ExportFail:
    errorNumber = Err.Number
    errorDescription = Err.Description

    Application.CutCopyMode = False
    Application.DisplayAlerts = priorDisplayAlerts

    If Not csvWorkbook Is Nothing Then
        csvWorkbook.Close SaveChanges:=False
    End If

    Err.Raise errorNumber, "ExportRangeToCsvUtf8", errorDescription

End Sub

from __future__ import annotations

import argparse
from pathlib import Path

import matplotlib.dates as mdates
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd


RATE_COLUMNS = ["ON", "1M", "3M", "6M"]


def detect_delimiter(path: Path) -> str:
    """Detect the delimiter from the header written by Excel."""
    with path.open("r", encoding="utf-8-sig", errors="replace") as handle:
        header = next((line for line in handle if line.strip()), "")

    candidates = [",", ";", "\t", "|"]
    delimiter = max(candidates, key=header.count)
    return delimiter if header.count(delimiter) else ","


def to_numeric_mixed(series: pd.Series, delimiter: str) -> pd.Series:
    """Convert values written with either decimal points or decimal commas."""
    text = series.astype("string").str.strip().str.replace("\u00a0", "", regex=False)
    if delimiter == ";":
        text = text.str.replace(",", ".", regex=False)
    return pd.to_numeric(text, errors="coerce")


def parse_excel_or_text_dates(series: pd.Series) -> pd.Series:
    """Accept ISO/text dates and Excel serial dates in the same column."""
    text = series.astype("string").str.strip()
    numeric = pd.to_numeric(text, errors="coerce")
    parsed = pd.Series(pd.NaT, index=series.index, dtype="datetime64[ns]")

    text_mask = numeric.isna() & text.notna()
    parsed.loc[text_mask] = pd.to_datetime(text.loc[text_mask], errors="coerce")

    numeric_mask = numeric.notna()
    parsed.loc[numeric_mask] = (
        pd.Timestamp("1899-12-30")
        + pd.to_timedelta(numeric.loc[numeric_mask], unit="D")
    )
    return parsed


def load_analysis_csv(path: Path) -> tuple[pd.DataFrame, pd.DataFrame]:
    delimiter = detect_delimiter(path)
    frame = pd.read_csv(path, sep=delimiter, encoding="utf-8-sig", dtype="string")

    required = {"StartDate", "ON", "1M", "3M", "6M", "Excess6M_bp"}
    missing = sorted(required.difference(frame.columns))
    if missing:
        raise ValueError(f"Missing required CSV columns: {', '.join(missing)}")

    frame["StartDate"] = parse_excel_or_text_dates(frame["StartDate"])
    for column in RATE_COLUMNS + ["Excess6M_bp"]:
        frame[column] = to_numeric_mixed(frame[column], delimiter)

    frame = (
        frame.dropna(subset=["StartDate"])
        .sort_values("StartDate")
        .drop_duplicates(subset=["StartDate"], keep="last")
        .reset_index(drop=True)
    )

    if frame.empty:
        raise ValueError("The CSV contains no usable dated observations.")

    rate_values = frame[RATE_COLUMNS].stack().abs()
    if rate_values.empty:
        raise ValueError("The CSV contains no usable rate observations.")

    # Excel analysis files normally store 5.25% as 0.0525. If the CSV already
    # stores percentage points (5.25), leave it unchanged.
    rate_multiplier = 100.0 if rate_values.quantile(0.95) <= 1.0 else 1.0
    frame[RATE_COLUMNS] = frame[RATE_COLUMNS] * rate_multiplier

    if "Valid6M" in frame.columns:
        valid = to_numeric_mixed(frame["Valid6M"], delimiter).eq(1)
    else:
        valid = frame["Excess6M_bp"].notna()

    realized = frame.loc[valid & frame["Excess6M_bp"].notna()].copy()
    if realized.empty:
        raise ValueError("No fully realized 6M observations were found.")

    return frame, realized


def build_slide(frame: pd.DataFrame, realized: pd.DataFrame, output: Path, dpi: int) -> None:
    colors = {
        "ON": "#B7BDC4",
        "1M": "#949BA4",
        "3M": "#59616B",
        "6M": "#17365D",
        "text": "#252A31",
        "muted": "#737B85",
        "grid": "#E5E8EC",
        "zero": "#8B929A",
    }

    plt.rcParams.update(
        {
            "font.family": "Arial",
            "axes.edgecolor": colors["grid"],
            "axes.labelcolor": colors["text"],
            "xtick.color": colors["muted"],
            "ytick.color": colors["muted"],
        }
    )

    figure = plt.figure(figsize=(13.333, 7.5), facecolor="white")
    top = figure.add_axes([0.075, 0.42, 0.88, 0.40])
    bottom = figure.add_axes([0.075, 0.13, 0.88, 0.20])

    figure.text(
        0.075,
        0.94,
        "Short-end repricing changed the value of locking 6M",
        fontsize=22,
        fontweight="bold",
        color=colors["text"],
        ha="left",
        va="top",
    )
    figure.add_artist(
        plt.Line2D([0.075, 0.13], [0.968, 0.968], color=colors["6M"], linewidth=3)
    )

    for column in RATE_COLUMNS:
        top.plot(
            frame["StartDate"],
            frame[column],
            color=colors[column],
            linewidth=2.0 if column == "6M" else 1.05,
            label=column,
            zorder=4 if column == "6M" else 2,
        )

    top.text(
        0,
        1.055,
        "SOFR rates | %",
        transform=top.transAxes,
        fontsize=11,
        fontweight="bold",
        color=colors["text"],
        ha="left",
    )
    top.text(
        1,
        1.055,
        f"{frame['StartDate'].min():%b %Y} – {frame['StartDate'].max():%b %Y}",
        transform=top.transAxes,
        fontsize=9,
        color=colors["muted"],
        ha="right",
    )

    legend = top.legend(
        loc="upper center",
        bbox_to_anchor=(0.69, 1.085),
        ncol=4,
        frameon=False,
        handlelength=2.2,
        columnspacing=1.5,
        fontsize=9,
    )
    for text, column in zip(legend.get_texts(), RATE_COLUMNS):
        text.set_color(colors[column])
        if column == "6M":
            text.set_fontweight("bold")

    latest_6m = frame.dropna(subset=["6M"]).iloc[-1]
    top.annotate(
        f"6M  {latest_6m['6M']:.2f}%",
        xy=(latest_6m["StartDate"], latest_6m["6M"]),
        xytext=(-6, -16),
        textcoords="offset points",
        ha="right",
        va="center",
        fontsize=9,
        fontweight="bold",
        color=colors["6M"],
        bbox={"facecolor": "white", "edgecolor": "none", "pad": 1.5},
    )

    bottom.plot(
        realized["StartDate"],
        realized["Excess6M_bp"],
        color=colors["6M"],
        linewidth=1.6,
        zorder=3,
    )
    bottom.axhline(0, color=colors["zero"], linewidth=0.9, zorder=2)

    bottom.text(
        0,
        1.09,
        "Realized 6M excess return vs ON | bp",
        transform=bottom.transAxes,
        fontsize=11,
        fontweight="bold",
        color=colors["text"],
        ha="left",
    )
    bottom.text(
        1,
        1.09,
        "Positive = 6M outperformed",
        transform=bottom.transAxes,
        fontsize=9,
        color=colors["muted"],
        ha="right",
    )

    last_realized = realized.iloc[-1]
    bottom.annotate(
        f"{last_realized['Excess6M_bp']:.0f} bp",
        xy=(last_realized["StartDate"], last_realized["Excess6M_bp"]),
        xytext=(7, 0),
        textcoords="offset points",
        fontsize=8.5,
        color=colors["6M"],
        va="center",
    )
    bottom.text(
        0.995,
        0.05,
        f"Realized through {last_realized['StartDate']:%b-%y}",
        transform=bottom.transAxes,
        fontsize=8.5,
        color=colors["muted"],
        ha="right",
        va="bottom",
    )

    top_max = float(np.nanmax(frame[RATE_COLUMNS].to_numpy()))
    top_limit = max(1.0, np.ceil((top_max + 0.15) * 2) / 2)
    top.set_ylim(0, top_limit)

    realized_min = float(realized["Excess6M_bp"].min())
    realized_max = float(realized["Excess6M_bp"].max())
    bottom_min = min(-5.0, np.floor(realized_min / 5) * 5)
    bottom_max = max(5.0, np.ceil(realized_max / 5) * 5)
    bottom.set_ylim(bottom_min, bottom_max)

    full_start = frame["StartDate"].min()
    full_end = frame["StartDate"].max()
    axis_start = pd.Timestamp(year=full_start.year, month=1, day=1)
    top.set_xlim(axis_start, full_end)
    bottom.set_xlim(axis_start, full_end)

    for axis in (top, bottom):
        axis.grid(axis="y", color=colors["grid"], linewidth=0.6)
        axis.spines["top"].set_visible(False)
        axis.spines["right"].set_visible(False)
        axis.spines["left"].set_visible(False)
        axis.tick_params(axis="both", which="both", length=0, labelsize=9)

    top.spines["bottom"].set_visible(False)
    top.tick_params(axis="x", labelbottom=False)
    bottom.spines["bottom"].set_color(colors["grid"])
    bottom.xaxis.set_major_locator(mdates.YearLocator())
    bottom.xaxis.set_major_formatter(mdates.DateFormatter("%Y"))

    figure.text(
        0.075,
        0.045,
        "6M return compared with compounded ON over the same realized maturity horizon.",
        fontsize=8,
        color=colors["muted"],
        ha="left",
    )
    figure.text(
        0.955,
        0.045,
        "Source: user-provided SOFR curve workbook and analysis dataset",
        fontsize=8,
        color=colors["muted"],
        ha="right",
    )

    output.parent.mkdir(parents=True, exist_ok=True)
    figure.savefig(output, dpi=dpi, facecolor="white")
    plt.close(figure)


def main() -> None:
    parser = argparse.ArgumentParser(
        description="Create the Slide 1 SOFR rate and realized 6M excess-return preview."
    )
    parser.add_argument(
        "csv_file",
        nargs="?",
        default="analysis_6m.csv",
        help="CSV exported by the Excel VBA macro (default: analysis_6m.csv).",
    )
    parser.add_argument(
        "--output",
        help="Output PNG path. Defaults to slide_01_sofr_preview.png beside the CSV.",
    )
    parser.add_argument("--dpi", type=int, default=300, help="PNG resolution (default: 300).")
    args = parser.parse_args()

    csv_path = Path(args.csv_file).expanduser().resolve()
    if not csv_path.exists():
        raise FileNotFoundError(f"CSV file not found: {csv_path}")

    output_path = (
        Path(args.output).expanduser().resolve()
