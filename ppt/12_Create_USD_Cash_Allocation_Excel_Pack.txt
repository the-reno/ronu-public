'
' PURPOSE
'   Creates the complete slide-ready data, charts, committee comments and audit trail

'   Designed for a workbook containing one required source sheet:
'     Curve          Date | ON | 1M | 3M | 6M
'   The historical evidence, inverted-path stress and forward allocation engine
'   are rebuilt from that curve. Forward scenario assumptions are explicit on

' OUTPUT SHEETS
'   PPT_Index          manual PowerPoint workflow and source map

'   PPT_S5_Decision    risk controls, execution inputs and evidence ownership
'   PPT_Audit          model checks and reference-value reconciliation
'
' USE
'   1. Save 13-aug.xlsx as an Excel Macro-Enabled Workbook (*.xlsm).
'   2. Press Alt+F11 and insert a standard Module.
'   3. Paste this entire TXT file into the Module.
'   4. Run USD_CreateCashAllocationExcelPack.
'   5. Copy the Excel charts and nearby comments into PowerPoint manually.
'====================================================================================

Public Sub USD_CreateCashAllocationExcelPack()

    Dim oldCalc As Long
    Dim oldScreen As Boolean
    Dim oldEvents As Boolean
    Dim errText As String

    On Error GoTo fail

    oldCalc = Application.Calculation
    oldScreen = Application.ScreenUpdating
    oldEvents = Application.EnableEvents
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    If ThisWorkbook.ProtectStructure Then _
        Err.Raise vbObjectError + 3099, , "Workbook structure is protected. Unprotect it before running the module."
    Application.StatusBar = "Building the USD cash-allocation Excel slide pack..."

    USD_ResetState
    USD_ValidateSourceWorkbook
    USD_LoadCurve
    USD_LoadForwardAssumptions
    USD_CalculateDailyEvidence
    USD_CalculateDirectionalStress
    USD_CalculateForwardAllocation

    USD_BuildIndex
    USD_BuildSlide1Sheet
    USD_BuildSlide2Sheet
    USD_BuildSlide3Sheet
    USD_BuildSlide4Sheet
    USD_BuildSlide5Sheet
    USD_BuildAuditSheet

    Application.CalculateFull
    ThisWorkbook.Worksheets(S_INDEX).Activate
    ThisWorkbook.Worksheets(S_INDEX).Range("A1").Select

clean_exit:
    Application.StatusBar = False
    Application.Calculation = oldCalc
    Application.EnableEvents = oldEvents
    Application.ScreenUpdating = oldScreen
    If Len(errText) = 0 Then
        MsgBox "Excel slide pack created." & vbCrLf & vbCrLf & _
               "No PowerPoint file was created." & vbCrLf & _
               "Use PPT_Index to copy charts and comments manually.", _
               vbInformation, "USD cash allocation"
    End If
    Exit Sub

fail:
    errText = Err.Description
    MsgBox "USD_CreateCashAllocationExcelPack stopped:" & vbCrLf & vbCrLf & errText, _
           vbCritical, "USD cash allocation"
    Resume clean_exit

End Sub

Private Sub USD_WriteHoldingExample(ByVal ws As Worksheet, ByVal targetRow As Long, _
                                    ByVal dailyIndex As Long, ByVal noteText As String)
    If dailyIndex = 0 Then Err.Raise vbObjectError + 3130, , "Required holding-period example was not found."
    ws.Cells(targetRow, 16).Value = mDailyDate(dailyIndex)
    ws.Cells(targetRow, 17).Value = mDailySpread(dailyIndex)
    ws.Cells(targetRow, 18).Value = mDailyRolledON(dailyIndex)
    ws.Cells(targetRow, 19).Value = mDailyLocked6M(dailyIndex)
    ws.Cells(targetRow, 20).Value = mDailyExcess(dailyIndex)
    ws.Cells(targetRow, 21).Value = IIf(mDailyExcess(dailyIndex) > 0#, "6M", "ON")
    ws.Cells(targetRow, 22).Value = noteText
End Sub

Private Function USD_DailyIndex(ByVal decisionDate As Double) As Long
    Dim i As Long
    For i = 1 To mDailyN
        If Abs(mDailyDate(i) - decisionDate) < 0.00001 Then USD_DailyIndex = i: Exit Function
    Next i
End Function

Private Function USD_GetOrCreateSheet(ByVal sheetName As String) As Worksheet
    If USD_SheetExists(sheetName) Then
        Set USD_GetOrCreateSheet = ThisWorkbook.Worksheets(sheetName)
    Else
        Set USD_GetOrCreateSheet = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
        USD_GetOrCreateSheet.Name = sheetName
    End If
End Function

Private Sub USD_ResetOutputSheet(ByVal ws As Worksheet)
    Dim i As Long
    On Error Resume Next
    For i = ws.ListObjects.Count To 1 Step -1
        ws.ListObjects(i).Unlist
    Next i
    For i = ws.Shapes.Count To 1 Step -1
        ws.Shapes(i).Delete
    Next i
    ws.Cells.UnMerge
    ws.Cells.Clear
    ws.Cells.ClearFormats
    On Error GoTo 0
End Sub

Private Sub USD_SetCanvas(ByVal ws As Worksheet, ByVal canvasColumns As Long)
    Dim i As Long
    With ws.Cells
        .Font.Name = FONT_NAME
        .Font.Size = 10
        .Font.Color = USD_Gray()
        .VerticalAlignment = xlCenter
    End With
    For i = 1 To canvasColumns
        ws.Columns(i).ColumnWidth = 9
    Next i
    ws.Rows("1:2").RowHeight = 27
    ws.Rows("3:39").RowHeight = 20
    ws.Tab.Color = USD_Blue()
    On Error Resume Next
    ws.Activate
    ActiveWindow.DisplayGridlines = False
    ActiveWindow.Zoom = 82
    On Error GoTo 0
End Sub

Private Sub USD_AddAudit(ByVal ws As Worksheet, ByRef targetRow As Long, _
                         ByVal checkName As String, ByVal actualValue As Variant, _
                         ByVal expectedValue As Variant, ByVal tolerance As Double, _
                         ByVal sourceText As String, ByVal riskNote As String, _
                         ByRef allPass As Boolean, Optional ByVal numberFormat As String = "General")
    Dim passed As Boolean
    If VarType(actualValue) = vbBoolean Or VarType(expectedValue) = vbBoolean Then
        passed = (CBool(actualValue) = CBool(expectedValue))
    ElseIf IsNumeric(actualValue) And IsNumeric(expectedValue) Then
        passed = (Abs(CDbl(actualValue) - CDbl(expectedValue)) <= tolerance)
    Else
        passed = (CStr(actualValue) = CStr(expectedValue))
    End If
    ws.Cells(targetRow, 1).Value = checkName
    ws.Cells(targetRow, 2).Value = actualValue
    ws.Cells(targetRow, 3).Value = expectedValue
    ws.Cells(targetRow, 4).Value = tolerance
    ws.Cells(targetRow, 5).Value = IIf(passed, "PASS", "REVIEW")
    ws.Cells(targetRow, 6).Value = sourceText
    ws.Cells(targetRow, 7).Value = riskNote
    ws.Cells(targetRow, 2).NumberFormat = numberFormat
    ws.Cells(targetRow, 3).NumberFormat = numberFormat
    If Not passed Then allPass = False
    targetRow = targetRow + 1
End Sub

Private Sub USD_LoadCurve()

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim v As Variant
    Dim i As Long
    Dim j As Long

    Set ws = ThisWorkbook.Worksheets("Curve")
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    If lastRow < 3 Then Err.Raise vbObjectError + 3102, , "Curve has no usable observations."

    If UCase$(Trim$(CStr(ws.Cells(1, 1).Value))) <> "DATE" Then _
        Err.Raise vbObjectError + 3103, , "Curve column A must be Date."
    If UCase$(Trim$(CStr(ws.Cells(1, 2).Value))) <> "ON" Then _
        Err.Raise vbObjectError + 3104, , "Curve column B must be ON."
    If UCase$(Trim$(CStr(ws.Cells(1, 3).Value))) <> "1M" Or _
       UCase$(Trim$(CStr(ws.Cells(1, 4).Value))) <> "3M" Or _
       UCase$(Trim$(CStr(ws.Cells(1, 5).Value))) <> "6M" Then _
        Err.Raise vbObjectError + 3105, , "Curve columns B:E must be ON, 1M, 3M and 6M."

    v = ws.Range("A2:E" & lastRow).Value2
    mCurveN = UBound(v, 1)
    ReDim mDate(1 To mCurveN)
    ReDim mRate(1 To 4, 1 To mCurveN)

    For i = 1 To mCurveN
        If Not IsNumeric(v(i, 1)) Then _
            Err.Raise vbObjectError + 3106, , "Curve contains a non-date value at row " & CStr(i + 1) & "."
        mDate(i) = CDbl(v(i, 1))
        If i > 1 Then
            If mDate(i) <= mDate(i - 1) Then _
                Err.Raise vbObjectError + 3107, , "Curve dates must be strictly increasing."
        End If
        For j = 1 To 4
            If Not IsNumeric(v(i, j + 1)) Then _
                Err.Raise vbObjectError + 3108, , "Curve contains a blank/non-numeric rate at row " & CStr(i + 1) & "."
            mRate(j, i) = CDbl(v(i, j + 1))
        Next j
    Next i

End Sub

Private Sub USD_CalculateDailyEvidence()

    Dim i As Long
    Dim n As Long
    Dim lastDate As Double
    Dim onRet As Double
    Dim sixRet As Double
    Dim meanX As Double
    Dim meanY As Double
    Dim sumXX As Double
    Dim sumYY As Double
    Dim sumXY As Double
    Dim dx As Double
    Dim dy As Double
    Dim hits As Long
    Dim sx() As Double
    Dim sy() As Double
    Dim q4Count As Long
    Dim q4Start As Long
    Dim q4Sum As Double
    Dim q4Hits As Long

    lastDate = mDate(mCurveN)
    For i = 1 To mCurveN
        If CDbl(DateAdd("m", 6, CDate(mDate(i)))) <= lastDate Then n = n + 1
    Next i
    If n = 0 Then Err.Raise vbObjectError + 3110, , "Curve does not contain a complete six-month horizon."

    mDailyN = n
    ReDim mDailyDate(1 To n)
    ReDim mDailySpread(1 To n)
    ReDim mDailyExcess(1 To n)
    ReDim mDailyRolledON(1 To n)
    ReDim mDailyLocked6M(1 To n)

    n = 0
    For i = 1 To mCurveN
        If CDbl(DateAdd("m", 6, CDate(mDate(i)))) <= lastDate Then
            n = n + 1
            onRet = USD_PathReturn(i, 1, 0, 0)
            sixRet = USD_PathReturn(i, 4, 6, 0)
            mDailyDate(n) = mDate(i)
            mDailySpread(n) = (mRate(4, i) - mRate(1, i)) * 10000#
            mDailyRolledON(n) = onRet
            mDailyLocked6M(n) = sixRet
            mDailyExcess(n) = (sixRet - onRet) * 10000#
            meanX = meanX + mDailySpread(n)
            meanY = meanY + mDailyExcess(n)
            If mDailyExcess(n) > 0# Then hits = hits + 1
        End If
    Next i

    meanX = meanX / mDailyN
    meanY = meanY / mDailyN
    For i = 1 To mDailyN
        dx = mDailySpread(i) - meanX
        dy = mDailyExcess(i) - meanY
        sumXX = sumXX + dx * dx
        sumYY = sumYY + dy * dy
        sumXY = sumXY + dx * dy
    Next i
    If sumXX = 0# Or sumYY = 0# Then _
        Err.Raise vbObjectError + 3111, , "Regression cannot be calculated from constant data."

    mSlope = sumXY / sumXX
    mIntercept = meanY - mSlope * meanX
    mCorr = sumXY / Sqr(sumXX * sumYY)
    mR2 = mCorr * mCorr
    mHitRate = hits / mDailyN

    ReDim sx(1 To mDailyN)
    ReDim sy(1 To mDailyN)
    For i = 1 To mDailyN
        sx(i) = mDailySpread(i)
        sy(i) = mDailyExcess(i)
    Next i
    USD_QuickSortPair sx, sy, 1, mDailyN
    q4Count = (mDailyN + 3) \ 4
    q4Start = mDailyN - q4Count + 1
    For i = q4Start To mDailyN
        q4Sum = q4Sum + sy(i)
        If sy(i) > 0# Then q4Hits = q4Hits + 1
    Next i
    mQ4Mean = q4Sum / q4Count
    mQ4Hit = q4Hits / q4Count

End Sub

Private Sub USD_CalculateDirectionalStress()

    Dim starts() As Long
    Dim count As Long
    Dim i As Long
    Dim monthKey As Long
    Dim priorKey As Long
    Dim tenor As Long
    Dim bestUp As Long
    Dim bestDown As Long
    Dim dailyMean As Double
    Dim meanX As Double
    Dim sumXX As Double
    Dim sumYY As Double
    Dim sumXY As Double
    Dim dx As Double
    Dim dy As Double
    Dim hits As Long

    ReDim mStressDailyExcess(1 To mDailyN)
    For i = 1 To mDailyN
        mStressDailyExcess(i) = _
            (USD_PathReturn(USD_FindExactIndex(mDailyDate(i)), 4, 6, 2) - _
             USD_PathReturn(USD_FindExactIndex(mDailyDate(i)), 1, 0, 2)) * 10000#
        dailyMean = dailyMean + mStressDailyExcess(i)
        meanX = meanX + mDailySpread(i)
        If mStressDailyExcess(i) > 0# Then hits = hits + 1
    Next i
    mStressDailyMean = dailyMean / mDailyN
    mStressDailyHit = hits / mDailyN
    meanX = meanX / mDailyN
    For i = 1 To mDailyN
        dx = mDailySpread(i) - meanX
        dy = mStressDailyExcess(i) - mStressDailyMean
        sumXX = sumXX + dx * dx
        sumYY = sumYY + dy * dy
        sumXY = sumXY + dx * dy
    Next i
    If sumXX > 0# And sumYY > 0# Then
        mStressSlope = sumXY / sumXX
        mStressCorr = sumXY / Sqr(sumXX * sumYY)
    End If

    ReDim starts(1 To mCurveN)
    For i = 1 To mCurveN
        If CDbl(DateAdd("m", 6, CDate(mDate(i)))) <= mDate(mCurveN) Then
            monthKey = Year(CDate(mDate(i))) * 100 + Month(CDate(mDate(i)))
            If monthKey <> priorKey Then
                count = count + 1
                starts(count) = i
                priorKey = monthKey
            End If
        End If
    Next i
    If count = 0 Then Err.Raise vbObjectError + 3112, , "No monthly scenario starts were available."

    mMonthlyN = count
    ReDim mMonthlyStart(1 To count)
    ReDim mUpReturn(1 To count, 1 To 4)
    ReDim mDownReturn(1 To count, 1 To 4)

    For i = 1 To count
        mMonthlyStart(i) = starts(i)
        For tenor = 1 To 4
            mUpReturn(i, tenor) = USD_PathReturn(starts(i), tenor, USD_TenorMonths(tenor), 1)
            mDownReturn(i, tenor) = USD_PathReturn(starts(i), tenor, USD_TenorMonths(tenor), 2)
            mUpExcess(tenor) = mUpExcess(tenor) + (mUpReturn(i, tenor) - mUpReturn(i, 1)) * 10000#
            mDownExcess(tenor) = mDownExcess(tenor) + (mDownReturn(i, tenor) - mDownReturn(i, 1)) * 10000#
        Next tenor
        bestUp = USD_BestTenor(mUpReturn, i)
        bestDown = USD_BestTenor(mDownReturn, i)
        mUpWins(bestUp) = mUpWins(bestUp) + 1
        mDownWins(bestDown) = mDownWins(bestDown) + 1
    Next i

    For tenor = 1 To 4
        mUpExcess(tenor) = mUpExcess(tenor) / count
        mDownExcess(tenor) = mDownExcess(tenor) / count
    Next tenor

End Sub

Private Sub USD_CalculateForwardAllocation()

    Dim scenario As Long
    Dim tenor As Long
    Dim monthNo As Long
    Dim mixNo As Long
    Dim onUnits As Long
    Dim oneUnits As Long
    Dim threeUnits As Long
    Dim sixUnits As Long
    Dim fade As Double
    Dim baseValue As Double
    Dim currentValue As Double
    Dim wealth As Double
    Dim stepMonths As Long
    Dim weightedCumulative As Double
    Dim bestWorst As Double
    Dim regretSum As Double

    mForwardStartDate = mDate(mCurveN)
    'Step 1: create five directly comparable paths from the latest observed curve.
    For scenario = 1 To 5
        fade = mForwardFadeInput(scenario)
        For monthNo = 0 To 36
            mForwardPath(scenario, 1, monthNo) = _
                USD_ForwardONRate(scenario, monthNo, mRate(1, mCurveN))
        Next monthNo

        baseValue = mForwardPath(scenario, 1, 0) + USD_ForwardPremium(scenario, 2) / 10000#
        For monthNo = 0 To 36
            currentValue = mForwardPath(scenario, 1, monthNo) + USD_ForwardPremium(scenario, 2) / 10000#
            mForwardPath(scenario, 2, monthNo) = _
                USD_Maximum(0#, currentValue + (mRate(2, mCurveN) - baseValue) * USD_Maximum(0#, 1# - monthNo / fade))
        Next monthNo

        baseValue = 0.65 * mForwardPath(scenario, 1, 0) + _
                    0.35 * USD_ForwardAverageON(scenario, 0, 2) + _
                    USD_ForwardPremium(scenario, 3) / 10000#
        For monthNo = 0 To 36
            currentValue = 0.65 * mForwardPath(scenario, 1, monthNo) + _
                           0.35 * USD_ForwardAverageON(scenario, monthNo, 2) + _
                           USD_ForwardPremium(scenario, 3) / 10000#
            mForwardPath(scenario, 3, monthNo) = _
                USD_Maximum(0#, currentValue + (mRate(3, mCurveN) - baseValue) * USD_Maximum(0#, 1# - monthNo / fade))
        Next monthNo

        baseValue = 0.5 * mForwardPath(scenario, 1, 0) + _
                    0.5 * USD_ForwardAverageON(scenario, 0, 5) + _
                    USD_ForwardPremium(scenario, 4) / 10000#
        For monthNo = 0 To 36
            currentValue = 0.5 * mForwardPath(scenario, 1, monthNo) + _
                           0.5 * USD_ForwardAverageON(scenario, monthNo, 5) + _
                           USD_ForwardPremium(scenario, 4) / 10000#
            mForwardPath(scenario, 4, monthNo) = _
                USD_Maximum(0#, currentValue + (mRate(4, mCurveN) - baseValue) * USD_Maximum(0#, 1# - monthNo / fade))
        Next monthNo
    Next scenario

    'Step 2: roll four independent sleeves over the 36-month horizon.
    For scenario = 1 To 5
        For tenor = 1 To 4
            wealth = 1#
            stepMonths = USD_ForwardRollMonths(tenor)
            For monthNo = 0 To 35 Step stepMonths
                wealth = wealth * (1# + mForwardPath(scenario, tenor, monthNo) * stepMonths / 12#)
            Next monthNo
            mForwardReturn(scenario, tenor) = wealth ^ (1# / 3#) - 1#
            If mForwardReturn(scenario, tenor) > mForwardBest(scenario) Then _
                mForwardBest(scenario) = mForwardReturn(scenario, tenor)
        Next tenor
    Next scenario

    'Step 3: enumerate all 286 mixes in 10% increments.
    ReDim mPortfolioWeight(1 To 286, 1 To 4)
    ReDim mPortfolioReturn(1 To 286, 1 To 5)
    ReDim mPortfolioRegret(1 To 286, 1 To 5)
    ReDim mPortfolioWorst(1 To 286)
    ReDim mPortfolioAverage(1 To 286)
    ReDim mPortfolioLowest(1 To 286)

    mixNo = 0
    For onUnits = 0 To 10
        For oneUnits = 0 To 10 - onUnits
            For threeUnits = 0 To 10 - onUnits - oneUnits
                sixUnits = 10 - onUnits - oneUnits - threeUnits
                mixNo = mixNo + 1
                mPortfolioWeight(mixNo, 1) = onUnits / 10#
                mPortfolioWeight(mixNo, 2) = oneUnits / 10#
                mPortfolioWeight(mixNo, 3) = threeUnits / 10#
                mPortfolioWeight(mixNo, 4) = sixUnits / 10#
            Next threeUnits
        Next oneUnits
    Next onUnits
    mPortfolioCount = mixNo
    If mPortfolioCount <> 286 Then _
        Err.Raise vbObjectError + 3120, , "Forward allocation grid did not produce 286 mixes."

    bestWorst = 1E+99
    For mixNo = 1 To mPortfolioCount
        regretSum = 0#
        mPortfolioLowest(mixNo) = 1E+99
        For scenario = 1 To 5
            weightedCumulative = 0#
            For tenor = 1 To 4
                weightedCumulative = weightedCumulative + _
                    mPortfolioWeight(mixNo, tenor) * ((1# + mForwardReturn(scenario, tenor)) ^ 3# - 1#)
            Next tenor
            mPortfolioReturn(mixNo, scenario) = (1# + weightedCumulative) ^ (1# / 3#) - 1#
            mPortfolioRegret(mixNo, scenario) = _
                (mForwardBest(scenario) - mPortfolioReturn(mixNo, scenario)) * 10000#
            regretSum = regretSum + mPortfolioRegret(mixNo, scenario)
            If mPortfolioRegret(mixNo, scenario) > mPortfolioWorst(mixNo) Then _
                mPortfolioWorst(mixNo) = mPortfolioRegret(mixNo, scenario)
            If mPortfolioReturn(mixNo, scenario) < mPortfolioLowest(mixNo) Then _
                mPortfolioLowest(mixNo) = mPortfolioReturn(mixNo, scenario)
        Next scenario
        mPortfolioAverage(mixNo) = regretSum / 5#
        If mPortfolioWorst(mixNo) < bestWorst Then
            bestWorst = mPortfolioWorst(mixNo)
            mRobustRow = mixNo
        End If
    Next mixNo

    For tenor = 1 To 4
        mMix(tenor) = mPortfolioWeight(mRobustRow, tenor)
    Next tenor
    mWeightedMonths = mMix(2) + 3# * mMix(3) + 6# * mMix(4)
    mMixWorstRegret = mPortfolioWorst(mRobustRow)
    mMixAverageRegret = mPortfolioAverage(mRobustRow)
    mMixLowestReturn = mPortfolioLowest(mRobustRow)
    For scenario = 1 To 5
        If mPortfolioReturn(mRobustRow, scenario) > mMixHighestReturn Then _
            mMixHighestReturn = mPortfolioReturn(mRobustRow, scenario)
    Next scenario

End Sub

Private Function USD_ForwardONRate(ByVal scenario As Long, ByVal monthNo As Long, _
                                   ByVal startON As Double) As Double
    Dim a1 As Double
    Dim a2 As Double
    Dim a3 As Double
    Dim a4 As Double
    Dim a5 As Double
    a1 = USD_ForwardAnchor(scenario, 1)
    a2 = USD_ForwardAnchor(scenario, 2)
    a3 = USD_ForwardAnchor(scenario, 3)
    a4 = USD_ForwardAnchor(scenario, 4)
    a5 = USD_ForwardAnchor(scenario, 5)
    If monthNo <= 1 Then
        USD_ForwardONRate = startON + (a1 - startON) * monthNo
    ElseIf monthNo <= 7 Then
        USD_ForwardONRate = a1 + (a2 - a1) * (monthNo - 1) / 6#
    ElseIf monthNo <= 17 Then
        USD_ForwardONRate = a2 + (a3 - a2) * (monthNo - 7) / 10#
    ElseIf monthNo <= 29 Then
        USD_ForwardONRate = a3 + (a4 - a3) * (monthNo - 17) / 12#
    Else
        USD_ForwardONRate = a4 + (a5 - a4) * (monthNo - 29) / 7#
    End If
End Function

Private Function USD_ForwardAnchor(ByVal scenario As Long, ByVal anchorNo As Long) As Double
    If scenario < 1 Or scenario > 5 Or anchorNo < 1 Or anchorNo > 5 Then _
        Err.Raise vbObjectError + 3121, , "Unknown forward scenario anchor."
    USD_ForwardAnchor = mForwardAnchorInput(scenario, anchorNo)
End Function

Private Function USD_ForwardPremium(ByVal scenario As Long, ByVal tenor As Long) As Double
    If scenario < 1 Or scenario > 5 Or tenor < 2 Or tenor > 4 Then _
        Err.Raise vbObjectError + 3122, , "Unknown forward scenario premium."
    USD_ForwardPremium = mForwardPremiumInput(scenario, tenor)
End Function

Private Function USD_ForwardAverageON(ByVal scenario As Long, ByVal startMonth As Long, _
                                      ByVal monthsAhead As Long) As Double
    Dim i As Long
    Dim finalMonth As Long
    Dim total As Double
    Dim count As Long
    finalMonth = startMonth + monthsAhead
    If finalMonth > 36 Then finalMonth = 36
    For i = startMonth To finalMonth
        total = total + mForwardPath(scenario, 1, i)
        count = count + 1
    Next i
    USD_ForwardAverageON = total / count
End Function

Private Function USD_ForwardRollMonths(ByVal tenor As Long) As Long
    Select Case tenor
        Case 1, 2: USD_ForwardRollMonths = 1
        Case 3: USD_ForwardRollMonths = 3
        Case 4: USD_ForwardRollMonths = 6
    End Select
End Function

Private Function USD_PathReturn( _
    ByVal startIndex As Long, ByVal tenorIndex As Long, _
    ByVal tenorMonths As Long, ByVal pathCode As Long) As Double

    Dim startDate As Date
    Dim maturityDate As Date
    Dim endIndex As Long
    Dim currentIndex As Long
    Dim nextIndex As Long
    Dim nextTarget As Date
    Dim factor As Double
    Dim k As Long
    Dim rateValue As Double

    startDate = CDate(mDate(startIndex))
    maturityDate = DateAdd("m", 6, startDate)
    endIndex = USD_FloorIndex(CDbl(maturityDate))
    If endIndex <= startIndex Then Exit Function
    factor = 1#

    If tenorMonths = 0 Then
        For currentIndex = startIndex To endIndex - 1
            rateValue = USD_PathRate(tenorIndex, currentIndex, pathCode)
            factor = factor * (1# + rateValue * (mDate(currentIndex + 1) - mDate(currentIndex)) / 360#)
        Next currentIndex
    Else
        currentIndex = startIndex
        k = 1
        Do While currentIndex < endIndex
            nextTarget = DateAdd("m", tenorMonths * k, startDate)
            If nextTarget >= maturityDate Then
                nextIndex = endIndex
            Else
                nextIndex = USD_FloorIndex(CDbl(nextTarget))
            End If
            If nextIndex <= currentIndex Then
                k = k + 1
            Else
                rateValue = USD_PathRate(tenorIndex, currentIndex, pathCode)
                factor = factor * (1# + rateValue * (mDate(nextIndex) - mDate(currentIndex)) / 360#)
                currentIndex = nextIndex
                If nextTarget < maturityDate Then k = k + 1
            End If
        Loop
    End If

    USD_PathReturn = factor - 1#

End Function

Private Function USD_PathRate(ByVal tenorIndex As Long, ByVal observation As Long, ByVal pathCode As Long) As Double
    Select Case pathCode
        Case 0
            USD_PathRate = mRate(tenorIndex, observation)
        Case 1
            USD_PathRate = STRESS_ON + mRate(tenorIndex, observation) - mRate(1, 1)
        Case 2
            USD_PathRate = STRESS_ON + 2# * mRate(tenorIndex, 1) - mRate(1, 1) - mRate(tenorIndex, observation)
        Case Else
            Err.Raise vbObjectError + 3122, , "Invalid directional path code."
    End Select
End Function

Private Function USD_BestTenor(ByRef returnsMatrix() As Double, ByVal rowIndex As Long) As Long
    Dim t As Long
    USD_BestTenor = 1
    For t = 2 To 4
        If returnsMatrix(rowIndex, t) > returnsMatrix(rowIndex, USD_BestTenor) Then USD_BestTenor = t
    Next t
End Function

Private Function USD_FloorIndex(ByVal targetDate As Double) As Long
    Dim lo As Long
    Dim hi As Long
    Dim mid As Long
    If targetDate <= mDate(1) Then USD_FloorIndex = 1: Exit Function
    If targetDate >= mDate(mCurveN) Then USD_FloorIndex = mCurveN: Exit Function
    lo = 1
    hi = mCurveN
    Do While lo <= hi
        mid = (lo + hi) \ 2
        If mDate(mid) = targetDate Then
            USD_FloorIndex = mid
            Exit Function
        ElseIf mDate(mid) < targetDate Then
            lo = mid + 1
        Else
            hi = mid - 1
        End If
    Loop
    USD_FloorIndex = hi
End Function

Private Function USD_FindExactIndex(ByVal targetDate As Double) As Long
    Dim idx As Long
    idx = USD_FloorIndex(targetDate)
    If Abs(mDate(idx) - targetDate) < 0.00001 Then USD_FindExactIndex = idx
End Function

Private Function USD_IsTrue(ByVal value As Variant) As Boolean
    If VarType(value) = vbBoolean Then
        USD_IsTrue = CBool(value)
    Else
        USD_IsTrue = (UCase$(Trim$(CStr(value))) = "TRUE" Or CDbl(Val(CStr(value))) = 1#)
    End If
End Function

Private Sub USD_QuickSortPair(ByRef x() As Double, ByRef y() As Double, ByVal first As Long, ByVal last As Long)
    Dim i As Long
    Dim j As Long
    Dim pivot As Double
    Dim tx As Double
    Dim ty As Double
    i = first
    j = last
    pivot = x((first + last) \ 2)
    Do While i <= j
        Do While x(i) < pivot: i = i + 1: Loop
        Do While x(j) > pivot: j = j - 1: Loop
        If i <= j Then
            tx = x(i): x(i) = x(j): x(j) = tx
            ty = y(i): y(i) = y(j): y(j) = ty
            i = i + 1
            j = j - 1
        End If
    Loop
    If first < j Then USD_QuickSortPair x, y, first, j
    If i < last Then USD_QuickSortPair x, y, i, last
End Sub
