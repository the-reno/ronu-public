Public Sub USD_CreateCashAllocationExcelPack()

    Dim oldCalc As Long
    Dim oldScreen As Boolean
    Dim oldEvents As Boolean
    Dim errText As String

    On Error GoTo fail

    oldCalc = Application.Calculation
    oldScreen = Application.ScreenUpdating
    oldEvents = Application.EnableEvents
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    If ThisWorkbook.ProtectStructure Then _
        Err.Raise vbObjectError + 3099, , "Workbook structure is protected. Unprotect it before running the module."
    Application.StatusBar = "Building the USD cash-allocation Excel slide pack..."

    USD_ResetState
    USD_ValidateSourceWorkbook
    USD_LoadCurve
    USD_CalculateDailyEvidence
    USD_CalculateDirectionalStress
    USD_LoadPortfolioResult

    USD_BuildIndex
    USD_BuildSlide1Sheet
    USD_BuildSlide2Sheet
    USD_BuildSlide3Sheet
    USD_BuildSlide4Sheet
    USD_BuildAuditSheet

    Application.CalculateFull
    ThisWorkbook.Worksheets(S_INDEX).Activate
    ThisWorkbook.Worksheets(S_INDEX).Range("A1").Select

clean_exit:
    Application.StatusBar = False
    Application.Calculation = oldCalc
    Application.EnableEvents = oldEvents
    Application.ScreenUpdating = oldScreen
    If Len(errText) = 0 Then
        MsgBox "Excel slide pack created." & vbCrLf & vbCrLf & _
               "No PowerPoint file was created." & vbCrLf & _
               "Use PPT_Index to copy charts and comments manually.", _
               vbInformation, "USD cash allocation"
    End If
    Exit Sub

fail:
    errText = Err.Description
    MsgBox "USD_CreateCashAllocationExcelPack stopped:" & vbCrLf & vbCrLf & errText, _
           vbCritical, "USD cash allocation"
    Resume clean_exit

End Sub

Private Sub USD_WriteHoldingExample(ByVal ws As Worksheet, ByVal targetRow As Long, _
                                    ByVal dailyIndex As Long, ByVal noteText As String)
    If dailyIndex = 0 Then Err.Raise vbObjectError + 3130, , "Required holding-period example was not found."
    ws.Cells(targetRow, 16).Value = mDailyDate(dailyIndex)
    ws.Cells(targetRow, 17).Value = mDailySpread(dailyIndex)
    ws.Cells(targetRow, 18).Value = mDailyRolledON(dailyIndex)
    ws.Cells(targetRow, 19).Value = mDailyLocked6M(dailyIndex)
    ws.Cells(targetRow, 20).Value = mDailyExcess(dailyIndex)
    ws.Cells(targetRow, 21).Value = IIf(mDailyExcess(dailyIndex) > 0#, "6M", "ON")
    ws.Cells(targetRow, 22).Value = noteText
End Sub

Private Function USD_DailyIndex(ByVal decisionDate As Double) As Long
    Dim i As Long
    For i = 1 To mDailyN
        If Abs(mDailyDate(i) - decisionDate) < 0.00001 Then USD_DailyIndex = i: Exit Function
    Next i
End Function

Private Function USD_GetOrCreateSheet(ByVal sheetName As String) As Worksheet
    If USD_SheetExists(sheetName) Then
        Set USD_GetOrCreateSheet = ThisWorkbook.Worksheets(sheetName)
    Else
        Set USD_GetOrCreateSheet = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
        USD_GetOrCreateSheet.Name = sheetName
    End If
End Function

Private Sub USD_ResetOutputSheet(ByVal ws As Worksheet)
    Dim i As Long
    On Error Resume Next
    For i = ws.ListObjects.Count To 1 Step -1
        ws.ListObjects(i).Unlist
    Next i
    For i = ws.Shapes.Count To 1 Step -1
        ws.Shapes(i).Delete
    Next i
    ws.Cells.UnMerge
    ws.Cells.Clear
    ws.Cells.ClearFormats
    On Error GoTo 0
End Sub

Private Sub USD_SetCanvas(ByVal ws As Worksheet, ByVal canvasColumns As Long)
    Dim i As Long
    With ws.Cells
        .Font.Name = FONT_NAME
        .Font.Size = 10
        .Font.Color = USD_Gray()
        .VerticalAlignment = xlCenter
    End With
    For i = 1 To canvasColumns
        ws.Columns(i).ColumnWidth = 9
    Next i
    ws.Rows("1:2").RowHeight = 27
    ws.Rows("3:39").RowHeight = 20
    ws.Tab.Color = USD_Blue()
    On Error Resume Next
    ws.Activate
    ActiveWindow.DisplayGridlines = False
    ActiveWindow.Zoom = 82
    On Error GoTo 0
End Sub

Private Sub USD_Title(ByVal ws As Worksheet, ByVal titleText As String, ByVal subtitleText As String)
    Dim endColumn As String
    If ws.Name = S_AUDIT Then
        endColumn = "G"
    ElseIf ws.Name = S_INDEX Then
        endColumn = "T"
    Else
        endColumn = "V"
    End If
    ws.Range("A1:" & endColumn & "1").Merge
    With ws.Range("A1:" & endColumn & "1")
        .Value = titleText
        .Interior.Color = USD_Navy()
        .Font.Color = RGB(255, 255, 255)
        .Font.Bold = True
        .Font.Size = 18
        .HorizontalAlignment = xlLeft
        .IndentLevel = 1
    End With
    ws.Range("A2:" & endColumn & "2").Merge
    With ws.Range("A2:" & endColumn & "2")
        .Value = subtitleText
        .Interior.Color = RGB(232, 238, 246)
        .Font.Color = USD_Navy()
        .Font.Italic = True
        .Font.Size = 11
        .HorizontalAlignment = xlLeft
        .IndentLevel = 1
    End With
End Sub

Private Sub USD_AddAudit(ByVal ws As Worksheet, ByRef targetRow As Long, _
                         ByVal checkName As String, ByVal actualValue As Variant, _
                         ByVal expectedValue As Variant, ByVal tolerance As Double, _
                         ByVal sourceText As String, ByVal riskNote As String, _
                         ByRef allPass As Boolean, Optional ByVal numberFormat As String = "General")
    Dim passed As Boolean
    If VarType(actualValue) = vbBoolean Or VarType(expectedValue) = vbBoolean Then
        passed = (CBool(actualValue) = CBool(expectedValue))
    ElseIf IsNumeric(actualValue) And IsNumeric(expectedValue) Then
        passed = (Abs(CDbl(actualValue) - CDbl(expectedValue)) <= tolerance)
    Else
        passed = (CStr(actualValue) = CStr(expectedValue))
    End If
    ws.Cells(targetRow, 1).Value = checkName
    ws.Cells(targetRow, 2).Value = actualValue
    ws.Cells(targetRow, 3).Value = expectedValue
    ws.Cells(targetRow, 4).Value = tolerance
    ws.Cells(targetRow, 5).Value = IIf(passed, "PASS", "REVIEW")
    ws.Cells(targetRow, 6).Value = sourceText
    ws.Cells(targetRow, 7).Value = riskNote
    ws.Cells(targetRow, 2).NumberFormat = numberFormat
    ws.Cells(targetRow, 3).NumberFormat = numberFormat
    If Not passed Then allPass = False
    targetRow = targetRow + 1
End Sub

Private Sub USD_LoadCurve()

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim v As Variant
    Dim i As Long
    Dim j As Long

    Set ws = ThisWorkbook.Worksheets("Curve")
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    If lastRow < 3 Then Err.Raise vbObjectError + 3102, , "Curve has no usable observations."

    If UCase$(Trim$(CStr(ws.Cells(1, 1).Value))) <> "DATE" Then _
        Err.Raise vbObjectError + 3103, , "Curve column A must be Date."
    If UCase$(Trim$(CStr(ws.Cells(1, 2).Value))) <> "ON" Then _
        Err.Raise vbObjectError + 3104, , "Curve column B must be ON."
    If UCase$(Trim$(CStr(ws.Cells(1, 3).Value))) <> "1M" Or _
       UCase$(Trim$(CStr(ws.Cells(1, 4).Value))) <> "3M" Or _
       UCase$(Trim$(CStr(ws.Cells(1, 5).Value))) <> "6M" Then _
        Err.Raise vbObjectError + 3105, , "Curve columns B:E must be ON, 1M, 3M and 6M."

    v = ws.Range("A2:E" & lastRow).Value2
    mCurveN = UBound(v, 1)
    ReDim mDate(1 To mCurveN)
    ReDim mRate(1 To 4, 1 To mCurveN)

    For i = 1 To mCurveN
        If Not IsNumeric(v(i, 1)) Then _
            Err.Raise vbObjectError + 3106, , "Curve contains a non-date value at row " & CStr(i + 1) & "."
        mDate(i) = CDbl(v(i, 1))
        If i > 1 Then
            If mDate(i) <= mDate(i - 1) Then _
                Err.Raise vbObjectError + 3107, , "Curve dates must be strictly increasing."
        End If
        For j = 1 To 4
            If Not IsNumeric(v(i, j + 1)) Then _
                Err.Raise vbObjectError + 3108, , "Curve contains a blank/non-numeric rate at row " & CStr(i + 1) & "."
            mRate(j, i) = CDbl(v(i, j + 1))
        Next j
    Next i

End Sub

Private Sub USD_CalculateDailyEvidence()

    Dim i As Long
    Dim n As Long
    Dim lastDate As Double
    Dim onRet As Double
    Dim sixRet As Double
    Dim meanX As Double
    Dim meanY As Double
    Dim sumXX As Double
    Dim sumYY As Double
    Dim sumXY As Double
    Dim dx As Double
    Dim dy As Double
    Dim hits As Long
    Dim sx() As Double
    Dim sy() As Double
    Dim q4Count As Long
    Dim q4Start As Long
    Dim q4Sum As Double
    Dim q4Hits As Long

    lastDate = mDate(mCurveN)
    For i = 1 To mCurveN
        If CDbl(DateAdd("m", 6, CDate(mDate(i)))) <= lastDate Then n = n + 1
    Next i
    If n = 0 Then Err.Raise vbObjectError + 3110, , "Curve does not contain a complete six-month horizon."

    mDailyN = n
    ReDim mDailyDate(1 To n)
    ReDim mDailySpread(1 To n)
    ReDim mDailyExcess(1 To n)
    ReDim mDailyRolledON(1 To n)
    ReDim mDailyLocked6M(1 To n)

    n = 0
    For i = 1 To mCurveN
        If CDbl(DateAdd("m", 6, CDate(mDate(i)))) <= lastDate Then
            n = n + 1
            onRet = USD_PathReturn(i, 1, 0, 0)
            sixRet = USD_PathReturn(i, 4, 6, 0)
            mDailyDate(n) = mDate(i)
            mDailySpread(n) = (mRate(4, i) - mRate(1, i)) * 10000#
            mDailyRolledON(n) = onRet
            mDailyLocked6M(n) = sixRet
            mDailyExcess(n) = (sixRet - onRet) * 10000#
            meanX = meanX + mDailySpread(n)
            meanY = meanY + mDailyExcess(n)
            If mDailyExcess(n) > 0# Then hits = hits + 1
        End If
    Next i

    meanX = meanX / mDailyN
    meanY = meanY / mDailyN
    For i = 1 To mDailyN
        dx = mDailySpread(i) - meanX
        dy = mDailyExcess(i) - meanY
        sumXX = sumXX + dx * dx
        sumYY = sumYY + dy * dy
        sumXY = sumXY + dx * dy
    Next i
    If sumXX = 0# Or sumYY = 0# Then _
        Err.Raise vbObjectError + 3111, , "Regression cannot be calculated from constant data."

    mSlope = sumXY / sumXX
    mIntercept = meanY - mSlope * meanX
    mCorr = sumXY / Sqr(sumXX * sumYY)
    mR2 = mCorr * mCorr
    mHitRate = hits / mDailyN

    ReDim sx(1 To mDailyN)
    ReDim sy(1 To mDailyN)
    For i = 1 To mDailyN
        sx(i) = mDailySpread(i)
        sy(i) = mDailyExcess(i)
    Next i
    USD_QuickSortPair sx, sy, 1, mDailyN
    q4Count = (mDailyN + 3) \ 4
    q4Start = mDailyN - q4Count + 1
    For i = q4Start To mDailyN
        q4Sum = q4Sum + sy(i)
        If sy(i) > 0# Then q4Hits = q4Hits + 1
    Next i
    mQ4Mean = q4Sum / q4Count
    mQ4Hit = q4Hits / q4Count

End Sub

Private Sub USD_CalculateDirectionalStress()

    Dim starts() As Long
    Dim count As Long
    Dim i As Long
    Dim monthKey As Long
    Dim priorKey As Long
    Dim tenor As Long
    Dim bestUp As Long
    Dim bestDown As Long

    ReDim starts(1 To mCurveN)
    For i = 1 To mCurveN
        If CDbl(DateAdd("m", 6, CDate(mDate(i)))) <= mDate(mCurveN) Then
            monthKey = Year(CDate(mDate(i))) * 100 + Month(CDate(mDate(i)))
            If monthKey <> priorKey Then
                count = count + 1
                starts(count) = i
                priorKey = monthKey
            End If
        End If
    Next i
    If count = 0 Then Err.Raise vbObjectError + 3112, , "No monthly scenario starts were available."

    mMonthlyN = count
    ReDim mMonthlyStart(1 To count)
    ReDim mUpReturn(1 To count, 1 To 4)
    ReDim mDownReturn(1 To count, 1 To 4)

    For i = 1 To count
        mMonthlyStart(i) = starts(i)
        For tenor = 1 To 4
            mUpReturn(i, tenor) = USD_PathReturn(starts(i), tenor, USD_TenorMonths(tenor), 1)
            mDownReturn(i, tenor) = USD_PathReturn(starts(i), tenor, USD_TenorMonths(tenor), 2)
            mUpExcess(tenor) = mUpExcess(tenor) + (mUpReturn(i, tenor) - mUpReturn(i, 1)) * 10000#
            mDownExcess(tenor) = mDownExcess(tenor) + (mDownReturn(i, tenor) - mDownReturn(i, 1)) * 10000#
        Next tenor
        bestUp = USD_BestTenor(mUpReturn, i)
        bestDown = USD_BestTenor(mDownReturn, i)
        mUpWins(bestUp) = mUpWins(bestUp) + 1
        mDownWins(bestDown) = mDownWins(bestDown) + 1
    Next i

    For tenor = 1 To 4
        mUpExcess(tenor) = mUpExcess(tenor) / count
        mDownExcess(tenor) = mDownExcess(tenor) / count
    Next tenor

End Sub

Private Sub USD_LoadPortfolioResult()

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim v As Variant
    Dim i As Long
    Dim robustFound As Boolean

    Set ws = ThisWorkbook.Worksheets("F_Portfolios")
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    If lastRow < 3 Then Err.Raise vbObjectError + 3120, , "F_Portfolios has no usable allocation grid."
    v = ws.Range("A2:L" & lastRow).Value2
    mPortfolioCount = UBound(v, 1)

    For i = 1 To UBound(v, 1)
        If Abs(CDbl(v(i, 1)) - 1#) < 0.0000001 And _
           Abs(CDbl(v(i, 2))) < 0.0000001 And _
           Abs(CDbl(v(i, 3))) < 0.0000001 And _
           Abs(CDbl(v(i, 4))) < 0.0000001 Then mONWorstRegret = CDbl(v(i, 9)) * 10000#

        If Abs(CDbl(v(i, 4)) - 1#) < 0.0000001 And _
           Abs(CDbl(v(i, 1))) < 0.0000001 And _
           Abs(CDbl(v(i, 2))) < 0.0000001 And _
           Abs(CDbl(v(i, 3))) < 0.0000001 Then mSixWorstRegret = CDbl(v(i, 9)) * 10000#

        If USD_IsTrue(v(i, 12)) Then
            mMix(1) = CDbl(v(i, 1))
            mMix(2) = CDbl(v(i, 2))
            mMix(3) = CDbl(v(i, 3))
            mMix(4) = CDbl(v(i, 4))
            mMixHistRegret = CDbl(v(i, 7)) * 10000#
            mMixInvRegret = CDbl(v(i, 8)) * 10000#
            mMixWorstRegret = CDbl(v(i, 9)) * 10000#
            mWeightedMonths = CDbl(v(i, 11))
            mRobustRow = i + 1
            robustFound = True
        End If
    Next i

    If Not robustFound Then _
        Err.Raise vbObjectError + 3121, , "F_Portfolios has no row marked Is_Robust = TRUE."

End Sub

Private Function USD_PathReturn( _
    ByVal startIndex As Long, ByVal tenorIndex As Long, _
    ByVal tenorMonths As Long, ByVal pathCode As Long) As Double

    Dim startDate As Date
    Dim maturityDate As Date
    Dim endIndex As Long
    Dim currentIndex As Long
    Dim nextIndex As Long
    Dim nextTarget As Date
    Dim factor As Double
    Dim k As Long
    Dim rateValue As Double

    startDate = CDate(mDate(startIndex))
    maturityDate = DateAdd("m", 6, startDate)
    endIndex = USD_FloorIndex(CDbl(maturityDate))
    If endIndex <= startIndex Then Exit Function
    factor = 1#

    If tenorMonths = 0 Then
        For currentIndex = startIndex To endIndex - 1
            rateValue = USD_PathRate(tenorIndex, currentIndex, pathCode)
            factor = factor * (1# + rateValue * (mDate(currentIndex + 1) - mDate(currentIndex)) / 360#)
        Next currentIndex
    Else
        currentIndex = startIndex
        k = 1
        Do While currentIndex < endIndex
            nextTarget = DateAdd("m", tenorMonths * k, startDate)
            If nextTarget >= maturityDate Then
                nextIndex = endIndex
            Else
                nextIndex = USD_FloorIndex(CDbl(nextTarget))
            End If
            If nextIndex <= currentIndex Then
                k = k + 1
            Else
                rateValue = USD_PathRate(tenorIndex, currentIndex, pathCode)
                factor = factor * (1# + rateValue * (mDate(nextIndex) - mDate(currentIndex)) / 360#)
                currentIndex = nextIndex
                If nextTarget < maturityDate Then k = k + 1
            End If
        Loop
    End If

    USD_PathReturn = factor - 1#

End Function

Private Function USD_PathRate(ByVal tenorIndex As Long, ByVal observation As Long, ByVal pathCode As Long) As Double
    Select Case pathCode
        Case 0
            USD_PathRate = mRate(tenorIndex, observation)
        Case 1
            USD_PathRate = STRESS_ON + mRate(tenorIndex, observation) - mRate(1, 1)
        Case 2
            USD_PathRate = STRESS_ON + 2# * mRate(tenorIndex, 1) - mRate(1, 1) - mRate(tenorIndex, observation)
        Case Else
            Err.Raise vbObjectError + 3122, , "Invalid directional path code."
    End Select
End Function

Private Function USD_TenorMonths(ByVal tenorIndex As Long) As Long
    Select Case tenorIndex
        Case 1: USD_TenorMonths = 0
        Case 2: USD_TenorMonths = 1
        Case 3: USD_TenorMonths = 3
        Case 4: USD_TenorMonths = 6
    End Select
End Function

Private Function USD_TenorName(ByVal tenorIndex As Long) As String
    Select Case tenorIndex
        Case 1: USD_TenorName = "ON"
        Case 2: USD_TenorName = "1M"
        Case 3: USD_TenorName = "3M"
        Case 4: USD_TenorName = "6M"
    End Select
End Function

Private Function USD_BestTenor(ByRef returnsMatrix() As Double, ByVal rowIndex As Long) As Long
    Dim t As Long
    USD_BestTenor = 1
    For t = 2 To 4
        If returnsMatrix(rowIndex, t) > returnsMatrix(rowIndex, USD_BestTenor) Then USD_BestTenor = t
    Next t
End Function

Private Function USD_FloorIndex(ByVal targetDate As Double) As Long
    Dim lo As Long
    Dim hi As Long
    Dim mid As Long
    If targetDate <= mDate(1) Then USD_FloorIndex = 1: Exit Function
    If targetDate >= mDate(mCurveN) Then USD_FloorIndex = mCurveN: Exit Function
    lo = 1
    hi = mCurveN
    Do While lo <= hi
        mid = (lo + hi) \ 2
        If mDate(mid) = targetDate Then
            USD_FloorIndex = mid
            Exit Function
        ElseIf mDate(mid) < targetDate Then
            lo = mid + 1
        Else
            hi = mid - 1
        End If
    Loop
    USD_FloorIndex = hi
End Function

Private Function USD_FindExactIndex(ByVal targetDate As Double) As Long
    Dim idx As Long
    idx = USD_FloorIndex(targetDate)
    If Abs(mDate(idx) - targetDate) < 0.00001 Then USD_FindExactIndex = idx
End Function

Private Function USD_IsTrue(ByVal value As Variant) As Boolean
    If VarType(value) = vbBoolean Then
        USD_IsTrue = CBool(value)
    Else
        USD_IsTrue = (UCase$(Trim$(CStr(value))) = "TRUE" Or CDbl(Val(CStr(value))) = 1#)
    End If
End Function

Private Sub USD_QuickSortPair(ByRef x() As Double, ByRef y() As Double, ByVal first As Long, ByVal last As Long)
    Dim i As Long
    Dim j As Long
    Dim pivot As Double
    Dim tx As Double
    Dim ty As Double
    i = first
    j = last
    pivot = x((first + last) \ 2)
    Do While i <= j
        Do While x(i) < pivot: i = i + 1: Loop
        Do While x(j) > pivot: j = j - 1: Loop
        If i <= j Then
            tx = x(i): x(i) = x(j): x(j) = tx
            ty = y(i): y(i) = y(j): y(j) = ty
            i = i + 1
            j = j - 1
        End If
    Loop
    If first < j Then USD_QuickSortPair x, y, first, j
    If i < last Then USD_QuickSortPair x, y, i, last
End Sub
