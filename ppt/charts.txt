Option Explicit

'====================================================================================
' USD CASH ALLOCATION - EXCEL DASHBOARD
'
' PURPOSE
'   Rebuilds the cash-allocation analysis directly from a single input sheet: Curve.
'   The workbook may start with ONLY the Curve tab.
'
' INPUT SHEET
'   Curve!A:E
'   A = Date | B = ON | C = 1M | D = 3M | E = 6M
'   Rates must be decimals (example: 0.055 = 5.50%).
'
' OUTPUT
'   USD_Dashboard  - visible, editable chart controls + presentation-ready charts
'   _USD_Calc      - hidden calculation data used by the charts
'
' MODEL LOGIC
'   1) Historical 6M decision: compare locked 6M vs ON rolled over the same 6M horizon.
'   2) Premium signal: 6M-ON spread at inception vs realized 6M excess return.
'   3) Directional stress: replay observed rate moves and the opposite-sign moves,
'      both rebased to the same starting ON level (default 5.50%).
'   4) Portfolio: test long-only ON/1M/3M/6M allocations and minimize the worse
'      average regret across historical and inverted paths.
'
' SOURCE LOGIC REVIEWED
'   https://github.com/the-reno/ronu-public/blob/main/ppt/11_Create_USD_Cash_Allocation_Slides_1_4.txt
'
' USE
'   1. Put the Curve data in a sheet named Curve.
'   2. Alt+F11 > Insert > Module > paste/import this module.
'   3. Run USD_BuildDashboard.
'   4. Edit BLUE cells in USD_Dashboard.
'   5. Run USD_ApplyStyle after color/arrow changes.
'   6. Run USD_RefreshDashboard after Curve/model-input changes.
'
' NOTES
'   - Charts remain native Excel charts so they can be copied manually to PowerPoint.
'   - Chart 1 annotations are embedded inside the chart; they copy with the chart.
'====================================================================================

'============================== PUBLIC MACROS ========================================

Public Sub USD_BuildDashboard()
    USD_RefreshDashboard
End Sub

Public Sub USD_RefreshDashboard()
    Dim oldCalc As XlCalculation
    Dim oldScreen As Boolean
    Dim oldAlerts As Boolean
    Dim wsDash As Worksheet

    On Error GoTo fail

    oldCalc = Application.Calculation
    oldScreen = Application.ScreenUpdating
    oldAlerts = Application.DisplayAlerts

    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.Calculation = xlCalculationManual
    Application.StatusBar = "USD cash allocation: validating Curve..."

    USD_LoadCurve
    Set wsDash = USD_EnsureDashboard
    USD_ReadModelSettings wsDash

    Application.StatusBar = "USD cash allocation: calculating realized evidence..."
    USD_CalculateDailyEvidence
    USD_CalculateMonthlyEvidence

    Application.StatusBar = "USD cash allocation: optimizing portfolio..."
    USD_OptimizePortfolio

    Application.StatusBar = "USD cash allocation: writing dashboard data..."
    USD_WriteCalcSheet
    USD_UpdateDashboardKPIs wsDash

    Application.StatusBar = "USD cash allocation: building charts..."
    USD_BuildCharts wsDash
    USD_ApplyStyleCore wsDash

    On Error Resume Next
    ThisWorkbook.Worksheets(CALC_SHEET).Visible = xlSheetVeryHidden
    On Error GoTo fail

    wsDash.Activate
    ActiveWindow.DisplayGridlines = False

    Application.StatusBar = False
    Application.Calculation = oldCalc
    Application.DisplayAlerts = oldAlerts
    Application.ScreenUpdating = oldScreen
    Exit Sub

fail:
    Application.StatusBar = False
    Application.Calculation = oldCalc
    Application.DisplayAlerts = oldAlerts
    Application.ScreenUpdating = oldScreen
    MsgBox "USD_RefreshDashboard stopped: " & Err.Description, vbCritical
End Sub

Public Sub USD_ApplyStyle()
    Dim ws As Worksheet
    On Error GoTo fail

    If Not USD_SheetExists(DASH_SHEET) Then
        MsgBox "Run USD_BuildDashboard first.", vbInformation
        Exit Sub
    End If

    Set ws = ThisWorkbook.Worksheets(DASH_SHEET)
    USD_ApplyStyleCore ws
    Exit Sub

fail:
    MsgBox "USD_ApplyStyle stopped: " & Err.Description, vbCritical
End Sub

Public Sub USD_ResetDashboardSettings()
    Dim ws As Worksheet
    If Not USD_SheetExists(DASH_SHEET) Then
        MsgBox "Run USD_BuildDashboard first.", vbInformation
        Exit Sub
    End If

    Set ws = ThisWorkbook.Worksheets(DASH_SHEET)
    USD_WriteDefaultSettings ws, True
    USD_ApplyStyle
End Sub

'============================== INPUT / SETTINGS =====================================

Private Sub USD_LoadCurve()
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim v As Variant
    Dim i As Long, j As Long

    If Not USD_SheetExists(CURVE_SHEET) Then _
        Err.Raise vbObjectError + 3000, , "Required sheet 'Curve' was not found."

    Set ws = ThisWorkbook.Worksheets(CURVE_SHEET)
    lastRow = ws.Cells(ws.Rows.count, "A").End(xlUp).Row
    If lastRow < 3 Then Err.Raise vbObjectError + 3001, , "Curve has no usable observations."

    If UCase$(Trim$(CStr(ws.Cells(1, 1).value))) <> "DATE" Then _
        Err.Raise vbObjectError + 3002, , "Curve column A must be Date."
    If UCase$(Trim$(CStr(ws.Cells(1, 2).value))) <> "ON" Or _
       UCase$(Trim$(CStr(ws.Cells(1, 3).value))) <> "1M" Or _
       UCase$(Trim$(CStr(ws.Cells(1, 4).value))) <> "3M" Or _
       UCase$(Trim$(CStr(ws.Cells(1, 5).value))) <> "6M" Then _
        Err.Raise vbObjectError + 3003, , "Curve columns B:E must be ON, 1M, 3M and 6M."

    v = ws.Range("A2:E" & lastRow).Value2
    mCurveN = UBound(v, 1)
    ReDim mDate(1 To mCurveN)
    ReDim mRate(1 To 4, 1 To mCurveN)

    For i = 1 To mCurveN
        If Not IsNumeric(v(i, 1)) Then _
            Err.Raise vbObjectError + 3004, , "Non-date value in Curve row " & CStr(i + 1) & "."
        mDate(i) = CDbl(v(i, 1))
        If i > 1 Then
            If mDate(i) <= mDate(i - 1) Then _
                Err.Raise vbObjectError + 3005, , "Curve dates must be strictly increasing."
        End If

        For j = 1 To 4
            If Not IsNumeric(v(i, j + 1)) Then _
                Err.Raise vbObjectError + 3006, , "Blank/non-numeric rate in Curve row " & CStr(i + 1) & "."
            mRate(j, i) = CDbl(v(i, j + 1))
        Next j
    Next i
End Sub

Private Function USD_EnsureDashboard() As Worksheet
    Dim ws As Worksheet
    Dim created As Boolean

    If USD_SheetExists(DASH_SHEET) Then
        Set ws = ThisWorkbook.Worksheets(DASH_SHEET)
    Else
        Set ws = ThisWorkbook.Worksheets.Add(Before:=ThisWorkbook.Worksheets(1))
        ws.Name = DASH_SHEET
        created = True
    End If

    USD_LayoutDashboard ws
    USD_WriteDefaultSettings ws, created
    USD_AddDashboardButtons ws

    Set USD_EnsureDashboard = ws
End Function

Private Sub USD_ReadModelSettings(ByVal ws As Worksheet)
    mStressON = DEFAULT_STRESS_ON
    mAllocationStep = DEFAULT_ALLOC_STEP

    If IsNumeric(ws.Range("E9").value) Then mStressON = CDbl(ws.Range("E9").value)
    If IsNumeric(ws.Range("E10").value) Then mAllocationStep = CDbl(ws.Range("E10").value)

    If mStressON <= 0 Or mStressON > 0.25 Then _
        Err.Raise vbObjectError + 3010, , "Stress ON anchor must be between 0% and 25%."

    If mAllocationStep <= 0 Or mAllocationStep > 0.5 Then _
        Err.Raise vbObjectError + 3011, , "Allocation step must be >0% and <=50%."

    If Abs((1# / mAllocationStep) - Round(1# / mAllocationStep, 0)) > 0.0000001 Then _
        Err.Raise vbObjectError + 3012, , "Allocation step must divide 100% exactly (for example 5%, 10%, 20%, 25%)."
End Sub

'============================== CALCULATIONS =========================================

Private Sub USD_CalculateDailyEvidence()
    Dim i As Long, n As Long
    Dim lastDate As Double
    Dim startDate As Date
    Dim onRet As Double, sixRet As Double
    Dim meanX As Double, meanY As Double
    Dim sumXX As Double, sumYY As Double, sumXY As Double
    Dim dx As Double, dy As Double
    Dim hits As Long
    Dim sx() As Double, sy() As Double
    Dim q4Count As Long, q4Start As Long
    Dim q4Sum As Double, q4Hits As Long

    lastDate = mDate(mCurveN)

    For i = 1 To mCurveN
        startDate = CDate(mDate(i))
        If CDbl(DateAdd("m", 6, startDate)) <= lastDate Then n = n + 1
    Next i

    If n = 0 Then Err.Raise vbObjectError + 3020, , "Curve does not contain a fully realized six-month horizon."

    mDailyN = n
    ReDim mDailyDate(1 To n)
    ReDim mDailySpread(1 To n)
    ReDim mDailyExcess(1 To n)
    ReDim mDailyONReturn(1 To n)
    ReDim mDaily6MReturn(1 To n)

    n = 0
    For i = 1 To mCurveN
        startDate = CDate(mDate(i))
        If CDbl(DateAdd("m", 6, startDate)) <= lastDate Then
            n = n + 1
            onRet = USD_PathReturn(i, 1, 0, 0)
            sixRet = USD_PathReturn(i, 4, 6, 0)

            mDailyDate(n) = mDate(i)
            mDailySpread(n) = (mRate(4, i) - mRate(1, i)) * 10000#
            mDailyExcess(n) = (sixRet - onRet) * 10000#
            mDailyONReturn(n) = onRet
            mDaily6MReturn(n) = sixRet

            meanX = meanX + mDailySpread(n)
            meanY = meanY + mDailyExcess(n)
            If mDailyExcess(n) > 0# Then hits = hits + 1
        End If
    Next i

    meanX = meanX / mDailyN
    meanY = meanY / mDailyN

    For i = 1 To mDailyN
        dx = mDailySpread(i) - meanX
        dy = mDailyExcess(i) - meanY
        sumXX = sumXX + dx * dx
        sumYY = sumYY + dy * dy
        sumXY = sumXY + dx * dy
    Next i

    If sumXX > 0# And sumYY > 0# Then
        mSlope = sumXY / sumXX
        mIntercept = meanY - mSlope * meanX
        mCorr = sumXY / Sqr(sumXX * sumYY)
        mR2 = mCorr * mCorr
    End If
    mHitRate = hits / mDailyN

    ReDim sx(1 To mDailyN)
    ReDim sy(1 To mDailyN)
    For i = 1 To mDailyN
        sx(i) = mDailySpread(i)
        sy(i) = mDailyExcess(i)
    Next i
    USD_QuickSortPair sx, sy, 1, mDailyN

    q4Count = (mDailyN + 3) \ 4
    q4Start = mDailyN - q4Count + 1
    For i = q4Start To mDailyN
        q4Sum = q4Sum + sy(i)
        If sy(i) > 0# Then q4Hits = q4Hits + 1
    Next i
    mQ4Mean = q4Sum / q4Count
    mQ4Hit = q4Hits / q4Count
End Sub

Private Sub USD_CalculateMonthlyEvidence()
    Dim starts() As Long
    Dim count As Long
    Dim i As Long, t As Long
    Dim monthKey As Long, priorKey As Long
    Dim best As Double

    ReDim starts(1 To mCurveN)

    For i = 1 To mCurveN
        If CDbl(DateAdd("m", 6, CDate(mDate(i)))) <= mDate(mCurveN) Then
            monthKey = Year(CDate(mDate(i))) * 100 + Month(CDate(mDate(i)))
            If monthKey <> priorKey Then
                count = count + 1
                starts(count) = i
                priorKey = monthKey
            End If
        End If
    Next i

    If count = 0 Then Err.Raise vbObjectError + 3021, , "No monthly six-month start dates were found."

    mMonthlyN = count
    ReDim mMonthlyStart(1 To count)
    ReDim mHistRet(1 To 4, 1 To count)
    ReDim mInvRet(1 To 4, 1 To count)
    ReDim mUpRet(1 To 4, 1 To count)

    For t = 1 To 4
        mUpExcess(t) = 0#
        mInvExcess(t) = 0#
    Next t
    mInv6MWins = 0

    For i = 1 To count
        mMonthlyStart(i) = mDate(starts(i))
        For t = 1 To 4
            mHistRet(t, i) = USD_PathReturn(starts(i), t, USD_TenorMonths(t), 0)
            mUpRet(t, i) = USD_PathReturn(starts(i), t, USD_TenorMonths(t), 1)
            mInvRet(t, i) = USD_PathReturn(starts(i), t, USD_TenorMonths(t), 2)
        Next t

        For t = 1 To 4
            mUpExcess(t) = mUpExcess(t) + (mUpRet(t, i) - mUpRet(1, i)) * 10000#
            mInvExcess(t) = mInvExcess(t) + (mInvRet(t, i) - mInvRet(1, i)) * 10000#
        Next t

        best = mInvRet(1, i)
        For t = 2 To 4
            If mInvRet(t, i) > best Then best = mInvRet(t, i)
        Next t
        If Abs(mInvRet(4, i) - best) < 0.0000000001 Then mInv6MWins = mInv6MWins + 1
    Next i

    For t = 1 To 4
        mUpExcess(t) = mUpExcess(t) / count
        mInvExcess(t) = mInvExcess(t) / count
    Next t
End Sub

Private Sub USD_OptimizePortfolio()
    Dim units As Long
    Dim a As Long, b As Long, c As Long, d As Long
    Dim i As Long, t As Long
    Dim w(1 To 4) As Double
    Dim histBest() As Double, invBest() As Double
    Dim histPort As Double, invPort As Double
    Dim histReg As Double, invReg As Double, worstReg As Double, meanReg As Double
    Dim bestWorst As Double, bestMean As Double
    Dim first As Boolean

    mAllocationCount = 0
    units = CLng(Round(1# / mAllocationStep, 0))
    If units < 2 Then Err.Raise vbObjectError + 3022, , "Allocation step is too large."

    ReDim histBest(1 To mMonthlyN)
    ReDim invBest(1 To mMonthlyN)

    For i = 1 To mMonthlyN
        histBest(i) = mHistRet(1, i)
        invBest(i) = mInvRet(1, i)
        For t = 2 To 4
            If mHistRet(t, i) > histBest(i) Then histBest(i) = mHistRet(t, i)
            If mInvRet(t, i) > invBest(i) Then invBest(i) = mInvRet(t, i)
        Next t
    Next i


    first = True

    For a = 0 To units
        For b = 0 To units - a
            For c = 0 To units - a - b
                d = units - a - b - c

                w(1) = a / units
                w(2) = b / units
                w(3) = c / units
                w(4) = d / units

                histReg = 0#: invReg = 0#
                For i = 1 To mMonthlyN
                    histPort = 0#: invPort = 0#
                    For t = 1 To 4
                        histPort = histPort + w(t) * mHistRet(t, i)
                        invPort = invPort + w(t) * mInvRet(t, i)
                    Next t
                    histReg = histReg + (histBest(i) - histPort)
                    invReg = invReg + (invBest(i) - invPort)
                Next i

                histReg = histReg / mMonthlyN
                invReg = invReg / mMonthlyN
                worstReg = WorksheetFunction.Max(histReg, invReg)
                meanReg = (histReg + invReg) / 2#

                mAllocationCount = mAllocationCount + 1


                If first Or worstReg < bestWorst - 0.000000000001 Or _
                   (Abs(worstReg - bestWorst) < 0.000000000001 And meanReg < bestMean) Then
                    first = False
                    bestWorst = worstReg
                    bestMean = meanReg
                    For t = 1 To 4
                        mRobustW(t) = w(t)
                    Next t
                    mRobustHistRegret = histReg
                    mRobustInvRegret = invReg
                    mRobustWorstRegret = worstReg
                    mWeightedMonths = w(2) + 3# * w(3) + 6# * w(4)
                End If

                If a = units And b = 0 And c = 0 Then mONWorstRegret = worstReg
                If d = units And a = 0 And b = 0 And c = 0 Then m6MWorstRegret = worstReg
            Next c
        Next b
    Next a

End Sub

'============================== RETURN ENGINE ========================================

Private Function USD_PathReturn( _
    ByVal startIndex As Long, _
    ByVal tenorIndex As Long, _
    ByVal tenorMonths As Long, _
    ByVal pathCode As Long) As Double

    Dim startDate As Date
    Dim maturityDate As Date
    Dim endIndex As Long
    Dim currentIndex As Long
    Dim nextIndex As Long
    Dim nextTarget As Date
    Dim factor As Double
    Dim k As Long
    Dim rateValue As Double

    startDate = CDate(mDate(startIndex))
    maturityDate = DateAdd("m", 6, startDate)
    endIndex = USD_FloorIndex(CDbl(maturityDate))
    If endIndex <= startIndex Then Exit Function

    factor = 1#

    If tenorMonths = 0 Then
        For currentIndex = startIndex To endIndex - 1
            rateValue = USD_PathRate(tenorIndex, currentIndex, pathCode)
            factor = factor * (1# + rateValue * (mDate(currentIndex + 1) - mDate(currentIndex)) / 360#)
        Next currentIndex
    Else
        currentIndex = startIndex
        k = 1
        Do While currentIndex < endIndex
            nextTarget = DateAdd("m", tenorMonths * k, startDate)
            If nextTarget >= maturityDate Then
                nextIndex = endIndex
            Else
                nextIndex = USD_FloorIndex(CDbl(nextTarget))
            End If

            If nextIndex <= currentIndex Then
                k = k + 1
            Else
                rateValue = USD_PathRate(tenorIndex, currentIndex, pathCode)
                factor = factor * (1# + rateValue * (mDate(nextIndex) - mDate(currentIndex)) / 360#)
                currentIndex = nextIndex
                If nextTarget < maturityDate Then k = k + 1
            End If
        Loop
    End If

    USD_PathReturn = factor - 1#
End Function

Private Function USD_PathRate(ByVal tenorIndex As Long, ByVal observation As Long, ByVal pathCode As Long) As Double
    Select Case pathCode
        Case 0      'Actual historical path
            USD_PathRate = mRate(tenorIndex, observation)
        Case 1      'Observed moves, rebased to common starting ON
            USD_PathRate = mStressON + mRate(tenorIndex, observation) - mRate(1, 1)
        Case 2      'Opposite-sign moves, same initial cross-tenor relationships
            USD_PathRate = mStressON + 2# * mRate(tenorIndex, 1) - mRate(1, 1) - mRate(tenorIndex, observation)
        Case Else
            Err.Raise vbObjectError + 3030, , "Invalid path code."
    End Select
End Function

Private Function USD_TenorMonths(ByVal tenorIndex As Long) As Long
    Select Case tenorIndex
        Case 1: USD_TenorMonths = 0
        Case 2: USD_TenorMonths = 1
        Case 3: USD_TenorMonths = 3
        Case 4: USD_TenorMonths = 6
    End Select
End Function

Private Function USD_FloorIndex(ByVal targetDate As Double) As Long
    Dim lo As Long, hi As Long, mid As Long

    If targetDate <= mDate(1) Then USD_FloorIndex = 1: Exit Function
    If targetDate >= mDate(mCurveN) Then USD_FloorIndex = mCurveN: Exit Function

    lo = 1: hi = mCurveN
    Do While lo <= hi
        mid = (lo + hi) \ 2
        If mDate(mid) = targetDate Then
            USD_FloorIndex = mid
            Exit Function
        ElseIf mDate(mid) < targetDate Then
            lo = mid + 1
        Else
            hi = mid - 1
        End If
    Loop
    USD_FloorIndex = hi
End Function

    Dim i As Long, t As Long

Private Sub USD_ApplyStyleCore(ByVal ws As Worksheet)

    Dim cPos As Long, cNeg As Long, cStress As Long
    Dim lineWidth As Double
    Dim ch As Chart

    cON = USD_HexColor(CStr(ws.Range("B4").value), RGB(148, 155, 164))
    c1M = USD_HexColor(CStr(ws.Range("B5").value), RGB(92, 135, 170))
    c3M = USD_HexColor(CStr(ws.Range("B6").value), RGB(31, 122, 107))
    c6M = USD_HexColor(CStr(ws.Range("B7").value), RGB(23, 54, 93))
    cPos = USD_HexColor(CStr(ws.Range("B8").value), RGB(78, 124, 60))
    cNeg = USD_HexColor(CStr(ws.Range("B9").value), RGB(166, 46, 46))
    cStress = USD_HexColor(CStr(ws.Range("B10").value), RGB(194, 118, 61))

    lineWidth = 2
    If IsNumeric(ws.Range("E6").value) Then lineWidth = CDbl(ws.Range("E6").value)
    If lineWidth <= 0 Then lineWidth = 2

    USD_PaintColorCells ws

    'Chart 1 - curve
    If USD_ChartExists(ws, "chtCurve") Then
        Set ch = ws.ChartObjects("chtCurve").Chart
        USD_SetLineSeries ch, 1, cON, lineWidth
        USD_SetLineSeries ch, 2, c1M, lineWidth
        USD_SetLineSeries ch, 3, c3M, lineWidth

        USD_SetDateAxisFromDashboard ch, ws
        USD_SetOptionalCurveYScale ch, ws
        USD_ApplyCurveAnnotations ch, ws
    End If

    'Chart 2 - premium vs realized
    If USD_ChartExists(ws, "chtPremiumTime") Then
        Set ch = ws.ChartObjects("chtPremiumTime").Chart
        USD_SetLineSeries ch, 1, c1M, lineWidth
        USD_SetLineSeries ch, 2, c6M, lineWidth + 0.25
        USD_SetDateAxisFromDashboard ch, ws
    End If

    'Chart 3 - scatter
    If USD_ChartExists(ws, "chtPremiumScatter") Then
        Set ch = ws.ChartObjects("chtPremiumScatter").Chart
        If ch.SeriesCollection.count > 0 Then
            ch.SeriesCollection(1).MarkerForegroundColor = c6M
            ch.SeriesCollection(1).MarkerBackgroundColor = c6M
        End If
        On Error Resume Next
        If ch.SeriesCollection(1).Trendlines.count > 0 Then ch.SeriesCollection(1).Trendlines(1).Format.Line.ForeColor.RGB = cNeg
        On Error GoTo 0
    End If

    'Chart 4 - directional stress
    If USD_ChartExists(ws, "chtStress") Then
        Set ch = ws.ChartObjects("chtStress").Chart
        If ch.SeriesCollection.count >= 2 Then
            ch.SeriesCollection(1).Format.Fill.ForeColor.RGB = c6M
            ch.SeriesCollection(1).Format.Line.Visible = msoFalse
            ch.SeriesCollection(2).Format.Fill.ForeColor.RGB = cStress
            ch.SeriesCollection(2).Format.Line.Visible = msoFalse
        End If
    End If

    'Chart 5 - regret
    If USD_ChartExists(ws, "chtRegret") Then
        Set ch = ws.ChartObjects("chtRegret").Chart
        If ch.SeriesCollection.count > 0 Then
            ch.SeriesCollection(1).Format.Fill.ForeColor.RGB = c6M
            ch.SeriesCollection(1).Format.Line.Visible = msoFalse
        End If
    End If

    'Chart 6 - mix
    If USD_ChartExists(ws, "chtMix") Then
        Set ch = ws.ChartObjects("chtMix").Chart
        If ch.SeriesCollection.count >= 4 Then
            ch.SeriesCollection(1).Format.Fill.ForeColor.RGB = cON
            ch.SeriesCollection(2).Format.Fill.ForeColor.RGB = c1M
            ch.SeriesCollection(3).Format.Fill.ForeColor.RGB = c3M
            ch.SeriesCollection(4).Format.Fill.ForeColor.RGB = c6M
            ch.SeriesCollection(1).Format.Line.Visible = msoFalse
            ch.SeriesCollection(2).Format.Line.Visible = msoFalse
            ch.SeriesCollection(3).Format.Line.Visible = msoFalse
            ch.SeriesCollection(4).Format.Line.Visible = msoFalse
        End If
    End If

    Dim minD As Double, maxD As Double
    minD = mDate(1): maxD = mDate(mCurveN)

    If IsDate(ws.Range("E4").value) Then minD = CDbl(CDate(ws.Range("E4").value))
    If IsDate(ws.Range("E5").value) Then maxD = CDbl(CDate(ws.Range("E5").value))

    If minD < mDate(1) Then minD = mDate(1)
    If maxD > mDate(mCurveN) Then maxD = mDate(mCurveN)
    If maxD <= minD Then
        minD = mDate(1)
        maxD = mDate(mCurveN)
    End If

    On Error Resume Next
    ch.Axes(xlCategory).MinimumScale = minD
    ch.Axes(xlCategory).MaximumScale = maxD
    ch.Axes(xlCategory).MajorUnit = WorksheetFunction.Max(30, (maxD - minD) / 8#)
    On Error GoTo 0
End Sub

Private Sub USD_SetOptionalCurveYScale(ByVal ch As Chart, ByVal ws As Worksheet)
    On Error Resume Next
    ch.Axes(xlValue).MinimumScaleIsAuto = True
    ch.Axes(xlValue).MaximumScaleIsAuto = True
    If IsNumeric(ws.Range("E7").value) And Len(CStr(ws.Range("E7").value)) > 0 Then ch.Axes(xlValue).MinimumScale = CDbl(ws.Range("E7").value)
    If IsNumeric(ws.Range("E8").value) And Len(CStr(ws.Range("E8").value)) > 0 Then ch.Axes(xlValue).MaximumScale = CDbl(ws.Range("E8").value)
    On Error GoTo 0
End Sub

Private Sub USD_ApplyCurveAnnotations(ByVal ch As Chart, ByVal ws As Worksheet)
    Dim i As Long, rowN As Long
    Dim showIt As Boolean
    Dim eventDate As Double, tenor As String, labelText As String
    Dim colorValue As Long
    Dim xOffset As Double, yOffset As Double
    Dim idx As Long, tenorIdx As Long
    Dim rateValue As Double
    Dim xMin As Double, xMax As Double, yMin As Double, yMax As Double
    Dim xPoint As Double, yPoint As Double
    Dim tx As Double, ty As Double
    Dim ln As Shape, tb As Shape

    On Error Resume Next
    For i = ch.Shapes.count To 1 Step -1
        If Left$(ch.Shapes(i).Name, 4) = "ANN_" Then ch.Shapes(i).Delete
    Next i
    On Error GoTo 0

    ch.Refresh
    DoEvents

    On Error Resume Next
    xMin = ch.Axes(xlCategory).MinimumScale
    xMax = ch.Axes(xlCategory).MaximumScale
    yMin = ch.Axes(xlValue).MinimumScale
    yMax = ch.Axes(xlValue).MaximumScale
    On Error GoTo 0

    If xMax <= xMin Or yMax <= yMin Then Exit Sub

    For rowN = 5 To 9
        showIt = USD_IsTrue(ws.Cells(rowN, "H").value)
        If showIt And IsDate(ws.Cells(rowN, "I").value) Then
            eventDate = CDbl(CDate(ws.Cells(rowN, "I").value))
            tenor = UCase$(Trim$(CStr(ws.Cells(rowN, "J").value)))
            labelText = CStr(ws.Cells(rowN, "K").value)
            colorValue = USD_HexColor(CStr(ws.Cells(rowN, "L").value), RGB(23, 54, 93))
            xOffset = USD_NumberOrZero(ws.Cells(rowN, "M").value)
            yOffset = USD_NumberOrZero(ws.Cells(rowN, "N").value)

            tenorIdx = USD_TenorIndex(tenor)
            If tenorIdx > 0 And eventDate >= xMin And eventDate <= xMax Then
                idx = USD_FloorIndex(eventDate)
                rateValue = mRate(tenorIdx, idx)

                If rateValue >= yMin And rateValue <= yMax Then
                    xPoint = ch.PlotArea.InsideLeft + (eventDate - xMin) / (xMax - xMin) * ch.PlotArea.InsideWidth
                    yPoint = ch.PlotArea.InsideTop + (yMax - rateValue) / (yMax - yMin) * ch.PlotArea.InsideHeight

                    tx = xPoint + xOffset - 55
                    ty = yPoint + yOffset

                    If tx < 2 Then tx = 2
                    If ty < 2 Then ty = 2
                    If tx > ch.ChartArea.Width - 112 Then tx = ch.ChartArea.Width - 112
                    If ty > ch.ChartArea.Height - 34 Then ty = ch.ChartArea.Height - 34

                    Set ln = ch.Shapes.AddLine(tx + 55, ty + 28, xPoint, yPoint)
                    ln.Name = "ANN_LINE_" & CStr(rowN)
                    With ln.Line
                        .ForeColor.RGB = colorValue
                        .Weight = 1.5
                        .EndArrowheadStyle = msoArrowheadTriangle
                    End With

                    Set tb = ch.Shapes.AddTextbox(msoTextOrientationHorizontal, tx, ty, 110, 28)
                    tb.Name = "ANN_TEXT_" & CStr(rowN)
                    tb.Fill.Visible = msoFalse
                    tb.Line.Visible = msoFalse
                    With tb.TextFrame2
                        .TextRange.Text = labelText
                        .TextRange.Font.Name = "Calibri"
                        .TextRange.Font.Size = 8.5
                        .TextRange.Font.Bold = msoTrue
                        .TextRange.Font.Fill.ForeColor.RGB = colorValue
                        .TextRange.ParagraphFormat.Alignment = msoAlignCenter
                        .VerticalAnchor = msoAnchorMiddle
                    End With
                End If
            End If
        End If
    Next rowN
End Sub

Private Sub USD_PaintColorCells(ByVal ws As Worksheet)
    Dim r As Long, c As Long, v As Long
    For r = 4 To 10
        v = USD_HexColor(CStr(ws.Cells(r, "B").value), RGB(255, 255, 255))
        ws.Cells(r, "B").Interior.Color = v
        ws.Cells(r, "B").Font.Color = USD_ContrastFont(v)
        ws.Cells(r, "B").Font.Bold = True
    Next r

    For r = 5 To 9
        v = USD_HexColor(CStr(ws.Cells(r, "L").value), RGB(255, 255, 255))
        ws.Cells(r, "L").Interior.Color = v
        ws.Cells(r, "L").Font.Color = USD_ContrastFont(v)
        ws.Cells(r, "L").Font.Bold = True
    Next r
End Sub

Private Function USD_TenorName(ByVal tenorIndex As Long) As String
    Select Case tenorIndex
        Case 1: USD_TenorName = "ON"
        Case 2: USD_TenorName = "1M"
        Case 3: USD_TenorName = "3M"
        Case 4: USD_TenorName = "6M"
    End Select
End Function

Private Function USD_TenorIndex(ByVal tenorName As String) As Long
    Select Case UCase$(Trim$(tenorName))
        Case "ON": USD_TenorIndex = 1
        Case "1M": USD_TenorIndex = 2
        Case "3M": USD_TenorIndex = 3
        Case "6M": USD_TenorIndex = 4
        Case Else: USD_TenorIndex = 0
    End Select
End Function

Private Function USD_IsTrue(ByVal value As Variant) As Boolean
    If VarType(value) = vbBoolean Then
        USD_IsTrue = CBool(value)
    Else
        USD_IsTrue = (UCase$(Trim$(CStr(value))) = "TRUE" Or Val(CStr(value)) = 1)
    End If
End Function

Private Function USD_NumberOrZero(ByVal value As Variant) As Double
    If IsNumeric(value) Then USD_NumberOrZero = CDbl(value) Else USD_NumberOrZero = 0#
End Function

Private Function USD_HexColor(ByVal hexText As String, ByVal fallback As Long) As Long
    Dim s As String
    Dim r As Long, g As Long, b As Long

    On Error GoTo bad
    s = Replace(Trim$(hexText), "#", "")
    If Len(s) <> 6 Then GoTo bad

    r = CLng("&H" & mid$(s, 1, 2))
    g = CLng("&H" & mid$(s, 3, 2))
    b = CLng("&H" & mid$(s, 5, 2))
    USD_HexColor = RGB(r, g, b)
    Exit Function
bad:
    USD_HexColor = fallback
End Function

Private Function USD_ContrastFont(ByVal fillColor As Long) As Long
    Dim r As Long, g As Long, b As Long
    Dim luminance As Double
    r = fillColor Mod 256
    g = (fillColor \ 256) Mod 256
    b = (fillColor \ 65536) Mod 256
    luminance = 0.299 * r + 0.587 * g + 0.114 * b
    If luminance < 145 Then USD_ContrastFont = RGB(255, 255, 255) Else USD_ContrastFont = RGB(37, 42, 49)
End Function

Private Sub USD_WriteRow(ByVal rng As Range, ByVal vals As Variant)
    Dim i As Long
    For i = LBound(vals) To UBound(vals)
        rng.Cells(1, i - LBound(vals) + 1).value = vals(i)
    Next i
End Sub

Private Sub USD_QuickSortPair(ByRef x() As Double, ByRef y() As Double, ByVal first As Long, ByVal last As Long)
    Dim i As Long, j As Long
    Dim pivot As Double, tx As Double, ty As Double

    i = first: j = last
    pivot = x((first + last) \ 2)

    Do While i <= j
        Do While x(i) < pivot: i = i + 1: Loop
        Do While x(j) > pivot: j = j - 1: Loop
        If i <= j Then
            tx = x(i): x(i) = x(j): x(j) = tx
            ty = y(i): y(i) = y(j): y(j) = ty
            i = i + 1
            j = j - 1
        End If
    Loop

    If first < j Then USD_QuickSortPair x, y, first, j
    If i < last Then USD_QuickSortPair x, y, i, last
End Sub
