' PURPOSE
'   Creates the four-slide presentation usd_cash_allocation_slides_1_4.pptx directly
'   from 13-aug.xlsx. The module reproduces the reference deck's content, narrative,
'   geometry, colors, labels and risk conclusions. All displayed curve, realized-
'   return, regression and allocation numbers are recalculated from the workbook.
'
' REQUIREMENTS
'   - Excel for Windows and Microsoft PowerPoint.
'   - Workbook sheets: Curve and F_Portfolios.
'   - No PowerPoint reference is required (late binding is used).
'
' USE
'   1. Open 13-aug.xlsx.
'   2. Press Alt+F11, insert a standard Module, and paste this entire file.
'   3. Run USD_CreateCashAllocationSlides_1_4.
'   4. The presentation is saved beside the workbook.
'====================================================================================

Private Sub USD_Slide1TableHeader(ByVal sld As Object)
    Dim widths As Variant
    Dim headers As Variant
    Dim i As Long
    Dim x As Double
    widths = Array(92#, 126#, 152#, 152#, 156#)
    headers = Array("START", "6M–ON AT START", "ROLLED ON", "LOCKED 6M", "6M EXCESS")
    x = 64
    For i = 0 To 4
        USD_AddText sld, "s1-table-header-" & CStr(i), CStr(headers(i)), x, 533, CDbl(widths(i)), 25, 14, True, USD_Gray600, IIf(i = 0, PP_ALIGN_LEFT, PP_ALIGN_CENTER)
        x = x + CDbl(widths(i))
    Next i
    USD_AddRule sld, "s1-table-header-rule", 64, 561, 678, 1, USD_Gray300
End Sub

Private Sub USD_Slide1TableRow( _
    ByVal sld As Object, ByVal rowId As String, ByVal top As Double, ByVal fillColor As Long, _
    ByVal label As String, ByVal premium As Double, ByVal onRet As Double, _
    ByVal sixRet As Double, ByVal excess As Double)

    Dim widths As Variant
    Dim values As Variant
    Dim i As Long
    Dim x As Double
    Dim color As Long

    widths = Array(92#, 126#, 152#, 152#, 156#)
    values = Array(label, USD_FmtSigned(premium, 1) & " bp", Format(onRet * 100#, "0.000") & "%", Format(sixRet * 100#, "0.000") & "%", USD_FmtSigned(excess, 1) & " bp")
    USD_AddRect sld, "s1-table-row-bg-" & rowId, 64, top, 678, 38, fillColor, USD_Grid, 0.5
    x = 64
    For i = 0 To 4
        If i = 4 Then
            If excess < 0# Then color = USD_Red Else color = USD_Teal
        ElseIf i = 1 Then
            color = USD_Navy2
        Else
            color = USD_Charcoal
        End If
        USD_AddText sld, "s1-table-cell-" & rowId & "-" & CStr(i), CStr(values(i)), x, top, CDbl(widths(i)), 38, IIf(i = 0, 17, 18), (i = 0 Or i = 4), color, IIf(i = 0, PP_ALIGN_LEFT, PP_ALIGN_CENTER)
        x = x + CDbl(widths(i))
    Next i
End Sub

Private Sub USD_ChartCurveHistory(ByVal sld As Object, ByVal l As Double, ByVal t As Double, ByVal w As Double, ByVal h As Double)

    Dim co As Object
    Dim ch As Object
    Dim data() As Variant
    Dim i As Long
    Dim j As Long
    Dim colors As Variant
    Dim weights As Variant

    USD_ResetChartSheet
    ReDim data(1 To mCurveN + 1, 1 To 5)
    data(1, 1) = "Date": data(1, 2) = "ON": data(1, 3) = "1M": data(1, 4) = "3M": data(1, 5) = "6M"
    For i = 1 To mCurveN
        data(i + 1, 1) = mDate(i)
        For j = 1 To 4
            data(i + 1, j + 1) = mRate(j, i) * 100#
        Next j
    Next i
    mTmp.Range("A1").Resize(mCurveN + 1, 5).Value = data

    Set co = USD_NewChart(XL_XY_LINES_NO_MARKERS, w, h)
    Set ch = co.Chart
    USD_DeleteAllSeries ch
    colors = Array(USD_Gray500, USD_Blue, USD_Teal, USD_Navy)
    weights = Array(1.5, 1.7, 1.9, 3#)
    For j = 1 To 4
        USD_AddXYSeries ch, CStr(data(1, j + 1)), mTmp.Range("A2:A" & CStr(mCurveN + 1)), mTmp.Range(mTmp.Cells(2, j + 1), mTmp.Cells(mCurveN + 1, j + 1)), CLng(colors(j - 1)), CDbl(weights(j - 1)), False, False
    Next j
    USD_FormatXYAxes ch, mDate(1), mDate(mCurveN), 366, 0, 6, 1, "yyyy", "0.0", 14
    USD_PasteChart sld, ch, "s1-sofr-curve-history", l, t, w, h

End Sub

Private Function USD_FmtSigned(ByVal value As Double, ByVal decimals As Long) As String
    Dim pattern As String
    pattern = "0"
    If decimals > 0 Then pattern = pattern & "." & String$(decimals, "0")
    If Abs(value) < 0.5 * 10# ^ (-decimals) Then
        USD_FmtSigned = Format(0#, pattern)
    ElseIf value > 0# Then
        USD_FmtSigned = "+" & Format(value, pattern)
    Else
        USD_FmtSigned = "−" & Format(Abs(value), pattern)
    End If
End Function

Private Function USD_FmtNumber(ByVal value As Double, ByVal decimals As Long) As String
    Dim pattern As String
    pattern = "0"
    If decimals > 0 Then pattern = pattern & "." & String$(decimals, "0")
    If value < 0# Then
        USD_FmtNumber = "−" & Format(Abs(value), pattern)
    Else
        USD_FmtNumber = Format(value, pattern)
    End If
End Function

Public Sub USD_CreateCashAllocationSlides_1_4_To(ByVal outputPath As String)

    Dim pptApp As Object
    Dim pres As Object
    Dim oldScreen As Boolean
    Dim oldAlerts As Boolean
    Dim savePath As String
    Dim errText As String

    On Error GoTo fail

    oldScreen = Application.ScreenUpdating
    oldAlerts = Application.DisplayAlerts
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.StatusBar = "Auditing workbook data for slides 1-4..."

    USD_ResetState

    USD_ValidateWorkbook
    USD_LoadCurve
    USD_CalculateDailyEvidence
    USD_CalculateDirectionalStress
    USD_LoadPortfolioResult

    Set mTmp = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
    mTmp.Name = USD_UniqueSheetName("_PPT_Chart_Data")

    Set pptApp = CreateObject("PowerPoint.Application")
    pptApp.Visible = True
    Set pres = pptApp.Presentations.Add
    pres.PageSetup.SlideWidth = SLIDE_W
    pres.PageSetup.SlideHeight = SLIDE_H

    Application.StatusBar = "Creating slide 1 of 4..."
    USD_BuildSlide1 pres.Slides.Add(1, PP_LAYOUT_BLANK)
    Application.StatusBar = "Creating slide 2 of 4..."
    USD_BuildSlide2 pres.Slides.Add(2, PP_LAYOUT_BLANK)
    Application.StatusBar = "Creating slide 3 of 4..."
    USD_BuildSlide3 pres.Slides.Add(3, PP_LAYOUT_BLANK)
    Application.StatusBar = "Creating slide 4 of 4..."
    USD_BuildSlide4 pres.Slides.Add(4, PP_LAYOUT_BLANK)

    savePath = USD_OutputPath(outputPath)
    If Len(Dir$(savePath)) > 0 Then
        If MsgBox("Replace the existing presentation?" & vbCrLf & savePath, _
                  vbQuestion + vbYesNo, "USD cash allocation") <> vbYes Then
            savePath = USD_TimestampedPath(savePath)
        Else
            Kill savePath
        End If
    End If
    pres.SaveAs savePath, PP_SAVE_OPENXML

    USD_DeleteTempSheet
    Application.DisplayAlerts = oldAlerts
    Application.ScreenUpdating = oldScreen
    Application.StatusBar = False
    MsgBox "PowerPoint created:" & vbCrLf & savePath, vbInformation
    Exit Sub

fail:
    errText = Err.Description
    On Error Resume Next
    USD_DeleteTempSheet
    If Not pres Is Nothing Then pres.Close
    If Not pptApp Is Nothing Then pptApp.Quit
    Application.DisplayAlerts = oldAlerts
    Application.ScreenUpdating = oldScreen
    Application.StatusBar = False
    On Error GoTo 0
    MsgBox "USD_CreateCashAllocationSlides_1_4 stopped: " & errText, vbCritical

End Sub

Private Sub USD_LoadCurve()

    Dim ws As Object
    Dim lastRow As Long
    Dim v As Variant
    Dim i As Long
    Dim j As Long

    Set ws = ThisWorkbook.Worksheets("Curve")
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    If lastRow < 3 Then Err.Raise vbObjectError + 2102, , "Curve has no usable observations."

    If UCase$(Trim$(CStr(ws.Cells(1, 1).Value))) <> "DATE" Then _
        Err.Raise vbObjectError + 2103, , "Curve column A must be Date."
    If UCase$(Trim$(CStr(ws.Cells(1, 2).Value))) <> "ON" Then _
        Err.Raise vbObjectError + 2104, , "Curve column B must be ON."
    If UCase$(Trim$(CStr(ws.Cells(1, 5).Value))) <> "6M" Then _
        Err.Raise vbObjectError + 2105, , "Curve columns B:E must be ON, 1M, 3M and 6M."

    v = ws.Range("A2:E" & lastRow).Value2
    mCurveN = UBound(v, 1)
    ReDim mDate(1 To mCurveN)
    ReDim mRate(1 To 4, 1 To mCurveN)

    For i = 1 To mCurveN
        If Not IsNumeric(v(i, 1)) Then _
            Err.Raise vbObjectError + 2106, , "Curve contains a non-date value at row " & CStr(i + 1) & "."
        mDate(i) = CDbl(v(i, 1))
        If i > 1 Then
            If mDate(i) <= mDate(i - 1) Then _
                Err.Raise vbObjectError + 2107, , "Curve dates must be strictly increasing."
        End If
        For j = 1 To 4
            If Not IsNumeric(v(i, j + 1)) Then _
                Err.Raise vbObjectError + 2108, , "Curve contains a blank/non-numeric rate at row " & CStr(i + 1) & "."
            mRate(j, i) = CDbl(v(i, j + 1))
        Next j
    Next i

End Sub

Private Sub USD_CalculateDailyEvidence()

    Dim i As Long
    Dim n As Long
    Dim lastDate As Double
    Dim startDate As Date
    Dim onRet As Double
    Dim sixRet As Double
    Dim sumX As Double
    Dim sumY As Double
    Dim sumXX As Double
    Dim sumYY As Double
    Dim sumXY As Double
    Dim dx As Double
    Dim dy As Double
    Dim hits As Long
    Dim sx() As Double
    Dim sy() As Double
    Dim q4Count As Long
    Dim q4Start As Long
    Dim q4Sum As Double
    Dim q4Hits As Long

    lastDate = mDate(mCurveN)
    For i = 1 To mCurveN
        startDate = CDate(mDate(i))
        If CDbl(DateAdd("m", 6, startDate)) <= lastDate Then n = n + 1
    Next i

    If n = 0 Then Err.Raise vbObjectError + 2110, , "Curve does not contain a six-month realized horizon."
    mDailyN = n
    ReDim mDailyDate(1 To n)
    ReDim mDailySpread(1 To n)
    ReDim mDailyExcess(1 To n)

    n = 0
    For i = 1 To mCurveN
        startDate = CDate(mDate(i))
        If CDbl(DateAdd("m", 6, startDate)) <= lastDate Then
            n = n + 1
            onRet = USD_PathReturn(i, 1, 0, 0)
            sixRet = USD_PathReturn(i, 4, 6, 0)
            mDailyDate(n) = mDate(i)
            mDailySpread(n) = (mRate(4, i) - mRate(1, i)) * 10000#
            mDailyExcess(n) = (sixRet - onRet) * 10000#
            sumX = sumX + mDailySpread(n)
            sumY = sumY + mDailyExcess(n)
            If mDailyExcess(n) > 0# Then hits = hits + 1
        End If
    Next i

    sumX = sumX / mDailyN
    sumY = sumY / mDailyN
    For i = 1 To mDailyN
        dx = mDailySpread(i) - sumX
        dy = mDailyExcess(i) - sumY
        sumXX = sumXX + dx * dx
        sumYY = sumYY + dy * dy
        sumXY = sumXY + dx * dy
    Next i

    If sumXX = 0# Or sumYY = 0# Then _
        Err.Raise vbObjectError + 2111, , "Regression cannot be calculated from constant data."
    mSlope = sumXY / sumXX
    mIntercept = sumY - mSlope * sumX
    mCorr = sumXY / Sqr(sumXX * sumYY)
    mR2 = mCorr * mCorr
    mHitRate = hits / mDailyN

    ReDim sx(1 To mDailyN)
    ReDim sy(1 To mDailyN)
    For i = 1 To mDailyN
        sx(i) = mDailySpread(i)
        sy(i) = mDailyExcess(i)
    Next i
    USD_QuickSortPair sx, sy, 1, mDailyN
    q4Count = (mDailyN + 3) \ 4
    q4Start = mDailyN - q4Count + 1
    For i = q4Start To mDailyN
        q4Sum = q4Sum + sy(i)
        If sy(i) > 0# Then q4Hits = q4Hits + 1
    Next i
    mQ4Mean = q4Sum / q4Count
    mQ4Hit = q4Hits / q4Count

End Sub

Private Sub USD_CalculateDirectionalStress()

    Dim starts() As Long
    Dim count As Long
    Dim i As Long
    Dim monthKey As Long
    Dim priorKey As Long
    Dim tenor As Long
    Dim upRet(1 To 4) As Double
    Dim downRet(1 To 4) As Double
    Dim upOn As Double
    Dim downOn As Double
    Dim best As Double
    Dim sixWins As Long

    For tenor = 1 To 4
        mUpExcess(tenor) = 0#
        mDownExcess(tenor) = 0#
    Next tenor
    mDownSixWins = 0

    ReDim starts(1 To mCurveN)
    For i = 1 To mCurveN
        If CDbl(DateAdd("m", 6, CDate(mDate(i)))) <= mDate(mCurveN) Then
            monthKey = Year(CDate(mDate(i))) * 100 + Month(CDate(mDate(i)))
            If monthKey <> priorKey Then
                count = count + 1
                starts(count) = i
                priorKey = monthKey
            End If
        End If
    Next i
    mMonthlyN = count

    For i = 1 To count
        For tenor = 1 To 4
            upRet(tenor) = USD_PathReturn(starts(i), tenor, USD_TenorMonths(tenor), 1)
            downRet(tenor) = USD_PathReturn(starts(i), tenor, USD_TenorMonths(tenor), 2)
        Next tenor
        upOn = upRet(1)
        downOn = downRet(1)
        For tenor = 1 To 4
            mUpExcess(tenor) = mUpExcess(tenor) + (upRet(tenor) - upOn) * 10000#
            mDownExcess(tenor) = mDownExcess(tenor) + (downRet(tenor) - downOn) * 10000#
        Next tenor
        best = downRet(1)
        For tenor = 2 To 4
            If downRet(tenor) > best Then best = downRet(tenor)
        Next tenor
        If Abs(downRet(4) - best) < 0.0000000001 Then sixWins = sixWins + 1
    Next i

    For tenor = 1 To 4
        mUpExcess(tenor) = mUpExcess(tenor) / count
        mDownExcess(tenor) = mDownExcess(tenor) / count
    Next tenor
    mDownSixWins = sixWins

End Sub

Private Sub USD_LoadPortfolioResult()

    Dim ws As Object
    Dim lastRow As Long
    Dim v As Variant
    Dim i As Long
    Dim robustFound As Boolean

    Set ws = ThisWorkbook.Worksheets("F_Portfolios")
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    If lastRow < 3 Then Err.Raise vbObjectError + 2120, , "F_Portfolios has no usable allocation grid."
    v = ws.Range("A2:L" & lastRow).Value2
    mPortfolioCount = UBound(v, 1)

    For i = 1 To UBound(v, 1)
        If Abs(CDbl(v(i, 1)) - 1#) < 0.0000001 And _
           Abs(CDbl(v(i, 2))) < 0.0000001 And _
           Abs(CDbl(v(i, 3))) < 0.0000001 And _
           Abs(CDbl(v(i, 4))) < 0.0000001 Then mONWorstRegret = CDbl(v(i, 9)) * 10000#

        If Abs(CDbl(v(i, 4)) - 1#) < 0.0000001 And _
           Abs(CDbl(v(i, 1))) < 0.0000001 And _
           Abs(CDbl(v(i, 2))) < 0.0000001 And _
           Abs(CDbl(v(i, 3))) < 0.0000001 Then mSixWorstRegret = CDbl(v(i, 9)) * 10000#

        If USD_IsTrue(v(i, 12)) Then
            mMix(1) = CDbl(v(i, 1))
            mMix(2) = CDbl(v(i, 2))
            mMix(3) = CDbl(v(i, 3))
            mMix(4) = CDbl(v(i, 4))
            mMixHistRegret = CDbl(v(i, 7)) * 10000#
            mMixInvRegret = CDbl(v(i, 8)) * 10000#
            mMixWorstRegret = CDbl(v(i, 9)) * 10000#
            mWeightedMonths = CDbl(v(i, 11))
            robustFound = True
        End If
    Next i

    If Not robustFound Then _
        Err.Raise vbObjectError + 2121, , "F_Portfolios does not contain a row marked Is_Robust = TRUE."

End Sub

Private Function USD_PathReturn( _
    ByVal startIndex As Long, _
    ByVal tenorIndex As Long, _
    ByVal tenorMonths As Long, _
    ByVal pathCode As Long) As Double

    Dim startDate As Date
    Dim maturityDate As Date
    Dim endIndex As Long
    Dim currentIndex As Long
    Dim nextIndex As Long
    Dim nextTarget As Date
    Dim factor As Double
    Dim k As Long
    Dim rateValue As Double

    startDate = CDate(mDate(startIndex))
    maturityDate = DateAdd("m", 6, startDate)
    endIndex = USD_FloorIndex(CDbl(maturityDate))
    If endIndex <= startIndex Then Exit Function
    factor = 1#

    If tenorMonths = 0 Then
        For currentIndex = startIndex To endIndex - 1
            rateValue = USD_PathRate(tenorIndex, currentIndex, pathCode)
            factor = factor * (1# + rateValue * (mDate(currentIndex + 1) - mDate(currentIndex)) / 360#)
        Next currentIndex
    Else
        currentIndex = startIndex
        k = 1
        Do While currentIndex < endIndex
            nextTarget = DateAdd("m", tenorMonths * k, startDate)
            If nextTarget >= maturityDate Then
                nextIndex = endIndex
            Else
                nextIndex = USD_FloorIndex(CDbl(nextTarget))
            End If
            If nextIndex <= currentIndex Then
                k = k + 1
            Else
                rateValue = USD_PathRate(tenorIndex, currentIndex, pathCode)
                factor = factor * (1# + rateValue * (mDate(nextIndex) - mDate(currentIndex)) / 360#)
                currentIndex = nextIndex
                If nextTarget < maturityDate Then k = k + 1
            End If
        Loop
    End If

    USD_PathReturn = factor - 1#

End Function

Private Function USD_PathRate(ByVal tenorIndex As Long, ByVal observation As Long, ByVal pathCode As Long) As Double
    Select Case pathCode
        Case 0
            USD_PathRate = mRate(tenorIndex, observation)
        Case 1
            USD_PathRate = STRESS_ON + mRate(tenorIndex, observation) - mRate(1, 1)
        Case 2
            USD_PathRate = STRESS_ON + 2# * mRate(tenorIndex, 1) - mRate(1, 1) - mRate(tenorIndex, observation)
        Case Else
            Err.Raise vbObjectError + 2122, , "Invalid directional path code."
    End Select
End Function

Private Function USD_TenorMonths(ByVal tenorIndex As Long) As Long
    Select Case tenorIndex
        Case 1: USD_TenorMonths = 0
        Case 2: USD_TenorMonths = 1
        Case 3: USD_TenorMonths = 3
        Case 4: USD_TenorMonths = 6
    End Select
End Function

Private Function USD_FloorIndex(ByVal targetDate As Double) As Long

    Dim lo As Long
    Dim hi As Long
    Dim mid As Long

    If targetDate <= mDate(1) Then
        USD_FloorIndex = 1
        Exit Function
    End If
    If targetDate >= mDate(mCurveN) Then
        USD_FloorIndex = mCurveN
        Exit Function
    End If

    lo = 1
    hi = mCurveN
    Do While lo <= hi
        mid = (lo + hi) \ 2
        If mDate(mid) = targetDate Then
            USD_FloorIndex = mid
            Exit Function
        ElseIf mDate(mid) < targetDate Then
            lo = mid + 1
        Else
            hi = mid - 1
        End If
    Loop
    USD_FloorIndex = hi

End Function

Private Function USD_FindExactIndex(ByVal targetDate As Double) As Long
    Dim idx As Long
    idx = USD_FloorIndex(targetDate)
    If Abs(mDate(idx) - targetDate) < 0.00001 Then USD_FindExactIndex = idx
End Function

Private Function USD_IsTrue(ByVal value As Variant) As Boolean
    If VarType(value) = vbBoolean Then
        USD_IsTrue = CBool(value)
    Else
        USD_IsTrue = (UCase$(Trim$(CStr(value))) = "TRUE" Or CDbl(Val(CStr(value))) = 1#)
    End If
End Function

Private Sub USD_QuickSortPair(ByRef x() As Double, ByRef y() As Double, ByVal first As Long, ByVal last As Long)

    Dim i As Long
    Dim j As Long
    Dim pivot As Double
    Dim tx As Double
    Dim ty As Double

    i = first
    j = last
    pivot = x((first + last) \ 2)
    Do While i <= j
        Do While x(i) < pivot: i = i + 1: Loop
        Do While x(j) > pivot: j = j - 1: Loop
        If i <= j Then
            tx = x(i): x(i) = x(j): x(j) = tx
            ty = y(i): y(i) = y(j): y(j) = ty
            i = i + 1
            j = j - 1
        End If
    Loop
    If first < j Then USD_QuickSortPair x, y, first, j
    If i < last Then USD_QuickSortPair x, y, i, last

End Sub
