The TABLE branch preserves the working start-date logic: exact curve date, otherwise the next available curve date; the effective end remains the latest curve date on or before the target. Pasted text.txt

Use:

=R_CURVE_FACTORS(A:F,K4,"TABLE")

or:

=R_CURVE_FACTORS(A:F,K4,"TRADE")

The previous syntax remains valid:

=R_CURVE_FACTORS(A:F,K4,FALSE)

I tested the logic with synthetic daily curve data, including a strategy date absent from the curve. For a strategy start of 1-Jan-2020, with the first available curve date on 2-Jan-2020, the first 6M trade produces:

Strategy Start = 1-Jan-2020
Start Date     = 2-Jan-2020
Target End     = 1-Jul-2020
Effective End  = latest curve date <= 1-Jul-2020

I could not execute the VBA inside Excel in this environment, so the testing covered the algorithm and boundary cases rather than Excel compilation.

Attribute VB_Name = "CurveFactors"
Option Explicit
' ============================================================================
' R_CURVE_FACTORS
'
' TABLE view:
'   =R_CURVE_FACTORS(A:F, K4, "TABLE")
'
' TRADE view:
'   =R_CURVE_FACTORS(A:F, K4, "TRADE")
'
' Without headers:
'   =R_CURVE_FACTORS(A:F, K4, "TABLE", FALSE)
'   =R_CURVE_FACTORS(A:F, K4, "TRADE", FALSE)
'
' Previous syntax remains supported:
'   =R_CURVE_FACTORS(A:F, K4)
'   =R_CURVE_FACTORS(A:F, K4, FALSE)
'
' Input:
'   Column 1 = Curve Date
'   Column 2 = ON
'   Column 3 = 1M
'   Column 4 = 2M
'   Column 5 = 3M
'   Column 6 = 6M
'
' TABLE VIEW
' ----------
' Keeps the existing working structure:
'
'   Start | Target End | Effective End | ON | 1M | 2M | 3M | 6M
'
' TRADE VIEW
' ----------
' Returns one row for each applicable tenor:
'
'   Strategy Start | Tenor | Start Date |
'   Target End | Effective End | Rate
'
' Trade-view date rules:
'
'   Strategy Start
'       Original input date, repeated on every trade row.
'
'   Scheduled Start
'       Original input date advanced by the monthly strategy position.
'
'   Start Date
'       Exact scheduled start date when present in the curve.
'       Otherwise, the next available curve date.
'
'   Target End
'       Calculated from the scheduled strategy date, not from the adjusted
'       curve start date.
'
'       ON = Scheduled Start + 1 calendar day
'       1M = Scheduled Start + 1 calendar month
'       2M = Scheduled Start + 2 calendar months
'       3M = Scheduled Start + 3 calendar months
'       6M = Scheduled Start + 6 calendar months
'
'   Effective End
'       Latest curve date less than or equal to Target End.
'
' Frequency:
'
'   ON = every monthly strategy position
'   1M = every monthly strategy position
'   2M = positions 1, 3, 5, 7...
'   3M = positions 1, 4, 7, 10...
'   6M = positions 1, 7, 13, 19...
' ============================================================================
Public Function R_CURVE_FACTORS( _
    ByVal CurveRange As Range, _
    ByVal RequestedStartDate As Variant, _
    Optional ByVal ViewMode As Variant = "TABLE", _
    Optional ByVal IncludeHeaders As Boolean = True _
) As Variant
    On Error GoTo InvalidInput
    Dim sourceData As Variant
    Dim curveDates() As Long
    Dim rates() As Double
    Dim onFactors() As Double
    Dim sourceFirstRow As Long
    Dim sourceLastRow As Long
    Dim firstDataRow As Long
    Dim lastDataRow As Long
    Dim rowCount As Long
    Dim curveCount As Long
    Dim requestedSerial As Long
    Dim selectedView As String
    Dim r As Long
    Dim c As Long
    Dim i As Long
    Application.Volatile False
    ' ------------------------------------------------------------------------
    ' Validate inputs
    ' ------------------------------------------------------------------------
    If CurveRange Is Nothing Then GoTo InvalidInput
    If CurveRange.Columns.Count < 6 Then GoTo InvalidInput
    If Not IsDate(RequestedStartDate) And _
       Not IsNumeric(RequestedStartDate) Then
        GoTo InvalidInput
    End If
    ' ------------------------------------------------------------------------
    ' Backward compatibility
    '
    ' Previous formula:
    '   =R_CURVE_FACTORS(A:F,K4,FALSE)
    '
    ' If the third argument is Boolean, treat it as IncludeHeaders and
    ' default the view to TABLE.
    ' ------------------------------------------------------------------------
    If VarType(ViewMode) = vbBoolean Then
        IncludeHeaders = CBool(ViewMode)
        selectedView = "TABLE"
    Else
        selectedView = UCase$(Trim$(CStr(ViewMode)))
        If Len(selectedView) = 0 Then
            selectedView = "TABLE"
        End If
    End If
    If selectedView <> "TABLE" And _
       selectedView <> "TRADE" Then
        GoTo InvalidInput
    End If
    ' ------------------------------------------------------------------------
    ' Read populated curve range
    ' ------------------------------------------------------------------------
    sourceFirstRow = CurveRange.Row
    sourceLastRow = CurveRange.Worksheet.Cells( _
        CurveRange.Worksheet.Rows.Count, _
        CurveRange.Column _
    ).End(xlUp).Row
    If sourceLastRow < sourceFirstRow Then GoTo InvalidInput
    sourceData = CurveRange.Worksheet.Range( _
        CurveRange.Worksheet.Cells( _
            sourceFirstRow, _
            CurveRange.Column _
        ), _
        CurveRange.Worksheet.Cells( _
            sourceLastRow, _
            CurveRange.Column + 5 _
        ) _
    ).Value2
    rowCount = UBound(sourceData, 1)
    ' ------------------------------------------------------------------------
    ' Find first data row
    ' Allows the source range to contain a header
    ' ------------------------------------------------------------------------
    firstDataRow = 0
    For r = 1 To rowCount
        If IsNumeric(sourceData(r, 1)) Then
            firstDataRow = r
            Exit For
        End If
    Next r
    If firstDataRow = 0 Then GoTo InvalidInput
    ' ------------------------------------------------------------------------
    ' Find last data row
    ' ------------------------------------------------------------------------
    lastDataRow = 0
    For r = rowCount To firstDataRow Step -1
        If IsNumeric(sourceData(r, 1)) Then
            lastDataRow = r
            Exit For
        End If
    Next r
    If lastDataRow < firstDataRow Then GoTo InvalidInput
    curveCount = lastDataRow - firstDataRow + 1
    If curveCount < 2 Then GoTo InvalidInput
    ReDim curveDates(1 To curveCount)
    ReDim rates(1 To curveCount, 1 To 5)
    ReDim onFactors(1 To curveCount)
    ' ------------------------------------------------------------------------
    ' Load and validate curve data
    ' ------------------------------------------------------------------------
    For i = 1 To curveCount
        r = firstDataRow + i - 1
        If Not IsNumeric(sourceData(r, 1)) Then
            GoTo InvalidInput
        End If
        curveDates(i) = CLng(Int(CDbl(sourceData(r, 1))))
        For c = 1 To 5
            If Not IsNumeric(sourceData(r, c + 1)) Then
                GoTo InvalidInput
            End If
            rates(i, c) = CDbl(sourceData(r, c + 1))
        Next c
        ' Curve dates must be strictly increasing.
        If i > 1 Then
            If curveDates(i) <= curveDates(i - 1) Then
                GoTo InvalidInput
            End If
        End If
    Next i
    ' ------------------------------------------------------------------------
    ' Build cumulative ON factors
    '
    ' This preserves the working calculation logic.
    ' ------------------------------------------------------------------------
    onFactors(1) = 1# + rates(1, 1) / 360#
    For i = 2 To curveCount
        onFactors(i) = onFactors(i - 1) * _
            (1# + _
            rates(i, 1) / 360# * _
            (curveDates(i) - curveDates(i - 1)))
    Next i
    requestedSerial = CLng(Int(CDbl(CDate(RequestedStartDate))))
    ' ------------------------------------------------------------------------
    ' Select output view
    ' ------------------------------------------------------------------------
    If selectedView = "TABLE" Then
        R_CURVE_FACTORS = BuildTableView( _
            curveDates, _
            rates, _
            onFactors, _
            curveCount, _
            requestedSerial, _
            IncludeHeaders _
        )
    Else
        R_CURVE_FACTORS = BuildTradeView( _
            curveDates, _
            rates, _
            onFactors, _
            curveCount, _
            requestedSerial, _
            IncludeHeaders _
        )
    End If
    Exit Function
InvalidInput:
    R_CURVE_FACTORS = CVErr(xlErrValue)
End Function
' ============================================================================
' TABLE VIEW
'
' This is the existing working logic.
' ============================================================================
Private Function BuildTableView( _
    ByRef curveDates() As Long, _
    ByRef rates() As Double, _
    ByRef onFactors() As Double, _
    ByVal curveCount As Long, _
    ByVal requestedSerial As Long, _
    ByVal IncludeHeaders As Boolean _
) As Variant
    On Error GoTo InvalidTable
    Dim startIndex As Long
    Dim endIndex As Long
    Dim effectiveStart As Long
    Dim effectiveEnd As Long
    Dim targetEnd As Date
    Dim targetSerial As Long
    Dim outputRows As Long
    Dim outputOffset As Long
    Dim onRate As Double
    Dim result() As Variant
    Dim i As Long
    ' ------------------------------------------------------------------------
    ' Original requested start-date rule
    '
    ' Exact date if available.
    ' Otherwise next available curve date.
    ' ------------------------------------------------------------------------
    startIndex = FirstDateGreaterThanOrEqual( _
        curveDates, _
        curveCount, _
        requestedSerial _
    )
    If startIndex = 0 Then GoTo InvalidTable
    effectiveStart = curveDates(startIndex)
    ' ------------------------------------------------------------------------
    ' Count monthly output periods
    ' ------------------------------------------------------------------------
    targetEnd = DateAdd("m", 1, CDate(effectiveStart))
    outputRows = 0
    endIndex = startIndex
    Do While CLng(targetEnd) <= curveDates(curveCount)
        targetSerial = CLng(targetEnd)
        endIndex = LastDateLessThanOrEqual( _
            curveDates, _
            curveCount, _
            targetSerial _
        )
        If endIndex = 0 Then GoTo InvalidTable
        If curveDates(endIndex) <= effectiveStart Then
            GoTo InvalidTable
        End If
        outputRows = outputRows + 1
        effectiveStart = curveDates(endIndex)
        targetEnd = DateAdd("m", 1, targetEnd)
    Loop
    If outputRows = 0 Then GoTo InvalidTable
    ' ------------------------------------------------------------------------
    ' Prepare table result
    ' ------------------------------------------------------------------------
    If IncludeHeaders Then
        outputOffset = 1
        ReDim result(1 To outputRows + 1, 1 To 8)
        result(1, 1) = "Start"
        result(1, 2) = "Target End"
        result(1, 3) = "Effective End"
        result(1, 4) = "ON"
        result(1, 5) = "1M"
        result(1, 6) = "2M"
        result(1, 7) = "3M"
        result(1, 8) = "6M"
    Else
        outputOffset = 0
        ReDim result(1 To outputRows, 1 To 8)
    End If
    ' ------------------------------------------------------------------------
    ' Reset to effective requested start
    ' ------------------------------------------------------------------------
    startIndex = FirstDateGreaterThanOrEqual( _
        curveDates, _
        curveCount, _
        requestedSerial _
    )
    effectiveStart = curveDates(startIndex)
    targetEnd = DateAdd("m", 1, CDate(effectiveStart))
    ' ------------------------------------------------------------------------
    ' Populate table output
    ' ------------------------------------------------------------------------
    For i = 1 To outputRows
        targetSerial = CLng(targetEnd)
        endIndex = LastDateLessThanOrEqual( _
            curveDates, _
            curveCount, _
            targetSerial _
        )
        effectiveEnd = curveDates(endIndex)
        onRate = AnnualizedONRate( _
            onFactors, _
            rates, _
            startIndex, _
            endIndex, _
            effectiveStart, _
            effectiveEnd _
        )
        result(i + outputOffset, 1) = CDate(effectiveStart)
        result(i + outputOffset, 2) = targetEnd
        result(i + outputOffset, 3) = CDate(effectiveEnd)
        ' ON: every row.
        result(i + outputOffset, 4) = onRate
        ' 1M: every row.
        result(i + outputOffset, 5) = rates(startIndex, 2)
        ' 2M: rows 1, 3, 5, 7...
        If (i - 1) Mod 2 = 0 Then
            result(i + outputOffset, 6) = rates(startIndex, 3)
        Else
            result(i + outputOffset, 6) = vbNullString
        End If
        ' 3M: rows 1, 4, 7, 10...
        If (i - 1) Mod 3 = 0 Then
            result(i + outputOffset, 7) = rates(startIndex, 4)
        Else
            result(i + outputOffset, 7) = vbNullString
        End If
        ' 6M: rows 1, 7, 13, 19...
        If (i - 1) Mod 6 = 0 Then
            result(i + outputOffset, 8) = rates(startIndex, 5)
        Else
            result(i + outputOffset, 8) = vbNullString
        End If
        effectiveStart = effectiveEnd
        startIndex = endIndex
        targetEnd = DateAdd("m", 1, targetEnd)
    Next i
    BuildTableView = result
    Exit Function
InvalidTable:
    BuildTableView = CVErr(xlErrValue)
End Function
' ============================================================================
' TRADE VIEW
'
' One row per individual tenor.
' ============================================================================
Private Function BuildTradeView( _
    ByRef curveDates() As Long, _
    ByRef rates() As Double, _
    ByRef onFactors() As Double, _
    ByVal curveCount As Long, _
    ByVal requestedSerial As Long, _
    ByVal IncludeHeaders As Boolean _
) As Variant
    On Error GoTo InvalidTrade
    Dim strategyStart As Date
    Dim scheduledStart As Date
    Dim targetEnd As Date
    Dim startIndex As Long
    Dim endIndex As Long
    Dim effectiveStart As Long
    Dim effectiveEnd As Long
    Dim strategyPosition As Long
    Dim tenorIndex As Long
    Dim tradeCount As Long
    Dim outputRow As Long
    Dim outputOffset As Long
    Dim rateValue As Double
    Dim result() As Variant
    strategyStart = CDate(requestedSerial)
    ' ------------------------------------------------------------------------
    ' First pass: count valid trade rows
    ' ------------------------------------------------------------------------
    strategyPosition = 1
    tradeCount = 0
    Do
        scheduledStart = DateAdd( _
            "m", _
            strategyPosition - 1, _
            strategyStart _
        )
        startIndex = FirstDateGreaterThanOrEqual( _
            curveDates, _
            curveCount, _
            CLng(scheduledStart) _
        )
        ' No remaining curve date exists.
        If startIndex = 0 Then Exit Do
        effectiveStart = curveDates(startIndex)
        For tenorIndex = 1 To 5
            If IncludeTenorAtPosition( _
                strategyPosition, _
                tenorIndex _
            ) Then
                targetEnd = TradeTargetEnd( _
                    scheduledStart, _
                    tenorIndex _
                )
                ' Only return trades whose target maturity is covered
                ' by the available curve history.
                If CLng(targetEnd) <= curveDates(curveCount) Then
                    endIndex = LastDateLessThanOrEqual( _
                        curveDates, _
                        curveCount, _
                        CLng(targetEnd) _
                    )
                    If endIndex > 0 Then
                        effectiveEnd = curveDates(endIndex)
                        ' Equal start and end are allowed for ON when the
                        ' original strategy date was not a curve date.
                        If effectiveEnd >= effectiveStart Then
                            tradeCount = tradeCount + 1
                        End If
                    End If
                End If
            End If
        Next tenorIndex
        strategyPosition = strategyPosition + 1
    Loop
    If tradeCount = 0 Then GoTo InvalidTrade
    ' ------------------------------------------------------------------------
    ' Prepare trade result
    ' ------------------------------------------------------------------------
    If IncludeHeaders Then
        outputOffset = 1
        ReDim result(1 To tradeCount + 1, 1 To 6)
        result(1, 1) = "Strategy Start"
        result(1, 2) = "Tenor"
        result(1, 3) = "Start Date"
        result(1, 4) = "Target End"
        result(1, 5) = "Effective End"
        result(1, 6) = "Rate"
    Else
        outputOffset = 0
        ReDim result(1 To tradeCount, 1 To 6)
    End If
    ' ------------------------------------------------------------------------
    ' Second pass: populate trade rows
    ' ------------------------------------------------------------------------
    strategyPosition = 1
    outputRow = 0
    Do
        scheduledStart = DateAdd( _
            "m", _
            strategyPosition - 1, _
            strategyStart _
        )
        startIndex = FirstDateGreaterThanOrEqual( _
            curveDates, _
            curveCount, _
            CLng(scheduledStart) _
        )
        If startIndex = 0 Then Exit Do
        effectiveStart = curveDates(startIndex)
        For tenorIndex = 1 To 5
            If IncludeTenorAtPosition( _
                strategyPosition, _
                tenorIndex _
            ) Then
                targetEnd = TradeTargetEnd( _
                    scheduledStart, _
                    tenorIndex _
                )
                If CLng(targetEnd) <= curveDates(curveCount) Then
                    endIndex = LastDateLessThanOrEqual( _
                        curveDates, _
                        curveCount, _
                        CLng(targetEnd) _
                    )
                    If endIndex > 0 Then
                        effectiveEnd = curveDates(endIndex)
                        If effectiveEnd >= effectiveStart Then
                            outputRow = outputRow + 1
                            result(outputRow + outputOffset, 1) = _
                                strategyStart
                            result(outputRow + outputOffset, 2) = _
                                TenorName(tenorIndex)
                            result(outputRow + outputOffset, 3) = _
                                CDate(effectiveStart)
                            result(outputRow + outputOffset, 4) = _
                                targetEnd
                            result(outputRow + outputOffset, 5) = _
                                CDate(effectiveEnd)
                            If tenorIndex = 1 Then
                                rateValue = AnnualizedONRate( _
                                    onFactors, _
                                    rates, _
                                    startIndex, _
                                    endIndex, _
                                    effectiveStart, _
                                    effectiveEnd _
                                )
                            Else
                                rateValue = rates( _
                                    startIndex, _
                                    tenorIndex _
                                )
                            End If
                            result(outputRow + outputOffset, 6) = _
                                rateValue
                        End If
                    End If
                End If
            End If
        Next tenorIndex
        strategyPosition = strategyPosition + 1
    Loop
    BuildTradeView = result
    Exit Function
InvalidTrade:
    BuildTradeView = CVErr(xlErrValue)
End Function
' ============================================================================
' Determines whether a tenor is active at a monthly strategy position.
'
' tenorIndex:
'   1 = ON
'   2 = 1M
'   3 = 2M
'   4 = 3M
'   5 = 6M
' ============================================================================
Private Function IncludeTenorAtPosition( _
    ByVal strategyPosition As Long, _
    ByVal tenorIndex As Long _
) As Boolean
    Select Case tenorIndex
        Case 1, 2
            ' ON and 1M every position.
            IncludeTenorAtPosition = True
        Case 3
            ' 2M at positions 1, 3, 5, 7...
            IncludeTenorAtPosition = _
                ((strategyPosition - 1) Mod 2 = 0)
        Case 4
            ' 3M at positions 1, 4, 7, 10...
            IncludeTenorAtPosition = _
                ((strategyPosition - 1) Mod 3 = 0)
        Case 5
            ' 6M at positions 1, 7, 13, 19...
            IncludeTenorAtPosition = _
                ((strategyPosition - 1) Mod 6 = 0)
        Case Else
            IncludeTenorAtPosition = False
    End Select
End Function
' ============================================================================
' Calculates the contractual target maturity from the scheduled strategy date.
'
' The target is intentionally not calculated from the adjusted curve date.
' ============================================================================
Private Function TradeTargetEnd( _
    ByVal scheduledStart As Date, _
    ByVal tenorIndex As Long _
) As Date
    Select Case tenorIndex
        Case 1
            TradeTargetEnd = DateAdd("d", 1, scheduledStart)
        Case 2
            TradeTargetEnd = DateAdd("m", 1, scheduledStart)
        Case 3
            TradeTargetEnd = DateAdd("m", 2, scheduledStart)
        Case 4
            TradeTargetEnd = DateAdd("m", 3, scheduledStart)
        Case 5
            TradeTargetEnd = DateAdd("m", 6, scheduledStart)
        Case Else
            Err.Raise 5
    End Select
End Function
' ============================================================================
' Returns the tenor label.
' ============================================================================
Private Function TenorName( _
    ByVal tenorIndex As Long _
) As String
    Select Case tenorIndex
        Case 1
            TenorName = "ON"
        Case 2
            TenorName = "1M"
        Case 3
            TenorName = "2M"
        Case 4
            TenorName = "3M"
        Case 5
            TenorName = "6M"
        Case Else
            TenorName = vbNullString
    End Select
End Function
' ============================================================================
' Calculates annualized compounded ON return.
'
' When Start Date and Effective End are equal, the source ON rate is returned.
' This can occur when the original strategy date is absent from the curve and
' the next available date is also the latest date on or before the ON target.
' ============================================================================
Private Function AnnualizedONRate( _
    ByRef onFactors() As Double, _
    ByRef rates() As Double, _
    ByVal startIndex As Long, _
    ByVal endIndex As Long, _
    ByVal effectiveStart As Long, _
    ByVal effectiveEnd As Long _
) As Double
    If effectiveEnd > effectiveStart Then
        AnnualizedONRate = _
            ((onFactors(endIndex) / onFactors(startIndex)) - 1#) / _
            (effectiveEnd - effectiveStart) * 360#
    Else
        AnnualizedONRate = rates(startIndex, 1)
    End If
End Function
' ============================================================================
' Returns the first index whose curve date is greater than or equal to
' targetDate.
'
' Returns zero when no valid date exists.
' ============================================================================
Private Function FirstDateGreaterThanOrEqual( _
    ByRef curveDates() As Long, _
    ByVal itemCount As Long, _
    ByVal targetDate As Long _
) As Long
    Dim low As Long
    Dim high As Long
    Dim middle As Long
    Dim answer As Long
    low = 1
    high = itemCount
    answer = 0
    Do While low <= high
        middle = low + (high - low) \ 2
        If curveDates(middle) >= targetDate Then
            answer = middle
            high = middle - 1
        Else
            low = middle + 1
        End If
    Loop
    FirstDateGreaterThanOrEqual = answer
End Function
' ============================================================================
' Returns the last index whose curve date is less than or equal to targetDate.
'
' Returns zero when no valid date exists.
' ============================================================================
Private Function LastDateLessThanOrEqual( _
    ByRef curveDates() As Long, _
    ByVal itemCount As Long, _
    ByVal targetDate As Long _
) As Long
    Dim low As Long
    Dim high As Long
    Dim middle As Long
    Dim answer As Long
    low = 1
    high = itemCount
    answer = 0
    Do While low <= high
        middle = low + (high - low) \ 2
        If curveDates(middle) <= targetDate Then
            answer = middle
            low = middle + 1
        Else
            high = middle - 1
        End If
    Loop
    LastDateLessThanOrEqual = answer
End Function
