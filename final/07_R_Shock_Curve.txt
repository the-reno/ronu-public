Option Explicit

'=====================================================================
' R_SHOCK_CURVE
'
' Excel usage:
'   =R_SHOCK_CURVE(curve_selection, shock_bp, factor)
'
' Examples:
'   =R_SHOCK_CURVE(Curve!A1:E1128, 0, 1)
'       -> original curve exactly
'
'   =R_SHOCK_CURVE(Curve!A1:E1128, 100, 1)
'       -> original curve shifted +100 bp
'
'   =R_SHOCK_CURVE(Curve!A1:E1128, 0, -1)
'       -> historical moves fully inverted around the initial curve
'
'   =R_SHOCK_CURVE(Curve!A1:E1128, 100, -1)
'       -> starting curve shifted +100 bp, then historical moves inverted
'
' INPUTS
' ------
' curve_selection:
'   Header-inclusive range containing Date and one or more rate columns.
'   Example headers: Date | ON | 1M | 3M | 6M
'   Rate values are expected in DECIMAL form.
'   Example: 5.50% = 0.0550.
'
' shock_bp:
'   Parallel shift applied to the INITIAL curve, in basis points.
'   +100 = shift every starting tenor +100 bp.
'   -50  = shift every starting tenor -50 bp.
'   0    = no initial shift.
'
' factor:
'   Multiplier applied to each tenor's historical ABSOLUTE rate move
'   from its own initial value.
'
'   +1.0 = reproduce historical moves exactly
'   +0.5 = same direction, half the historical moves
'    0.0 = no subsequent movement; flat path at shocked initial curve
'   -0.5 = inverted direction, half the historical moves
'   -1.0 = fully inverted historical moves
'   -2.0 = inverted direction, double the historical moves
'
' METHOD
' ------
' For tenor j at time t:
'
'   Anchor_j = first available value for tenor j
'
'   Shocked_j(t) = Anchor_j
'                  + ShockBp / 10000
'                  + Factor * (Historical_j(t) - Anchor_j)
'
' Therefore:
'   - shock_bp controls WHERE the path starts;
'   - factor controls HOW the path moves after the start;
'   - factor +1 and shock 0 reproduces the original curve exactly;
'   - factor -1 mirrors each tenor's historical moves around its
'     shocked starting level;
'   - initial cross-tenor curve shape is preserved under a parallel
'     shock because the same bp shift is applied to every tenor.
'
' Excel 365 / dynamic-array Excel:
'   Enter the formula in one cell and the entire shocked curve spills.
'
' No dependency on any other VBA module.
'=====================================================================

Public Function R_SHOCK_CURVE( _
    ByVal curveSelection As Range, _
    ByVal shockBp As Double, _
    ByVal factor As Double) As Variant

    Dim src As Variant
    Dim result() As Variant

    Dim nRows As Long
    Dim nCols As Long
    Dim dateCol As Long
    Dim anchorRow As Long

    Dim r As Long
    Dim c As Long

    Dim headerText As String
    Dim anchorRate As Double
    Dim histNow As Double
    Dim shockRate As Double

    On Error GoTo fail

    If curveSelection Is Nothing Then
        R_SHOCK_CURVE = CVErr(xlErrRef)
        Exit Function
    End If

    src = curveSelection.Value2

    If Not IsArray(src) Then
        R_SHOCK_CURVE = CVErr(xlErrValue)
        Exit Function
    End If

    nRows = UBound(src, 1)
    nCols = UBound(src, 2)

    If nRows < 2 Or nCols < 2 Then
        R_SHOCK_CURVE = CVErr(xlErrValue)
        Exit Function
    End If

    '-------------------------------------------------------------
    ' Locate Date header.
    '-------------------------------------------------------------
    For c = 1 To nCols
        headerText = UCase$(Trim$(CStr(src(1, c))))
        If headerText = "DATE" Then
            dateCol = c
            Exit For
        End If
    Next c

    If dateCol = 0 Then
        R_SHOCK_CURVE = CVErr(xlErrValue)
        Exit Function
    End If

    '-------------------------------------------------------------
    ' First row with a usable date becomes the common anchor row.
    ' All numeric rate columns are shocked from their value on this row.
    '-------------------------------------------------------------
    For r = 2 To nRows
        If Len(CStr(src(r, dateCol))) > 0 Then
            If IsNumeric(src(r, dateCol)) Or IsDate(src(r, dateCol)) Then
                anchorRow = r
                Exit For
            End If
        End If
    Next r

    If anchorRow = 0 Then
        R_SHOCK_CURVE = CVErr(xlErrValue)
        Exit Function
    End If

    shockRate = shockBp / 10000#

    ReDim result(1 To nRows, 1 To nCols)

    '-------------------------------------------------------------
    ' Preserve headers exactly.
    '-------------------------------------------------------------
    For c = 1 To nCols
        result(1, c) = src(1, c)
    Next c

    '-------------------------------------------------------------
    ' Build shocked curve.
    '
    ' Date column is unchanged.
    ' Any column numeric on the anchor row is treated as a rate column.
    ' Non-rate / metadata columns are copied unchanged.
    ' Blank rate cells remain blank.
    '-------------------------------------------------------------
    For r = 2 To nRows

        For c = 1 To nCols

            If c = dateCol Then

                result(r, c) = src(r, c)

            ElseIf IsNumeric(src(anchorRow, c)) Then

                anchorRate = CDbl(src(anchorRow, c))

                If IsNumeric(src(r, c)) Then
                    histNow = CDbl(src(r, c))

                    result(r, c) = _
                        anchorRate + shockRate + _
                        factor * (histNow - anchorRate)

                ElseIf Len(CStr(src(r, c))) = 0 Then
                    result(r, c) = vbNullString
                Else
                    result(r, c) = CVErr(xlErrValue)
                End If

            Else

                result(r, c) = src(r, c)

            End If

        Next c

    Next r

    R_SHOCK_CURVE = result
    Exit Function

fail:
    R_SHOCK_CURVE = CVErr(xlErrValue)

End Function
