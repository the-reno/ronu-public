Option Explicit

'=====================================================================
' SIMPLE POWERPOINT EXPORT FOR THE FINAL CASH TENOR ANALYSIS
'
' PURPOSE
'   Builds a five-slide PowerPoint from the Excel sheets created by
'   FINAL_BuildAll. Existing Excel charts are copied as portable
'   enhanced-metafile pictures and each sheet's dynamic conclusion in
'   cell A2 is added as the visible risk comment and as speaker notes.
'
' REQUIREMENTS
'   - Excel for Windows.
'   - Microsoft PowerPoint installed.
'   - The FINAL VBA package in this repository.
'   - No PowerPoint reference is required; the code uses late binding.
'
' USE
'   1. Paste this file into a standard VBA module.
'   2. Run FINAL_BuildAll at least once, or let this macro run it.
'   3. Run FINAL_CreatePowerPoint.
'
' OUTPUT
'   Cash_Tenor_Risk_Committee.pptx in the workbook folder.
'   If that file already exists, a timestamp is added.
'=====================================================================

Private Const PPT_LAYOUT_BLANK As Long = 12
Private Const PPT_SAVE_OPENXML As Long = 24
Private Const PPT_PASTE_ENHANCED_METAFILE As Long = 2

Private Const MSO_FALSE As Long = 0
Private Const MSO_TRUE As Long = -1
Private Const MSO_SHAPE_RECTANGLE As Long = 1
Private Const MSO_TEXT_HORIZONTAL As Long = 1

Private Const SLIDE_W As Double = 960
Private Const SLIDE_H As Double = 540

Public Sub FINAL_CreatePowerPoint()
    FINAL_CreatePowerPointTo vbNullString
End Sub

Public Sub FINAL_CreatePowerPointTo(ByVal outputPath As String)

    Dim pptApp As Object
    Dim pres As Object
    Dim sld As Object
    Dim ws As Worksheet
    Dim slideSheets As Variant
    Dim sheetName As Variant
    Dim slideIndex As Long
    Dim savePath As String
    Dim oldScreen As Boolean

    On Error GoTo fail

    oldScreen = Application.ScreenUpdating
    Application.ScreenUpdating = False
    Application.StatusBar = "Creating PowerPoint presentation..."

    slideSheets = Array( _
        "S1_Historical", _
        "S2_Direction", _
        "S3_Regret", _
        "S4_Robust", _
        "S5_Execution")

    If Not FINAL_PPT_AllSheetsExist(slideSheets) Then
        On Error Resume Next
        Application.Run "FINAL_BuildAll"
        On Error GoTo fail

        If Not FINAL_PPT_AllSheetsExist(slideSheets) Then
            Err.Raise vbObjectError + 700, , _
                "The presentation sheets were not found. Import the FINAL modules, " & _
                "compile the VBA project, and run FINAL_BuildAll."
        End If
    End If

    Set pptApp = CreateObject("PowerPoint.Application")
    pptApp.Visible = True

    Set pres = pptApp.Presentations.Add
    pres.PageSetup.SlideWidth = SLIDE_W
    pres.PageSetup.SlideHeight = SLIDE_H

    slideIndex = 0

    For Each sheetName In slideSheets
        slideIndex = slideIndex + 1
        Set ws = ThisWorkbook.Worksheets(CStr(sheetName))
        Set sld = pres.Slides.Add(slideIndex, PPT_LAYOUT_BLANK)

        FINAL_PPT_BuildSlide sld, ws, slideIndex
    Next sheetName

    savePath = FINAL_PPT_OutputPath(outputPath)
    pres.SaveAs savePath, PPT_SAVE_OPENXML

    Application.StatusBar = False
    Application.ScreenUpdating = oldScreen

    MsgBox "PowerPoint created:" & vbCrLf & savePath, vbInformation
    Exit Sub

fail:
    Application.StatusBar = False
    Application.ScreenUpdating = oldScreen

    On Error Resume Next
    If Not pres Is Nothing Then pres.Close
    If Not pptApp Is Nothing Then pptApp.Quit
    On Error GoTo 0

    MsgBox "FINAL_CreatePowerPoint stopped: " & Err.Description, vbCritical

End Sub

Private Sub FINAL_PPT_BuildSlide( _
    ByVal sld As Object, _
    ByVal ws As Worksheet, _
    ByVal slideIndex As Long)

    Dim titleText As String
    Dim commentText As String
    Dim chartCount As Long
    Dim i As Long
    Dim co As ChartObject
    Dim pasted As Object
    Dim cellLeft As Double
    Dim cellTop As Double
    Dim cellWidth As Double
    Dim cellHeight As Double

    titleText = Trim$(CStr(ws.Range("A1").Value))
    If Len(titleText) = 0 Then titleText = FINAL_PPT_DefaultTitle(ws.Name)

    commentText = Trim$(CStr(ws.Range("A2").Value))
    If Len(commentText) = 0 Then commentText = FINAL_PPT_DefaultComment(ws.Name)

    FINAL_PPT_AddHeader sld, titleText, slideIndex

    chartCount = ws.ChartObjects.Count

    If chartCount = 0 Then
        FINAL_PPT_CopyCanvas sld, ws
    Else
        For i = 1 To chartCount
            Set co = ws.ChartObjects(i)

            FINAL_PPT_ChartCell chartCount, i, _
                cellLeft, cellTop, cellWidth, cellHeight

            co.Chart.CopyPicture xlScreen, xlPicture
            DoEvents

            Set pasted = sld.Shapes.PasteSpecial(PPT_PASTE_ENHANCED_METAFILE)
            FINAL_PPT_FitShape pasted(1), _
                cellLeft, cellTop, cellWidth, cellHeight
        Next i
    End If

    FINAL_PPT_AddComment sld, commentText
    FINAL_PPT_AddFooter sld, ws.Name
    FINAL_PPT_SetNotes sld, commentText

End Sub

Private Sub FINAL_PPT_AddHeader( _
    ByVal sld As Object, _
    ByVal titleText As String, _
    ByVal slideIndex As Long)

    Dim shp As Object

    Set shp = sld.Shapes.AddShape(MSO_SHAPE_RECTANGLE, 48, 20, 100, 4)
    shp.Fill.ForeColor.RGB = RGB(23, 54, 93)
    shp.Line.Visible = MSO_FALSE

    Set shp = sld.Shapes.AddTextbox(MSO_TEXT_HORIZONTAL, 48, 30, 820, 44)
    With shp.TextFrame.TextRange
        .Text = titleText
        .Font.Name = "Arial"
        .Font.Size = 25
        .Font.Bold = MSO_TRUE
        .Font.Color.RGB = RGB(37, 42, 49)
    End With
    shp.TextFrame.WordWrap = MSO_TRUE

    Set shp = sld.Shapes.AddTextbox(MSO_TEXT_HORIZONTAL, 888, 34, 24, 26)
    With shp.TextFrame.TextRange
        .Text = CStr(slideIndex)
        .Font.Name = "Arial"
        .Font.Size = 13
        .Font.Bold = MSO_TRUE
        .Font.Color.RGB = RGB(115, 123, 133)
        .ParagraphFormat.Alignment = 3
    End With

    Set shp = sld.Shapes.AddTextbox(MSO_TEXT_HORIZONTAL, 48, 76, 864, 22)
    With shp.TextFrame.TextRange
        .Text = "USD CASH ALLOCATION | RISK COMMITTEE"
        .Font.Name = "Arial"
        .Font.Size = 10
        .Font.Bold = MSO_TRUE
        .Font.Color.RGB = RGB(115, 123, 133)
    End With

End Sub

Private Sub FINAL_PPT_ChartCell( _
    ByVal chartCount As Long, _
    ByVal chartIndex As Long, _
    ByRef cellLeft As Double, _
    ByRef cellTop As Double, _
    ByRef cellWidth As Double, _
    ByRef cellHeight As Double)

    Const left0 As Double = 48
    Const top0 As Double = 104
    Const totalW As Double = 864
    Const totalH As Double = 330
    Const gap As Double = 16

    If chartCount = 1 Then
        cellLeft = left0
        cellTop = top0
        cellWidth = totalW
        cellHeight = totalH

    ElseIf chartCount = 2 Then
        cellWidth = (totalW - gap) / 2
        cellHeight = totalH
        cellTop = top0
        cellLeft = left0 + (chartIndex - 1) * (cellWidth + gap)

    ElseIf chartCount = 3 Then
        If chartIndex <= 2 Then
            cellWidth = (totalW - gap) / 2
            cellHeight = (totalH - gap) / 2
            cellTop = top0
            cellLeft = left0 + (chartIndex - 1) * (cellWidth + gap)
        Else
            cellLeft = left0
            cellTop = top0 + (totalH + gap) / 2
            cellWidth = totalW
            cellHeight = (totalH - gap) / 2
        End If

    Else
        cellWidth = (totalW - gap) / 2
        cellHeight = (totalH - gap) / 2
        cellLeft = left0 + ((chartIndex - 1) Mod 2) * (cellWidth + gap)
        cellTop = top0 + Int((chartIndex - 1) / 2) * (cellHeight + gap)
    End If

End Sub

Private Sub FINAL_PPT_FitShape( _
    ByVal shp As Object, _
    ByVal cellLeft As Double, _
    ByVal cellTop As Double, _
    ByVal cellWidth As Double, _
    ByVal cellHeight As Double)

    Dim scaleFactor As Double

    shp.LockAspectRatio = MSO_TRUE

    scaleFactor = cellWidth / shp.Width
    If shp.Height * scaleFactor > cellHeight Then
        scaleFactor = cellHeight / shp.Height
    End If

    shp.Width = shp.Width * scaleFactor
    shp.Height = shp.Height * scaleFactor
    shp.Left = cellLeft + (cellWidth - shp.Width) / 2
    shp.Top = cellTop + (cellHeight - shp.Height) / 2

End Sub

Private Sub FINAL_PPT_CopyCanvas(ByVal sld As Object, ByVal ws As Worksheet)

    Dim pasted As Object

    ws.Range("A1:O18").CopyPicture xlScreen, xlPicture
    DoEvents

    Set pasted = sld.Shapes.PasteSpecial(PPT_PASTE_ENHANCED_METAFILE)
    FINAL_PPT_FitShape pasted(1), 48, 104, 864, 330

End Sub

Private Sub FINAL_PPT_AddComment(ByVal sld As Object, ByVal commentText As String)

    Dim box As Object
    Dim label As Object
    Dim body As Object

    Set box = sld.Shapes.AddShape(MSO_SHAPE_RECTANGLE, 48, 447, 864, 62)
    box.Fill.ForeColor.RGB = RGB(23, 54, 93)
    box.Line.Visible = MSO_FALSE

    Set label = sld.Shapes.AddTextbox(MSO_TEXT_HORIZONTAL, 64, 454, 118, 18)
    With label.TextFrame.TextRange
        .Text = "RISK COMMENT"
        .Font.Name = "Arial"
        .Font.Size = 10
        .Font.Bold = MSO_TRUE
        .Font.Color.RGB = RGB(145, 212, 196)
    End With

    Set body = sld.Shapes.AddTextbox(MSO_TEXT_HORIZONTAL, 190, 452, 704, 48)
    With body.TextFrame.TextRange
        .Text = commentText
        .Font.Name = "Arial"
        .Font.Size = 14
        .Font.Bold = MSO_TRUE
        .Font.Color.RGB = RGB(255, 255, 255)
    End With
    body.TextFrame.WordWrap = MSO_TRUE

    On Error Resume Next
    body.TextFrame2.AutoSize = 2
    On Error GoTo 0

End Sub

Private Sub FINAL_PPT_AddFooter(ByVal sld As Object, ByVal sourceSheet As String)

    Dim shp As Object

    Set shp = sld.Shapes.AddShape(MSO_SHAPE_RECTANGLE, 48, 518, 864, 1)
    shp.Fill.ForeColor.RGB = RGB(229, 232, 236)
    shp.Line.Visible = MSO_FALSE

    Set shp = sld.Shapes.AddTextbox(MSO_TEXT_HORIZONTAL, 48, 521, 864, 14)
    With shp.TextFrame.TextRange
        .Text = "Source: " & ThisWorkbook.Name & " | " & sourceSheet
        .Font.Name = "Arial"
        .Font.Size = 8
        .Font.Color.RGB = RGB(115, 123, 133)
        .ParagraphFormat.Alignment = 3
    End With

End Sub

Private Sub FINAL_PPT_SetNotes(ByVal sld As Object, ByVal noteText As String)

    Dim shp As Object

    On Error Resume Next

    For Each shp In sld.NotesPage.Shapes
        If shp.Type = 14 Then
            If shp.PlaceholderFormat.Type = 2 Then
                shp.TextFrame.TextRange.Text = noteText
                Exit For
            End If
        End If
    Next shp

    On Error GoTo 0

End Sub

Private Function FINAL_PPT_AllSheetsExist(ByVal sheetNames As Variant) As Boolean

    Dim item As Variant

    For Each item In sheetNames
        If Not FINAL_PPT_SheetExists(CStr(item)) Then Exit Function
    Next item

    FINAL_PPT_AllSheetsExist = True

End Function

Private Function FINAL_PPT_SheetExists(ByVal sheetName As String) As Boolean

    Dim ws As Worksheet

    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(sheetName)
    FINAL_PPT_SheetExists = Not ws Is Nothing
    On Error GoTo 0

End Function

Private Function FINAL_PPT_DefaultTitle(ByVal sheetName As String) As String

    Select Case sheetName
        Case "S1_Historical"
            FINAL_PPT_DefaultTitle = "Historical evidence"
        Case "S2_Direction"
            FINAL_PPT_DefaultTitle = "Direction changes the tenor winner"
        Case "S3_Regret"
            FINAL_PPT_DefaultTitle = "Manage the cost of being wrong"
        Case "S4_Robust"
            FINAL_PPT_DefaultTitle = "Use a robust allocation reference"
        Case "S5_Execution"
            FINAL_PPT_DefaultTitle = "Price, allocate and review"
        Case Else
            FINAL_PPT_DefaultTitle = sheetName
    End Select

End Function

Private Function FINAL_PPT_DefaultComment(ByVal sheetName As String) As String

    Select Case sheetName
        Case "S1_Historical"
            FINAL_PPT_DefaultComment = _
                "Historical return is path-dependent; do not overstate a narrow winner."
        Case "S2_Direction"
            FINAL_PPT_DefaultComment = _
                "The preferred tenor reverses when the direction of rates reverses."
        Case "S3_Regret"
            FINAL_PPT_DefaultComment = _
                "The relevant risk is the return cost of choosing the wrong tenor."
        Case "S4_Robust"
            FINAL_PPT_DefaultComment = _
                "Use the robust mix as a risk reference, not as a rate forecast."
        Case "S5_Execution"
            FINAL_PPT_DefaultComment = _
                "Execute consistently: price the alternatives, allocate and review."
        Case Else
            FINAL_PPT_DefaultComment = _
                "Review the result in the context of liquidity and rate-direction risk."
    End Select

End Function

Private Function FINAL_PPT_OutputPath(ByVal requestedPath As String) As String

    Dim folder As String
    Dim candidate As String

    If Len(Trim$(requestedPath)) > 0 Then
        FINAL_PPT_OutputPath = requestedPath
        Exit Function
    End If

    folder = ThisWorkbook.Path
    If Len(folder) = 0 Then folder = Application.DefaultFilePath

    candidate = folder & Application.PathSeparator & _
        "Cash_Tenor_Risk_Committee.pptx"

    If Len(Dir$(candidate)) > 0 Then
        candidate = folder & Application.PathSeparator & _
            "Cash_Tenor_Risk_Committee_" & _
            Format$(Now, "yyyymmdd_hhnnss") & ".pptx"
    End If

    FINAL_PPT_OutputPath = candidate

End Function
