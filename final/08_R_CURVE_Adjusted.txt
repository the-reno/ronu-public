Option Explicit

'=======================================================================
' RATES ENGINE - R_CURVE (FINAL v3.2)
'
' Preferred syntax:
'   =R_CURVE(start, table, tenor, adj, roll, [type], [n], [maturity])
'
' Examples:
'   =R_CURVE(B2,RateTable,"6M",0,FALSE)
'       One 6M investment only.
'
'   =R_CURVE(B2,RateTable,"6M",0,FALSE,"FACTOR",0)-1
'       Return of one 6M investment.
'
'   =R_CURVE(B2,RateTable,"1M",0,TRUE)
'       Roll 1M repeatedly through the curve.
'
'   =R_CURVE(B2,RateTable,"1M",0,TRUE,"FACTOR",0,C2)-1
'       Roll 1M until explicit maturity C2.
'
' Legacy syntax is retained:
'   =R_CURVE(B2,RateTable,"1M",0,"FACTOR",0,C2)-1
'   In legacy form Roll defaults to TRUE.
'
' start:
'   Contractual anchor date. If unavailable, START moves FORWARD to the
'   first available curve date, but future roll targets remain anchored
'   to the ORIGINAL requested start.
'
' adj:
'   1 = snap forward to next available curve date
'   0 = snap backward to previous available curve date
'
' roll:
'   FALSE = one investment only. Maturity = requested start + one tenor.
'   TRUE  = reinvest/roll the selected tenor.
'
' type:
'   ALL (default), DATE, RATE, DAYS, FACTOR, HELP
'
' n:
'   omitted = all rows; 1 = START; 0 = MATURITY; -1 = prior row
'
' maturity:
'   optional explicit maturity when Roll=TRUE.
'
' ACT/360 simple accrual by period:
'   Factor(next) = Factor * (1 + Rate * Days / 360)
'=======================================================================

Public Function R_CURVE(ByVal start As Variant, ByVal table As Range, _
                        ByVal tenor As Variant, ByVal adj As Variant, _
                        Optional ByVal roll As Variant, _
                        Optional ByVal outType As Variant, _
                        Optional ByVal n As Variant, _
                        Optional ByVal maturity As Variant) As Variant

    Dim v As Variant
    Dim nR As Long, nC As Long, nD As Long
    Dim d() As Double
    Dim col As Long

    Dim f As String, q As Long
    Dim s As Double
    Dim t As String
    Dim doRoll As Boolean

    Dim hasN As Boolean, nArg As Variant
    Dim hasMat As Boolean, matArg As Variant

    Dim startIdx As Long
    Dim effectiveStart As Double

    Dim sched() As Long
    Dim cnt As Long
    Dim matIdx As Long
    Dim matS As Double

    Dim m As Long
    Dim dS() As Double
    Dim rt() As Variant
    Dim dy() As Variant
    Dim fc() As Variant

    Dim g As Double
    Dim bad As Boolean
    Dim rateValue As Variant

    Dim i As Long, k As Long
    Dim tgt As Double
    Dim out() As Variant
    Dim j As Long

    '===================================================================
    ' OPTIONAL ARGUMENT PARSER
    '
    ' New:
    '   roll, type, n, maturity
    '
    ' Legacy:
    '   type, n, maturity
    '   If fifth argument is an output-type string, Roll = TRUE.
    '===================================================================
    doRoll = True
    t = "ALL"
    hasN = False
    hasMat = False

    If IsMissing(roll) Then
        doRoll = True

    ElseIf IsRollValue(roll) Then

        doRoll = RollValue(roll)

        If Not IsMissing(outType) Then
            If Not IsEmpty(outType) And Len(Trim$(CStr(outType))) > 0 Then
                t = UCase$(Trim$(CStr(outType)))
            End If
        End If

        If Not IsMissing(n) Then
            If Not IsEmpty(n) Then
                hasN = True
                nArg = n
            End If
        End If

        If Not IsMissing(maturity) Then
            If Not IsEmpty(maturity) Then
                hasMat = True
                matArg = maturity
            End If
        End If

    ElseIf IsOutputType(roll) Then

        ' Legacy form: fifth argument is Type, Roll defaults TRUE.
        doRoll = True
        t = UCase$(Trim$(CStr(roll)))

        If Not IsMissing(outType) Then
            If Not IsEmpty(outType) Then
                hasN = True
                nArg = outType
            End If
        End If

        If Not IsMissing(n) Then
            If Not IsEmpty(n) Then
                hasMat = True
                matArg = n
            End If
        End If

        ' An eighth argument is not valid in legacy form.
        If Not IsMissing(maturity) Then
            If Not IsEmpty(maturity) Then
                R_CURVE = "#ARGS"
                Exit Function
            End If
        End If

    Else
        R_CURVE = "#ROLL"
        Exit Function
    End If

    If t = "HELP" Then
        R_CURVE = HelpBlock()
        Exit Function
    End If

    If Not IsOutputType(t) Then
        R_CURVE = "#TYPE"
        Exit Function
    End If

    ' ---- scalar inputs ------------------------------------------------
    s = ToSerial(start)
    If s < 0 Then
        R_CURVE = "#START"
        Exit Function
    End If

    If Not ParseTenor(tenor, f, q) Then
        R_CURVE = "#TENOR"
        Exit Function
    End If

    If Not IsNumeric(adj) Then
        R_CURVE = "#ADJ"
        Exit Function
    End If

    If CDbl(adj) <> 0 And CDbl(adj) <> 1 Then
        R_CURVE = "#ADJ"
        Exit Function
    End If

    ' ---- read and validate table -------------------------------------
    v = table.Value2

    If Not IsArray(v) Then
        R_CURVE = "#TABLE"
        Exit Function
    End If

    nR = UBound(v, 1)
    nC = UBound(v, 2)

    If nR < 2 Or nC < 2 Then
        R_CURVE = "#TABLE"
        Exit Function
    End If

    nD = nR - 1
    ReDim d(1 To nD)

    For i = 2 To nR

        If IsEmpty(v(i, 1)) Or Not IsNumeric(v(i, 1)) Then
            R_CURVE = "#TABLE"
            Exit Function
        End If

        d(i - 1) = CDbl(v(i, 1))

        If d(i - 1) <= 0 Then
            R_CURVE = "#TABLE"
            Exit Function
        End If

        If i > 2 Then
            If d(i - 1) <= d(i - 2) Then
                R_CURVE = "#ORDER"
                Exit Function
            End If
        End If

    Next i

    If s > d(nD) Then
        R_CURVE = "#NODATE"
        Exit Function
    End If

    ' Requested start may be missing or before the first curve date.
    ' START moves forward; contractual anchor s is retained.
    startIdx = FirstOnOrAfter(d, nD, s)

    If startIdx = 0 Then
        R_CURVE = "#NODATE"
        Exit Function
    End If

    effectiveStart = d(startIdx)

    ' ---- tenor column -------------------------------------------------
    col = 0

    For i = 2 To nC
        If VarType(v(1, i)) = vbString Then
            If UCase$(Trim$(CStr(v(1, i)))) = UCase$(Trim$(CStr(tenor))) Then
                col = i
                Exit For
            End If
        End If
    Next i

    If col = 0 Then
        R_CURVE = "#TENOR"
        Exit Function
    End If

    ReDim sched(1 To nD + 1)

    '===================================================================
    ' ROLL = FALSE
    ' One investment only: effective START -> requested start + tenor.
    '===================================================================
    If Not doRoll Then

        If hasMat Then
            R_CURVE = "#MAT"
            Exit Function
        End If

        tgt = Target(s, f, q)

        If tgt <= d(nD) And tgt >= d(1) Then

            matIdx = Snap(d, nD, tgt, CDbl(adj) = 1)

            If matIdx = 0 Then
                R_CURVE = "#MAT"
                Exit Function
            End If

            matS = d(matIdx)

        ElseIf tgt > d(nD) Then

            ' Future fixing is unnecessary because the rate is fixed at START.
            matS = tgt

        Else
            R_CURVE = "#MAT"
            Exit Function
        End If

        If matS <= effectiveStart Then
            R_CURVE = "#MAT"
            Exit Function
        End If

        m = 2
        ReDim dS(1 To m)
        ReDim rt(1 To m)
        ReDim dy(1 To m)
        ReDim fc(1 To m)

        dS(1) = effectiveStart
        dS(2) = matS
        fc(1) = 1#

        rateValue = v(startIdx + 1, col)
        dy(1) = matS - effectiveStart

        If IsEmpty(rateValue) Or Not IsNumeric(rateValue) Then
            rt(1) = "#RATE"
            fc(2) = "#RATE"
        Else
            rt(1) = CDbl(rateValue)
            fc(2) = 1# + CDbl(rateValue) * CDbl(dy(1)) / 360#
        End If

        dy(2) = 0
        rt(2) = vbNullString

        ' Informational maturity fixing only if exact curve date exists.
        If matS <= d(nD) Then
            matIdx = ExactDateIdx(d, nD, matS)
            If matIdx > 0 Then
                If IsNumeric(v(matIdx + 1, col)) Then
                    rt(2) = CDbl(v(matIdx + 1, col))
                End If
            End If
        End If

    '===================================================================
    ' ROLL = TRUE, NO EXPLICIT MATURITY
    ' Roll through curve then one final full tenor.
    '===================================================================
    ElseIf Not hasMat Then

        cnt = 1
        sched(1) = startIdx
        k = 1

        Do

            tgt = Target(s, f, k * q)
            If tgt > d(nD) Then Exit Do

            i = Snap(d, nD, tgt, CDbl(adj) = 1)

            If i >= startIdx Then
                If i > sched(cnt) Then
                    cnt = cnt + 1
                    sched(cnt) = i
                End If
            End If

            k = k + 1

        Loop

        matS = Target(d(sched(cnt)), f, q)
        m = cnt + 1

        ReDim dS(1 To m)
        ReDim rt(1 To m)
        ReDim dy(1 To m)
        ReDim fc(1 To m)

        g = 1#
        bad = False
        fc(1) = 1#

        For i = 1 To cnt

            dS(i) = d(sched(i))
            rateValue = v(sched(i) + 1, col)

            If IsEmpty(rateValue) Or Not IsNumeric(rateValue) Then
                rt(i) = "#RATE"
                bad = True
            Else
                rt(i) = CDbl(rateValue)
            End If

            If i < cnt Then
                dy(i) = d(sched(i + 1)) - d(sched(i))
            Else
                dy(i) = matS - d(sched(i))
            End If

            If bad Then
                fc(i + 1) = "#RATE"
            Else
                g = g * (1# + CDbl(rateValue) * CDbl(dy(i)) / 360#)
                fc(i + 1) = g
            End If

        Next i

        dS(m) = matS
        rt(m) = vbNullString
        dy(m) = 0

        i = ExactDateIdx(d, nD, matS)
        If i > 0 Then
            If IsNumeric(v(i + 1, col)) Then
                rt(m) = CDbl(v(i + 1, col))
            End If
        End If

    '===================================================================
    ' ROLL = TRUE + EXPLICIT MATURITY
    '===================================================================
    Else

        matS = ToSerial(matArg)

        If matS < 0 Then
            R_CURVE = "#MAT"
            Exit Function
        End If

        If matS > d(nD) Then
            R_CURVE = "#MAT"
            Exit Function
        End If

        matIdx = Snap(d, nD, matS, CDbl(adj) = 1)

        If matIdx = 0 Then
            R_CURVE = "#MAT"
            Exit Function
        End If

        matS = d(matIdx)

        If matS <= effectiveStart Then
            R_CURVE = "#MAT"
            Exit Function
        End If

        cnt = 1
        sched(1) = startIdx
        k = 1

        Do

            tgt = Target(s, f, k * q)
            If tgt > matS Then Exit Do

            i = Snap(d, nD, tgt, CDbl(adj) = 1)

            If i >= startIdx And i <= matIdx Then
                If i > sched(cnt) Then
                    cnt = cnt + 1
                    sched(cnt) = i
                End If
            End If

            k = k + 1

        Loop

        If sched(cnt) < matIdx Then
            cnt = cnt + 1
            sched(cnt) = matIdx
        End If

        If cnt < 2 Then
            R_CURVE = "#NODATE"
            Exit Function
        End If

        m = cnt
        ReDim dS(1 To m)
        ReDim rt(1 To m)
        ReDim dy(1 To m)
        ReDim fc(1 To m)

        g = 1#
        bad = False
        fc(1) = 1#

        For i = 1 To m

            dS(i) = d(sched(i))
            rateValue = v(sched(i) + 1, col)

            If IsEmpty(rateValue) Or Not IsNumeric(rateValue) Then
                rt(i) = "#RATE"
                If i < m Then bad = True
            Else
                rt(i) = CDbl(rateValue)
            End If

            If i < m Then

                dy(i) = d(sched(i + 1)) - d(sched(i))

                If bad Then
                    fc(i + 1) = "#RATE"
                Else
                    g = g * (1# + CDbl(rateValue) * CDbl(dy(i)) / 360#)
                    fc(i + 1) = g
                End If

            Else
                dy(i) = 0
            End If

        Next i

    End If

    '===================================================================
    ' RETURN
    '===================================================================
    If Not hasN Then

        If t = "ALL" Then

            ReDim out(0 To m, 1 To 4)
            out(0, 1) = "Date"
            out(0, 2) = "Rate"
            out(0, 3) = "Days"
            out(0, 4) = "Factor"

            For i = 1 To m
                out(i, 1) = CDate(dS(i))
                out(i, 2) = rt(i)
                out(i, 3) = dy(i)
                out(i, 4) = fc(i)
            Next i

            R_CURVE = out

        Else

            ReDim out(1 To m, 1 To 1)

            For i = 1 To m
                out(i, 1) = OnePick(t, dS(i), rt(i), dy(i), fc(i))
            Next i

            R_CURVE = out

        End If

    Else

        j = PickIdx(nArg, m)

        If j = 0 Then
            R_CURVE = "#N"
            Exit Function
        End If

        If t = "ALL" Then

            ReDim out(1 To 1, 1 To 4)
            out(1, 1) = CDate(dS(j))
            out(1, 2) = rt(j)
            out(1, 3) = dy(j)
            out(1, 4) = fc(j)
            R_CURVE = out

        Else
            R_CURVE = OnePick(t, dS(j), rt(j), dy(j), fc(j))
        End If

    End If

End Function

'======================================================================
' FUNCTION WIZARD REGISTRATION
'======================================================================
Public Sub R_CURVE_Register()

    On Error GoTo fail

    If Not ThisWorkbook Is ActiveWorkbook Then ThisWorkbook.Activate

    Application.MacroOptions Macro:="R_CURVE", _
        Description:="Rate curve engine. Roll=FALSE is one investment; Roll=TRUE reinvests the tenor. ACT/360.", _
        Category:="Rates Engine", _
        ArgumentDescriptions:=Array( _
            "Requested start date. If unavailable, START moves forward while roll dates stay anchored to the requested date.", _
            "Header-inclusive curve range: Date, ON, 1M, 3M, 6M, ...", _
            "Tenor/rate column: ON, 1D, 1M, 3M, 6M, 2Y, ...", _
            "Date adjustment: 1 = next available date; 0 = previous available date.", _
            "TRUE = roll/reinvest tenor; FALSE = one investment only. Optional; default TRUE.", _
            "Output: ALL, DATE, RATE, DAYS, FACTOR, HELP.", _
            "Row picker: omit=all; 1=START; 0=MATURITY; -1=prior row.", _
            "Optional explicit maturity when Roll=TRUE.")

    MsgBox "R_CURVE help registered.", vbInformation
    Exit Sub

fail:
    MsgBox "Registration FAILED: " & Err.Number & " - " & Err.Description, vbExclamation

End Sub

Public Sub Auto_Open()

    On Error Resume Next

    If Not ThisWorkbook Is ActiveWorkbook Then ThisWorkbook.Activate

    Application.MacroOptions Macro:="R_CURVE", _
        Description:="Rate curve engine. Roll=FALSE is one investment; Roll=TRUE reinvests the tenor."

    On Error GoTo 0

End Sub

'======================================================================
' HELPERS
'======================================================================
Private Function IsRollValue(ByVal x As Variant) As Boolean

    IsRollValue = False

    If IsError(x) Or IsArray(x) Or IsObject(x) Then Exit Function

    If VarType(x) = vbBoolean Then
        IsRollValue = True
    ElseIf IsNumeric(x) Then
        If CDbl(x) = 0# Or CDbl(x) = 1# Then IsRollValue = True
    End If

End Function

Private Function RollValue(ByVal x As Variant) As Boolean

    If VarType(x) = vbBoolean Then
        RollValue = CBool(x)
    Else
        RollValue = (CDbl(x) <> 0#)
    End If

End Function

Private Function IsOutputType(ByVal x As Variant) As Boolean

    Dim txt As String

    IsOutputType = False

    If IsError(x) Or IsArray(x) Or IsObject(x) Then Exit Function

    txt = UCase$(Trim$(CStr(x)))

    Select Case txt
        Case "ALL", "DATE", "RATE", "DAYS", "FACTOR", "HELP"
            IsOutputType = True
    End Select

End Function

Private Function OnePick(ByVal t As String, ByVal dateSerial As Double, _
                         ByVal rate As Variant, ByVal dd As Variant, _
                         ByVal fact As Variant) As Variant

    Select Case t
        Case "DATE":   OnePick = CDate(dateSerial)
        Case "RATE":   OnePick = rate
        Case "DAYS":   OnePick = dd
        Case "FACTOR": OnePick = fact
    End Select

End Function

Private Function ToSerial(ByVal x As Variant) As Double

    On Error GoTo bad

    If IsObject(x) Then x = x.Value
    If IsError(x) Then GoTo bad
    If IsArray(x) Then GoTo bad

    If IsNumeric(x) Then
        If CDbl(x) > 0# Then
            ToSerial = CDbl(x)
        Else
            ToSerial = -1#
        End If
    ElseIf IsDate(x) Then
        ToSerial = CDbl(CDate(x))
    Else
        ToSerial = -1#
    End If

    Exit Function

bad:
    ToSerial = -1#

End Function

Private Function ParseTenor(ByVal tenor As Variant, ByRef f As String, ByRef q As Long) As Boolean

    Dim tenorText As String
    Dim num As String

    ParseTenor = False

    If IsError(tenor) Then Exit Function

    tenorText = UCase$(Trim$(CStr(tenor)))

    If tenorText = "ON" Then
        f = "D"
        q = 1
        ParseTenor = True
        Exit Function
    End If

    If Len(tenorText) < 2 Then Exit Function

    f = Right$(tenorText, 1)

    If f <> "D" And f <> "M" And f <> "Y" Then Exit Function

    num = Left$(tenorText, Len(tenorText) - 1)

    If num Like "*[!0-9]*" Then Exit Function

    q = CLng(num)

    If q < 1 Then Exit Function

    ParseTenor = True

End Function

Private Function Target(ByVal startSerial As Double, ByVal f As String, _
                        ByVal periods As Long) As Double

    Select Case f
        Case "D"
            Target = startSerial + periods
        Case "M"
            Target = CDbl(DateAdd("m", periods, CDate(startSerial)))
        Case "Y"
            Target = CDbl(DateAdd("yyyy", periods, CDate(startSerial)))
    End Select

End Function

Private Function FirstOnOrAfter(d() As Double, ByVal nD As Long, _
                                ByVal tgt As Double) As Long

    Dim i As Long

    FirstOnOrAfter = 0

    For i = 1 To nD
        If d(i) >= tgt Then
            FirstOnOrAfter = i
            Exit Function
        End If
    Next i

End Function

Private Function Snap(d() As Double, ByVal nD As Long, ByVal tgt As Double, _
                      ByVal snapNext As Boolean) As Long

    Dim i As Long

    Snap = 0

    If tgt < d(1) Or tgt > d(nD) Then Exit Function

    If snapNext Then

        For i = 1 To nD
            If d(i) >= tgt Then
                Snap = i
                Exit Function
            End If
        Next i

    Else

        For i = nD To 1 Step -1
            If d(i) <= tgt Then
                Snap = i
                Exit Function
            End If
        Next i

    End If

End Function

Private Function ExactDateIdx(d() As Double, ByVal nD As Long, _
                              ByVal tgt As Double) As Long

    Dim i As Long

    ExactDateIdx = 0

    For i = 1 To nD
        If d(i) = tgt Then
            ExactDateIdx = i
            Exit Function
        End If
    Next i

End Function

Private Function PickIdx(ByVal n As Variant, ByVal m As Long) As Long

    Dim p As Long

    PickIdx = 0

    If VarType(n) = vbBoolean Then Exit Function
    If Not IsNumeric(n) Then Exit Function
    If CDbl(n) <> Int(CDbl(n)) Then Exit Function

    p = CLng(n)

    If p >= 1 Then
        If p <= m Then PickIdx = p
    Else
        If m + p >= 1 Then PickIdx = m + p
    End If

End Function
