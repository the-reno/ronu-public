Option Explicit

'=====================================================================
' SLIDE 1 - HISTORICAL EVIDENCE
'
' Message:
'   Start with what actually happened.
'   Short tenors can reprice as rates move; longer tenors lock earlier.
'
' Presentation choice:
'   1) Show the realized ON rate path.
'   2) Compare tenor performance as EXCESS RETURN VS ON in bp.
'      The relative chart makes economically small differences visible
'      and avoids overstating a narrow historical winner.
'=====================================================================

Public Sub FINAL_Slide1_Historical()

    Dim ws As Worksheet
    Dim wsCurve As Worksheet

    Dim lastCurveRow As Long
    Dim r As Long
    Dim outRow As Long

    Dim avgON As Double
    Dim avg1M As Double
    Dim avg3M As Double
    Dim avg6M As Double

    Dim bp1M As Double
    Dim bp3M As Double
    Dim bp6M As Double
    Dim maxAbsBp As Double
    Dim bpAxisMax As Double

    Dim winner As String
    Dim headline As String
    Dim conclusion As String
    Dim sixMText As String

    Dim chPath As ChartObject
    Dim chRelative As ChartObject

    Dim minRate As Double
    Dim maxRate As Double
    Dim axisMin As Double
    Dim axisMax As Double
    Dim span As Double
    Dim stepRate As Double

    Set ws = FINAL_GetOrCreateSheet(FINAL_S1, True)
    Set wsCurve = ThisWorkbook.Worksheets(FINAL_CURVE)

    avgON = FINAL_ScenarioAverage("Historical", 4)
    avg1M = FINAL_ScenarioAverage("Historical", 5)
    avg3M = FINAL_ScenarioAverage("Historical", 6)
    avg6M = FINAL_ScenarioAverage("Historical", 7)

    bp1M = (avg1M - avgON) * 10000#
    bp3M = (avg3M - avgON) * 10000#
    bp6M = (avg6M - avgON) * 10000#

    winner = FINAL_BestTenorByAverage("Historical")

    If bp6M < 0 Then
        sixMText = "6M lagged ON by " & Format$(Abs(bp6M), "0.0") & " bp"
    ElseIf bp6M > 0 Then
        sixMText = "6M exceeded ON by " & Format$(bp6M, "0.0") & " bp"
    Else
        sixMText = "6M matched ON"
    End If

    If winner = "1M" And Abs(bp1M) <= 2# Then
        headline = "Short tenors delivered similar realized returns; 1M edged ON while 6M lagged"
        conclusion = _
            "Across " & FINAL_ScenarioCount("Historical") & " monthly 6M starts, 1M exceeded ON by " & _
            Format$(bp1M, "0.0") & " bp on average; " & sixMText & "."

    ElseIf winner = "ON" Then
        headline = "Rolling short protected returns during the realized rate cycle"
        conclusion = _
            "ON produced the highest average realized 6M return; " & sixMText & "."

    Else
        headline = winner & " produced the highest average realized 6M return over the historical sample"
        conclusion = _
            "Historical winner = " & winner & ". Relative performance versus ON is shown in basis points to keep the economic difference clear."
    End If

    FINAL_PrepareSlideSheet ws, headline, conclusion

    '-------------------------------------------------------------
    ' DATA 1: historical ON path
    '-------------------------------------------------------------
    ws.Range("A19:B19").Value = Array("Date", "ON")
    lastCurveRow = FINAL_LastRow(wsCurve, 1)

    outRow = 20

    For r = 2 To lastCurveRow
        ws.Cells(outRow, 1).Value = wsCurve.Cells(r, 1).Value
        ws.Cells(outRow, 2).Value = wsCurve.Cells(r, 2).Value
        outRow = outRow + 1
    Next r

    ws.Range("A20:A" & outRow - 1).NumberFormat = "mmm-yy"
    ws.Range("B20:B" & outRow - 1).NumberFormat = "0.00%"

    '-------------------------------------------------------------
    ' DATA 2: historical excess return vs ON
    '-------------------------------------------------------------
    ws.Range("D19:E19").Value = Array("Tenor", "Excess vs ON (bp)")

    ws.Range("D20").Value = "ON": ws.Range("E20").Value = 0
    ws.Range("D21").Value = "1M": ws.Range("E21").Value = bp1M
    ws.Range("D22").Value = "3M": ws.Range("E22").Value = bp3M
    ws.Range("D23").Value = "6M": ws.Range("E23").Value = bp6M

    ws.Range("E20:E23").NumberFormat = "0.0 ""bp"""

    ws.Range("G19").Value = "Historical winner"
    ws.Range("H19").Value = winner
    ws.Range("G20").Value = "Monthly 6M starts"
    ws.Range("H20").Value = FINAL_ScenarioCount("Historical")
    ws.Range("G21").Value = "Analysis through"
    ws.Range("H21").Value = wsCurve.Cells(lastCurveRow, 1).Value
    ws.Range("H21").NumberFormat = "dd-mmm-yyyy"

    ws.Range("A19:H19").Font.Bold = True

    '-------------------------------------------------------------
    ' CHART 1: historical ON path
    '-------------------------------------------------------------
    Set chPath = ws.ChartObjects.Add(Left:=15, Top:=78, Width:=555, Height:=285)

    With chPath.Chart
        .ChartType = xlLine
        .SetSourceData ws.Range("A19:B" & outRow - 1)
        FINAL_StyleChart chPath.Chart, "Historical overnight rate path", False

        FINAL_SetSeriesColor .SeriesCollection(1), FINAL_ColorDark()
        .SeriesCollection(1).Format.Line.Weight = 2.25
        .SeriesCollection(1).MarkerStyle = xlMarkerStyleNone

        .Axes(xlValue).TickLabels.NumberFormat = "0.0%"
        .Axes(xlValue).HasTitle = True
        .Axes(xlValue).AxisTitle.Text = "Rate"

        On Error Resume Next
        .Axes(xlCategory).CategoryType = xlTimeScale
        .Axes(xlCategory).MajorUnit = 6
        .Axes(xlCategory).MajorUnitScale = xlMonths
        .Axes(xlCategory).TickLabels.NumberFormat = "mmm-yy"
        On Error GoTo 0
    End With

    minRate = WorksheetFunction.Min(ws.Range("B20:B" & outRow - 1))
    maxRate = WorksheetFunction.Max(ws.Range("B20:B" & outRow - 1))
    span = maxRate - minRate

    If span < 0.01 Then
        stepRate = 0.0025
    Else
        stepRate = 0.005
    End If

    axisMin = WorksheetFunction.Floor_Math(minRate - span * 0.08, stepRate)
    axisMax = WorksheetFunction.Ceiling_Math(maxRate + span * 0.08, stepRate)

    If minRate >= 0 And axisMin < 0 Then axisMin = 0
    If axisMax <= axisMin Then axisMax = axisMin + stepRate

    With chPath.Chart.Axes(xlValue)
        .MinimumScale = axisMin
        .MaximumScale = axisMax
        .MajorUnit = stepRate
    End With

    '-------------------------------------------------------------
    ' CHART 2: relative realized return vs ON
    ' Horizontal bars make the small +/- bp differences readable.
    '-------------------------------------------------------------
    maxAbsBp = WorksheetFunction.Max(Abs(bp1M), Abs(bp3M), Abs(bp6M), 1#)
    bpAxisMax = FINAL_NiceBpMax(maxAbsBp)

    Set chRelative = ws.ChartObjects.Add(Left:=590, Top:=78, Width:=335, Height:=285)

    With chRelative.Chart
        .ChartType = xlBarClustered
        .SetSourceData ws.Range("D19:E23")
        FINAL_StyleChart chRelative.Chart, "Historical excess return vs ON", False

        FINAL_SetSeriesColor .SeriesCollection(1), FINAL_ColorLightGray()

        On Error Resume Next
        .SeriesCollection(1).Points(1).Format.Fill.ForeColor.RGB = FINAL_ColorGray()
        .SeriesCollection(1).Points(1).Format.Line.ForeColor.RGB = FINAL_ColorGray()
        .SeriesCollection(1).Points(2).Format.Fill.ForeColor.RGB = FINAL_ColorAccent()
        .SeriesCollection(1).Points(2).Format.Line.ForeColor.RGB = FINAL_ColorAccent()
        .SeriesCollection(1).Points(3).Format.Fill.ForeColor.RGB = FINAL_ColorGray()
        .SeriesCollection(1).Points(3).Format.Line.ForeColor.RGB = FINAL_ColorGray()
        .SeriesCollection(1).Points(4).Format.Fill.ForeColor.RGB = FINAL_ColorRisk()
        .SeriesCollection(1).Points(4).Format.Line.ForeColor.RGB = FINAL_ColorRisk()
        On Error GoTo 0

        .SeriesCollection(1).ApplyDataLabels
        .SeriesCollection(1).DataLabels.NumberFormat = "0.0 ""bp"""
        .SeriesCollection(1).DataLabels.Position = xlLabelPositionOutsideEnd

        With .Axes(xlValue)
            .MinimumScale = -bpAxisMax
            .MaximumScale = bpAxisMax
            .MajorUnit = bpAxisMax / 3#
            .HasTitle = True
            .AxisTitle.Text = "Excess return vs ON (bp)"
            .CrossesAt = 0
        End With

        On Error Resume Next
        .Axes(xlCategory).ReversePlotOrder = True
        On Error GoTo 0
    End With

    ws.Columns("A:H").AutoFit

End Sub
