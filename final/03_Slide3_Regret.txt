Option Explicit

'=====================================================================
' SLIDE 3 - DEFINE THE RISK
'
' Message:
'   The decision risk is the cost of choosing the wrong tenor.
'
' Definition:
'   Regret = best single-tenor return in an observation
'            minus selected strategy return.
'
' Lower is better.
'=====================================================================

Public Sub FINAL_Slide3_Regret()

    Dim ws As Worksheet

    Dim histONRegret As Double
    Dim hist6MRegret As Double
    Dim invONRegret As Double
    Dim inv6MRegret As Double

    Dim histONBp As Double
    Dim hist6MBp As Double
    Dim invONBp As Double
    Dim inv6MBp As Double

    Dim headline As String
    Dim conclusion As String

    Dim ch As ChartObject
    Dim axisMax As Double
    Dim maxBp As Double

    Set ws = FINAL_GetOrCreateSheet(FINAL_S3, True)

    histONRegret = FINAL_ScenarioRegretAverage("Historical", 10)
    hist6MRegret = FINAL_ScenarioRegretAverage("Historical", 13)

    invONRegret = FINAL_ScenarioRegretAverage("Inverted", 10)
    inv6MRegret = FINAL_ScenarioRegretAverage("Inverted", 13)

    histONBp = histONRegret * 10000#
    hist6MBp = hist6MRegret * 10000#
    invONBp = invONRegret * 10000#
    inv6MBp = inv6MRegret * 10000#

    headline = "The relevant risk is the cost of choosing the wrong tenor"
    conclusion = _
        "100% ON: " & Format$(histONBp, "0.0") & " bp historical / " & Format$(invONBp, "0.0") & _
        " bp inverted; 100% 6M: " & Format$(hist6MBp, "0.0") & " bp historical / " & _
        Format$(inv6MBp, "0.0") & " bp inverted. Protection is directional."

    FINAL_PrepareSlideSheet ws, headline, conclusion

    '-------------------------------------------------------------
    ' DATA
    '-------------------------------------------------------------
    ws.Range("A19:C19").Value = Array("Strategy", "Historical", "Inverted")

    ws.Range("A20").Value = "100% ON"
    ws.Range("A21").Value = "100% 6M"

    ws.Range("B20").Value = histONBp
    ws.Range("B21").Value = hist6MBp

    ws.Range("C20").Value = invONBp
    ws.Range("C21").Value = inv6MBp

    ws.Range("A19:C19").Font.Bold = True
    ws.Range("B20:C21").NumberFormat = "0.0 ""bp"""

    '-------------------------------------------------------------
    ' DEFINITION BOX
    '-------------------------------------------------------------
    ws.Range("K5:N5").Merge
    ws.Range("K5").Value = "REGRET"

    ws.Range("K6:N8").Merge
    ws.Range("K6").Value = "Best tenor return" & vbLf & "minus" & vbLf & "selected strategy return"

    ws.Range("K10:N10").Merge
    ws.Range("K10").Value = "LOWER IS BETTER"

    With ws.Range("K5:N5")
        .Font.Bold = True
        .Font.Size = 12
        .Interior.Color = RGB(243, 244, 246)
        .HorizontalAlignment = xlCenter
    End With

    With ws.Range("K6:N8")
        .WrapText = True
        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlCenter
        .Font.Size = 12
    End With

    With ws.Range("K10:N10")
        .Font.Bold = True
        .Font.Size = 12
        .HorizontalAlignment = xlCenter
        .Font.Color = FINAL_ColorAccent2()
    End With

    '-------------------------------------------------------------
    ' CHART
    '-------------------------------------------------------------
    maxBp = WorksheetFunction.Max(ws.Range("B20:C21"))
    axisMax = FINAL_NiceBpMax(maxBp)

    Set ch = ws.ChartObjects.Add(Left:=40, Top:=80, Width:=650, Height:=300)

    With ch.Chart

        .ChartType = xlBarClustered
        .SetSourceData ws.Range("A19:C21")

        FINAL_StyleChart ch.Chart, "Average cost of being wrong - lower is better", True

        FINAL_SetSeriesColor .SeriesCollection(1), FINAL_ColorDark()
        FINAL_SetSeriesColor .SeriesCollection(2), FINAL_ColorAccent()

        .SeriesCollection(1).ApplyDataLabels
        .SeriesCollection(2).ApplyDataLabels

        .SeriesCollection(1).DataLabels.NumberFormat = "0.0 ""bp"""
        .SeriesCollection(2).DataLabels.NumberFormat = "0.0 ""bp"""
        .SeriesCollection(1).DataLabels.Position = xlLabelPositionOutsideEnd
        .SeriesCollection(2).DataLabels.Position = xlLabelPositionOutsideEnd

        With .Axes(xlValue)
            .MinimumScale = 0
            .MaximumScale = axisMax
            .MajorUnit = axisMax / 4#
            .HasTitle = True
            .AxisTitle.Text = "Average regret (bp)"
        End With

        On Error Resume Next
        .Axes(xlCategory).ReversePlotOrder = True
        On Error GoTo 0

    End With

    ws.Columns("K:N").ColumnWidth = 15

End Sub
