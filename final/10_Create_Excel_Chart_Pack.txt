Option Explicit

'=====================================================================
' EXCEL CHART PACK / MODEL REVIEW
'
' Purpose
'   Refresh the existing cash-tenor model and prepare its Excel output
'   for manual transfer into PowerPoint.
'
' What this module does
'   1) Runs FINAL_BuildAll, which rebuilds the validated calculation
'      sheets and the five presentation chart sheets.
'   2) Creates Presentation_Index with the narrative, source map,
'      robust allocation and independent model checks.
'   3) Standardizes the five chart sheets for clear viewing, printing
'      and audit. Charts stay above row 19; supporting data stays below.
'
' What this module does NOT do
'   - It does not duplicate or overwrite the return engine.
'   - It does not create a PowerPoint file.
'   - It does not replace actual deposit-premium inputs on Final_Input.
'
' Required modules
'   00_Master through 08_R_CURVE_Adjusted
'
' Run
'   FINAL_CreateExcelChartPack
'=====================================================================

Private Const FINAL_XL_INDEX As String = "Presentation_Index"
Private Const FINAL_XL_TOL As Double = 0.0000000001

Public Sub FINAL_CreateExcelChartPack()

    Dim oldCalc As XlCalculation
    Dim oldScreen As Boolean
    Dim oldEvents As Boolean
    Dim wsIndex As Worksheet

    On Error GoTo fail

    oldCalc = Application.Calculation
    oldScreen = Application.ScreenUpdating
    oldEvents = Application.EnableEvents

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    Application.StatusBar = "Refreshing model and preparing Excel chart pack..."

    'Run each existing model stage directly so an error cannot leave a stale
    'chart pack looking current. FINAL_BuildAll intentionally handles its own
    'errors for interactive use; this review module needs errors to propagate.
    FINAL_XL_RunWorkbookMacro "FINAL_EnsureSetup"
    FINAL_XL_RunWorkbookMacro "FINAL_BuildCoreData"
    FINAL_XL_RunWorkbookMacro "FINAL_Slide1_Historical"
    FINAL_XL_RunWorkbookMacro "FINAL_Slide2_Direction"
    FINAL_XL_RunWorkbookMacro "FINAL_Slide3_Regret"
    FINAL_XL_RunWorkbookMacro "FINAL_Slide4_Robust"
    FINAL_XL_RunWorkbookMacro "FINAL_Slide5_Execution"
    FINAL_XL_RunWorkbookMacro "FINAL_UpdateWordingSummary"

    FINAL_XL_RequireOutputs
    FINAL_XL_StandardizePresentationSheets

    Set wsIndex = FINAL_XL_GetOrCreateSheet(FINAL_XL_INDEX)
    FINAL_XL_BuildIndex wsIndex
    FINAL_XL_StandardizeIndex wsIndex

    wsIndex.Activate
    wsIndex.Range("A1").Select

clean_exit:
    Application.StatusBar = False
    Application.Calculation = oldCalc
    Application.EnableEvents = oldEvents
    Application.ScreenUpdating = oldScreen
    Exit Sub

fail:
    MsgBox "FINAL_CreateExcelChartPack stopped: " & Err.Description, vbCritical
    Resume clean_exit

End Sub

Private Sub FINAL_XL_BuildIndex(ByVal ws As Worksheet)

    Dim slideNames As Variant
    Dim questions As Variant
    Dim sourceMaps As Variant
    Dim handoffNotes As Variant
    Dim i As Long
    Dim r As Long
    Dim wsSlide As Worksheet
    Dim robustRow As Long
    Dim wsPort As Worksheet
    Dim qaRow As Long
    Dim passCount As Long
    Dim warningCount As Long
    Dim failCount As Long

    FINAL_XL_ClearSheet ws

    slideNames = Array("S1_Historical", "S2_Direction", "S3_Regret", "S4_Robust", "S5_Execution")

    questions = Array( _
        "What did the curve price, and what was actually earned?", _
        "Did the initial term premium predict the realized holding-period outcome?", _
        "What changes if the rate path moves in the opposite direction?", _
        "Which allocation limits regret without relying on one rate forecast?", _
        "How is the risk framework translated into an executable deposit program?")

    sourceMaps = Array( _
        "Curve; F_Curve_Normalized; F_Core_Returns", _
        "F_Curve_Normalized; F_Core_Returns", _
        "F_Core_Returns", _
        "F_Core_Returns; F_Portfolios; Final_Control", _
        "Final_Input; S5_Execution support table")

    handoffNotes = Array( _
        "Copy both charts and retain the event/return context in the speaker notes.", _
        "Keep the zero line visible; distinguish priced spread from realized excess return.", _
        "Keep historical and inverted outcomes on the same basis-point scale.", _
        "Use the robust allocation as a risk limit, not as a point forecast.", _
        "Replace the placeholder only with actual premium and institution data.")

    With ws
        .Range("A1:J1").Merge
        .Range("A1").Value = "EXCEL CHART PACK | CASH TENOR ALLOCATION"

        .Range("A2:J2").Merge
        .Range("A2").Value = _
            "Manual PowerPoint workflow: copy the charts from S1-S5. Each chart sheet keeps its committee message at the top and the auditable support table from row 19 downward."

        .Range("A4:D4").Merge
        .Range("A4").Value = "MODEL SCOPE"

        .Range("A5").Value = "Analysis start"
        .Range("A6").Value = "Analysis end"
        .Range("A7").Value = "Historical observations"
        .Range("A8").Value = "Inverted observations"
        .Range("A9").Value = "Common holding horizon"

        .Range("B5").Value = ThisWorkbook.Worksheets("Final_Control").Range("B3").Value
        .Range("B6").Value = ThisWorkbook.Worksheets("Final_Control").Range("B4").Value
        .Range("B7").Value = FINAL_XL_ScenarioCount("Historical")
        .Range("B8").Value = FINAL_XL_ScenarioCount("Inverted")
        .Range("B9").Value = "6 months"
        .Range("B5:B6").NumberFormat = "dd-mmm-yyyy"

        .Range("F4:J4").Merge
        .Range("F4").Value = "ROBUST ALLOCATION | MINIMAX REGRET"

        .Range("F5:J5").Value = Array("ON", "1M", "3M", "6M", "Max directional regret")
    End With

    Set wsPort = ThisWorkbook.Worksheets("F_Portfolios")
    robustRow = FINAL_XL_FindRobustRow(wsPort)

    If robustRow > 0 Then
        ws.Range("F6").Value = wsPort.Cells(robustRow, 1).Value
        ws.Range("G6").Value = wsPort.Cells(robustRow, 2).Value
        ws.Range("H6").Value = wsPort.Cells(robustRow, 3).Value
        ws.Range("I6").Value = wsPort.Cells(robustRow, 4).Value
        ws.Range("J6").Value = wsPort.Cells(robustRow, 9).Value * 10000#
        ws.Range("F6:I6").NumberFormat = "0%"
        ws.Range("J6").NumberFormat = "0.0 ""bp"""
    End If

    ws.Range("F8:J8").Merge
    ws.Range("F8").Value = _
        "Decision rule: minimize the larger of historical and inverted average regret; break ties by mean regret, then shorter weighted tenor."

    ws.Range("A11:J11").Merge
    ws.Range("A11").Value = "PRESENTATION FLOW AND AUDIT MAP"

    ws.Range("A12:I12").Value = Array( _
        "Order", "Sheet", "Committee question", "Key conclusion", _
        "Charts", "Audit range", "Primary sources", "Status", "Manual handoff")

    For i = LBound(slideNames) To UBound(slideNames)

        r = 13 + i
        Set wsSlide = ThisWorkbook.Worksheets(CStr(slideNames(i)))

        ws.Cells(r, 1).Value = i + 1
        ws.Hyperlinks.Add Anchor:=ws.Cells(r, 2), Address:="", _
            SubAddress:="'" & CStr(slideNames(i)) & "'!A1", _
            TextToDisplay:=CStr(slideNames(i))
        ws.Cells(r, 3).Value = CStr(questions(i))
        ws.Cells(r, 4).Value = CStr(wsSlide.Range("A2").Value)
        ws.Cells(r, 5).Value = wsSlide.ChartObjects.Count
        ws.Cells(r, 6).Value = FINAL_XL_AuditRangeText(wsSlide)
        ws.Cells(r, 7).Value = CStr(sourceMaps(i))
        ws.Cells(r, 8).Value = FINAL_XL_SlideStatus(wsSlide, CStr(slideNames(i)))
        ws.Cells(r, 9).Value = CStr(handoffNotes(i))

    Next i

    ws.Range("A20:J20").Merge
    ws.Range("A20").Value = "INDEPENDENT MODEL REVIEW"

    ws.Range("A21:E21").Value = Array("Check", "Status", "Evidence", "Source", "Committee relevance")
    qaRow = 22

    FINAL_XL_RunAudits ws, qaRow, passCount, warningCount, failCount

    ws.Range("F21:J21").Merge
    ws.Range("F21").Value = "REVIEW CONCLUSION"

    ws.Range("F22:J24").Merge

    If failCount > 0 Then
        ws.Range("F22").Value = _
            "NEEDS REVISION | " & failCount & " core check(s) failed. Do not use the charts in committee materials until the failed items are resolved."
        ws.Range("F22").Interior.Color = RGB(254, 226, 226)
        ws.Range("F22").Font.Color = RGB(153, 27, 27)
    ElseIf warningCount > 0 Then
        ws.Range("F22").Value = _
            "READY WITH INPUT CAVEAT | Core calculations passed " & passCount & _
            " checks. Complete the execution inputs before using Slide 5."
        ws.Range("F22").Interior.Color = RGB(255, 247, 237)
        ws.Range("F22").Font.Color = RGB(154, 52, 18)
    Else
        ws.Range("F22").Value = _
            "READY TO SHARE | Core calculations and chart-sheet controls passed " & passCount & _
            " checks. Conclusions remain scenario-based risk evidence, not forecasts."
        ws.Range("F22").Interior.Color = RGB(236, 253, 245)
        ws.Range("F22").Font.Color = RGB(6, 95, 70)
    End If

    ws.Range("F26:J26").Merge
    ws.Range("F26").Value = "MODEL INTERPRETATION"

    ws.Range("F27:J31").Merge
    ws.Range("F27").Value = _
        "The model compares ON, 1M, 3M and 6M over one common six-month horizon. " & _
        "The historical path measures realized repricing; the inverted path is a directional stress built by reversing observed rate changes from a positive anchor. " & _
        "The recommended mix minimizes forecast dependence by controlling worst directional average regret. " & _
        "A zero weight is a grid outcome, not a permanent prohibition: allocations should be re-run when curves, liquidity needs or counterparty limits change."

End Sub

'=====================================================================
' AUDIT TESTS
'=====================================================================

Private Sub FINAL_XL_RunAudits( _
    ByVal ws As Worksheet, _
    ByRef nextRow As Long, _
    ByRef passCount As Long, _
    ByRef warningCount As Long, _
    ByRef failCount As Long)

    FINAL_XL_AddAudit ws, nextRow, "Required sheets", _
        FINAL_XL_AllRequiredSheetsExist(), _
        "Curve, calculation and S1-S5 output sheets present", _
        "Workbook sheet inventory", _
        "Prevents an incomplete committee pack", _
        passCount, failCount

    FINAL_XL_AddAudit ws, nextRow, "Curve dates", _
        FINAL_XL_CurveDatesAscending(), _
        "Strictly ascending normalized dates", _
        "F_Curve_Normalized!A2:A...", _
        "Protects roll-date and maturity sequencing", _
        passCount, failCount

    FINAL_XL_AddAudit ws, nextRow, "Required tenor rates", _
        FINAL_XL_CurveRatesComplete(), _
        "ON, 1M, 3M and 6M are numeric", _
        "F_Curve_Normalized!B2:E...", _
        "Avoids silent gaps in return calculations", _
        passCount, failCount

    FINAL_XL_AddAudit ws, nextRow, "Inverted curve level", _
        FINAL_XL_InvertedRatesNonNegative(), _
        "All displayed stress rates are non-negative", _
        "F_Curve_Inverted!B2:E...", _
        "Keeps historical and inverse paths comparable without negative-rate fields", _
        passCount, failCount

    FINAL_XL_AddAudit ws, nextRow, "Scenario symmetry", _
        FINAL_XL_ScenarioCountsValid(), _
        FINAL_XL_ScenarioCount("Historical") & " observations per direction", _
        "F_Core_Returns!A2:A...", _
        "Prevents one direction from dominating by sample size", _
        passCount, failCount

    FINAL_XL_AddAudit ws, nextRow, "Common six-month horizon", _
        FINAL_XL_HorizonsValid(), _
        "Every maturity is later than its decision date", _
        "F_Core_Returns!B2:C...", _
        "Makes tenor returns economically comparable", _
        passCount, failCount

    FINAL_XL_AddAudit ws, nextRow, "Return and regret reconciliation", _
        FINAL_XL_ReturnsReconcile(), _
        "Best return and four regret fields independently recomputed", _
        "F_Core_Returns!D2:M...", _
        "Confirms the message is supported by the return engine", _
        passCount, failCount

    FINAL_XL_AddAudit ws, nextRow, "Portfolio grid", _
        FINAL_XL_PortfolioCountValid(), _
        "286 long-only allocations at 10% increments", _
        "F_Portfolios!A2:L287", _
        "Confirms the full constrained search space", _
        passCount, failCount

    FINAL_XL_AddAudit ws, nextRow, "Portfolio weights", _
        FINAL_XL_PortfolioWeightsValid(), _
        "Every allocation sums to 100% and contains no negative weight", _
        "F_Portfolios!A2:D...", _
        "Avoids leverage or unallocated-cash assumptions", _
        passCount, failCount

    FINAL_XL_AddAudit ws, nextRow, "Robust solution", _
        FINAL_XL_RobustSolutionValid(), _
        "Exactly one row is flagged and it has the minimum max directional regret", _
        "F_Portfolios!G2:L...", _
        "Supports the allocation recommendation without a rate forecast", _
        passCount, failCount

    FINAL_XL_AddAudit ws, nextRow, "Chart sheet structure", _
        FINAL_XL_CoreSlidesValid(), _
        "S1-S4 contain headline, conclusion, charts and support data", _
        "S1_Historical:S4_Robust", _
        "Keeps the committee narrative traceable", _
        passCount, failCount

    If FINAL_XL_ExecutionInputsComplete() Then
        FINAL_XL_AddAudit ws, nextRow, "Execution inputs", True, _
            "Actual premium history is available from Jan-2025", _
            "Final_Input!A10:C...", _
            "Slide 5 can be used", _
            passCount, failCount
    Else
        FINAL_XL_AddWarning ws, nextRow, "Execution inputs", _
            "Premium history is not yet complete", _
            "Final_Input!A10:C...", _
            "Treat Slide 5 as a template until actual observations are entered", _
            warningCount
    End If

End Sub

Private Sub FINAL_XL_AddAudit( _
    ByVal ws As Worksheet, _
    ByRef r As Long, _
    ByVal checkName As String, _
    ByVal passed As Boolean, _
    ByVal evidence As String, _
    ByVal sourceText As String, _
    ByVal relevance As String, _
    ByRef passCount As Long, _
    ByRef failCount As Long)

    ws.Cells(r, 1).Value = checkName

    If passed Then
        ws.Cells(r, 2).Value = "PASS"
        ws.Cells(r, 2).Interior.Color = RGB(236, 253, 245)
        ws.Cells(r, 2).Font.Color = RGB(6, 95, 70)
        passCount = passCount + 1
    Else
        ws.Cells(r, 2).Value = "FAIL"
        ws.Cells(r, 2).Interior.Color = RGB(254, 226, 226)
        ws.Cells(r, 2).Font.Color = RGB(153, 27, 27)
        failCount = failCount + 1
    End If

    ws.Cells(r, 3).Value = evidence
    ws.Cells(r, 4).Value = sourceText
    ws.Cells(r, 5).Value = relevance
    r = r + 1

End Sub

Private Sub FINAL_XL_AddWarning( _
    ByVal ws As Worksheet, _
    ByRef r As Long, _
    ByVal checkName As String, _
    ByVal evidence As String, _
    ByVal sourceText As String, _
    ByVal relevance As String, _
    ByRef warningCount As Long)

    ws.Cells(r, 1).Value = checkName
    ws.Cells(r, 2).Value = "INPUT REQUIRED"
    ws.Cells(r, 2).Interior.Color = RGB(255, 247, 237)
    ws.Cells(r, 2).Font.Color = RGB(154, 52, 18)
    ws.Cells(r, 3).Value = evidence
    ws.Cells(r, 4).Value = sourceText
    ws.Cells(r, 5).Value = relevance

    warningCount = warningCount + 1
    r = r + 1

End Sub

Private Function FINAL_XL_AllRequiredSheetsExist() As Boolean

    Dim names As Variant
    Dim i As Long

    names = Array( _
        "Curve", "Final_Control", "Final_Input", _
        "F_Curve_Normalized", "F_Curve_Inverted", _
        "F_Core_Returns", "F_Portfolios", _
        "S1_Historical", "S2_Direction", "S3_Regret", _
        "S4_Robust", "S5_Execution")

    For i = LBound(names) To UBound(names)
        If Not FINAL_XL_SheetExists(CStr(names(i))) Then Exit Function
    Next i

    FINAL_XL_AllRequiredSheetsExist = True

End Function

Private Function FINAL_XL_CurveDatesAscending() As Boolean

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim r As Long
    Dim previousDate As Double

    Set ws = ThisWorkbook.Worksheets("F_Curve_Normalized")
    lastRow = FINAL_XL_LastDataRow(ws, 1)

    If lastRow < 2 Then Exit Function

    For r = 2 To lastRow
        If Not IsDate(ws.Cells(r, 1).Value) Then Exit Function
        If previousDate > 0 Then
            If CDbl(CDate(ws.Cells(r, 1).Value)) <= previousDate Then Exit Function
        End If
        previousDate = CDbl(CDate(ws.Cells(r, 1).Value))
    Next r

    FINAL_XL_CurveDatesAscending = True

End Function

Private Function FINAL_XL_CurveRatesComplete() As Boolean

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim r As Long
    Dim c As Long

    Set ws = ThisWorkbook.Worksheets("F_Curve_Normalized")
    lastRow = FINAL_XL_LastDataRow(ws, 1)

    For r = 2 To lastRow
        For c = 2 To 5
            If Not IsNumeric(ws.Cells(r, c).Value) Then Exit Function
            If CDbl(ws.Cells(r, c).Value) <= -1# Then Exit Function
        Next c
    Next r

    FINAL_XL_CurveRatesComplete = (lastRow >= 2)

End Function

Private Function FINAL_XL_InvertedRatesNonNegative() As Boolean

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim r As Long
    Dim c As Long

    Set ws = ThisWorkbook.Worksheets("F_Curve_Inverted")
    lastRow = FINAL_XL_LastDataRow(ws, 1)

    For r = 2 To lastRow
        For c = 2 To 5
            If Not IsNumeric(ws.Cells(r, c).Value) Then Exit Function
            If CDbl(ws.Cells(r, c).Value) < -FINAL_XL_TOL Then Exit Function
        Next c
    Next r

    FINAL_XL_InvertedRatesNonNegative = (lastRow >= 2)

End Function

Private Function FINAL_XL_ScenarioCountsValid() As Boolean

    Dim nHist As Long
    Dim nInv As Long

    nHist = FINAL_XL_ScenarioCount("Historical")
    nInv = FINAL_XL_ScenarioCount("Inverted")

    FINAL_XL_ScenarioCountsValid = (nHist > 0 And nHist = nInv)

End Function

Private Function FINAL_XL_HorizonsValid() As Boolean

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim r As Long

    Set ws = ThisWorkbook.Worksheets("F_Core_Returns")
    lastRow = FINAL_XL_LastDataRow(ws, 1)

    For r = 2 To lastRow
        If Not IsDate(ws.Cells(r, 2).Value) Then Exit Function
        If Not IsDate(ws.Cells(r, 3).Value) Then Exit Function
        If CDate(ws.Cells(r, 3).Value) <= CDate(ws.Cells(r, 2).Value) Then Exit Function
    Next r

    FINAL_XL_HorizonsValid = (lastRow >= 2)

End Function

Private Function FINAL_XL_ReturnsReconcile() As Boolean

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim r As Long
    Dim c As Long
    Dim bestReturn As Double
    Dim tenorReturn As Double

    Set ws = ThisWorkbook.Worksheets("F_Core_Returns")
    lastRow = FINAL_XL_LastDataRow(ws, 1)

    For r = 2 To lastRow

        For c = 4 To 7
            If Not IsNumeric(ws.Cells(r, c).Value) Then Exit Function
        Next c

        bestReturn = CDbl(ws.Cells(r, 4).Value)
        For c = 5 To 7
            If CDbl(ws.Cells(r, c).Value) > bestReturn Then bestReturn = CDbl(ws.Cells(r, c).Value)
        Next c

        If Abs(CDbl(ws.Cells(r, 9).Value) - bestReturn) > FINAL_XL_TOL Then Exit Function

        For c = 4 To 7
            If Abs(CDbl(ws.Cells(r, c + 6).Value) - _
                (bestReturn - CDbl(ws.Cells(r, c).Value))) > FINAL_XL_TOL Then Exit Function

            If CDbl(ws.Cells(r, c + 6).Value) < -FINAL_XL_TOL Then Exit Function
        Next c

        tenorReturn = FINAL_XL_TenorReturn(ws, r, CStr(ws.Cells(r, 8).Value))
        If Abs(tenorReturn - bestReturn) > FINAL_XL_TOL Then Exit Function

    Next r

    FINAL_XL_ReturnsReconcile = (lastRow >= 2)

End Function

Private Function FINAL_XL_PortfolioCountValid() As Boolean

    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("F_Portfolios")

    FINAL_XL_PortfolioCountValid = (FINAL_XL_LastDataRow(ws, 1) - 1 = 286)

End Function

Private Function FINAL_XL_PortfolioWeightsValid() As Boolean

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim r As Long
    Dim c As Long
    Dim totalWeight As Double
    Dim w As Double

    Set ws = ThisWorkbook.Worksheets("F_Portfolios")
    lastRow = FINAL_XL_LastDataRow(ws, 1)

    For r = 2 To lastRow
        totalWeight = 0

        For c = 1 To 4
            If Not IsNumeric(ws.Cells(r, c).Value) Then Exit Function
            w = CDbl(ws.Cells(r, c).Value)
            If w < -FINAL_XL_TOL Or w > 1# + FINAL_XL_TOL Then Exit Function
            totalWeight = totalWeight + w
        Next c

        If Abs(totalWeight - 1#) > FINAL_XL_TOL Then Exit Function
    Next r

    FINAL_XL_PortfolioWeightsValid = (lastRow >= 2)

End Function

Private Function FINAL_XL_RobustSolutionValid() As Boolean

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim r As Long
    Dim robustCount As Long
    Dim robustRow As Long
    Dim minimumMaxRegret As Double
    Dim rowMax As Double

    Set ws = ThisWorkbook.Worksheets("F_Portfolios")
    lastRow = FINAL_XL_LastDataRow(ws, 1)
    minimumMaxRegret = 1E+99

    For r = 2 To lastRow
        If CBool(ws.Cells(r, 12).Value) Then
            robustCount = robustCount + 1
            robustRow = r
        End If

        rowMax = CDbl(ws.Cells(r, 7).Value)
        If CDbl(ws.Cells(r, 8).Value) > rowMax Then rowMax = CDbl(ws.Cells(r, 8).Value)

        If Abs(CDbl(ws.Cells(r, 9).Value) - rowMax) > FINAL_XL_TOL Then Exit Function
        If CDbl(ws.Cells(r, 9).Value) < minimumMaxRegret Then minimumMaxRegret = CDbl(ws.Cells(r, 9).Value)
    Next r

    If robustCount <> 1 Then Exit Function
    If Abs(CDbl(ws.Cells(robustRow, 9).Value) - minimumMaxRegret) > FINAL_XL_TOL Then Exit Function

    FINAL_XL_RobustSolutionValid = True

End Function

Private Function FINAL_XL_CoreSlidesValid() As Boolean

    Dim names As Variant
    Dim expectedCharts As Variant
    Dim i As Long
    Dim ws As Worksheet

    names = Array("S1_Historical", "S2_Direction", "S3_Regret", "S4_Robust")
    expectedCharts = Array(2, 1, 1, 3)

    For i = LBound(names) To UBound(names)
        Set ws = ThisWorkbook.Worksheets(CStr(names(i)))

        If Len(Trim$(CStr(ws.Range("A1").Value))) = 0 Then Exit Function
        If Len(Trim$(CStr(ws.Range("A2").Value))) = 0 Then Exit Function
        If ws.ChartObjects.Count <> CLng(expectedCharts(i)) Then Exit Function
        If Application.WorksheetFunction.CountA(ws.Rows(19)) = 0 Then Exit Function
    Next i

    FINAL_XL_CoreSlidesValid = True

End Function

Private Function FINAL_XL_ExecutionInputsComplete() As Boolean

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim r As Long

    Set ws = ThisWorkbook.Worksheets("Final_Input")
    lastRow = FINAL_XL_LastDataRow(ws, 1)

    For r = 10 To lastRow
        If IsDate(ws.Cells(r, 1).Value) Then
            If CDate(ws.Cells(r, 1).Value) >= DateSerial(2025, 1, 1) Then
                If IsNumeric(ws.Cells(r, 2).Value) Or IsNumeric(ws.Cells(r, 3).Value) Then
                    FINAL_XL_ExecutionInputsComplete = True
                    Exit Function
                End If
            End If
        End If
    Next r

End Function

'=====================================================================
' PRESENTATION SHEET FORMATTING
'=====================================================================

Private Sub FINAL_XL_StandardizePresentationSheets()

    Dim names As Variant
    Dim colors As Variant
    Dim i As Long
    Dim ws As Worksheet

    names = Array("S1_Historical", "S2_Direction", "S3_Regret", "S4_Robust", "S5_Execution")
    colors = Array( _
        RGB(31, 78, 121), RGB(68, 114, 196), RGB(127, 127, 127), _
        RGB(192, 80, 77), RGB(112, 173, 71))

    For i = LBound(names) To UBound(names)

        Set ws = ThisWorkbook.Worksheets(CStr(names(i)))
        ws.Tab.Color = CLng(colors(i))

        ws.Activate

        With ActiveWindow
            .DisplayGridlines = False
            .Zoom = 85
            .FreezePanes = False
            .SplitColumn = 0
            .SplitRow = 18
            .FreezePanes = True
        End With

        With ws.PageSetup
            .Orientation = xlLandscape
            .Zoom = False
            .FitToPagesWide = 1
            .FitToPagesTall = False
            .LeftMargin = Application.InchesToPoints(0.25)
            .RightMargin = Application.InchesToPoints(0.25)
            .TopMargin = Application.InchesToPoints(0.35)
            .BottomMargin = Application.InchesToPoints(0.35)
            .PrintArea = ws.Range("A1", FINAL_XL_LastUsedCell(ws)).Address
        End With

        With ws.Rows(19)
            .Font.Bold = True

    If Not FINAL_XL_AllRequiredSheetsExist() Then
        Err.Raise vbObjectError + 500, , _
            "The model refresh did not create all required sheets. Review the FINAL_BuildAll error message and the Curve input."
    End If

Private Sub FINAL_XL_RunWorkbookMacro(ByVal procedureName As String)

    Dim qualifiedName As String

    qualifiedName = _
        "'" & Replace(ThisWorkbook.Name, "'", "''") & "'!" & procedureName

    Application.Run qualifiedName

End Sub

Private Function FINAL_XL_GetOrCreateSheet(ByVal sheetName As String) As Worksheet

    If FINAL_XL_SheetExists(sheetName) Then
        Set FINAL_XL_GetOrCreateSheet = ThisWorkbook.Worksheets(sheetName)
    Else
        Set FINAL_XL_GetOrCreateSheet = _
            ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
        FINAL_XL_GetOrCreateSheet.Name = sheetName
    End If

End Function

Private Function FINAL_XL_SheetExists(ByVal sheetName As String) As Boolean

    Dim ws As Worksheet

    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(sheetName)
    FINAL_XL_SheetExists = Not ws Is Nothing
    On Error GoTo 0

End Function

Private Sub FINAL_XL_ClearSheet(ByVal ws As Worksheet)

    Dim i As Long

    ws.Cells.UnMerge
    ws.Cells.Clear

    For i = ws.Shapes.Count To 1 Step -1
        ws.Shapes(i).Delete
    Next i

End Sub

Private Function FINAL_XL_LastDataRow(ByVal ws As Worksheet, ByVal colNum As Long) As Long
    FINAL_XL_LastDataRow = ws.Cells(ws.Rows.Count, colNum).End(xlUp).Row
End Function

Private Function FINAL_XL_LastUsedCell(ByVal ws As Worksheet) As Range

    Dim lastRowCell As Range
    Dim lastColCell As Range

    Set lastRowCell = ws.Cells.Find( _
        What:="*", After:=ws.Cells(1, 1), LookIn:=xlFormulas, _
        LookAt:=xlPart, SearchOrder:=xlByRows, SearchDirection:=xlPrevious)

    Set lastColCell = ws.Cells.Find( _
        What:="*", After:=ws.Cells(1, 1), LookIn:=xlFormulas, _
        LookAt:=xlPart, SearchOrder:=xlByColumns, SearchDirection:=xlPrevious)

    If lastRowCell Is Nothing Or lastColCell Is Nothing Then
        Set FINAL_XL_LastUsedCell = ws.Range("A1")
    Else
        Set FINAL_XL_LastUsedCell = ws.Cells(lastRowCell.Row, lastColCell.Column)
    End If

End Function

Private Function FINAL_XL_FindRobustRow(ByVal ws As Worksheet) As Long

    Dim lastRow As Long
    Dim r As Long

    lastRow = FINAL_XL_LastDataRow(ws, 1)

    For r = 2 To lastRow
        If CBool(ws.Cells(r, 12).Value) Then
            FINAL_XL_FindRobustRow = r
            Exit Function
        End If
    Next r

End Function

Private Function FINAL_XL_ScenarioCount(ByVal scenarioName As String) As Long

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim r As Long

    Set ws = ThisWorkbook.Worksheets("F_Core_Returns")
    lastRow = FINAL_XL_LastDataRow(ws, 1)

    For r = 2 To lastRow
        If CStr(ws.Cells(r, 1).Value) = scenarioName Then
            FINAL_XL_ScenarioCount = FINAL_XL_ScenarioCount + 1
        End If
    Next r

End Function

Private Function FINAL_XL_TenorReturn( _
    ByVal ws As Worksheet, _
    ByVal r As Long, _
    ByVal tenorName As String) As Double

    Select Case UCase$(Trim$(tenorName))
        Case "ON": FINAL_XL_TenorReturn = CDbl(ws.Cells(r, 4).Value)
        Case "1M": FINAL_XL_TenorReturn = CDbl(ws.Cells(r, 5).Value)
        Case "3M": FINAL_XL_TenorReturn = CDbl(ws.Cells(r, 6).Value)
        Case "6M": FINAL_XL_TenorReturn = CDbl(ws.Cells(r, 7).Value)
        Case Else
            FINAL_XL_TenorReturn = -1E+99
    End Select

End Function

Private Function FINAL_XL_AuditRangeText(ByVal ws As Worksheet) As String

    Dim lastCell As Range
    Set lastCell = FINAL_XL_LastUsedCell(ws)

    If lastCell.Row < 19 Then
        FINAL_XL_AuditRangeText = "No support table"
    Else
        FINAL_XL_AuditRangeText = "A19:" & lastCell.Address(False, False)
    End If

End Function

Private Function FINAL_XL_SlideStatus(ByVal ws As Worksheet, ByVal sheetName As String) As String

    If Len(Trim$(CStr(ws.Range("A1").Value))) = 0 Or _
       Len(Trim$(CStr(ws.Range("A2").Value))) = 0 Then
        FINAL_XL_SlideStatus = "CHECK CONTENT"
        Exit Function
    End If

    If sheetName = "S5_Execution" And ws.ChartObjects.Count = 0 Then
        FINAL_XL_SlideStatus = "INPUT REQUIRED"
    ElseIf ws.ChartObjects.Count = 0 Then
        FINAL_XL_SlideStatus = "CHECK CHART"
    Else
        FINAL_XL_SlideStatus = "READY"
    End If

End Function
