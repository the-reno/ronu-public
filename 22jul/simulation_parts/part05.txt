    End With

    gClosestPattern = ClosestPatternName(distance)

    If gNextPolicyEffective > gDates(gLastIdx) Then
        totalDays = DateDiff("d", CDate(gDates(gLastIdx)), DateAdd("m", 1, CDate(gDates(gLastIdx))))
        beforeDays = DateDiff("d", CDate(gDates(gLastIdx)), CDate(gNextPolicyEffective))
        afterDays = totalDays - beforeDays
        If afterDays > 0 Then
            impliedPostON = (gRates(2, gLastIdx) * totalDays - gRates(1, gLastIdx) * beforeDays) / afterDays
            WriteMergedLine ws, 20, 1, 12, "Pure-expectations 1M illustration: implied post-meeting ON near " & Format(impliedPostON, "0.000%") & ". This is an upper-bound signal, not a forecast."
        End If

Private Sub BuildPatternStats(ByVal decisionName As String, ByRef output() As Variant, ByVal outputRow As Long, _
                              ByRef aggregateMean As Double, ByRef aggregateStd As Double)

    Dim e As Long, eventIdx As Long, preIdx As Long, t As Long
    Dim countValue As Long, slopeValues() As Double, tenorTotal(2 To 5) As Double

    ReDim slopeValues(1 To Application.Max(1, gFOMCCount))

    For e = 1 To gFOMCCount
        If gFOMCDecision(e) = decisionName Then
            eventIdx = FirstCurveIndexGE(gFOMCDates(e))
            If eventIdx > gAnalysisStartIdx + 10 And eventIdx <= gLastIdx Then
                countValue = countValue + 1
                preIdx = eventIdx - 10
                For t = 2 To 5
                    tenorTotal(t) = tenorTotal(t) + (gRates(t, preIdx) - gRates(1, preIdx)) * 10000#
                    slopeValues(countValue) = slopeValues(countValue) + (gRates(t, preIdx) - gRates(1, preIdx)) * 10000# / 4#
                Next t
            End If
        End If
    Next e

    output(outputRow, 1) = decisionName
    output(outputRow, 2) = countValue
    If countValue > 0 Then
        For t = 2 To 5
            output(outputRow, t + 1) = tenorTotal(t) / countValue
        Next t
        aggregateMean = MeanArray(slopeValues, countValue)
        aggregateStd = SampleStdArray(slopeValues, countValue)
        output(outputRow, 7) = aggregateMean
        output(outputRow, 8) = aggregateStd
    End If

End Sub


Private Function AverageFOMCReaction(ByVal decisionName As String, ByRef averageReaction() As Double) As Long

    Dim e As Long, eventIdx As Long, t As Long, countValue As Long

    For e = 1 To gFOMCCount
        If gFOMCDecision(e) = decisionName Then
            eventIdx = FirstCurveIndexGE(gFOMCDates(e))
            If eventIdx > gAnalysisStartIdx And eventIdx + 1 <= gLastIdx Then
                countValue = countValue + 1
                For t = 1 To 5
                    averageReaction(t) = averageReaction(t) + (gRates(t, eventIdx + 1) - gRates(t, eventIdx - 1)) * 10000#
                Next t
            End If
        End If
    Next e

    If countValue > 0 Then
        For t = 1 To 5
            averageReaction(t) = averageReaction(t) / countValue
        Next t
    End If
    AverageFOMCReaction = countValue

End Function


Private Function ClosestPatternName(ByRef distance() As Double) As String
    Dim bestIdx As Long, i As Long
    bestIdx = 1
    For i = 2 To 3
        If distance(i) < distance(bestIdx) Then bestIdx = i
    Next i
    Select Case bestIdx
        Case 1: ClosestPatternName = "Hike"
        Case 2: ClosestPatternName = "Hold"
        Case 3: ClosestPatternName = "Cut"
    End Select
End Function

Private Function ReadConfigDate(ByVal values As Object, ByVal key As String, ByVal defaultValue As Double) As Double
    If values.Exists(key) And IsDate(values(key)) Then
        ReadConfigDate = CDbl(CDate(values(key)))
    Else
        ReadConfigDate = defaultValue
    End If
End Function


Private Function ReadConfigOptionalDate(ByVal values As Object, ByVal key As String) As Double
    If values.Exists(key) And IsDate(values(key)) Then ReadConfigOptionalDate = CDbl(CDate(values(key)))
End Function


Private Function ReadConfigNumber(ByVal values As Object, ByVal key As String, ByVal defaultValue As Double) As Double
    If values.Exists(key) And IsNumeric(values(key)) Then
        ReadConfigNumber = CDbl(values(key))
    Else
        ReadConfigNumber = defaultValue
    End If
End Function
