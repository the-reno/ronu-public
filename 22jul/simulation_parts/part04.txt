
                portfolioCount = portfolioCount + 1
                portReturn(portfolioCount) = expectedReturn * 10000#
                portRisk(portfolioCount) = volatility * 10000#
                For i = 1 To 4
                    portWeights(i, portfolioCount) = weights(i)
                Next i

                If volatility > 0 Then
                    currentIR = expectedReturn / volatility
                    If bestIRIdx = 0 Or currentIR > bestIR Then
                        bestIR = currentIR
                        bestIRIdx = portfolioCount
                    End If
                End If
            Next c
        Next b
    Next a

    ReDim Preserve portReturn(1 To portfolioCount)
    ReDim Preserve portRisk(1 To portfolioCount)
    ReDim Preserve portWeights(1 To 4, 1 To portfolioCount)
    ReDim efficient(1 To portfolioCount)

    SortPortfoliosByRisk portRisk, portReturn, portWeights, portfolioCount

    bestIRIdx = 0
    bestIR = -1E+99
    For i = 1 To portfolioCount
        If portRisk(i) > 0 Then
            currentIR = portReturn(i) / portRisk(i)
            If bestIRIdx = 0 Or currentIR > bestIR Then
                bestIR = currentIR
                bestIRIdx = i
            End If
        End If
    Next i

    bestReturnAtRisk = -1E+99
    For i = 1 To portfolioCount
        If portReturn(i) > bestReturnAtRisk + 0.0000001 Then
            efficient(i) = True
            efficientCount = efficientCount + 1
            bestReturnAtRisk = portReturn(i)
        End If
    Next i

    ReDim pointRows(1 To efficientCount + 1, 1 To 7)
    pointRows(1, 1) = scenarioName
    pointRows(1, 2) = "Risk (bp)"
    pointRows(1, 3) = "Return (bp)"
    pointRows(1, 4) = "1M Weight"
    pointRows(1, 5) = "2M Weight"
    pointRows(1, 6) = "3M Weight"
    pointRows(1, 7) = "6M Weight"

    j = 1
    For i = 1 To portfolioCount
        If efficient(i) Then
            j = j + 1
            pointRows(j, 1) = scenarioName
            pointRows(j, 2) = portRisk(i)
            pointRows(j, 3) = portReturn(i)
            pointRows(j, 4) = portWeights(1, i)
            pointRows(j, 5) = portWeights(2, i)
            pointRows(j, 6) = portWeights(3, i)
            pointRows(j, 7) = portWeights(4, i)
        End If
    Next i

    ws.Range("J" & startRow).Resize(efficientCount + 1, 7).Value = pointRows
    ApplyHeader ws.Range("J" & startRow & ":P" & startRow)
    ws.Range("K" & startRow + 1 & ":L" & startRow + efficientCount).NumberFormat = "0.00;[Red](0.00);-"
    ws.Range("M" & startRow + 1 & ":P" & startRow + efficientCount).NumberFormat = "0%"

    Set chartRange = ws.Range("K" & startRow & ":L" & startRow + efficientCount)
    Set chartObj = ws.ChartObjects.Add(ws.Range("R" & startRow).Left, ws.Range("R" & startRow).Top, 520, 250)
    With chartObj.Chart
        .ChartType = xlXYScatterLinesNoMarkers
        .SetSourceData chartRange
        .HasTitle = True
        .ChartTitle.Text = scenarioName & " Efficient Frontier"
        .HasLegend = False
        .Axes(xlCategory).HasTitle = True
        .Axes(xlCategory).AxisTitle.Text = "Tracking Error (bp)"
        .Axes(xlValue).HasTitle = True
        .Axes(xlValue).AxisTitle.Text = "Expected Excess Return (bp)"
    End With

    summaryOutput(summaryRow, 1) = scenarioName
    summaryOutput(summaryRow, 2) = selectedCount
    summaryOutput(summaryRow, 3) = TenorName(bestPure + 1)
    summaryOutput(summaryRow, 4) = means(bestPure) * 10000#
    summaryOutput(summaryRow, 5) = WeightTextFromArray(portWeights, bestIRIdx)
    summaryOutput(summaryRow, 6) = portReturn(bestIRIdx)
    summaryOutput(summaryRow, 7) = portRisk(bestIRIdx)
    summaryOutput(summaryRow, 8) = bestIR

    If mode = 0 Then
        gOverallBestIRText = WeightTextFromArray(portWeights, bestIRIdx)
        gOverallBestIR = bestIR
    End If

    BuildOneFrontier = startRow + efficientCount

End Function


Private Function RegimeMatch(ByVal mode As Long, ByVal onChange As Double) As Boolean
    Select Case mode
        Case 0: RegimeMatch = True
        Case 1: RegimeMatch = (onChange > gRisingThreshold)
        Case 2: RegimeMatch = (onChange <= gRisingThreshold And onChange >= gFallingThreshold)
        Case 3: RegimeMatch = (onChange < gFallingThreshold)
    End Select
End Function


Private Sub ComputeMeanCovariance(ByRef selected() As Long, ByVal selectedCount As Long, _
                                  ByRef means() As Double, ByRef covariance() As Double)

    Dim i As Long, j As Long, k As Long, monthIdx As Long
    Dim xi As Double, xj As Double

    For i = 1 To 4
        For k = 1 To selectedCount
            monthIdx = selected(k)
            means(i) = means(i) + (gMonthlyReturns(i + 1, monthIdx) - gMonthlyReturns(1, monthIdx))
        Next k
        means(i) = means(i) / selectedCount * 12#
    Next i

    For i = 1 To 4
        For j = 1 To 4
            For k = 1 To selectedCount
                monthIdx = selected(k)
                xi = gMonthlyReturns(i + 1, monthIdx) - gMonthlyReturns(1, monthIdx)
                xj = gMonthlyReturns(j + 1, monthIdx) - gMonthlyReturns(1, monthIdx)
                covariance(i, j) = covariance(i, j) + (xi - means(i) / 12#) * (xj - means(j) / 12#)
            Next k
            covariance(i, j) = covariance(i, j) / (selectedCount - 1) * 12#
        Next j
    Next i

End Sub


Private Function WeightTextFromArray(ByRef weights() As Double, ByVal portfolioIdx As Long) As String
    Dim t As Long, textValue As String
    For t = 1 To 4
        If weights(t, portfolioIdx) > 0.0001 Then
            If Len(textValue) > 0 Then textValue = textValue & " / "
            textValue = textValue & Format(weights(t, portfolioIdx), "0%") & " " & TenorName(t + 1)
        End If
    Next t
    WeightTextFromArray = textValue
End Function
