' ============================================================================
' PUBLIC MACROS
' ============================================================================

Public Sub RunSOFRSimulation()

    Dim oldCalc As XlCalculation
    Dim started As Double

    On Error GoTo CleanFail

    started = Timer
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    oldCalc = Application.Calculation
    Application.Calculation = xlCalculationManual
    Application.DisplayAlerts = False

    Application.StatusBar = "SOFR model: validating inputs..."
    ValidateInputSheets
    LoadCurveData
    LoadConfiguration
    LoadFOMCData
    BuildMonthlyData

    Application.StatusBar = "SOFR model: historical curve..."
    BuildHistoricalCurveSheet

    Application.StatusBar = "SOFR model: premium versus realized..."
    BuildPremiumSheet

    Application.StatusBar = "SOFR model: starting-period returns..."
    BuildStartingPeriodsSheet

    Application.StatusBar = "SOFR model: risk and return..."
    BuildRiskReturnSheet

    Application.StatusBar = "SOFR model: efficient frontier..."
    BuildEfficientFrontierSheet

    Application.StatusBar = "SOFR model: next FOMC scenarios..."
    BuildNextFOMCSheet

    Application.StatusBar = "SOFR model: executive summary..."
    BuildExecutiveSheet
    BuildSourcesSheet

    ThisWorkbook.Worksheets("Executive").Activate
    Application.CalculateFull

    MsgBox "SOFR simulation completed." & vbCrLf & _
           "Curve period: " & Format(CDate(gDates(1)), "dd-mmm-yyyy") & _
           " to " & Format(CDate(gDates(gLastIdx)), "dd-mmm-yyyy") & vbCrLf & _
           "Elapsed time: " & Format(Timer - started, "0.0") & " seconds", vbInformation

CleanExit:
    Application.DisplayAlerts = True
    Application.Calculation = oldCalc
    Application.EnableEvents = True
    Application.ScreenUpdating = True
    Application.StatusBar = False
    Exit Sub

CleanFail:
    MsgBox "SOFR simulation stopped:" & vbCrLf & Err.Description, vbCritical
    Resume CleanExit

End Sub


' Compatibility alias for older buttons or workbooks.
Public Sub RunSOFRAnalysis()
    RunSOFRSimulation
End Sub


Private Sub ValidateInputSheets()

    Dim missing As String

    If Not SheetExists("Curve") Then missing = missing & vbCrLf & "- Curve"
    If Not SheetExists("Config") Then missing = missing & vbCrLf & "- Config"
    If Not SheetExists("FOMC") Then missing = missing & vbCrLf & "- FOMC"

    If Len(missing) > 0 Then
        Err.Raise vbObjectError + 1000, "RunSOFRSimulation", _
                  "Required input sheets are missing:" & missing & vbCrLf & vbCrLf & _
                  "Run BuildSOFRTemplate first."
    End If

End Sub


' ============================================================================
' DATA LOADING AND VALIDATION
' ============================================================================

Private Sub LoadCurveData()

    Dim ws As Worksheet
    Dim lastRow As Long, r As Long, n As Long
    Dim tempDates() As Double, tempRates() As Double
    Dim rateValue As Double, t As Long
    Dim uniqueCount As Long

    Set ws = ThisWorkbook.Worksheets("Curve")
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    If lastRow < 3 Then Err.Raise vbObjectError + 100, , "Curve requires at least two observations."

    ReDim tempDates(1 To lastRow - 1)
    ReDim tempRates(1 To TENOR_COUNT, 1 To lastRow - 1)

    For r = 2 To lastRow
        If IsDate(ws.Cells(r, 1).Value) Then
            If RowHasAllRates(ws, r) Then
                n = n + 1
                tempDates(n) = CDbl(CDate(ws.Cells(r, 1).Value))
                For t = 1 To TENOR_COUNT
                    rateValue = CDbl(ws.Cells(r, t + 1).Value)
                    tempRates(t, n) = NormalizeRate(rateValue)
                Next t
            End If
        End If
    Next r

    If n < 2 Then Err.Raise vbObjectError + 101, , "Curve contains fewer than two complete valid rows."

    QuickSortCurve tempDates, tempRates, 1, n

    ReDim gDates(1 To n)
    ReDim gRates(1 To TENOR_COUNT, 1 To n)

    uniqueCount = 0
    For r = 1 To n
        If uniqueCount = 0 Then
            uniqueCount = 1
            gDates(uniqueCount) = tempDates(r)
        ElseIf tempDates(r) <> gDates(uniqueCount) Then
            uniqueCount = uniqueCount + 1
            gDates(uniqueCount) = tempDates(r)
        End If
        For t = 1 To TENOR_COUNT
            gRates(t, uniqueCount) = tempRates(t, r)
        Next t
    Next r

    ReDim Preserve gDates(1 To uniqueCount)
    ReDim Preserve gRates(1 To TENOR_COUNT, 1 To uniqueCount)

    gCurveCount = uniqueCount
    gLastIdx = uniqueCount

    If gDates(gLastIdx) <= gDates(1) Then Err.Raise vbObjectError + 102, , "Curve dates are invalid."

End Sub
