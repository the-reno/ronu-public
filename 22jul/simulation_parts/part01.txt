Private Function RowHasAllRates(ByVal ws As Worksheet, ByVal rowNumber As Long) As Boolean
    Dim c As Long
    For c = 2 To 6
        If Not IsNumeric(ws.Cells(rowNumber, c).Value) Then Exit Function
    Next c
    RowHasAllRates = True
End Function


Private Function NormalizeRate(ByVal value As Double) As Double
    If Abs(value) > 1# Then
        NormalizeRate = value / 100#
    Else
        NormalizeRate = value
    End If
End Function


Private Sub LoadConfiguration()

    Dim ws As Worksheet
    Dim values As Object
    Dim lastRow As Long, r As Long
    Dim firstDate As Double, lastDate As Double

    Set ws = ThisWorkbook.Worksheets("Config")
    Set values = CreateObject("Scripting.Dictionary")

    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    For r = 4 To lastRow
        If Len(Trim$(CStr(ws.Cells(r, 1).Value))) > 0 Then
            values(Trim$(CStr(ws.Cells(r, 1).Value))) = ws.Cells(r, 2).Value
        End If
    Next r

    firstDate = gDates(1)
    lastDate = gDates(gLastIdx)

    gAnalysisStart = ReadConfigDate(values, "Analysis Start Date", firstDate)
    gAnalysisEnd = ReadConfigDate(values, "Analysis End Date", lastDate)
    gRiskStart = ReadConfigDate(values, "Risk Start Date", Application.Max(firstDate, CDbl(DateSerial(2023, 1, 1))))

    If gAnalysisStart < firstDate Then gAnalysisStart = firstDate
    If gAnalysisEnd > lastDate Or gAnalysisEnd = 0 Then gAnalysisEnd = lastDate
    If gAnalysisStart > gAnalysisEnd Then Err.Raise vbObjectError + 110, , "Analysis start is after analysis end."
    If gRiskStart < firstDate Then gRiskStart = firstDate
    If gRiskStart > gAnalysisEnd Then gRiskStart = gAnalysisStart

    gAnalysisStartIdx = FirstCurveIndexGE(gAnalysisStart)
    gLastIdx = LastCurveIndexLE(gAnalysisEnd)
    gRiskStartIdx = FirstCurveIndexGE(gRiskStart)

    gRisingThreshold = ReadConfigNumber(values, "Rising Regime Threshold (bp)", 5#)
    gFallingThreshold = ReadConfigNumber(values, "Falling Regime Threshold (bp)", -5#)
    gWeightStep = ReadConfigNumber(values, "Frontier Weight Step", 0.05)
    gNotional = ReadConfigNumber(values, "Analysis Notional", 100#)

    If gWeightStep <= 0 Or gWeightStep > 0.5 Then gWeightStep = 0.05
    If gNotional <= 0 Then gNotional = 100#

    gNextFOMCStart = ReadConfigOptionalDate(values, "Next FOMC Start Date")
    gNextFOMCDecision = ReadConfigOptionalDate(values, "Next FOMC Decision Date")
    gNextPolicyEffective = ReadConfigOptionalDate(values, "Next Policy Effective Date")
    gTargetLower = ReadConfigRate(values, "Current Target Lower")
    gTargetUpper = ReadConfigRate(values, "Current Target Upper")
    gBaseCase = ReadConfigText(values, "Presentation Base Case", "Hold")
    gCurveSource = ReadConfigText(values, "Curve Source", "User-provided local workbook")

    LoadStartingPeriods ws

End Sub


Private Sub LoadStartingPeriods(ByVal ws As Worksheet)

    Dim lastRow As Long, r As Long, countValue As Long
    Dim labelText As String

    lastRow = ws.Cells(ws.Rows.Count, 4).End(xlUp).Row
    If lastRow < 4 Then lastRow = 4

    ReDim gStartLabels(1 To Application.Max(1, lastRow - 3))
    ReDim gStartDates(1 To Application.Max(1, lastRow - 3))

    For r = 4 To lastRow
        labelText = Trim$(CStr(ws.Cells(r, 4).Value))
        If Len(labelText) > 0 And IsDate(ws.Cells(r, 5).Value) Then
            countValue = countValue + 1
            gStartLabels(countValue) = labelText
            gStartDates(countValue) = CDbl(CDate(ws.Cells(r, 5).Value))
        End If
    Next r

    If countValue = 0 Then
        countValue = 1
        gStartLabels(1) = Format(CDate(gDates(gAnalysisStartIdx)), "mmm-yy")
        gStartDates(1) = gDates(gAnalysisStartIdx)
    End If

    ReDim Preserve gStartLabels(1 To countValue)
    ReDim Preserve gStartDates(1 To countValue)
    gStartCount = countValue

End Sub


Private Sub LoadFOMCData()

    Dim ws As Worksheet
    Dim lastRow As Long, r As Long, n As Long
    Dim tempDate() As Double, tempDecision() As String
    Dim tempAction() As Double, tempSEP() As String

    Set ws = ThisWorkbook.Worksheets("FOMC")
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    If lastRow < 2 Then
        gFOMCCount = 0
        Exit Sub
    End If

    ReDim tempDate(1 To lastRow - 1)
    ReDim tempDecision(1 To lastRow - 1)
    ReDim tempAction(1 To lastRow - 1)
    ReDim tempSEP(1 To lastRow - 1)

    For r = 2 To lastRow
        If IsDate(ws.Cells(r, 1).Value) Then
            n = n + 1
            tempDate(n) = CDbl(CDate(ws.Cells(r, 1).Value))
            tempDecision(n) = NormalizeDecision(CStr(ws.Cells(r, 2).Value))
            If IsNumeric(ws.Cells(r, 3).Value) Then tempAction(n) = CDbl(ws.Cells(r, 3).Value)
            tempSEP(n) = CStr(ws.Cells(r, 4).Value)
        End If
    Next r

    If n = 0 Then
        gFOMCCount = 0
        Exit Sub
    End If

    QuickSortFOMC tempDate, tempDecision, tempAction, tempSEP, 1, n

    ReDim gFOMCDates(1 To n)
    ReDim gFOMCDecision(1 To n)
    ReDim gFOMCAction(1 To n)
    ReDim gFOMCSEP(1 To n)

    For r = 1 To n
        gFOMCDates(r) = tempDate(r)
        gFOMCDecision(r) = tempDecision(r)
        gFOMCAction(r) = tempAction(r)
        gFOMCSEP(r) = tempSEP(r)
    Next r
    gFOMCCount = n

    If gNextFOMCDecision = 0 Then
        For r = 1 To n
            If gFOMCDates(r) > gDates(gLastIdx) Then
                gNextFOMCDecision = gFOMCDates(r)
                Exit For
            End If
        Next r
    End If

End Sub


Private Function NormalizeDecision(ByVal decisionText As String) As String
    decisionText = UCase$(Trim$(decisionText))
    Select Case decisionText
        Case "HIKE", "INCREASE", "RAISE": NormalizeDecision = "Hike"
        Case "CUT", "DECREASE", "LOWER": NormalizeDecision = "Cut"
        Case "HOLD", "UNCHANGED", "NO CHANGE": NormalizeDecision = "Hold"
        Case Else: NormalizeDecision = ""
    End Select
End Function

Private Function TenorName(ByVal tenorIndex As Long) As String
    Select Case tenorIndex
        Case 1: TenorName = "ON"
        Case 2: TenorName = "1M"
        Case 3: TenorName = "2M"
        Case 4: TenorName = "3M"
        Case 5: TenorName = "6M"
    End Select
End Function


Private Function TenorMonths(ByVal tenorIndex As Long) As Long
    Select Case tenorIndex
        Case 2: TenorMonths = 1
        Case 3: TenorMonths = 2
        Case 4: TenorMonths = 3
        Case 5: TenorMonths = 6
    End Select
End Function


Private Function TargetMaturity(ByVal startDate As Double, ByVal tenorIndex As Long) As Double
    If tenorIndex = 1 Then
        TargetMaturity = startDate + 1
    Else
        TargetMaturity = CDbl(DateAdd("m", TenorMonths(tenorIndex), CDate(startDate)))
    End If
End Function


Private Function FirstCurveIndexGE(ByVal targetDate As Double) As Long

    Dim low As Long, high As Long, mid As Long, answer As Long

    low = 1
    high = gCurveCount
    answer = 0

    Do While low <= high
        mid = (low + high) \ 2
        If gDates(mid) >= targetDate Then
            answer = mid
            high = mid - 1
        Else
            low = mid + 1
        End If
    Loop

    FirstCurveIndexGE = answer

End Function


Private Function LastCurveIndexLE(ByVal targetDate As Double) As Long

    Dim low As Long, high As Long, mid As Long, answer As Long

    low = 1
    high = gCurveCount
    answer = 0

    Do While low <= high
        mid = (low + high) \ 2
        If gDates(mid) <= targetDate Then
            answer = mid
            low = mid + 1
        Else
            high = mid - 1
        End If
    Loop

    LastCurveIndexLE = answer

End Function


Private Function RolledONFactor(ByVal startIdx As Long, ByVal endIdx As Long) As Double

    Dim i As Long, factorValue As Double, dayCount As Long

    factorValue = 1#
    For i = startIdx To endIdx - 1
        dayCount = CLng(gDates(i + 1) - gDates(i))
        factorValue = factorValue * (1# + gRates(1, i) * dayCount / DAY_BASIS)
    Next i
    RolledONFactor = factorValue

End Function


Private Function RollingResult(ByVal startIdx As Long, ByVal tenorIndex As Long) As Double

    Dim baseValue As Double, currentIdx As Long, maturityIdx As Long
    Dim lockedRate As Double, dayCount As Long, elapsedDays As Long

    baseValue = 1#
    currentIdx = startIdx
    lockedRate = gRates(tenorIndex, currentIdx)
    maturityIdx = FirstCurveIndexGE(TargetMaturity(gDates(currentIdx), tenorIndex))

    Do While maturityIdx > 0 And maturityIdx <= gLastIdx And currentIdx < maturityIdx
        dayCount = CLng(gDates(maturityIdx) - gDates(currentIdx))
        baseValue = baseValue * (1# + lockedRate * dayCount / DAY_BASIS)
        currentIdx = maturityIdx
        If currentIdx >= gLastIdx Then
            RollingResult = baseValue
            Exit Function
        End If
        lockedRate = gRates(tenorIndex, currentIdx)
        maturityIdx = FirstCurveIndexGE(TargetMaturity(gDates(currentIdx), tenorIndex))
    Loop

    elapsedDays = CLng(gDates(gLastIdx) - gDates(currentIdx))
    RollingResult = baseValue * (1# + lockedRate * elapsedDays / DAY_BASIS)

End Function


Private Sub BuildWealthPath(ByVal startIdx As Long, ByVal tenorIndex As Long, ByRef values() As Double)

    Dim baseValue As Double, currentIdx As Long, maturityIdx As Long
    Dim lockedRate As Double, i As Long, dayCount As Long, elapsedDays As Long

    ReDim values(1 To gLastIdx)
    baseValue = 1#
    currentIdx = startIdx
    lockedRate = gRates(tenorIndex, currentIdx)
    maturityIdx = FirstCurveIndexGE(TargetMaturity(gDates(currentIdx), tenorIndex))

    For i = startIdx To gLastIdx
        Do While maturityIdx > 0 And i >= maturityIdx And currentIdx < i
            dayCount = CLng(gDates(maturityIdx) - gDates(currentIdx))
            baseValue = baseValue * (1# + lockedRate * dayCount / DAY_BASIS)
            currentIdx = maturityIdx
            If currentIdx >= gLastIdx Then Exit Do
            lockedRate = gRates(tenorIndex, currentIdx)
            maturityIdx = FirstCurveIndexGE(TargetMaturity(gDates(currentIdx), tenorIndex))
        Loop
        elapsedDays = CLng(gDates(i) - gDates(currentIdx))
        values(i) = baseValue * (1# + lockedRate * elapsedDays / DAY_BASIS)
    Next i

End Sub

Private Sub BuildMonthlyData()

    Dim monthEndIdx() As Long, countValue As Long
    Dim i As Long, previousKey As String, currentKey As String
    Dim t As Long, m As Long
    Dim paths() As Double, allPaths() As Double

    ReDim monthEndIdx(1 To gLastIdx - gRiskStartIdx + 1)

    For i = gRiskStartIdx To gLastIdx
        currentKey = Format$(CDate(gDates(i)), "yyyymm")
        If i = gRiskStartIdx Then previousKey = currentKey

        If currentKey <> previousKey Then
            countValue = countValue + 1
            monthEndIdx(countValue) = i - 1
            previousKey = currentKey
        End If
    Next i

    countValue = countValue + 1
    monthEndIdx(countValue) = gLastIdx

    If countValue < 3 Then Err.Raise vbObjectError + 120, , "Risk analysis requires at least three month-end observations."

    gMonthCount = countValue - 1
    ReDim gMonthDates(1 To gMonthCount)
    ReDim gMonthlyReturns(1 To TENOR_COUNT, 1 To gMonthCount)
    ReDim gMonthlyONChange(1 To gMonthCount)
    ReDim allPaths(1 To TENOR_COUNT, 1 To gLastIdx)

    For t = 1 To TENOR_COUNT
        BuildWealthPath gRiskStartIdx, t, paths
        For i = gRiskStartIdx To gLastIdx
            allPaths(t, i) = paths(i)
        Next i
    Next t

    For m = 1 To gMonthCount
        gMonthDates(m) = gDates(monthEndIdx(m + 1))
        gMonthlyONChange(m) = (gRates(1, monthEndIdx(m + 1)) - gRates(1, monthEndIdx(m))) * 10000#
        For t = 1 To TENOR_COUNT
            gMonthlyReturns(t, m) = allPaths(t, monthEndIdx(m + 1)) / allPaths(t, monthEndIdx(m)) - 1#
        Next t
    Next m

End Sub


Private Function MeanArray(ByRef values() As Double, ByVal countValue As Long) As Double
    Dim i As Long, total As Double
    If countValue <= 0 Then Exit Function
    For i = 1 To countValue
        total = total + values(i)
    Next i
    MeanArray = total / countValue
End Function


Private Function SampleStdArray(ByRef values() As Double, ByVal countValue As Long) As Double
    Dim i As Long, avg As Double, total As Double
    If countValue <= 1 Then Exit Function
    avg = MeanArray(values, countValue)
    For i = 1 To countValue
        total = total + (values(i) - avg) ^ 2
    Next i
    SampleStdArray = Sqr(total / (countValue - 1))
End Function


Private Function MedianArray(ByRef values() As Double, ByVal countValue As Long) As Double
    MedianArray = PercentileArray(values, countValue, 0.5)
End Function


Private Function PercentileArray(ByRef values() As Double, ByVal countValue As Long, ByVal percentileValue As Double) As Double

    Dim copyValues() As Double
    Dim position As Double, lowerIdx As Long, upperIdx As Long, fractionValue As Double
    Dim i As Long

    If countValue <= 0 Then Exit Function
    ReDim copyValues(1 To countValue)
    For i = 1 To countValue
        copyValues(i) = values(i)
    Next i
    QuickSortDouble copyValues, 1, countValue

    If percentileValue <= 0 Then
        PercentileArray = copyValues(1)
        Exit Function
    ElseIf percentileValue >= 1 Then
        PercentileArray = copyValues(countValue)
        Exit Function
    End If

    position = 1# + (countValue - 1#) * percentileValue
    lowerIdx = Int(position)
    upperIdx = lowerIdx + 1
    If upperIdx > countValue Then upperIdx = countValue
    fractionValue = position - lowerIdx
    PercentileArray = copyValues(lowerIdx) + fractionValue * (copyValues(upperIdx) - copyValues(lowerIdx))

End Function


Private Function CorrelationArrays(ByRef x() As Double, ByRef y() As Double, ByVal countValue As Long) As Double

    Dim meanX As Double, meanY As Double
    Dim sumXY As Double, sumXX As Double, sumYY As Double
    Dim i As Long

    If countValue <= 1 Then Exit Function
    meanX = MeanArray(x, countValue)
    meanY = MeanArray(y, countValue)

    For i = 1 To countValue
        sumXY = sumXY + (x(i) - meanX) * (y(i) - meanY)
        sumXX = sumXX + (x(i) - meanX) ^ 2
        sumYY = sumYY + (y(i) - meanY) ^ 2
    Next i

    If sumXX > 0 And sumYY > 0 Then CorrelationArrays = sumXY / Sqr(sumXX * sumYY)

End Function

Private Sub BuildHistoricalCurveSheet()

    Dim ws As Worksheet
    Dim monthEndIdx() As Long, n As Long, i As Long, r As Long, t As Long
    Dim currentKey As String, previousKey As String
    Dim output() As Variant
    Dim summary() As Variant
    Dim chartObj As ChartObject

    Set ws = PrepareOutputSheet("Historical Curve", "1. Historical Curve and FOMC Meetings")

    ReDim monthEndIdx(1 To gLastIdx - gAnalysisStartIdx + 1)
    For i = gAnalysisStartIdx To gLastIdx
        currentKey = Format$(CDate(gDates(i)), "yyyymm")
        If i = gAnalysisStartIdx Then previousKey = currentKey
        If currentKey <> previousKey Then
            n = n + 1
            monthEndIdx(n) = i - 1
            previousKey = currentKey
        End If
    Next i
    n = n + 1
    monthEndIdx(n) = gLastIdx

    ReDim output(1 To n + 1, 1 To 6)
    output(1, 1) = "Date"
    For t = 1 To TENOR_COUNT
        output(1, t + 1) = TenorName(t)
    Next t
    For r = 1 To n
        output(r + 1, 1) = CDate(gDates(monthEndIdx(r)))
        For t = 1 To TENOR_COUNT
            output(r + 1, t + 1) = gRates(t, monthEndIdx(r))
        Next t
    Next r

    ws.Range("A4").Resize(n + 1, 6).Value = output
    ApplyHeader ws.Range("A4:F4")
